"""Pure guard checks; fixtures are synthetic, not model acceptance or accuracy."""
import ast
import copy
import hashlib
import json
from pathlib import Path

L = Path(__file__).resolve().parent
body = (L / 'contract.py.txt').read_text()
ns = {}
exec(compile(body, 'R7_contract_only', 'exec'), ns)
count = 0


def rejects(fn, *args, **kwargs):
    global count
    try:
        fn(*args, **kwargs)
    except (ValueError, KeyError, AssertionError):
        count += 1
    else:
        raise AssertionError('Mutation was not rejected')


nb = {'nbformat': 4, 'cells': [{'cell_type': 'code', 'source': ['%%writefile /tmp/run.py\n', 'x = 1\n']}, {'cell_type': 'markdown', 'source': 'note'}, {'cell_type': 'code', 'source': 'print(x)\n'}]}
original = json.dumps(nb)
assert ns['exact_cells'](original, json.dumps(nb, sort_keys=True, indent=2))
count += 1
other = copy.deepcopy(nb)
other['cells'][0]['source'] = ''.join(other['cells'][0]['source'])
assert ns['exact_cells'](original, json.dumps(other))
count += 1
other['cells'][0]['source'] += 'x = 2\n'
rejects(ns['exact_cells'], original, json.dumps(other))
other = copy.deepcopy(nb)
other['cells'][0], other['cells'][2] = other['cells'][2], other['cells'][0]
rejects(ns['exact_cells'], original, json.dumps(other))
rejects(ns['cells'], json.dumps({'nbformat': 4, 'cells': []}))

    rejects(ns['metadata_contract'], {**metadata, key: wrong})
