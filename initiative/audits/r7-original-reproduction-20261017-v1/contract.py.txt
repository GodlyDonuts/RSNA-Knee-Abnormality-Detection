def cells(text):
    value = json.loads(text)
    if value.get("nbformat") != 4 or not isinstance(value.get("cells"), list):
        raise ValueError("Native original Notebook required")
    result = []
    for cell in value["cells"]:
        if cell.get("cell_type") == "code":
            content = cell["source"]
            if isinstance(content, list) and all(isinstance(x, str) for x in content):
                content = "".join(content)
            if not isinstance(content, str):
                raise ValueError("Code-cell bytes required")
            result.append(content)
    if not result:
        raise ValueError("Nonempty original code required")
    return result


def exact_cells(original, returned):
    left, right = cells(original), cells(returned)
    if left != right:
        raise ValueError("Every executed code cell and its order must stay byte identical")
    raw = json.dumps(left, ensure_ascii=False, separators=(",", ":")).encode()
    return hashlib.sha256(raw).hexdigest()

def budget(quota, *, unfinished_main, known_main_reserved, unfinished_sampling,
           blend_unfinished, scout_unfinished):
    def duration(value):
        if not isinstance(value, str) or not value.endswith("s"):
            raise ValueError("Fresh authoritative seconds required")
        seconds = float(value[:-1])
        if not math.isfinite(seconds) or seconds < 0:
            raise ValueError("Finite nonnegative duration required")
        return seconds
    if any(type(x) is not int or x < 0 for x in (unfinished_main, known_main_reserved, unfinished_sampling)):
        raise ValueError("All actual unfinished liabilities required")
    if type(blend_unfinished) is not bool or type(scout_unfinished) is not bool:
        raise ValueError("Explicit sibling obligations required")
    total, minimum, used, reserved = [duration(quota[k]) for k in ("totalTimeAllowed", "minimumTimeAllowed", "timeUsed", "timeReserved")]
    if not total >= minimum > 0:
        raise ValueError("Actual allowance required")
    allowed = total if quota.get("isPayToScaleEnabled") is True else minimum
    available = allowed - used - reserved + min(reserved, known_main_reserved)
    required = unfinished_main + unfinished_sampling + 3600 + 32400
    required += 32400 if blend_unfinished else 0
    required += 1800 if scout_unfinished else 0
    return {"allowed": available >= required, "funded_VM_seconds": available,
            "required_VM_seconds": required, "own_maximum_VM_seconds": 32400,
            "fixed_blend_VM_liability": 32400 if blend_unfinished else 0,
            "scout_VM_liability": 1800 if scout_unfinished else 0,
            "serving_reserve_seconds": 3600, "fresh_GPU_quota": quota,
            "no_account_or_billing_change": True}
