"""Freeze three personally reasoned OA decisions from the complete neutral source."""
import hashlib
import json
import shutil
from datetime import datetime, timezone
from pathlib import Path

P = Path(__file__).resolve().parent
W = P.parents[2]
D = P.parent
R = W / 'work/recovered'
AGENT = '/root/quality_temporal_review'
RUBRIC = '138c5cc33b92901d57dbf5fbe7e3a9878b73d25b0af6934908d0052cd6786014'


def read(path):
    return json.loads(path.read_text())


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def bind(path):
    return {'path': str(path.relative_to(W)), 'sha256': sha(path), 'bytes': path.stat().st_size}


def write(name, value):
    with (P / name).open('x') as f:
        f.write(json.dumps(value, ensure_ascii=False, indent=2) + '\n')

    assert shutil.disk_usage(W).free >= 20 * 1024**3
    supplied = D / 'cartilage-observed-002-root-review-v1/additional-neutral-full-source-packet.json'
    assert sha(supplied) == '9ce2fca78fbcd5709b5f6591dd0a0594e8931639c854bebe40f9455609be842a'
    sources = read(supplied)
    assert len(sources) == 1
    source = sources[0]
    assert source['study_uid'] == '1.2.826.0.1.3680043.8.498.21012635480218712816069264326295410954'
    original = R / source['source_path']
    assert sha(original) == source['source_sha256']
    assert original.stem == source['batch'] == 'batch-029' and source['index'] == 16
    assert read(original)[source['index']] == {'study_uid': source['study_uid'], 'report': source['report']}
    rubric = R / 'protocols/report_weak_v2.md'

    prior_complete = D / 'cartilage-observed-002-postfreeze-v1/completion-manifest.json'

    assert sha(prior_dispositions) == '4b94e52f83e52f51515690fc2fd103e2ed7740e1719dbeba2541123bc512bc0f'
    assert sha(prior_complete) == 'b64a1d6b67b30004aceaadf6e9f7142d2fa9d8e4fb1a0890663d4b0f298549b2'
    prior_packet = D / 'cartilage-observed-002-source-review-v1/neutral-full-reports.json'
    assert sum(r['study_uid'] == source['study_uid'] for r in read(prior_packet)) == 1
    with (P / 'neutral-full-sources.json').open('xb') as f:
        f.write(supplied.read_bytes())
    targets = ['Medial OA', 'Lateral OA', 'PF OA']
    states = ['negative', 'uncertain', 'uncertain']
    evidence = [
        'Cartílagos del compartimento\nfemorotibial medial sin alteraciones.',
        'La\nlínea úlceras condrales de espesor parcial y total del aspecto posterior de carga del\ncompartimento femorotibial lateral sin cambios óseos secundarios.',
        'Úlceras condrales de espesor total de distribución difusa de la\ntróclea femoral. Úlceras condral de espesor parcial de la faceta lateral de la rótula.',
    ]
    reasons = [
        'Medial femorotibial cartilage is explicitly without alterations. The separately described lateral and trochlear/patellar lesions do not propagate into the medial compartment.',
        'Partial- and full-thickness ulcers establish cartilage abnormality and focal full-depth involvement, but no qualifying roughly1 cm cartilage-loss region is measured or clearly described. Posterior weightbearing location is anatomical location, not lesion extent. The summary calls lesions focal grade3 without a named grading framework; that cannot fill the missing extent or negate the full-depth body finding. Preserve uncertainty rather than label every ulcer positive or use an unnamed numeric grade as negative.',
        'The body establishes full-thickness trochlear ulcers distributed diffusely and partial-thickness lateral patellar ulcers. Diffuse distribution of plural ulcers does not establish that a qualifying roughly1 cm region has greater-than50 percent loss; individual lesion extent or confluent regional loss is not specified. The summary grades2 and3 are unnamed and cannot be assigned a calibrated depth convention or used to resolve the body wording. I retain PF uncertainty on the missing region extent, without requiring two measured dimensions or inferring a contradiction solely from those numbers.',
    ]
    assert len(targets) == len(states) == len(evidence) == len(reasons) == 3
    for q in evidence:
        assert q and q in source['report']
    now = datetime.now(timezone.utc).isoformat()
    context = {
        'schema_version': 1, 'agent_id': AGENT, 'session_id': AGENT,
        'stage': 'additional_cartilage002_complete_source_reread_before_new_root_proposal',
        'recorded_at_utc': now, 'supplied_neutral_packet': bind(supplied),
        'rubric_version': 'report-weak-v2.0', 'rubric': bind(rubric),
        'prior_same_source_pass': bind(prior_source), 'prior_same_source_dispositions': bind(prior_dispositions),
        'prior_same_source_completion': bind(prior_complete), 'prior_neutral_packet': bind(prior_packet),
        'historical_exposure': 'This is a reread after my complete first cartilage002 source150 and postfreeze comparison, including this exact report. I have prior all260 cartilage first-source/OA comparisons, qualified000 full-source judgments and later additional adjudications, cartilage000/001 additional rereads and decisions, bone000 through007 source/comparison context, original/inherited report-review context, and technical064:14/46 source exposure. Historical same-source judgments are bound by hash but were not reopened before this new three-judgment freeze; root new case-specific002 findings/proposal and current selected annotation values have not been opened in this reread.',

    }
    write('source-pass-context.json', context)
    row = {

        'source_path': source['source_path'], 'source_index_zero_based': source['index'],
        'source_sha256': source['source_sha256'],
        'report_sha256': hashlib.sha256(source['report'].encode()).hexdigest(),
        'rubric_version': 'report-weak-v2.0', 'rubric_sha256': RUBRIC,
        'target_order': targets, 'states': states, 'evidence': evidence, 'rationales': reasons,
        'complete_source_personally_read': True,
