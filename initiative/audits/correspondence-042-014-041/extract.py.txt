from pathlib import Path
import hashlib
import json
import pandas as pd

STAGE = Path(__file__).resolve().parent
WORK = STAGE.parent.parent
REPO = WORK / "recovered"
REV = "52f995f4568e36bfa5fbbaf69a89af41226b4f5d"

def sha(path):
    h = hashlib.sha256()
    with path.open('rb') as f:
        for block in iter(lambda: f.read(1024 * 1024), b''):
            h.update(block)
    return h.hexdigest()

def bind(path):
    return {'path': str(path.relative_to(WORK)), 'sha256': sha(path), 'bytes': path.stat().st_size}

def write(name, value):
    path = STAGE / name
    with path.open('x') as f:
        f.write(json.dumps(value, ensure_ascii=False, indent=2) + '\n')
    return path

reports = json.loads((REPO / 'initiative/reviews/inputs/batch-042.json').read_text())
cases = {f'042{i:02d}': reports[i] for i in [14, 41]}
ids = {r['study_uid']: case for case, r in cases.items()}
headers = {case: [] for case in cases}
artifact = WORK / 'header-endpoint-full-v1/endpoint-headers.jsonl'
summary_path = artifact.parent / 'summary.json'
summary = json.loads(summary_path.read_text())
assert sha(artifact) == summary['endpoint_artifact_sha256'] == 'd5569c33e453bb1ccdf4600c39e2093821af7ffc1e7e9691f3a5d0ff97e258f3'
assert summary['complete_endpoint_coverage'] and not summary['failures']
for line in artifact.open():
    record = json.loads(line)
    if record['study_uid'] in ids:
        headers[ids[record['study_uid']]].append(record)
copies = []
for path in [REPO / 'initiative/source/train.csv', WORK / 'kaggle_current/train.csv']:
    assert sha(path) == '8ca2203c0e9d61c080c7a314c7cdb51c1b03a1d9eb4770819f7f34af53ef4e33'
    frame = pd.read_csv(path, usecols=['StudyInstanceUID', 'Report'], dtype=str, keep_default_na=False)
    copies.append((path, frame))
for case, source in cases.items():
    uid = source['study_uid']
    references = [bind(STAGE / f'{case}-source-preread.json'), bind(REPO / 'initiative/reviews/inputs/batch-042.json'), bind(artifact), bind(summary_path), bind(artifact.parent / 'manifest.json'), bind(Path(__file__).resolve())]
    copy_checks = []
    for path, frame in copies:
        match = frame.loc[frame.StudyInstanceUID.eq(uid)]
        assert len(match) == 1 and match.iloc[0].Report == source['report']
        copy_checks.append({**bind(path), 'one_uid_row': True, 'report_exact_match': True})
    copy_path = write(f'{case}-source-copy-assessment.json', {'study_uid': uid, 'source_report_sha256': hashlib.sha256(source['report'].encode()).hexdigest(), 'source_copies': copy_checks, 'scope': 'Blinded input and archived train.csv match the existing authenticated current Kaggle train.csv exactly. No fresh download; only UID and Report columns materialized. No clinical targets.'})
    references.append(bind(copy_path))
    inventory = pd.read_parquet(REPO / 'initiative/source/inventory', filters=[('study_uid', '==', uid)])
    assert set(inventory.split) == {'train_series'}
    records = inventory.sort_values(['series_uid', 'sop_uid']).to_dict('records')
    lookup = {r['path']: r for r in records}
    selected = headers[case]
    for record in selected:
        expected = lookup[record['source_path']]
        for a, b in [('study_uid', 'study_uid'), ('series_uid', 'series_uid'), ('sop_uid', 'sop_uid'), ('source_sha256', 'sha256'), ('source_bytes', 'bytes'), ('storage', 'storage')]:
            assert record[a] == expected[b]
        assert record['reached_pixel_data_header']
        assert not record['full_file_sha256_verified']
    inventory_path = write(f'{case}-inventory.json', {'study_uid': uid, 'source_revision': REV, 'n_series': int(inventory.series_uid.nunique()), 'n_slices': len(inventory), 'source_bytes': int(inventory.bytes.sum()), 'records': records})
    references.append(bind(inventory_path))
    endpoint_path = STAGE / f'{case}-endpoint-headers.jsonl'
    with endpoint_path.open('x') as f:
        for record in selected:
            f.write(json.dumps(record, ensure_ascii=False, sort_keys=True) + '\n')
    references.append(bind(endpoint_path))
    series = []
    for index, (sid, group) in enumerate(inventory.groupby('series_uid', sort=True)):
        endpoint = [r for r in selected if r['series_uid'] == sid]
        assert len(endpoint) == 2 and {r['sop_uid'] for r in endpoint} == {group.sop_uid.min(), group.sop_uid.max()}
        cache_path = WORK / 'cache256-v3/train_series' / uid / f'{sid}.npz'
        meta_path = cache_path.with_suffix('.json')
        meta = json.loads(meta_path.read_text())
        assert meta['source_revision'] == REV
        assert (meta['study_uid'], meta['series_uid'], meta['split']) == (uid, sid, 'train_series')
        assert meta['n_slices'] == len(group) == len(meta['sop_uids']) == len(set(meta['sop_uids']))
        assert dict(zip(meta['sop_uids'], meta['slice_sha256'])) == dict(zip(group.sop_uid, group.sha256))
        assert meta['path_fingerprint'] == hashlib.sha256('\n'.join(sorted(group.path)).encode()).hexdigest()
        assert meta['cache_bytes'] == cache_path.stat().st_size and meta['cache_sha256'] == sha(cache_path)
        references.extend([bind(meta_path), {'path':str(cache_path.relative_to(WORK)), 'sha256':meta['cache_sha256'], 'bytes':meta['cache_bytes']}])
        item = {'series_index': index, 'series_uid': sid, 'n_inventory_slices': len(group), 'n_endpoint_headers': len(endpoint), 'header_fields': {key: sorted({r['header_fields'][key] for r in endpoint}) for key in sorted(endpoint[0]['header_fields'])}, 'patient_id_sha256': sorted({r['patient_id_sha256'] for r in endpoint}), 'frame_of_reference_sha256': sorted({r['frame_of_reference_sha256'] for r in endpoint}), 'cache_metadata': {key:meta[key] for key in ['cache_version','ordering','warnings','original_shape','resized_shape','orientation','pixel_spacing','positions','n_slices','side','cache_sha256']}, 'cache_sop_source_sha_and_file_sha_verified': True}
        series.append(item)
    series_path = write(f'{case}-series-technical-metadata.json', series)
    references.append(bind(series_path))
    source_binding = write(f'{case}-technical-bindings.json', {'stage':'source_endpoint_cache_metadata_verification','agent_id':'/root/image_pipeline_audit','study_uid':uid,'source_revision':REV,'n_series':len(series),'n_inventory_slices':len(inventory),'source_bytes':int(inventory.bytes.sum()),'n_exact_lexical_endpoint_headers':len(selected),'n_patient_hashes_at_endpoints':len({r['patient_id_sha256'] for r in selected}),'n_frame_hashes_at_endpoints':len({r['frame_of_reference_sha256'] for r in selected}),'scope':'Exact full-study inventory and earlier full-cache source/UID/hash metadata checked; all cached NPZ bytes rehashed. Endpoint headers are source-bound partial HTTP prefixes, not all native headers. No original DICOM redecode or native image interpretation at this stage.','input_bindings':references})
    print(json.dumps({'case':case, 'n_series':len(series), 'n_slices':len(inventory), 'n_endpoints':len(selected), 'source_bytes':int(inventory.bytes.sum()),'n_patient_hashes':len({r['patient_id_sha256'] for r in selected}),'n_frame_hashes':len({r['frame_of_reference_sha256'] for r in selected}),'binding_sha256':sha(source_binding)}))
    for record in series:
        print(json.dumps({'series_index':record['series_index'],'series_uid':record['series_uid'],'n_slices':record['n_inventory_slices'],'cache_ordering':record['cache_metadata']['ordering'],'cache_warnings':record['cache_metadata']['warnings'], **{key:value for key,value in record['header_fields'].items() if key in ['Laterality','ImageLaterality','FrameLaterality','SeriesDescription','SeriesNumber','AcquisitionNumber','Rows','Columns','PixelSpacing','ImageOrientationPatient','PatientPosition']}},ensure_ascii=False))
