def ref(path):
    h = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            h.update(block)
    return {"path": str(path.relative_to(WORK)), "sha256": h.hexdigest(), "bytes": path.stat().st_size}


def references(value):
    if isinstance(value, dict):
        if {"path", "sha256", "bytes"} <= value.keys():
            yield {k: value[k] for k in ("path", "sha256", "bytes")}
        for item in value.values():
            yield from references(item)
    elif isinstance(value, list):
        for item in value:
            yield from references(item)

        assert agent["panel_manifest"] == root["panel_manifest"] == ref(manifest_path)
        assert agent["images"] == root["images"]
        assert agent["selected_series_personally_viewed"] == root["series_personally_viewed"] == manifest["series_sampled"]
        assert agent["matrices_personally_viewed"] == root["samples_personally_viewed"] == manifest["cached_panels"]
