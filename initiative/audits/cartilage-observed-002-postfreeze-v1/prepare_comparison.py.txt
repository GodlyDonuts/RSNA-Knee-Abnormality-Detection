"""Mechanical post-freeze comparison; no new semantic decisions are automated."""
import hashlib
import json
import sys
from datetime import datetime, timezone
from pathlib import Path
STAGE=Path(__file__).resolve().parent
WORKSPACE=STAGE.parents[2]
REPO=WORKSPACE/'work/recovered'
SOURCE=STAGE.parent/'cartilage-observed-002-source-review-v1'
sys.path.insert(0,str(REPO))
from rsna.adjudications import inspect_adjudications
from rsna.labels import source_reports_fingerprint
from rsna.schema import TARGETS

def sha(p): return hashlib.sha256(p.read_bytes()).hexdigest()
def read(p): return json.loads(p.read_text())
def bind(p): return {'path':str(p.relative_to(WORKSPACE)), 'sha256':sha(p)}
def write(p,x):
    with p.open('x') as f: f.write(json.dumps(x,ensure_ascii=False,indent=2)+'\n')

def main():
    assert sha(SOURCE/'source-freeze.json')=='043253f7442c7fb9727f4d73a603a8589dcc9c963c112f07414e312384fae411'
    reports={r['study_uid']:r['report'] for p in sorted((REPO/'initiative/reviews/inputs').glob('batch-*.json')) for r in read(p)}
    assert len(reports)==4407 and source_reports_fingerprint(reports)=='5c18838a09e9eec6522966555694fca83a9bae67e4d7c5c68592207ee4b60ed3'
    registry=REPO/'initiative/reviews-v2/adjudications/registry.json'
    pinned=sha(registry)
    assert pinned=='64b6b8b51efb693738e8a3f70743afe4b160c5b3acd1f4ec3300cc699e8fee12'
    replay=inspect_adjudications(reports,REPO/'initiative/reviews-v2/reviewed',REPO/'protocols/report_weak_v2.md',REPO)
    changes={(x['study_uid'],x['target']):x for x in replay['changes']}
    assert len(changes)==118 and sha(registry)==pinned
    with (STAGE/'registry-at-comparison.json').open('xb') as f:f.write(registry.read_bytes())
    own=[json.loads(l) for l in (SOURCE/'source-assessment-frozen.jsonl').read_text().splitlines()]
    sources=read(SOURCE/'neutral-full-reports.json')
    records=[]
    for s,o in zip(sources,own,strict=True):
        rp=REPO/'initiative/reviews-v2/reviewed'/f"{s['batch']}-reviewed.jsonl"
        original=next(x for x in map(json.loads,rp.read_text().splitlines()) if x['study_uid']==s['study_uid'])
        for k,target in enumerate(o['target_order']):
            j=TARGETS.index(target)
            correction=changes.get((s['study_uid'],target))
            after=correction['after'] if correction else {'state':original['states'][j],'evidence':original['evidence'][j]}
            records.append({'packet_index':o['packet_index'],'priority_index':o['priority_index'],'study_uid':s['study_uid'],'batch':s['batch'],'source_index':s['index'],'target':target,'source_path':s['source_path'],'source_sha256':s['source_sha256'],'source_report_sha256':o['source_report_sha256'],'original_review_path':str(rp.relative_to(REPO)),'original_review_sha256':sha(rp),'original_state':original['states'][j],'original_evidence':original['evidence'][j],'original_notes':original['notes'],'accepted_correction':correction,'baseline_effective_state':after['state'],'baseline_effective_evidence':after['evidence'],'own_frozen_state':o['states'][k],'own_frozen_evidence':o['evidence'][k],'own_frozen_rationale':o['rationales'][k]})
    write(STAGE/'own-source-vs-118-baseline.json',{'schema_version':1,'agent_id':'/root/quality_temporal_review','created_at_utc':datetime.now(timezone.utc).isoformat(),'stage':'mechanical_comparison_after_own_source_freeze','source_freeze':bind(SOURCE/'source-freeze.json'),'source_assessment':bind(SOURCE/'source-assessment-frozen.jsonl'),'neutral_source_packet':bind(SOURCE/'neutral-full-reports.json'),'registry':bind(registry),'preserved_registry_snapshot':bind(STAGE/'registry-at-comparison.json'),'registered_changes':len(changes),'scope_reports':50,'scope_targets':150,'records':records,'new_clinical_dispositions_from_this_mechanical_comparison':False,'second_reader_case_findings_opened':False,'gold_accessed':False,'mechanical_comparison_helper':bind(Path(__file__).resolve())})
    print(json.dumps({'cells':len(records),'initial_state_differences':sum(x['baseline_effective_state']!=x['own_frozen_state'] for x in records),'existing_correction_overlap':sum(x['accepted_correction'] is not None for x in records)},indent=2))
if __name__=='__main__':main()
