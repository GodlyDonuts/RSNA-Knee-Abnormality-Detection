CHECKED = {}


class Mismatch(Exception):
    pass


def require(condition, message):
    if not condition:
        raise Mismatch(message)


def same(actual, expected, context):
    if actual != expected:

        def short(value):
            encoded = json.dumps(value, sort_keys=True, default=str)
            return (
                value
                if len(encoded) < 300
                else {
                    "serialized_sha256": hashlib.sha256(encoded.encode()).hexdigest(),
                    "characters": len(encoded),
                }
            )

        raise Mismatch({"check": context, "actual": short(actual), "expected": short(expected)})


def read(path):
    return json.loads(path.read_text())


def sha(path):
    path = path.resolve()
    require(path.is_relative_to(WORK.resolve()), "Reference escapes workspace")
    require(
        path.suffix.lower() not in {".npz", ".npy", ".dcm", ".png", ".jpg", ".jpeg"},
        "Payload/image access forbidden",
    )
    h = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            h.update(block)
    return h.hexdigest()


def ref(path):
    path = path.resolve()
    return {"path": str(path.relative_to(WORK)), "sha256": sha(path), "bytes": path.stat().st_size}


def refs(value):
    if isinstance(value, dict):
        if (
            {"path", "sha256", "bytes"} <= value.keys()
            and "sop_uid" not in value
            and "storage" not in value
        ):
            path = (WORK / value["path"]).resolve()
            require(path.is_relative_to(WORK.resolve()), "Reference escapes workspace")
            same(path.stat().st_size, value["bytes"], "Referenced size: " + value["path"])
            key = str(path.relative_to(WORK))
            if key not in CHECKED:
                CHECKED[key] = sha(path)
            same(CHECKED[key], value["sha256"], "Referenced hash: " + key)
        for item in value.values():
            refs(item)
    elif isinstance(value, list):
        for item in value:
            refs(item)


def write_new(path, obj):
    payload = (json.dumps(obj, ensure_ascii=False, indent=2) + "\n").encode()
    require(
        shutil.disk_usage(WORK).free >= FLOOR + len(payload) + 2**20,
        "20 GiB floor would be breached",
    )
    require(path.parent == STAGE, "Output must remain in crosscheck staging")
    with path.open("xb") as stream:
        stream.write(payload)


def terminal_checks(report, source, root, correction):
    raw = report.encode()
    same(root.get("terminal_utf8_bytes_hex"), raw[-16:].hex(), "Root terminal 16 bytes")
    if "source_terminal_bytes" in source:
        expected = {
            "last_24_utf8_bytes_hex": raw[-24:].hex(),
            "last_40_characters_repr": repr(report[-40:]),
            "trailing_whitespace_repr": repr(report[len(report.rstrip()) :]),
            "ends_with_lf": report.endswith("\n"),
        }
        same(source["source_terminal_bytes"], expected, "Independent terminal byte record")
    if correction:
        same(
            correction["source_report_sha256"],
            hashlib.sha256(raw).hexdigest(),
            "Correction source digest",
        )
        expected = {
            "terminal_24_utf8_bytes_hex": raw[-24:].hex(),
            "terminal_40_characters_repr": repr(report[-40:]),
            "ends_with_lf": raw.endswith(b"\n"),
            "ends_with_cr": raw.endswith(b"\r"),
            "ends_with_ascii_space": raw.endswith(b" "),
            "trailing_ascii_whitespace_bytes_hex": raw[len(raw.rstrip(b" \t\r\n\v\f")) :].hex(),
        }
        for key, value in expected.items():
            same(correction[key], value, "Preserved correction: " + key)
        same(
            correction["original_completeness_limit"],
            source["completeness_limit"],
            "Correction binds original clause",
        )
        same(
            correction["correction_required"],
            correction["original_completeness_limit"] != correction["corrected_completeness_limit"],
            "Correction declaration",
        )


def pair_values(left, right, lm, rm):
    lf, rf = left["header_endpoint_values"], right["header_endpoint_values"]
    return {
        "series_numbers": [int(lf["SeriesNumber"][0]), int(rf["SeriesNumber"][0])],
        "series_uids": [left["series_uid"], right["series_uid"]],
        "n_slices": [lm["n_slices"], rm["n_slices"]],
        "literal_laterality": [lf["Laterality"], rf["Laterality"]],
        "same_native_shape": lm["original_shape"] == rm["original_shape"],
        "same_pixel_spacing": lm["pixel_spacing"] == rm["pixel_spacing"],
        "same_iop": lm["orientation"] == rm["orientation"],
        "max_abs_iop_component_difference": float(
            np.max(np.abs(np.asarray(lm["orientation"]) - np.asarray(rm["orientation"])))
        ),
        "same_all_ordered_origins": lm["positions"] == rm["positions"],
        "native_fingerprint_lists_equal": lm["pixel_sha256"] == rm["pixel_sha256"],
        "native_fingerprint_intersection": len(set(lm["pixel_sha256"]) & set(rm["pixel_sha256"])),
        "frame_hash_sets_equal": left["frame_of_reference_hashes"]
        == right["frame_of_reference_hashes"],
    }

def main():
    require(
        shutil.disk_usage(WORK).free > FLOOR + 2**20, "Insufficient reserved disk before replay"
    )
    for name, expected in PINS.items():
        actual = sha(WORK / name)
        same(actual, expected, "Pinned original " + name)
        CHECKED[name] = actual
    groups = [WORK / "annotation-staging" / f"technical-tier4-{name}-six-v1" for name in GROUPS]
    studies = {
        row["study_uid"]
        for group in groups
        for row in read(group / "source-only-preread.json")["records"]
    }
    require(len(studies) == 24, "Expected 24 unique source studies")
    inventory = pd.read_parquet(
        WORK / "cache256-v3-official-recovery-v1/inventory_effective.parquet",
        filters=[("study_uid", "in", sorted(studies))],
    )
    endpoints = []
    with (WORK / "header-endpoint-full-v1/endpoint-headers.jsonl").open() as stream:
        for line in stream:
            row = json.loads(line)
            if row["study_uid"] in studies:
                endpoints.append(row)
    lineage = {}
    with (WORK / "cache256-v3-official-recovery-v1/cache_recovery_lineage.jsonl").open() as stream:
        for line in stream:
            row = json.loads(line)
            if row["study_uid"] in studies:
                require(row["series_uid"] not in lineage, "Duplicate lineage series")
                lineage[row["series_uid"]] = row
    audit = pd.read_parquet(
        WORK / "dicom-audit-v3-official-recovery-v1/series_cache_index.parquet"
    ).set_index("series_uid")
    official = pd.read_csv(REPO / "initiative/source/train_series.csv").set_index(
        "SeriesInstanceUID"
    )
    report_tables = [
        pd.read_csv(
            WORK / name, usecols=["StudyInstanceUID", "Report"], keep_default_na=False
        ).set_index("StudyInstanceUID")
        for name in ("recovered/initiative/source/train.csv", "kaggle_current/train.csv")
    ]
    require(
        all(table.index.is_unique for table in report_tables), "Duplicate official report study"
    )
    outputs = []
    for name in GROUPS:
        try:
            result = replay_group(
                name, inventory, endpoints, lineage, audit, official, report_tables
            )
        except (Mismatch, KeyError, ValueError, TypeError, OSError) as error:
            result = {
                "schema_version": 1,
                "stage": "independent_contract_metadata_replay_mismatch",
                "agent_id": ASSESSOR,
                "group": name,
                "status": "mismatch",
                "error_type": type(error).__name__,
                "details": error.args[0] if error.args else "unspecified",
                "training_ready": False,
            }
        path = STAGE / f"{name}-metadata-replay.json"
        write_new(path, result)
        outputs.append(ref(path))
        print(
            json.dumps({"group": name, "status": result["status"], "result": outputs[-1]}),
            flush=True,
        )
    write_new(
        STAGE / "metadata-replay-index.json",
        {
            "schema_version": 1,
            "stage": "independent_contract_metadata_crosscheck_index",
            "agent_id": ASSESSOR,
            "results": outputs,
            "unique_file_bindings_verified": len(CHECKED),
            "checked_file_bindings": CHECKED,
            "pinned_originals": PINS,
            "adapted_from": [
                ref(WORK / "annotation-staging/root_verify_tier4_metadata_v9.py"),
                ref(WORK / "annotation-staging/root_verify_tier4_field_extents_v9.py"),
            ],
            "free_bytes_after": shutil.disk_usage(WORK).free,
            "required_floor_bytes": FLOOR,
            "training_ready": False,
        },
    )


if __name__ == "__main__":
    main()
