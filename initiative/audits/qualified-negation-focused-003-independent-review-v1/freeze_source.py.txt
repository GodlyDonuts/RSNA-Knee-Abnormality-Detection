"""Package600 personally authored target decisions and verify immutable source evidence."""
import hashlib
import json
import runpy
from collections import Counter
from datetime import datetime, timezone
from pathlib import Path

S = Path(__file__).resolve().parent
W = S.parents[2]
R = W / 'work/recovered'
AGENT = '/root/last_twelve_visual_review'
TARGETS = ['ACL', 'MCL', 'Medial Meniscus', 'Lateral Meniscus', 'Medial OA', 'Lateral OA', 'PF OA', 'Effusion', 'Synovitis', "Baker's", 'Contusion', 'Fracture']


def obj(p):
    return json.loads(p.read_bytes())


def ref(p):
    b = p.read_bytes()
    return {'path': str(p.relative_to(W)), 'sha256': hashlib.sha256(b).hexdigest(), 'bytes': len(b)}


def write(name, value):
    p = S / name
    with p.open('x') as f:
        json.dump(value, f, ensure_ascii=False, indent=2)
        f.write('\n')
    return p


packet = obj(S / 'neutral-source-packet.json')
binding = obj(S / 'source-bindings-and-exposure.json')
rubric = R / 'protocols/report_weak_v2.md'
assert ref(S / 'neutral-source-packet.json')['sha256'] == 'a64451d4ed5bc9d442ee760a65b2ca3748f504c43dc5f13b66d7fdc3bff1cfe8'
assert ref(rubric)['sha256'] == '138c5cc33b92901d57dbf5fbe7e3a9878b73d25b0af6934908d0052cd6786014'
assert (S / 'neutral-source-packet.json').read_bytes() == (W / binding['source_packet']['path']).read_bytes()
for prior in binding['prior_neutral_packet_bindings']:
    assert ref(W / prior['path']) == prior
rows = {}
helpers = sorted(S.glob('decisions_*.py'))
assert len(helpers) == 10
for h in helpers:
    authored = runpy.run_path(str(h))['ROWS']
    assert not rows.keys() & authored.keys()
    rows.update(authored)
assert len(packet) == 50 and set(rows) == set(range(50))
checks, out = [], []
for i, p in enumerate(packet):
    b = binding['bindings'][i]
    sp = R / p['source_path']
    assert ref(sp)['sha256'] == p['source_sha256']
    assert obj(sp)[p['index']] == {'study_uid': p['study_uid'], 'report': p['report']}
    assert b['study_uid'] == p['study_uid'] and b['source_index_zero_based'] == p['index']
    assert b['report_utf8_sha256'] == hashlib.sha256(p['report'].encode()).hexdigest()
    assert len(rows[i]) == 12
    for target, (state, quote, reason) in zip(TARGETS, rows[i]):
        assert state in {'positive', 'negative', 'uncertain', 'not_mentioned'}
        assert bool(quote) == (state != 'not_mentioned') and quote in p['report']
        assert reason.strip() and 'PLACEHOLDER' not in reason
        checks.append({'packet_index': i, 'study_uid': p['study_uid'], 'target': target,
                       'quote_start_character': p['report'].find(quote) if quote else None,
                       'quote_characters': len(quote),
                       'quote_utf8_sha256': hashlib.sha256(quote.encode()).hexdigest() if quote else None,
                       'exact_contiguous_or_missing_empty': True})
    out.append({'agent_id': AGENT, 'session_id': AGENT, 'packet_index': i,
                'case': f"{p['batch'].removeprefix('batch-')}:{p['index']}",
                'study_uid': p['study_uid'], 'source_path': p['source_path'],
                'source_sha256': p['source_sha256'], 'source_index_zero_based': p['index'],
                'report_utf8_sha256': b['report_utf8_sha256'], 'target_order': TARGETS,
                'states': [x[0] for x in rows[i]], 'evidence': [x[1] for x in rows[i]],
                'target_reasoning': [x[2] for x in rows[i]],
                'prior_exact_neutral_overlaps': [x for x in binding['exact_prior_packet_overlaps'] if x['packet_index'] == i],
                'notes': 'Independent full-source wording003 second pass. Historical exact source/clinical overlap and broader contextual exposure disclosed. No first wording003 findings or fresh current003 case annotations opened before freeze. Decisions remain unintegrated.'})
assert len(out) == len({x['study_uid'] for x in out}) == 50
assert len(checks) == len({(x['study_uid'], x['target']) for x in checks}) == 600
assessment = S / 'independent-assessment.jsonl'
with assessment.open('x') as f:
    for row in out:
        f.write(json.dumps(row, ensure_ascii=False) + '\n')
revision = write('pre-freeze-own-reconsiderations.json', {
    'agent_id': AGENT, 'at_utc': datetime.now(timezone.utc).isoformat(),
    'scope': 'Own source-only review before freezing, without first003/current003 case findings.',

         'initial_authored': {'state': 'negative', 'evidence': 'Gelenkkapsel nicht verdickt, kein wesentlicher Erguss.',
                             'reason': 'Explicit absence of joint-capsular thickening is a negative thickening assessment; no contradictory synovial inflammation is described.'},
         'final_source': {'state': rows[16][8][0], 'evidence': rows[16][8][1], 'reason': rows[16][8][2]},
         'reason': 'On own anatomical-scope check, joint capsule without thickening was not a specific synovial-lining assessment. Changed to not_mentioned with empty evidence; no clinical feedback was opened.'},
        {'packet_index': 45, 'study_uid': packet[45]['study_uid'], 'target': 'Medial Meniscus',

         'initial_evidence': 'Il y a une déchirure horizontale impliquant le corps et la corne postérieure s’étendant à la surface articulaire inférieure au niveau du corps. Il y a légère extrusion du ménisque au niveau du corps. Il n’y a pas de fragment méniscal déplacé. Le ménisque externe est sans particularité.',
         'final_evidence': rows[45][2][1],
         'reason': 'Expanded the quote backward to include explicit medial meniscus identity and omitted the unnecessary lateral sentence. State unchanged.'},
    ],
    'mechanical_validation_history': 'All first300 quotation checks passed. After the two authored refinements, all600 quotation checks passed. No source text was normalized or repaired.',
})
exposure = write('completed-source-exposure-and-chronology.json', {

    'preparation': ref(S / 'source-bindings-and-exposure.json'),
    'full_source_reading': 'All50 complete neutral original reports were personally read in ten sequential groups of five.     'rubric_use': 'The full unchanged rubric was personally reread during the immediately preceding legacy-three source task and reused here; its bytes/hash were rechecked. No rubric revision or invented numerical grading policy was applied.',
    'prior_exposure': binding['historical_exposure'],

validation = write('validation.json', {
    'agent_id': AGENT, 'at_utc': datetime.now(timezone.utc).isoformat(),

    'stage': 'qualified_negation_focused003_independent_second_full_source_50_reports_600_targets_v1',

manifest = write('final-file-list.json', {'agent_id': AGENT,
    'files': [ref(p) for p in sorted(S.rglob('*')) if p.is_file()], 'self_excluded': True})

    assert ref(W / item['path']) == item
