records = []
for n, target, state, quote, reason in decisions:
    r = packet['reports'][n]
    assert target in r['targets_for_this_assessment']
    assert (state == 'not_mentioned' and quote == '') or (quote and quote in r['report'])
    src = R / r['source_path']
    assert sha(src) == r['source_sha256']
    original = json.loads(src.read_text())
    if isinstance(original, dict):
        original = original.get('reports', original.get('records'))
    raw = original[r['source_index_zero_based']]
    assert raw['study_uid'] == r['study_uid'] and raw['report'] == r['report']
    assert hashlib.sha256(r['report'].encode()).hexdigest() == r['report_sha256']

        'target':target,'state':state,'source_quote':quote,'reason':reason,
