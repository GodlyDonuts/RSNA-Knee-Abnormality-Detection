def sha(p): return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def read(p): return json.loads(Path(p).read_text())
def bind(p): return {'path':str(Path(p).relative_to(W)), 'sha256':sha(p)}
def archive(p):
    p=Path(p)
    rel = ('initiative/audits/' + str(p.relative_to(S.parent))) if p.is_relative_to(S.parent) else str(p.relative_to(R))
    return {'path':rel,'sha256':sha(p)}
def write(n,d):
    if (S/n).exists():
        assert read(S/n)==d, n
        return
    with (S/n).open('x') as f: f.write(json.dumps(d,ensure_ascii=False,indent=2)+'\n')
