def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def pin(path, expected=None):
    path = path.resolve()
    assert path.is_relative_to(W) and path.is_file() and not path.is_symlink()
    row = {'path': str(path.relative_to(W)), 'sha256': sha(path), 'bytes': path.stat().st_size}
    assert expected is None or row['sha256'] == expected, ('binding_hash', row['path'])
    assert bound.setdefault(row['path'], row) == row
    return row


def ref(row):
    name = Path(row['path'])
    path = name if name.is_absolute() else (R / name if str(name).startswith(('initiative/', 'protocols/')) else W / name)
    actual = pin(path, row['sha256'])
    assert 'bytes' not in row or row['bytes'] == actual['bytes']
    return path


def read(path):
    pin(path)
    return json.loads(path.read_text())


def refs(value):
    if isinstance(value, dict):
        if isinstance(value.get('path'), str) and 'sha256' in value:
            ref(value)
        for v in value.values():
            refs(v)
    elif isinstance(value, list):
        for v in value:
            refs(v)


def idx(rows):
    out = {(r['study_uid'], r['target']): r for r in rows}
    assert len(out) == len(rows)
    return out


def ann(state, evidence):
    return {'state': state, 'evidence': evidence}


def git(*args):
    return subprocess.check_output(['git', *args], cwd=R, text=True).strip()


def quote(value, uid):
    global quote_checks
    assert value['state'] in {'positive', 'negative', 'uncertain', 'not_mentioned'}
    if value['state'] == 'not_mentioned':
        assert value['evidence'] == ''
    else:
        assert value['evidence'] and value['evidence'] in reports[uid]
        quote_checks += 1


def write(name, value):
    raw = (json.dumps(value, ensure_ascii=False, indent=2) + '\n').encode()
    scanner.Scanner(secrets).feed(raw)
    with (P / name).open('xb') as out:
        out.write(raw)
    return {'path': str((P / name).relative_to(W)), 'sha256': sha(P / name), 'bytes': len(raw)}
