"""Append late metadata bindings and seal the existing passed technical verification."""
from datetime import datetime, timezone
import hashlib
import importlib.util
import json
from pathlib import Path
import shutil
import sys

P = Path(__file__).resolve().parent
W = P.parents[2]
A = P.parent / 'qualified-negation-focused-004-independent-context-amendment-v1'
AGENT = '/root/duplicate_consistency_review'


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def pin(path):
    return {'path': str(path.relative_to(W)), 'sha256': sha(path), 'bytes': path.stat().st_size}


def verify(row):
    path = Path(row['path'])
    path = path if path.is_absolute() else W / path
    assert path.is_relative_to(W) and path.is_file() and not path.is_symlink()
    assert sha(path) == row['sha256'] and path.stat().st_size == row['bytes']


def write(name, value):
    raw = (json.dumps(value, ensure_ascii=False, indent=2) + '\n').encode()
    scanner.Scanner(secrets).feed(raw)
    with (P / name).open('xb') as out:
        out.write(raw)
    return pin(P / name)


scanner_path = P.parent / 'checkpoint-delta-helper-v2-v1/checkpoint_delta.py'
assert sha(scanner_path) == '58defa944734a6010f74feaf78362649dd9f23244fe5164f0587278197848dc8'
spec = importlib.util.spec_from_file_location('qualified004_final_scanner', scanner_path)
scanner = importlib.util.module_from_spec(spec)
sys.modules[spec.name] = scanner
spec.loader.exec_module(scanner)
secrets = scanner.secret_values(W / '.env')
assert secrets
assert sha(P / 'technical-verification.json') == '98a82378f968fa7dd207e563bcbce7418c672a177861242d599c8b1f55b28e51'
core = json.loads((P / 'technical-verification.json').read_text())
assert core['status'] == 'passed_independent_technical_verification_no_new_clinical_assessment'
for row in core['bound_inputs']:
    verify(row)
verify(core['actual491_inspection'])
supplement_path = A / 'completion-supplement-v2.json'
assert sha(supplement_path) == '57a615fdaa6bb6c55d22f74db4c947eb58283a7424a5aebc502daa9ae7f92610'
supplement = json.loads(supplement_path.read_text())
assert supplement['agent_id'] == '/root/last_twelve_visual_review'
assert supplement['context_amendments'] == 2 and supplement['clinical_state_changes'] == 0 and not supplement['registry_changed']
assert len(supplement['files']) == 16
for row in supplement['files']:
    verify(row)
verify(supplement['supersedes_manifest_scope_only'])

assert {W / row['path'] for row in supplement['files']} | {supplement_path} == {path for path in A.rglob('*') if path.is_file()}

assert sha(preservation) == 'b3467a123a3d680b60ac6e8105e1e3de3608d66180227ce4512fe5baeecf5bb6'

write('preserved-attempt-note.json', {
    'schema_version': 1,
    'agent_id': AGENT,
    'session_id': AGENT,
    'initial_helper': pin(P / 'verify_adjudication-attempt-01.py.txt'),
    'initial_failure_log': pin(P / 'attempt-01.log'),
    'final_helper': pin(P / 'verify_adjudication.py'),
