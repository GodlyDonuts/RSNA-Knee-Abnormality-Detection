owner = owner.replace("str(L/'run_phase_capacity_once.py')", "str(L/'run_phase_status_recovery_once.py')")

owner = owner.replace("prefix=role+'-epoch-'+str(epoch);prepared_path=L/(prefix+'-prepared.json')", """prefix=role+'-epoch-'+str(epoch);prepared_path=L/(prefix+'-prepared.json')
   # Reuse actual already preserved original epochs without re-downloading weights.

checks = []
for value in [{}, {'status': None}, {'status': 'QUEUED'}, {'status': 'RUNNING'}, {'status': 'COMPLETE'}, {'status': 'ERROR'}, {'status': 'CANCELLED'}]:

    checks.append({'response': value, 'classification': category})
