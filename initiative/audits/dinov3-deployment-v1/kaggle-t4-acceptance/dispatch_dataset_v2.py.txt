"""Stream the selected credential privately to one CPU-only coordinator, never argv/logs."""
import hashlib
import json
import shlex
import subprocess
from pathlib import Path

O=Path(__file__).resolve().parents[2]
STAGE=Path(__file__).resolve().parent


def main():
    receipt=STAGE/'final-project-cpu-receipt.json'
    expected=hashlib.sha256(receipt.read_bytes()).hexdigest()
    c=json.loads((O/'artifacts/newton-access-control/connection.json').read_text())
    ssh=['ssh','-S',c['control_socket'],'-o','BatchMode=yes','-o','StrictHostKeyChecking=yes','-o','ControlMaster=no',c['user']+'@'+c['host']]
    code=(STAGE/'publish_private_wheels.py').read_text()
    install='import sys;from pathlib import Path;p=Path.home()/"rsna-knee/kaggle-wheelhouse-v1/gpu-probe-input-v2";p.mkdir(mode=448);(p/"publish_private_wheels.py").write_text(sys.stdin.read())'
    subprocess.run(ssh+['python3 -c '+shlex.quote(install)],input=code,text=True,check=True,capture_output=True)
    intent={'purpose':'one_private_wheelhouse_dataset_cpu_coordinator','expected_receipt_sha256':expected,
            'publisher_sha256':hashlib.sha256(code.encode()).hexdigest(),'competition_submission':False,
            'token_transport':'SSH standard input only; no persistent credential file; never command-line arguments'}
    with (STAGE/'coordinator-dispatch-intent-v2.json').open('x') as f:json.dump(intent,f,indent=2)
    env=O/'.env'
    if env.stat().st_mode & 0o077:raise RuntimeError('Private credential mode required')
    token=None
    for line in env.read_text().splitlines():
        key,sep,value=line.partition('=')
        if sep and key.strip()=='KAGGLE_API_TOKEN':token=value.strip().strip('\"\'')
    if not token or not token.startswith('KGAT_'):raise RuntimeError('Selected credential absent')
    command='module load python/python-3.11.4-gcc-12.2.0; unset PYTHONPATH; export PYTHONNOUSERSITE=1; exec srun --account=skattel --partition=normal --constraint=h100 --gres=none --cpus-per-task=2 --mem=8G --time=00:30:00 python3.12 "$HOME/rsna-knee/kaggle-wheelhouse-v1/gpu-probe-input-v2/publish_private_wheels.py"'
    r=subprocess.run(ssh+['bash -l -c '+shlex.quote(command)],input=json.dumps({'token':token,'expected_receipt_sha256':expected}),text=True,stdout=subprocess.PIPE,stderr=subprocess.PIPE)
    # Only our JSON records are projected; never raw SDK/transport stderr.
    records=[]
    for line in r.stdout.splitlines():
        try:value=json.loads(line)
        except ValueError:continue
        if isinstance(value,dict) and value.get('status'):records.append(value)
    result={'returncode':r.returncode,'records':records,'retry_automatically':False}
    (STAGE/'coordinator-dispatch-result-v2.json').write_text(json.dumps(result,indent=2))
    print(json.dumps(result))


if __name__=='__main__':
    try:main()
    except Exception as exc:
        print(json.dumps({'status':'blocked','error_type':type(exc).__name__,'retry_automatically':False}));raise SystemExit(1) from None
