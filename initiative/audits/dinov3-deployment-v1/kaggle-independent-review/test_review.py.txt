@pytest.mark.parametrize('extra', ['traversal','absolute','duplicate','hardlink','payload_tamper','manifest_tamper'])
def test_archive_structure_or_member_tamper_blocks(tmp_path, extra):
    bundle,inputs,manifest=f.fixture_archive(tmp_path)
    archive=inputs/'wheelhouse-cp312-cu130.bin'
    if extra in {'payload_tamper','manifest_tamper'}:
        with tarfile.open(archive)as handle:
            contents={item.name:handle.extractfile(item).read()for item in handle.getmembers()}
        name='wheelhouse/fixture.whl' if extra=='payload_tamper' else 'wheelhouse/wheelhouse.json'
        contents[name]+=b' '
        with tarfile.open(archive,'w')as handle:
            for name,data in contents.items():
                item=tarfile.TarInfo(name);item.size=len(data);handle.addfile(item,io.BytesIO(data))
    else:
        with tarfile.open(archive,'a')as handle:
            name={'traversal':'wheelhouse/../../escape','absolute':'/escape','duplicate':'wheelhouse/fixture.whl','hardlink':'wheelhouse/link'}[extra]
            item=tarfile.TarInfo(name)
            if extra=='hardlink':
                item.type=tarfile.LNKTYPE;item.linkname='wheelhouse/fixture.whl';handle.addfile(item)
            else:
                item.size=1;handle.addfile(item,io.BytesIO(b'x'))
    # Simulates a newly bound but invalid archive; payload/structure checks must still fail.
    manifest['shared_wheelhouse']['archive_sha256']=w.digest(archive)
    manifest['shared_wheelhouse']['archive_bytes']=archive.stat().st_size
    with pytest.raises((ValueError,FileExistsError)):
        n.resolve_wheels(bundle,manifest,inputs,tmp_path/'output')
    assert not(tmp_path/'escape').exists()

@pytest.mark.parametrize('case',['later_target','duplicate_page','foreign_ref','bad_rows','unbounded','valid_omitted_empty'])
def test_owned_listing_403_refusals_and_exact_empty_semantics(case):
    remote=w.Remote.__new__(w.Remote);remote.username='owner';calls=[]
    def call(service,method,body):
        calls.append((method,body))
        if method=='GetKernel':raise w.RemoteError('Kaggle GetKernel HTTP 403')
        page=body['page']
        if case=='later_target':return {'kernels':[{'ref':'owner/previous' if page==1 else 'owner/target'}]}
        if case=='duplicate_page':return {'kernels':[{'ref':'owner/repeated'}]}
        if case=='foreign_ref':return {'kernels':[{'ref':'someone/target'}]}
        if case=='bad_rows':return {'kernels':None}
        if case=='unbounded':return {'kernels':[{'ref':f'owner/page-{page}'}]}
        return {'kernels':[{'ref':'owner/previous'}]} if page==1 else {}
    remote.call=call
    if case=='valid_omitted_empty':
        result=remote.require_absent_kernel('owner/target')
        assert result['pages']==2 and result['count']==1
        assert result['refs_sha256']==w.object_hash(['owner/previous'])
    else:
        with pytest.raises((ValueError,w.RemoteError)):
            remote.require_absent_kernel('owner/target')
    assert len(calls)<=101
    assert all(method!='SaveKernel' for method,_ in calls)
