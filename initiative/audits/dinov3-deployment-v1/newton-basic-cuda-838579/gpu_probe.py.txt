"""Bounded allocated-device visibility probe; never expands CUDA visibility."""
import argparse
import json
import os
import pathlib
import stat
import subprocess
import sys


def run(command, env=None, timeout=15):
    try:
        result=subprocess.run(command,capture_output=True,text=True,env=env,timeout=timeout)
        return {'returncode':result.returncode,'stdout':result.stdout[-12000:],'stderr':result.stderr[-4000:]}
    except subprocess.TimeoutExpired:
        return {'status':'timeout','seconds':timeout}


def main():
    parser=argparse.ArgumentParser();parser.add_argument('--phase',required=True);args=parser.parse_args()
    if not os.environ.get('SLURM_JOB_ID') or not os.environ.get('CUDA_VISIBLE_DEVICES'):
        raise RuntimeError('Allocated visible devices required')
    record={'phase':args.phase,'environment':{k:os.environ.get(k) for k in ['SLURM_JOB_ID','SLURM_STEP_ID','SLURM_JOB_GPUS','SLURM_STEP_GPUS','SLURM_GPUS_ON_NODE','SLURM_GPUS_PER_TASK','CUDA_VISIBLE_DEVICES','CUDA_DEVICE_ORDER','LD_LIBRARY_PATH','LD_PRELOAD','PYTHONHOME']},'cpu_affinity':sorted(os.sched_getaffinity(0)),'devices':[]}
    for path in sorted(pathlib.Path('/dev').glob('nvidia*')):
        value=path.stat()
        record['devices'].append({'name':path.name,'mode':oct(stat.S_IMODE(value.st_mode)),'major':os.major(value.st_rdev),'minor':os.minor(value.st_rdev),'read_write':os.access(path,os.R_OK|os.W_OK)})
    record['nvidia_smi']=run(['nvidia-smi','--query-gpu=index,uuid,pci.bus_id,name,driver_version','--format=csv,noheader'])
    record['cgroup']=pathlib.Path('/proc/self/cgroup').read_text()
    driver=r'''
import ctypes,json
from pathlib import Path
r={};library=ctypes.CDLL('libcuda.so.1');r['cuInit']=library.cuInit(0)
count=ctypes.c_int();r['cuDeviceGetCount']=library.cuDeviceGetCount(ctypes.byref(count));r['count']=count.value
r['mapped_driver_paths']=sorted({x.split()[-1] for x in Path('/proc/self/maps').read_text().splitlines() if 'libcuda' in x or 'libnvidia' in x})
print(json.dumps(r))
'''
    record['driver']=run([sys.executable,'-c',driver])
    torch_probe=r'''
import json,torch
from pathlib import Path
r={'torch':str(torch.__version__),'cuda_build':torch.version.cuda,'device_count':torch.cuda.device_count()}
try:
 torch.cuda.init();r['init']='passed';r['devices']=[]
 for i in range(torch.cuda.device_count()):
  with torch.cuda.device(i):
   x=torch.ones(1,device='cuda');torch.cuda.synchronize();r['devices'].append({'index':i,'name':torch.cuda.get_device_name(i),'sum':float(x.sum())})
except Exception as e:r['error_type']=type(e).__name__;r['error']=str(e)
r['mapped_driver_paths']=sorted({x.split()[-1] for x in Path('/proc/self/maps').read_text().splitlines() if 'libcuda' in x or 'libnvidia' in x})
print(json.dumps(r))
'''
    record['torch_original_order']=run([sys.executable,'-c',torch_probe],timeout=20)
    ordered=os.environ.copy();ordered['CUDA_DEVICE_ORDER']='PCI_BUS_ID'
    assert ordered['CUDA_VISIBLE_DEVICES']==os.environ['CUDA_VISIBLE_DEVICES']
    record['torch_pci_order']=run([sys.executable,'-c',torch_probe],env=ordered,timeout=20)
    print(json.dumps(record,sort_keys=True),flush=True)


if __name__=='__main__':main()
