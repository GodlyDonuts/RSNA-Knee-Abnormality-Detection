def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def module(path, name):
    spec = importlib.util.spec_from_file_location(name, path)
    obj = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(obj)
    return obj


def same(left, right):
    import torch
    if isinstance(left, torch.Tensor):
        return isinstance(right, torch.Tensor) and left.dtype == right.dtype and left.shape == right.shape and torch.equal(left, right)
    if type(left) is not type(right):
        return False
    if isinstance(left, dict):
        return left.keys() == right.keys() and all(same(v, right[k]) for k, v in left.items())
    if isinstance(left, (list, tuple)):
        return len(left) == len(right) and all(same(a, b) for a, b in zip(left, right, strict=True))
    return left == right


def paired_metrics(labels, models, library, *, draws=4000, seed=20261017):
    if type(draws) is not int or draws < 100 or type(seed) is not int or not 0 <= seed < 2**32:
        raise ValueError('A fixed positive bootstrap plan is required')
    if not len(labels) or set(models) != set(ROLES) or not labels.index.is_unique or labels.leakage_group.isna().any() or labels.leakage_group.eq('').any():
        raise ValueError('Three exact model roles and complete unique component membership required')
    arrays = {}
    for role, frame in models.items():
        if not frame.index.is_unique or set(frame.index) != set(labels.index):
            raise ValueError('Prediction study membership differs')
        values = frame.loc[labels.index, list(TARGETS)].to_numpy(float)
        if not np.isfinite(values).all() or (values < 0).any() or (values > 1).any():
            raise ValueError('Finite probabilities required')
        arrays[role] = values
    plans, points, targets = [], {r: [] for r in ROLES}, []
    for j, target in enumerate(TARGETS):
        observed = labels[target + '__mask'].to_numpy()
        if not np.isin(observed, [0, 1]).all():
            raise ValueError('Original binary observation masks required')
        mask = observed == 1
        truth = labels.loc[mask, target].to_numpy(float)
        if not np.isfinite(truth).all() or not np.isin(truth, [0, 1]).all():
            raise ValueError('Original observed binary labels required')
        plan = {r: library.auc_plan(truth, x[mask, j]) if len(truth) else None for r, x in arrays.items()}
        values = {r: library.weighted_auc(p, np.ones(len(truth))) if p is not None else None for r, p in plan.items()}
        for role in ROLES:
            points[role].append(values[role])
        targets.append({'target': target, 'auc': values, 'observed_studies': int(mask.sum()), 'positive_studies': int(truth.sum()), 'negative_studies': int(len(truth) - truth.sum()), 'positive_components': int(labels.loc[mask & labels[target].eq(1), 'leakage_group'].nunique()), 'negative_components': int(labels.loc[mask & labels[target].eq(0), 'leakage_group'].nunique())})
        plans.append((mask, plan))
    macro = {r: float(np.mean(v)) if all(x is not None for x in v) else None for r, v in points.items()}
    rng = np.random.default_rng(seed)
    samples = {a + '_minus_' + b: {'targets': [[] for _ in TARGETS], 'macro': []} for a, b in CONTRASTS}
    draw_sha = hashlib.sha256()
    for _ in range(draws):
        weights = library.component_weights(labels.leakage_group.to_list(), rng)
        draw_sha.update(weights.astype('<i8').tobytes())
        differences = {key: [] for key in samples}
        for j, (mask, plan) in enumerate(plans):
            values = {r: library.weighted_auc(p, weights[mask]) if p is not None else None for r, p in plan.items()}
            for a, b in CONTRASTS:
                key = a + '_minus_' + b
                delta = None if values[a] is None or values[b] is None else values[a] - values[b]
                samples[key]['targets'][j].append(delta)
                differences[key].append(delta)
        for key, values in differences.items():
            samples[key]['macro'].append(float(np.mean(values)) if all(x is not None for x in values) else None)
    contrasts = {}
    for a, b in CONTRASTS:
        key = a + '_minus_' + b
        contrasts[key] = {'macro_delta': None if macro[a] is None or macro[b] is None else macro[a] - macro[b], 'macro_uncertainty': library.interval(samples[key]['macro'], draws), 'targets': [{'target': t, 'delta': None if points[a][j] is None or points[b][j] is None else points[a][j] - points[b][j], 'uncertainty': library.interval(samples[key]['targets'][j], draws)} for j, t in enumerate(TARGETS)]}
    return {'macro_auc': macro, 'targets': targets, 'contrasts': contrasts, 'studies': len(labels), 'components': int(labels.leakage_group.nunique()), 'bootstrap_draws': draws, 'bootstrap_seed': seed, 'paired_whole_component_draws_sha256': draw_sha.hexdigest(), 'undefined_policy': 'Keep all12 targets; count each undefined draw, never drop a target, repair classes or redraw.', 'uncertainty_scope': 'Conditional on defined draws, these fitted models, this one seed/fold and weak-label development pool; not clinical truth or training/search uncertainty.'}
