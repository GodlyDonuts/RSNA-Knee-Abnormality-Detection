class GuardError(Exception):
    pass


def require(value, code):
    if not value:
        raise GuardError(code)


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def data(value):
    return (json.dumps(value, ensure_ascii=False, indent=2) + '\n').encode()


def read(path, expected=None):
    require(path.is_file() and not path.is_symlink(), 'input_path')
    require(expected is None or sha(path) == expected, 'input_hash')
    return json.loads(path.read_text())


def idx(rows):
    result = {(row['study_uid'], row['target']): row for row in rows}
    require(len(result) == len(rows), 'duplicate_keys')
    return result


def safe(root, name):
    relative = Path(name)
    require(not relative.is_absolute() and '..' not in relative.parts and str(relative) == name and '\\' not in name, 'unsafe_relative_path')
    path = root / relative
    require(path.resolve().is_relative_to(root.resolve()) and not path.is_symlink(), 'path_outside_scope_or_symlink')
    return path


def check_file(path, record):
    require(path.is_file() and not path.is_symlink(), 'missing_or_symlink_input')
    require(path.stat().st_size == record['bytes'] and sha(path) == record['sha256'], 'input_bytes_changed')
    require('mode' not in record or stat.S_IMODE(path.stat().st_mode) == record['mode'], 'input_mode_changed')


def module(path, name, expected):
    require(sha(path) == expected, 'module_hash')
    spec = importlib.util.spec_from_file_location(name, path)
    result = importlib.util.module_from_spec(spec)
    sys.modules[name] = result
    spec.loader.exec_module(result)
    return result


def equal_or_new(path, raw, mode=0o644):
    if path.exists():
        require(path.is_file() and not path.is_symlink() and path.read_bytes() == raw and stat.S_IMODE(path.stat().st_mode) == mode, 'existing_archive_differs')
        return 'reused_equal'
    return 'new'


def put(path, raw, mode=0o644):
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open('xb') as out:
        out.write(raw)
        out.flush()
        os.fsync(out.fileno())
        os.fchmod(out.fileno(), mode)

def execute_root(plan):
    c, R, W, I = plan['context'], plan['R'], plan['W'], plan['I']
    registry_path = c['registry_path']
    original_registry = registry_path.read_bytes()
    require(sha(registry_path) == BASE_REGISTRY, 'registry_prewrite_fence')
    require(shutil.disk_usage(W).free - plan['counts']['new_archive_bytes'] - RESERVE > FLOOR, 'prewrite_disk_floor')
    now = datetime.now(timezone.utc).isoformat()

    def reference(path, digest=None):
        return {'path': str(path.relative_to(R)), 'sha256': digest or sha(path)}

    decisions, wrappers = [], []
    for bundle in plan['bundles']:
        decision = {'schema_version': 1, 'status': 'accepted', 'stage': 'independent_adjudication_decision', 'agent_id': '/root', 'session_id': '/root', 'adjudicator_agent_id': '/root', 'adjudicator_session_id': '/root', 'proposer_agent_id': bundle['proposer'], 'proposer_session_id': bundle['proposer'], 'accepted_at_utc': now, 'source_reports_sha256': CANONICAL, 'rubric_sha256': RUBRIC, 'proposal_sha256': sha(bundle['proposal_path']), 'independent_assessment_sha256': sha(bundle['assessment_path']), **{field: sha(path) for field, path in plan['supports'].items()}, 'accepted_changes': bundle['accepted_rows_for_root_execution'], 'n_changes': len(bundle['changes']), 'prepared_by': AGENT, 'executed_by': '/root', 'original_artifacts_preserved': True, 'decision_statement': 'Actual root execution serializes root personally authored and frozen support only. All63 FIRST and73 SECOND original objects, two original-author evidence amendments,27 root reconsiderations and nine SECOND/four FIRST state rejections remain exact. Four additional root proposals are excluded and retain their current annotations. Original proposal reasons remain separate from root reasons.', 'global_semantic_qa_closed': False, 'training_ready': False, 'gold_values_inspected': False}
        wrapper = {'schema_version': 1, 'id': f"qualified-negation-focused-004-{bundle['name']}-20260922", 'source_reports_sha256': CANONICAL, 'rubric_sha256': RUBRIC, 'proposal': reference(plan['archive'](bundle['proposal_path']), sha(bundle['proposal_path'])), 'independent_assessment': reference(plan['archive'](bundle['assessment_path']), sha(bundle['assessment_path'])), 'decision': reference(bundle['decision_path'], hashlib.sha256(data(decision)).hexdigest()), 'supporting_evidence': {field: reference(plan['archive'](path), sha(path)) for field, path in plan['supports'].items()}, 'changes': bundle['changes']}
        decisions.append((bundle['decision_path'], data(decision)))
        wrappers.append((bundle['case_path'], data(wrapper)))
    for _, raw in decisions + wrappers:
        require(len(raw) < 100_000_000, 'generated_member_size_limit')
        plan['scanner'].Scanner(plan['secrets']).feed(raw)
    for path, row in plan['entries'].items():
        check_file(path, row)
    require(registry_path.read_bytes() == original_registry, 'original_registry_changed')
    # First mutation: reached only after root explicitly supplies the execute flag.
    preservation = []
    for path, item in sorted(plan['planned'].items()):
        action = equal_or_new(path, item['raw'], item['mode'])
        if action == 'new':
            put(path, item['raw'], item['mode'])
        require(equal_or_new(path, item['raw'], item['mode']) == 'reused_equal', 'archive_after_write')
        preservation.append({**reference(path), 'bytes': len(item['raw']), 'mode': item['mode'], 'action': action, 'original_paths': item['original_paths']})
    for path, raw in decisions + wrappers:
        put(path, raw)
    put(I / 'registry-before491.json', original_registry)
    put(I / 'complete491-inspection-before.json', data(c['runtime']))
    retained = {'agent_id': '/root', 'session_id': '/root', 'prepared_by': AGENT, 'original_first63': c['first_doc'], 'original_second73_raw': c['raw_second'], 'original_second73_normalized': c['second_doc'], 'two_original_author_amendment_objects': list(c['amendments'].values()), 'first_all63_dispositions': c['disposition']['first_proposal_state_support'], 'second_all73_dispositions': c['disposition']['second_proposal_support'], 'second9_rejections': c['disposition']['rejected_second_proposals'], 'pending_additional_root4': c['disposition']['additional_root_proposals'], 'additional_changes_included': False, 'additional_current_annotations': [{'study_uid': key[0], 'target': key[1], 'annotation': c['effective'][key]} for key in c['pending']], 'selected_prior_accepted_overlap': []}
    plan['scanner'].Scanner(plan['secrets']).feed(data(retained))
    put(I / 'whole-original-proposals-rejections-and-pending.json', data(retained))
    put(I / 'archive-preservation.json', data({'agent_id': '/root', 'session_id': '/root', 'prepared_by': AGENT, 'files': preservation, 'equal_existing_files_reused': True, 'originals_overwritten': False}))

    updated_registry = data({**registry, 'cases': registry['cases'] + [reference(path) for path, _ in wrappers]})
    temporary = registry_path.with_name('registry.wording004-new.json')
    put(temporary, updated_registry)
    require(registry_path.read_bytes() == original_registry, 'registry_switch_fence')
    os.replace(temporary, registry_path)
    try:
        from rsna.adjudications import inspect_adjudications
        from rsna.schema import TARGETS
        after = inspect_adjudications(c['reports'], R / 'initiative/reviews-v2/reviewed', R / 'protocols/report_weak_v2.md', R)
        changes = idx(after['changes'])
        require(after['n_changes'] == 569 and all(changes[key] == value for key, value in c['old'].items()) and set(changes) - set(c['old']) == set(c['new']), 'exact569_and_prior491_objects')
        require(after['semantic_qa'] == c['runtime']['semantic_qa'] and after['pending_semantic_qa'] == c['runtime']['pending_semantic_qa'], 'three_pending_QA_gates_unchanged')
        for key, change in c['new'].items():
            require(changes[key]['before'] == change['before'] and changes[key]['after'] == change['after'] and changes[key]['reason'] == change['reason'], 'new78_exact_states_quotes_reasons')
        final = {**c['original'], **{key: change['after'] for key, change in changes.items()}}
        for key, row in c['rows'].items():
            expected = c['effective'][key] if key in c['pending'] else {'state': row['root_postfreeze_state'], 'evidence': row['root_postfreeze_evidence']}
            require(final[key] == expected, 'root548_exact_and_pending4_current_preserved')
        for row in c['retirement']['restored_original_annotations']:
            key = row['study_uid'], row['target']
            require(key not in changes and final[key] == row['restored_annotation'], 'retirements_preserved')
        require(not any(any(len({final[uid, target]['state'] for uid in uids}) > 1 for target in TARGETS) for uids in c['groups'].values()), 'normalized_duplicate_conflict')
        for path, row in plan['entries'].items():
            if path != registry_path:
                check_file(path, row)
        require(registry_path.read_bytes() == updated_registry, 'postwrite_registry_fence')
        require(shutil.disk_usage(W).free > FLOOR, 'postwrite_disk_floor')
    except Exception:
        if registry_path.read_bytes() == updated_registry:
            rollback = registry_path.with_name('registry.wording004-rollback.json')
            put(rollback, original_registry)
            os.replace(rollback, registry_path)
        for path, raw in wrappers:
            if path.exists() and path.read_bytes() == raw:
                os.replace(path, I / ('failed-' + path.name))
        raise
    inspection_raw = data(after)
    plan['scanner'].Scanner(plan['secrets']).feed(inspection_raw)
    put(I / 'complete569-inspection.json', inspection_raw)
    receipt = {'schema_version': 1, 'agent_id': '/root', 'session_id': '/root', 'executed_by': '/root', 'prepared_by': AGENT, 'registered_at_utc': now, 'registry': reference(registry_path), 'registered_total': 569, 'supported_first': 14, 'supported_second_unamended': 62, 'supported_original_author_amendments': 2, 'new_correction_cases': 4, 'new_changes': 78, 'all491_whole_prior_objects_preserved': True, 'all_original_case_bytes_preserved': True, 'selected_prior_overlap': 0, 'prior_retirements_preserved': 3, 'root552_initial_source_judgments_preserved': True, 'root_final_exact_state_and_quote_matches': 548, 'additional_root4_current_annotations_preserved': True, 'additional_changes_included': 0, 'raw_first63_second73_objects_preserved': True, 'first_state_rejections_preserved': 4, 'second_rejections_preserved': 9, 'explicit_root_state_reconsiderations_preserved': 27, 'global_semantic_QA_gates_pending': 3, 'duplicate_groups': 54, 'duplicate_reports': 204, 'duplicate_conflicts': 0, 'new_clinical_judgments_created_by_helper': 0, 'gold_values_opened': False, 'training_ready': False, 'network_calls': 0}
    receipt_raw = data(receipt)
    plan['scanner'].Scanner(plan['secrets']).feed(receipt_raw)
    put(I / 'complete569-integration-validation.json', receipt_raw)
    put(I / 'integrate-wording004.py.txt', Path(__file__).read_bytes())
    print(json.dumps({'status': 'root_core78_integration_verified', 'registered_total': 569, 'receipt': reference(I / 'complete569-integration-validation.json')}))


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--workspace', required=True, type=Path)
    parser.add_argument('--execute-root-reviewed-decision', action='store_true')
    args = parser.parse_args()
    plan = prepare_plan(args.workspace)
    if args.execute_root_reviewed_decision:
        execute_root(plan)
    else:
        print(json.dumps({'mode': 'read_only_preflight_no_R_writes', 'counts': plan['counts'], 'additional_root_changes_included': 0}))


if __name__ == '__main__':
    try:
        main()
    except Exception as exc:
        print(json.dumps({'status': 'integration_not_confirmed', 'error_class': type(exc).__name__, 'guard_code': str(exc) if isinstance(exc, GuardError) else None}))
        raise SystemExit(2) from None
