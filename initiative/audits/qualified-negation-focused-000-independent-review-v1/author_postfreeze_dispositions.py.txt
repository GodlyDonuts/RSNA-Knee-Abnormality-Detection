def read(path):
    return json.loads(path.read_text())


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def binding(path):
    return {"path": str(path.relative_to(WORKSPACE)), "sha256": sha(path), "bytes": path.stat().st_size}


def write(name, value):
    with (OUT / name).open("x", encoding="utf-8") as handle:
        json.dump(value, handle, ensure_ascii=False, indent=2)
        handle.write("\n")

def independent_reason(cell, key):
    if key in RECONSIDERATIONS:
        return RECONSIDERATIONS[key]["reason"]
    target = cell["target"]
    if target == "Effusion":
        return "The complete report has only a qualified denial of this joint-fluid target, with no separate unqualified absence or explicitly below-threshold amount. Significant/relevant/evident is not a defined moderate/large cutoff. I independently support masking this insufficiently graded report evidence."
    if target == "Contusion":
        return "The complete report supplies only significance- or suspicion-qualified marrow normality, without a separate unqualified normal-marrow statement or definite exclusion of impact edema. The frozen contusion criterion has no significance exception. I independently support uncertain, without asserting a traumatic diagnosis."
    if target == "Baker's":
        return ("In this English coordination significant can govern both effusion and popliteal cyst. No separate cyst absence/size statement resolves the modifier scope; I independently support uncertainty." if key == (26,9) else
                "The source denies pathological popliteal cysts, without an unqualified cyst absence or explicitly below-moderate amount. Pathological is not a defined size cutoff in the frozen rubric. I independently support uncertainty.")
    if target in {"Medial OA", "Lateral OA", "PF OA"}:
        return "The exact compartment has only qualified cartilage preservation/absence, without an unqualified normal statement or a depth/extent measurement excluding the threshold. A significance/majority/grossness/definiteness qualifier or mild OA diagnosis cannot supply the omitted cartilage-loss criterion. I independently support uncertainty."
    if target in {"ACL", "MCL"}:
        return "The full report provides only significance-qualified ligament normality, without an explicit intact-fiber statement or a below-threshold injury grade. Anatomical scope is clear but severity is not established or excluded. I independently support uncertain."
    if key == (31,11):
        return "A suspected focal osteochondral lesion does not itself mention a fracture line, cortical disruption, acute injury or fracture diagnosis. Fracture is not informatively mentioned; I independently support not_mentioned with empty evidence, preserving masking."
    if key == (41,11):
        return "No gross fracture is narrower than absence of every acute fracture line or cortical disruption, and the full source has no separate unqualified fracture denial. I independently support uncertain."
    if key == (12,2):
        return "The abnormal medial meniscocapsular focus is only thought not to represent a tear. Normal remainder does not resolve that specific focus, and it is not isolated intrasubstance degeneration. I independently support uncertain."
    if key == (16,3):
        return "The lateral-compartment statement excludes only noteworthy pathology and does not separately establish normal lateral meniscal structure or no tear. I independently support uncertainty rather than infer a normal meniscus from a qualified aggregate."
    if key == (32,2):
        return "Uncharacterized medial meniscal signal elevation does not state surface relation, normal morphology, or a definite isolated-degeneration explanation. I independently support uncertainty."
    if key == (41,3):
        return "Lateral meniscal contusion with no definite tear is equivocal for this meniscal target. Meniscal contusion is not itself a bone contusion or an explicit isolated-degeneration finding. I independently support uncertainty."
    if key in {(2,2),(2,3),(27,3)}:
        return "The named meniscus has only a definite/significant/discrete-qualified tear denial and no independent normal morphology or isolated-degeneration description that resolves it. I independently support uncertainty."
    raise AssertionError((key, target))

            "source_first_exposure_caveat": "Both actual source freezes and their historical exposures are bound. The first reader authored discovery; neither record is relabeled as a fully historically blind clinical read.",
