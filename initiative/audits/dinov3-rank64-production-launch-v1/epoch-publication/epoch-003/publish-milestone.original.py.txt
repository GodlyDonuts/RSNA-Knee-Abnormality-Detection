"""Publish the new epoch-three handoff once, then verify every private byte."""

import hashlib
import json
import os
import shutil
import subprocess
import tempfile
from pathlib import Path

os.environ["HF_HUB_DISABLE_PROGRESS_BARS"] = "1"
from huggingface_hub import CommitOperationAdd, HfApi, hf_hub_download

root = Path.cwd()
audit = root / "initiative/audits/dinov3-rank64-production-launch-v1/epoch-publication/epoch-003"
assert not (audit / "hf-milestone-intent.json").exists()
assert (root / ".env").stat().st_mode & 0o077 == 0
values = {}
for line in (root / ".env").read_text().splitlines():
    key, sep, value = line.partition("=")
    if sep and not key.lstrip().startswith("#"):
        values[key.strip()] = value.strip().strip("\"'")


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def save(name, value):
    with (audit / name).open("x") as stream:
        json.dump(value, stream, sort_keys=True, indent=2)
        stream.write("\n")
        stream.flush()
        os.fsync(stream.fileno())


verified = json.loads((audit / "verified.json").read_text())
assert verified["status"] == "private_epoch_fully_verified"
assert verified["files_verified_by_download"] == 4
assert verified["revision"] == "06618b46ce73bec9ec254b8b3bc570bb1c39b6cb"
paths = sorted(p for p in audit.iterdir() if p.is_file())
paths += [
    root / "docs/codex-continuation.md",
    root / "initiative/audits/dinov3-rank64-production-launch-v1/README.md",
]
assert all(not p.is_symlink() for p in paths)
files = {
    p.relative_to(root).as_posix(): {"bytes": p.stat().st_size, "sha256": sha(p)} for p in paths
}
assert sum(record["bytes"] for record in files.values()) < 4 * 1024**2
assert shutil.disk_usage(root).free > 20 * 1024**3
assert not subprocess.check_output(["git", "status", "--porcelain", "--", *files], text=True)
head = subprocess.check_output(["git", "rev-parse", "HEAD"], text=True).strip()
repo = "GodlyDonuts/RSNA-Data"
prefix = "experiments/dinov3-v1/milestones/rank64-e3-preservation-20260930"
api = HfApi(token=values["HF_TOKEN"])
info = api.repo_info(repo, repo_type="dataset")
assert info.private is True
assert not api.get_paths_info(
    repo,
    [prefix + "/manifest.json", *[prefix + "/" + n for n in files]],
    repo_type="dataset",
    revision=info.sha,
)
manifest = {
    "schema_version": 1,
    "source_git_commit": head,
    "files": files,
    "rank64_completed_epoch": 3,
    "optimizer_steps": 843,
    "epoch_private_revision": verified["revision"],
    "new_accuracy_result": False,
}
mp = root / "artifacts/rank64-e3-preservation-v1/manifest.json"
with mp.open("x") as stream:
    json.dump(manifest, stream, sort_keys=True, indent=2)
    stream.write("\n")
save(
    "hf-milestone-intent.json",
    {
        "repo_id": repo,
        "prefix": prefix,
        "parent_commit": info.sha,
        "source_git_commit": head,
        "files": files,
        "private_required": True,
        "one_commit_only": True,
        "manifest_sha256": sha(mp),
    },
)
operations = [
    CommitOperationAdd(path_in_repo=prefix + "/" + name, path_or_fileobj=str(root / name))
    for name in files
] + [CommitOperationAdd(path_in_repo=prefix + "/manifest.json", path_or_fileobj=str(mp))]
commit = api.create_commit(
    repo_id=repo,
    repo_type="dataset",
    parent_commit=info.sha,
    operations=operations,
    commit_message="Preserve verified rank64 epoch-three publication and continuation evidence",
)
save("hf-milestone-commit.json", {"revision": commit.oid, "prefix": prefix})
assert api.repo_info(repo, repo_type="dataset", revision=commit.oid).private is True
expected = {**files, "manifest.json": {"bytes": mp.stat().st_size, "sha256": sha(mp)}}
with tempfile.TemporaryDirectory(prefix="rsna-rank64-e3-receipts-") as temporary:
    for name, record in expected.items():
        downloaded = Path(
            hf_hub_download(
                repo,
                filename=prefix + "/" + name,
                repo_type="dataset",
                revision=commit.oid,
                token=values["HF_TOKEN"],
                local_dir=temporary,
                force_download=True,
            )
        )
        assert downloaded.stat().st_size == record["bytes"] and sha(downloaded) == record["sha256"]
save(
    "hf-milestone-verified.json",
    {
        "status": "private_full_download_size_sha_verified",
        "repo_id": repo,
        "prefix": prefix,
        "revision": commit.oid,
        "source_git_commit": head,
        "file_count": len(expected),
        "files": expected,
    },
)
print(
    json.dumps(
        {
            "status": "private_full_download_size_sha_verified",
            "revision": commit.oid,
            "files": len(expected),
        }
    )
)
