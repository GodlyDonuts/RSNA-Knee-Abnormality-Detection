def digest(raw):
    return hashlib.sha256(raw).hexdigest()

def ref(path):
    raw = path.read_bytes()
    return {'path': str(path.relative_to(W)), 'sha256': digest(raw), 'bytes': len(raw)}

def obj(path):
    return json.loads(path.read_bytes())

def jsonl(path):
    return [json.loads(line) for line in path.read_text().splitlines()]

def write(name, data):
    path = S / name
    with path.open('x', encoding='utf-8') as handle:
        handle.write(json.dumps(data, ensure_ascii=False, indent=2) + '\n')
    return path

    review_path = R/row['reviewed_artifact_path']
    assert ref(review_path)['sha256'] == row['reviewed_artifact_sha256']
    if review_path not in review_cache:
        review_cache[review_path] = {x['study_uid']: x for x in jsonl(review_path) if x['study_uid'] in ids}
    original = review_cache[review_path][row['study_uid']]
    target_index = ALL_TARGETS.index(row['target'])
    original_annotation = {'state': original['states'][target_index], 'evidence': original['evidence'][target_index]}
    assert original_annotation == {'state': row['original_state'], 'evidence': row['original_evidence']}
    base_annotation = {'state': row['inspection_effective_state'], 'evidence': row['inspection_effective_evidence']}
    # Authenticate any pre-existing34 correction rather than confusing its
    # effective state with the original reviewed state.
    if row['existing_correction_case'] is None:
        assert base_annotation == original_annotation
    else:
        assert key in matching_current
        assert matching_current[key]['case'] == row['existing_correction_case']
        assert matching_current[key]['change']['before'] == original_annotation
        assert matching_current[key]['change']['after'] == base_annotation
    current = dict(original_annotation)
    if key in matching_current:
        change = matching_current[key]['change']
        assert change['before'] == original_annotation
        current = dict(change['after'])
    first_record = post_index[key]
    assert base_annotation == {'state': first_record['baseline_effective_state'], 'evidence': first_record['baseline_effective_evidence']}
    rebase_records.append({'study_uid':key[0], 'target':key[1], 'baseline34':base_annotation,
