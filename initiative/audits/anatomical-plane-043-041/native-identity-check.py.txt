"""Confirm full native metadata identity consistency for completed bounded cases."""

from collections import defaultdict,Counter
import hashlib,json,sys
base=Path.cwd().parent;sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
for folder in sys.argv[1:]:
 p=base/f'plane-{folder}/all-slice-headers-and-pixels.jsonl';rows=[json.loads(x) for x in p.read_text().splitlines()];by_series=defaultdict(list)
 for r in rows:by_series[r['series_uid']].append(r)

 out=base/'annotation-staging'/f'native-{folder}-identity-check.json'

Does not establish real patient identity, anatomical side, report linkage or cross-corpus uniqueness.','input_bindings':[{'path':str(f.relative_to(base)),'sha256':sha(f),'bytes':f.stat().st_size} for f in [p,Path(__file__).resolve()]]}
 with out.open('x') as f:f.write(json.dumps(a,indent=2)+'\n')
 print(json.dumps({'case':folder,'n_full_native_records':len(rows),'n_series':len(by_series),'sha256':sha(out)}))
