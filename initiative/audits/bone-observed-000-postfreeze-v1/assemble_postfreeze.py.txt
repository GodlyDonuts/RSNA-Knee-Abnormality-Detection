"""Assemble individually reasoned post-freeze decisions; never register them."""
from pathlib import Path
from datetime import datetime, timezone
from collections import Counter
import hashlib
import json
import sys
import authored_postfreeze_decisions as manual

S=Path(__file__).resolve().parent
W=S.parents[2]
R=W/'work/recovered'
F=W/'work/annotation-staging/bone-observed-000-source-review-v1'
sys.path.insert(0,str(R))
from rsna.adjudications import inspect_adjudications
from rsna.schema import TARGETS

REGISTRY='64b6b8b51efb693738e8a3f70743afe4b160c5b3acd1f4ec3300cc699e8fee12'
RUBRIC='138c5cc33b92901d57dbf5fbe7e3a9878b73d25b0af6934908d0052cd6786014'


def read(p):return json.loads(p.read_text())


def sha(p):
    h=hashlib.sha256()
    with p.open('rb') as f:
        for b in iter(lambda:f.read(1024*1024),b''):h.update(b)
    return h.hexdigest()


def ref(p):return dict(path=str(p.relative_to(W)),sha256=sha(p),bytes=p.stat().st_size)


def write(name,obj):
    p=S/name;assert not p.exists();p.write_text(json.dumps(obj,ensure_ascii=False,indent=2)+'\n');return ref(p)


def state_mask(state):return int(state in {'positive','negative'})


def main():
    assert sha(F/'source-freeze.json')=='b11ee841930a1fefcde029e8b1fa84a8b167d15761b4fef094b8abac2a0fc8ce'
    assert sha(F/'source-assessment.json')=='bd414d34a1ba8e69b40084d18e4087fa18083f00a1e81ce73ce44bb434e70d50'
    for name,info in read(F/'artifact-manifest.json')['files'].items():assert sha(F/name)==info['sha256']
    comparison=read(S/'all100-comparison.json');rows=comparison['rows'];assert len(rows)==100
    packet=read(F/'neutral-source-packet.json');sources={x['study_uid']:x for x in packet['reports']}
    by_key={(x['ordinal'],x['target']):x for x in rows};assert len(by_key)==100
    disagreements={k for k,x in by_key.items() if not x['state_agreement']}
    assert disagreements==set(manual.PROPOSE_STATE)|set(manual.RECONSIDER)|set(manual.RETAIN_DISAGREEMENT)
    quote_diffs={k for k,x in by_key.items() if x['state_agreement'] and not x['quote_agreement']}
    assert quote_diffs==set(manual.PROPOSE_EVIDENCE)|set(manual.EQUIVALENT_EVIDENCE)
    protected={}
    for row in rows:
        for field,hash_field in [('source_path','source_sha256'),('artifact_path','artifact_sha256')]:
            assert sha(R/row[field])==row[hash_field];protected[row[field]]=row[hash_field]
        src=read(R/row['source_path'])[row['source_index_zero_based']]
        assert src['study_uid']==row['study_uid'] and src['report']==sources[row['study_uid']]['report']
        original=json.loads((R/row['artifact_path']).read_text().splitlines()[row['review_index_zero_based']])
        assert original['study_uid']==row['study_uid'];i=list(TARGETS).index(row['target'])
        assert row['original']==dict(state=original['states'][i],evidence=original['evidence'][i])
    registry_path=R/'initiative/reviews-v2/adjudications/registry.json';assert sha(registry_path)==REGISTRY
    registry=read(registry_path)
    for x in registry['cases']+registry['semantic_qa']:
        assert sha(R/x['path'])==x['sha256'];protected[x['path']]=x['sha256']
    protected['initiative/reviews-v2/adjudications/registry.json']=REGISTRY
    protected['protocols/report_weak_v2.md']=RUBRIC
    for p in [R/'initiative/reviews-v2/index.json',R/'initiative/reviews-v2/provenance.json']:
        protected[str(p.relative_to(R))]=sha(p)
    for name in sorted({row['artifact_path'] for row in rows}):
        stem=Path(name).stem;p=R/'initiative/reviews-v2/provenance'/f'{stem}.json'
        assert p.exists();protected[str(p.relative_to(R))]=sha(p)
    all_reports={x['study_uid']:x['report'] for p in sorted((R/'initiative/reviews/inputs').glob('batch-*.json')) for x in read(p)}
    runtime=inspect_adjudications(all_reports,R/'initiative/reviews-v2/reviewed',R/'protocols/report_weak_v2.md',R)
    assert runtime==read(S/'current-adjudications-inspection.json') and runtime['n_changes']==118
    changes={(x['study_uid'],x['target']):x for x in runtime['changes']}
    dispositions=[];proposals=[];reconsiderations=[];retained_disagreements=[];retained_masks=[]
    for x in rows:
        k=(x['ordinal'],x['target']);before=x['effective'];frozen=x['source_frozen']
        expected=changes[(x['study_uid'],x['target'])]['after'] if (x['study_uid'],x['target']) in changes else x['original']
        assert before==expected
        after=dict(state=frozen['state'],evidence=frozen['evidence'])
        action='retain_current_agreement';rationale=frozen['rationale'];proposal=None
        if k in manual.PROPOSE_STATE or k in manual.PROPOSE_EVIDENCE:
            action='propose_state_change' if k in manual.PROPOSE_STATE else 'propose_evidence_expansion'
            rationale=(manual.PROPOSE_STATE|manual.PROPOSE_EVIDENCE)[k]
            assert x['existing_correction'] is None, 'existing accepted key requires separate supersession handling'
            proposal={field:x[field] for field in ['ordinal','study_uid','target','source_path','source_sha256','report_sha256','artifact_path','artifact_sha256','review_index_zero_based']}
            proposal.update(index=x['source_index_zero_based'],before_state=before['state'],before_evidence=before['evidence'],
                original_state=x['original']['state'],original_evidence=x['original']['evidence'],
                proposed_state=after['state'],source_quote=after['evidence'],
                source_context_quote=after['evidence'] or before['evidence'],reason=rationale,
                proposal_type='state_change' if k in manual.PROPOSE_STATE else 'evidence_expansion_only',
                source_frozen_state=frozen['state'],source_frozen_evidence=frozen['evidence'],
                before_observation_mask=state_mask(before['state']),proposed_observation_mask=state_mask(after['state']),
                source_assessment_sha256=sha(F/'source-assessment.json'),source_freeze_sha256=sha(F/'source-freeze.json'),
                baseline_registry_sha256=REGISTRY,status='proposed_not_accepted_or_registered',
                proposer_agent_id='/root/duplicate_consistency_review')
            proposals.append(proposal)
        elif k in manual.RECONSIDER:
            action='own_reconsideration_retain_existing_mask';rationale=manual.RECONSIDER[k]
            after=dict(state='not_mentioned',evidence='');assert before==after
        elif k in manual.RETAIN_DISAGREEMENT:
            action='retain_mask_with_unresolved_source_scope_disagreement';rationale=manual.RETAIN_DISAGREEMENT[k]
            assert state_mask(before['state'])==0
        elif k in manual.EQUIVALENT_EVIDENCE:
            action='retain_current_sufficient_evidence';rationale=manual.EQUIVALENT_EVIDENCE[k]
        else:
            assert x['state_agreement'] and x['quote_agreement']
        disp={field:x[field] for field in ['ordinal','study_uid','target','source_path','source_sha256','source_index_zero_based','report_sha256','artifact_path','artifact_sha256']}
        disp.update(original=x['original'],current_effective=before,source_frozen=dict(state=frozen['state'],evidence=frozen['evidence'],rationale=frozen['rationale']),
            postfreeze_assessor_judgment=after,action=action,reason=rationale,proposal_issued=proposal is not None,
            own_state_reconsidered=k in manual.RECONSIDER,unresolved_source_scope_disagreement=k in manual.RETAIN_DISAGREEMENT,
            existing_correction_preserved=x['existing_correction'],current_observation_mask=state_mask(before['state']),
            proposed_observation_mask=state_mask(after['state']) if proposal else state_mask(before['state']),
            mask_scope='Report target-state observation only; does not override correspondence, leakage, quarantine or final-release masking.')
        dispositions.append(disp)
        if k in manual.RECONSIDER:reconsiderations.append(disp)
        if k in manual.RETAIN_DISAGREEMENT:retained_disagreements.append(disp)
        if before['state'] in {'uncertain','not_mentioned'} and proposal is None:retained_masks.append(disp)
    assert len(proposals)==8 and len(reconsiderations)==3 and len(retained_disagreements)==2
    for x in dispositions:
        report=sources[x['study_uid']]['report']
        for field in ['current_effective','source_frozen','postfreeze_assessor_judgment']:
            st,q=x[field]['state'],x[field]['evidence'];assert q=='' if st=='not_mentioned' else q and q in report
    for x in proposals:
        assert x['source_context_quote'] in sources[x['study_uid']]['report']
        assert x['before_state']!=x['proposed_state'] if x['proposal_type']=='state_change' else x['before_state']==x['proposed_state'] and x['before_evidence']!=x['source_quote']
    disposition_ref=write('all100-postfreeze-dispositions.json',dict(schema_version=1,agent_id='/root/duplicate_consistency_review',n_cells=100,
        source_freeze=ref(F/'source-freeze.json'),source_assessment=ref(F/'source-assessment.json'),comparison=ref(S/'all100-comparison.json'),
        baseline_registry_sha256=REGISTRY,records=dispositions,clinical_annotations_changed=False))
    proposal_ref=write('proposed-errata.json',dict(schema_version=1,proposer_agent_id='/root/duplicate_consistency_review',session_id='/root/duplicate_consistency_review',
        status='proposed_only_requires_distinct_source_assessment_and_actual_acceptance',n_reports_assessed=50,n_cells_assessed=100,
        n_proposed_changes=len(proposals),n_proposed_state_changes=5,n_proposed_evidence_expansions=3,
        source_freeze_sha256=sha(F/'source-freeze.json'),source_assessment_sha256=sha(F/'source-assessment.json'),rubric_sha256=RUBRIC,
        source_reports_sha256='5c18838a09e9eec6522966555694fca83a9bae67e4d7c5c68592207ee4b60ed3',registry_sha256=REGISTRY,
        comparison_sha256=sha(S/'all100-comparison.json'),dispositions_sha256=sha(S/'all100-postfreeze-dispositions.json'),
        prior_discovery_exposure_disclosed=True,independent_acceptance=None,proposed_errata=proposals))
    write('own-reconsiderations.json',dict(schema_version=1,agent_id='/root/duplicate_consistency_review',n_cells=3,
        scope='Own post-freeze changes of interpretation only; preserve original source freeze; existing masks retained.',records=reconsiderations))
    write('retained-mask-disagreements.json',dict(schema_version=1,agent_id='/root/duplicate_consistency_review',n_cells=2,
        scope='Existing masks retained; frozen negative interpretation and source-scope disagreement remain visible; no unmasking proposal.',records=retained_disagreements))
    write('all-retained-existing-masks.json',dict(schema_version=1,agent_id='/root/duplicate_consistency_review',n_cells=len(retained_masks),
        scope='Every currently masked cell without a proposed change; includes agreements, own reconsiderations and unresolved disagreements.',records=retained_masks))
    attention_uids={x['study_uid'] for x in dispositions if x['proposal_issued'] or x['own_state_reconsidered'] or x['unresolved_source_scope_disagreement']}
    attention=[x for x in packet['reports'] if x['study_uid'] in attention_uids];assert len(attention)==13
    attention_ref=write('neutral-attention-full-reports.json',dict(schema_version=1,scope='Complete original sources only. Read both bone targets and freeze judgments before opening any comparison or proposal.',
        rubric_sha256=RUBRIC,source_reports_sha256='5c18838a09e9eec6522966555694fca83a9bae67e4d7c5c68592207ee4b60ed3',reports=attention))
    allowed=set(packet['reports'][0]);assert all(set(x)==allowed for x in attention)
    bindings_ref=write('exact-originals-and-provenance-bindings.json',dict(schema_version=1,agent_id='/root/duplicate_consistency_review',
        source_freeze=ref(F/'source-freeze.json'),source_manifest=ref(F/'artifact-manifest.json'),neutral_source_packet=ref(F/'neutral-source-packet.json'),
        protected_repository_artifacts=protected,original_and_effective_cell_bindings=ref(S/'all100-comparison.json'),
        inspection_baseline=ref(S/'current-adjudications-inspection.json'),n_registered_corrections=118,n_pending_semantic_qa=3,
        original_files_not_copied_or_changed=True,source_stage_unchanged=True))
    for name,digest in protected.items():assert sha(R/name)==digest
    for name,info in read(F/'artifact-manifest.json')['files'].items():assert sha(F/name)==info['sha256']
    summary=dict(schema_version=1,agent_id='/root/duplicate_consistency_review',completed_at_utc=datetime.now(timezone.utc).isoformat(),
        n_full_sources_personally_read=50,n_source_cells_frozen=100,n_cells_compared=100,n_state_agreements=90,n_state_differences=10,
        n_same_state_quote_differences=15,n_proposed_changes=8,n_proposed_state_changes=5,n_proposed_evidence_expansions=3,
        n_own_reconsiderations=3,n_retained_mask_disagreements=2,n_existing_masks_retained_without_proposal=len(retained_masks),
        n_existing_registered_corrections_in_scope_preserved=sum(x['existing_correction'] is not None for x in rows),
        n_attention_reports=13,n_protected_repository_artifacts=len(protected),
        actions=dict(Counter(x['action'] for x in dispositions)),proposal_transition_counts=dict(Counter(x['before_state']+'->'+x['proposed_state'] for x in proposals)),
        observation_mask_impact_if_all_proposals_later_accepted=dict(observed_to_masked=sum(x['before_observation_mask']==1 and x['proposed_observation_mask']==0 for x in proposals),
            masked_to_observed=sum(x['before_observation_mask']==0 and x['proposed_observation_mask']==1 for x in proposals),
            masked_to_masked=sum(x['before_observation_mask']==0 and x['proposed_observation_mask']==0 for x in proposals),
            observed_evidence_only=sum(x['proposal_type']=='evidence_expansion_only' for x in proposals),
            note='No proposals applied; final training eligibility masks are not inferred.'),
        source_freeze=ref(F/'source-freeze.json'),source_assessment=ref(F/'source-assessment.json'),dispositions=disposition_ref,
        proposals=proposal_ref,neutral_attention_packet=attention_ref,bindings=bindings_ref,
        current_registry_sha256=REGISTRY,all_three_global_qa_gates_still_pending=True,clinical_annotations_changed=False,
        source_stage_unchanged=True,gold_values_inspected=False,repository_mutations=0,remote_calls=0,
        limitations='First bone-focused source assessor has prior inventory/source-review exposure. No distinct independent clinical acceptance has occurred. Source interpretations, unresolved scope disagreements and proposed observation changes require separate review.')
    write('postfreeze-summary.json',summary)
    write('validation-receipt.json',dict(schema_version=1,status='passed',n_source_reports=50,n_cells=100,n_proposals=8,
        exact_source_file_hashes_uids_indices_reports_verified=True,original_review_artifact_hashes_and_record_cells_verified=True,
        current118_inspection_replayed_exactly=True,all_frozen_current_and_proposed_quotes_exact=True,
        all_discrepancies_have_personally_authored_disposition=True,all100_cells_have_dispositions=True,
        proposal_mask_impacts_recomputed=True,existing_accepted_correction_preserved=True,
        source_freeze_and_all_bound_files_unchanged=True,neutral_attention_packet_source_only_allowlist_verified=True,
        original_archives_referenced_without_recopying=True,n_protected_artifacts_unchanged=len(protected),
        global_qa_closed=False,clinical_annotations_changed=False,gold_inspected=False,repository_mutations=0,remote_calls=0))
    print(json.dumps({k:summary[k] for k in ['n_cells_compared','n_proposed_changes','n_proposed_state_changes','n_proposed_evidence_expansions',
        'n_own_reconsiderations','n_retained_mask_disagreements','n_existing_masks_retained_without_proposal','n_attention_reports',
        'proposal_transition_counts','observation_mask_impact_if_all_proposals_later_accepted']},indent=2))


if __name__=='__main__':main()
