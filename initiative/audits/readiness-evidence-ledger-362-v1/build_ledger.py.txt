def sha(data):
    return hashlib.sha256(data).hexdigest()


def read(path):
    return json.loads(path.read_text())


def string_values(value):
    if isinstance(value, str):
        yield value
    elif isinstance(value, dict):
        for child in value.values():
            yield from string_values(child)
    elif isinstance(value, list):
        for child in value:
            yield from string_values(child)


def actor(*objects):
    agent = session = None
    for ob in objects:
        if not isinstance(ob, dict):
            continue
        for key in ("agent_id", "assessor_agent_id", "assessor_id", "reader_id"):
            if isinstance(ob.get(key), str):
                agent = agent or ob[key]
        for key in ("session_id", "assessor_session_id"):
            if isinstance(ob.get(key), str):
                session = session or ob[key]
        if isinstance(ob.get("assessor"), dict):
            a, s = actor(ob["assessor"])
            agent, session = agent or a, session or s
    return agent, session


def records(path):
    if path.suffix == ".jsonl":
        return {}, [(f"line:{i}", json.loads(line)) for i, line in enumerate(path.read_text().splitlines(), 1)]
    doc = read(path)
    if isinstance(doc, list):
        return {}, [(f"/{i}", r) for i, r in enumerate(doc)]
    for key in ("records", "reports", "judgments"):
        if isinstance(doc.get(key), list):
            return doc, [(f"/{key}/{i}", r) for i, r in enumerate(doc[key])]
    raise ValueError(f"Unsupported record container: {path.name}")


def validate_judgment(report, state, evidence):
    if state not in STATES or not isinstance(evidence, str):
        raise ValueError("Invalid state/evidence type")
    if state == "not_mentioned":
        if evidence:
            raise ValueError("Missing state has nonempty evidence")
        return None
    if not evidence or evidence not in report:
        raise ValueError("Evidence is not an exact contiguous original source quote")
    return report.index(evidence)

    def bind(p, expected=None):
        p = Path(p).resolve()
        if not p.is_relative_to(w) or not p.is_file():
            raise ValueError("Artifact outside workspace or missing")
        # No official label table is ever a ledger input.
        if p == r / "initiative/source/train.csv" or p.name == ".env":
            raise ValueError("Forbidden sensitive/gold input")
        digest = sha(p.read_bytes())
        if expected and digest != expected:
            raise ValueError(f"Changed bound artifact: {p.name}")
        path = str(p.relative_to(w))
        if path not in artifact_ids:
            aid = f"a{len(artifacts):04d}"
            artifact_ids[path] = aid
            artifacts.append({"id": aid, "path": path, "sha256": digest, "bytes": p.stat().st_size})
        return artifact_ids[path]

    def resolve(p, owner):
        p = Path(p)
        choices = [p] if p.is_absolute() else [w / p, w / "work" / p, r / p, owner / p]
        for q in choices:
            if q.is_file():
                return q.resolve()
        raise ValueError(f"Unresolved artifact basename: {p.name}")

    def bound_ref(d, owner):
        p = resolve(d["path"], owner)
        bind(p, d.get("sha256"))
        return p

    def choose(folder, names):
        for name in names:
            if (folder / name).is_file():
                return folder / name
        raise ValueError(f"Missing named assessment in {folder.name}")

    def source_check(row, allowed=None):
        uid = row["study_uid"]
        if uid not in source or (allowed is not None and uid not in allowed):
            raise ValueError("UID outside exact source packet")
        canonical = source[uid]
        for key in ("report_sha256", "source_report_sha256", "report_utf8_sha256"):
            if row.get(key) and row[key] != canonical["report_sha256"]:
                raise ValueError("Report hash mismatch")
        if row.get("report") is not None and row["report"] != canonical["report"]:
            raise ValueError("Full report byte mismatch")
        for key in ("source_sha256", "source_batch_sha256", "source_file_sha256"):
            if row.get(key) and row[key] != canonical["source_sha256"]:
                raise ValueError("Source batch hash mismatch")
        if row.get("source_path") and Path(row["source_path"]).name != Path(canonical["source_path"]).name:
            raise ValueError("Source batch path mismatch")
        for key in ("source_index_zero_based", "source_index"):
            if key in row and row[key] != canonical["source_index_zero_based"]:
                raise ValueError("Source index mismatch")
        return canonical

    def emit(stage, aid, selector, row, target, state, quote, agent, session, allowed=None, reason=None, raw_fields=None):
        original = source_check(row, allowed)
        if target not in stage["targets"] or target not in targets:
            raise ValueError("Target outside actual reader scope")
        offset = validate_judgment(original["report"], state, quote)
        eid = f"e{len(evidence):06d}"
        item = {"id": eid, "study_uid": row["study_uid"], "target": target, "report_sha256": original["report_sha256"], "stage": stage["id"], "kind": stage["kind"], "artifact": aid, "selector": selector, "agent_id": agent, "session_id": session, "session_recorded": session is not None, "state": state, "evidence": quote, "evidence_sha256": sha(quote.encode()), "first_exact_quote_start": offset}
        if reason is not None:
            item["rationale"] = reason
        if raw_fields:
            item["original_selected_fields"] = raw_fields
        if not agent:
            raise ValueError("Missing actual author identity")
        evidence.append(item)
        cell_records[(row["study_uid"], target)].append(eid)
        scope_memberships[stage["id"]].add((row["study_uid"], target))
        return eid

    def newstage(sid, kind, actual_targets, anchor=None, author=None, session=None, family=None):

            stage["anchor_artifact"] = bind(anchor)
        stage.update(agent_id=author, session_id=session, session_missing_is_not_inferred=session is None)
        stages.append(stage)
        return stage

                key = (row["study_uid"], target)
                if key in seen:
                    raise ValueError("Duplicate source cell in stage")
                seen.add(key)
                emit(stage, aid, loc, row, target, state, quote, author, session, allowed, reason)
        if allowed is not None and seen != {(u, t) for u in allowed for t in stage["targets"]}:
            raise ValueError("Source pass does not cover its exact packet/target scope")
        stage["assessment_artifacts"] = stage.get("assessment_artifacts", []) + [aid]
