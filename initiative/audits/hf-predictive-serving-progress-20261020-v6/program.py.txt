def paid_unique_npz(roots):
    identities = {}
    for root in roots:
        for p in Path(root).rglob("*.npz"):
            try:
                st = p.lstat()
            except FileNotFoundError:
                continue  # A missing transient file grants no paid-byte credit.
            assert not p.is_symlink() and st.st_uid == os.getuid()
            identities[(st.st_dev, st.st_ino)] = st.st_size
    return sum(identities.values())
