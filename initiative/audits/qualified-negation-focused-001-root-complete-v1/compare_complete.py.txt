from pathlib import Path

freeze=G/'complete-source-freeze.json';assert sha(freeze)=='a745d9e7c210282cf8ea075e0564804b4b75c616de4d1d411a938576b9e102c6'

reports={r['study_uid']:r['report'] for p in sorted((R/'initiative/reviews/inputs').glob('batch-*.json')) for r in read(p)}

  assert p['before_state']==original['state'] and p['before_evidence']==original['evidence']

assert not any(r['existing_correction'] for r in records)

  print(r['case'],r['target'],'base',r['effective_baseline']['state'],'root',r['root_frozen']['state'],'first',r['first_postfreeze_state'],'second',r['second_postfreeze']['state'],'PRIMARY',bool(r['primary_proposal']),'HELD',r['second_explicitly_held_for_root'])
