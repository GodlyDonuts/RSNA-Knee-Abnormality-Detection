"""Authorized local build plus actual peer streaming proof; no network operations."""
import argparse
import json
from final_release_bindings import *

def bind_actual_execution(proof):
    """Replace execution provenance before the unchanged builder saves its proof."""
    old_identity = proof['verifier_agent_id']
    require(old_identity == HELPER_DEFAULT_IDENTITY, 'unexpected_shared_default_identity')
    return dict(proof, verifier_agent_id=PEER, actual_executor=PEER,
        generated_helper_default_identity=old_identity,
        helper_default_is_not_evidence_of_actual_execution=True,
        helper_sha256=HELPER_SHA, peer_runner=ref(Path(__file__)),
        preparation_author_agent_id=WORKFLOW_AUTHOR,
        workflow_revision_author_agent_id=WORKFLOW_REVISER,
        actual_role='Actual local builder and peer verifier: the unchanged shared verification function streamed the complete base and newly built delta within build_package. Only execution-provenance fields were replaced before the first proof save. The peer-restoration-proof.json name is a byte-identical alias of this same execution, not a second verification.',
        protected_source576_inventory=ref(SOURCE_INVENTORY),
        source576_exact_target_git_identities_verified=576,
        process_exposure='Opaque byte streaming and credential/authenticated-URL scanning across all base/delta payloads. No clinical choices, gold-label interpretation or pixel review. Disclose this mechanical exposure in subsequent source review.')

def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--actual-executor', choices=[PEER], required=True)
    args = parser.parse_args()
    require(args.actual_executor == PEER, 'actual_executor_required')
    future_pins_guard()
    h = helper()
    package = WORK / 'hf-preparation-deltas' / TARGET_GIT
    require(not package.exists(), 'new_package_required')
    require(h.git(R, 'rev-parse', TARGET_GIT + '^{tree}').decode().strip() == TARGET_TREE, 'target_tree_changed')
    rows = h.tree(R, TARGET_GIT)
    require(len(rows) == TARGET_FILE_COUNT, 'target_file_count_changed')
    require(h.git(R, 'merge-base', PREVIOUS_GIT, TARGET_GIT).decode().strip() == PREVIOUS_GIT, 'target_not_descendant_of_previous569')
    registry = h.git(R, 'show', TARGET_GIT + ':initiative/reviews-v2/adjudications/registry.json')
    require(hashlib.sha256(registry).hexdigest() == REGISTRY_SHA, 'registry569_binding')
    source576_guard(h, rows)
    eligibility = release_guard(h, rows)
    original_verify = h.verify_package
    def actual_verify(*positional, **keywords):
        return dict(bind_actual_execution(original_verify(*positional, **keywords)),
                    final_release_eligibility=eligibility)
    h.verify_package = actual_verify
    base = WORK / 'hf-preparation-checkpoints' / BASE_GIT
    h.disk_guard(package.parent)
    proof = h.build_package(R, TARGET_GIT, base / 'checkpoint.tar.gz', base / 'manifest.json', package, secrets=h.secret_values(W / '.env'))
    saved = read(package / 'restoration-proof.json')
    require(saved == proof and saved['verifier_agent_id'] == saved['actual_executor'] == PEER, 'saved_actual_identity_mismatch')
    verify_args = argparse.Namespace(target_commit=TARGET_GIT, target_tree=TARGET_TREE, expected_file_count=TARGET_FILE_COUNT, package=package, manifest_sha256=sha(package / 'manifest.json'))
    _, manifest, _, fresh_eligibility = local_guard(verify_args, h)
    require(fresh_eligibility == eligibility, "eligibility_changed_during_build")
    proof_guard(proof, verify_args, manifest, PEER)
    h.Scanner(h.secret_values(W / '.env')).feed((package / 'restoration-proof.json').read_bytes())
    with (package / 'peer-restoration-proof.json').open('xb') as f:
        f.write((package / 'restoration-proof.json').read_bytes())
    require(sha(package / 'restoration-proof.json') == sha(package / 'peer-restoration-proof.json'), 'proof_alias_not_exact')
    receipt = {'stage': 'actual_local_final_release_build_and_peer_streaming_verification',
        'actual_executor': PEER, 'target_git_commit': TARGET_GIT, 'target_git_tree': TARGET_TREE,
        'registered_corrections': N_CORRECTIONS, 'registry_sha256': REGISTRY_SHA,
        'n_result_files_verified': proof['n_result_files_verified'],
        'n_delta_members_verified': proof['n_delta_members_verified'],
        'manifest': ref(package / 'manifest.json'), 'delta_archive': ref(package / 'delta.tar.gz'),
        'restoration_proof': ref(package / 'restoration-proof.json'),
        'peer_proof_alias': ref(package / 'peer-restoration-proof.json'),
        'proof_alias_scope': 'Identical bytes, one actual streaming execution; separate root replay remains required.',
        'builder': ref(Path(__file__)), 'shared_helper': ref(HELPER),
        'binding_module': ref(STAGE / 'final_release_bindings.py'),
        'disk_after': h.disk_guard(package), 'remote_calls': 0, 'remote_mutations': 0,
        'repository_writes': False, 'final_release_eligibility': eligibility, 'training_ready': False, 'publication_performed': False}
    save(package / 'peer-build-receipt.json', receipt)
    print(json.dumps(receipt, indent=2))

if __name__ == '__main__':
    try:
        main()
    except Exception as error:
        print(json.dumps(error_result(error)))
        raise SystemExit(2) from None
