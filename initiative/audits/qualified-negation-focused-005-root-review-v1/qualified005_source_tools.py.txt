"""Freeze individually authored root judgments against unchanged neutral sources."""
from pathlib import Path
from datetime import datetime, timezone
import hashlib, json
W = Path(__file__).resolve().parent.parent
R = W / 'work/recovered'
G = W / 'work/annotation-staging/qualified-negation-focused-005-root-review-v1'
N = W / 'work/annotation-staging/qualified-negation-focused-005-independent-postfreeze-v1'
PINS = ['b2f72ecdc4c66ea660ea065eb07026fe94e05eacbfa72ed60e6c7d9e4ca4fc0b', 'e99c9e8dd14d7b36106f87704e291c2d6aeddd27b7d24cbb65470d326cce1dbd', 'fcb90bf1e3537715edb4cc3f1c66972ad87fec2ef3ad3730b0aa005ea2b84d76', '9bad189c8c97d687b0a940f8a85a5553388b136fb75aeba92b2c29ba79694a4c', 'fa832b40d8ccb506a195a978e2ee4bc276f7ce83400ff36aea8b466ed6e9d0ac']
TARGETS = ['ACL', 'MCL', 'Medial Meniscus', 'Lateral Meniscus', 'Medial OA', 'Lateral OA', 'PF OA', 'Effusion', 'Synovitis', "Baker's", 'Contusion', 'Fracture']
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
def freeze(name, batch, offset, judgments):
    packet = N / f'neutral-root-attention-{batch:03d}.json'
    assert sha(packet) == PINS[batch]
    rows = json.loads(packet.read_text())['reports'][offset:offset + len(judgments)]
    assert len(rows) == len(judgments)
    records = []
    for row, (letters, evidence, notes) in zip(rows, judgments):
        source = R / row['source_path']
        assert sha(source) == row['source_sha256']
        assert json.loads(source.read_text())[row['index']] == {'study_uid': row['study_uid'], 'report': row['report']}
        states = [dict(P='positive', N='negative', U='uncertain', X='not_mentioned')[x] for x in letters.split()]
        assert len(states) == len(evidence) == 12
        for state, quote in zip(states, evidence):
            assert (state == 'not_mentioned' and quote == '') or (state != 'not_mentioned' and quote and quote in row['report'])
        records.append(dict(case=source.stem.removeprefix('batch-') + ':' + str(row['index']).zfill(2), study_uid=row['study_uid'], source_path=row['source_path'], source_sha256=row['source_sha256'], source_index_zero_based=row['index'], report_sha256=hashlib.sha256(row['report'].encode()).hexdigest(), target_order=TARGETS, states=states, evidence=evidence, notes=notes))
    G.mkdir(parents=True, exist_ok=True)
    previous = [x for p in G.glob('*-source-assessment.json') for x in json.loads(p.read_text())['records']]
    assert not {x['study_uid'] for x in records} & {x['study_uid'] for x in previous}
    assert sha(R / 'protocols/report_weak_v2.md') == '138c5cc33b92901d57dbf5fbe7e3a9878b73d25b0af6934908d0052cd6786014'
    assert sha(N / 'neutral-root-packet-index.json') == 'fd9ff5c333497df8a30d4204fff23afe1eb20f325a1acf8c0304fb30ba41c9ec'
    doc = dict(stage='partial_root_qualified005_source_assessment_before_current_reader_case_findings', agent_id='/root', session_id='/root', at_utc=datetime.now(timezone.utc).isoformat(), neutral_packet=dict(path=str(packet.relative_to(W)), sha256=sha(packet)), packet_rows_zero_based=list(range(offset, offset + len(rows))), rubric_sha256=sha(R / 'protocols/report_weak_v2.md'), canonical_all4407_source_reports_sha256='5c18838a09e9eec6522966555694fca83a9bae67e4d7c5c68592207ee4b60ed3', exposure='Prior full4407 report work, wording000/001/002/003/004, cartilage000–005 source/adjudication and technical/image work can include historical overlaps. These complete sources are personally reread before current wording005 case-specific reader findings or selected current annotation rows. This is not first-ever blindness. No gold values opened.', full_reports=len(records), target_judgments=len(records) * 12, records=records, source_index_uid_report_hash_and_exact_quotes_verified=True, validation_helper_sha256=sha(Path(__file__)), current005_reader_case_findings_opened=False, training_ready=False)
    out = G / (name + '-source-assessment.json')
    with out.open('x') as f:
        f.write(json.dumps(doc, ensure_ascii=False, indent=2) + '\n')
    print(json.dumps(dict(part=name, sha256=sha(out), total_root_sources_frozen=len(previous) + len(records), reader_case_findings_opened=False)))
