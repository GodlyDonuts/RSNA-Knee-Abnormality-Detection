def sha(p):
    return hashlib.sha256(p.read_bytes()).hexdigest()
def pin(p):
    p = p.resolve()
    value = {'path': str(p.relative_to(W)), 'sha256': sha(p), 'bytes': p.stat().st_size}
    old = protected.setdefault(value['path'], value)
    assert old == value, ('input changed', value['path'])
    return value

def read(p):
    pin(p)
    return json.loads(p.read_text())

def lines(p):
    pin(p)
    return [json.loads(line) for line in p.read_text().splitlines()]

def validate_refs(x):
    if isinstance(x, dict):
        if isinstance(x.get('path'), str) and 'sha256' in x:
            check(x)
        for v in x.values():
            validate_refs(v)
    elif isinstance(x, list):
        for v in x:
            validate_refs(v)

def check_dict_manifest(stage, name):
    d = read(stage / name)
    for name, digest in d['files'].items():
        assert pin(stage / name)['sha256'] == digest
    return d

def indexed(rows):
    out = {(r['study_uid'], r['target']): r for r in rows}
    assert len(out) == len(rows)
    return out

def annotation(state, evidence):
    return {'state': state, 'evidence': evidence}

def validate_annotation(a, report, tag):
    assert a['state'] in {'negative', 'positive', 'uncertain', 'not_mentioned'}
    assert isinstance(a['evidence'], str)
    if a['state'] == 'not_mentioned':
        assert a['evidence'] == ''
    else:
        assert a['evidence'] and a['evidence'] in report, ('noncontiguous or changed quote', tag)
        quote_checks[tag] += 1

def write(name, data):
    path = P / name
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open('x') as f:
        json.dump(data, f, ensure_ascii=False, indent=2)
        f.write('\n')
    return {'path': str(path.relative_to(W)), 'sha256': sha(path), 'bytes': path.stat().st_size}

def count_field(d, *keys):
    values = [d[k] for k in keys if k in d]
    assert values and len(set(values)) == 1
    return values[0]
