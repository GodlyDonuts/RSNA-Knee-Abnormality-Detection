sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
pins={'semantic-supervision-membership-builder-569-v1':'43711273b4ab570e66e601952c85aa21a1362ef208caff51ae652f95b0b91f8e','semantic-supervision-membership-independent-assessment-569-v1':'9edb2ffecc5af7b1b82ec04465b41b3f7ac18378237e9aed716fbdc14cfb39a7'}

  refs[p]=dict(sha256=sha(p),bytes=p.stat().st_size)

p=B/'semantic-supervision-root-membership-decision-569-v1/root-membership-decision.json';assert sha(p)=='972a86a9d4f07029a46139e2c41fd388d9742ac4d3494019af717bee0f27863f';refs[p]=dict(sha256=sha(p),bytes=p.stat().st_size)
