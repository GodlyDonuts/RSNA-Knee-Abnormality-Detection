def span(index, first, last):
    report = SOURCES[index]["report"]
    start = report.index(first)
    stop = report.index(last, start) + len(last)
    return report[start:stop]


def add(index, states, quotes, reasons):
    source = SOURCES[index]
    assert index == len(OUT) and len(states.split()) == len(quotes) == len(reasons) == 3
    expanded = [STATES[state] for state in states.split()]
    for state, quote, reason in zip(expanded, quotes, reasons, strict=True):
        assert reason.strip()
        assert (state == "not_mentioned" and quote == "") or (state != "not_mentioned" and quote and quote in source["report"]), (index, state, quote)
