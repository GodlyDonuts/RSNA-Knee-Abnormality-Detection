def sha(p): return hashlib.sha256(p.read_bytes()).hexdigest()
def read(p, expected=None):
    if expected: assert sha(p) == expected, p
    return json.loads(p.read_text())
def bind(p): return {"path": str(p.relative_to(R)), "sha256": sha(p)}
def save(p, obj):
    p.parent.mkdir(parents=True, exist_ok=True)
    with p.open("x") as f: f.write(json.dumps(obj, ensure_ascii=False, indent=2) + "\n")
def archived(p):
    if p.is_relative_to(S):
        out = R / "initiative/audits" / p.relative_to(S)
        return out.with_suffix(".py.txt") if out.suffix == ".py" else out
    assert p.is_relative_to(R)
    return p
