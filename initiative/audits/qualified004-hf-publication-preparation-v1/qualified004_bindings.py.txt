class GuardError(Exception):
    """Fixed nonsensitive codes only."""

def require(value, code):
    if not value:
        raise GuardError(code)

def sha(path):
    require(path.is_file() and not path.is_symlink(), 'missing_or_symlink_file')
    h = hashlib.sha256()
    with path.open('rb') as stream:
        for block in iter(lambda: stream.read(2**20), b''):
            h.update(block)
    return h.hexdigest()

def read(path):
    require(path.is_file() and not path.is_symlink(), 'missing_or_symlink_json')
    return json.loads(path.read_text())

def ref(path):
    return {'local_path': str(path.relative_to(W)), 'sha256': sha(path), 'bytes': path.stat().st_size}

def save(path, value):
    require(path.resolve().is_relative_to(WORK) and not path.resolve().is_relative_to(R), 'output_scope')
    with path.open('x') as f:
        f.write(json.dumps(value, indent=2) + '\n')

def module(name, path, expected):
    require(sha(path) == expected, 'dependency_hash_changed')
    spec = importlib.util.spec_from_file_location(name, path)
    obj = importlib.util.module_from_spec(spec)
    sys.modules[name] = obj
    spec.loader.exec_module(obj)
    return obj

def error_result(error):
    # Never serialize exception payloads from HTTP/Hub/subprocess dependencies.
    return {'status': 'not_completed', 'error_class': type(error).__name__,
            'guard_code': str(error) if type(error) is GuardError else 'external_or_unexpected_error',
            'training_ready': False}
