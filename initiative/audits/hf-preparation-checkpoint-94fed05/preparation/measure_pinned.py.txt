"""Measure a deterministic pinned Git package without writing the tar to disk."""
from pathlib import Path
import gzip,hashlib,io,json,subprocess,tarfile,shutil,datetime
STAGE=Path(__file__).resolve().parent;WORK=STAGE.parents[1];REPO=WORK/'recovered';COMMIT='94fed05d353d180e30c968bc0561e05c00ae6880'

def selected_files():
 result=[]
 for line in subprocess.check_output(['git','ls-tree','-rl',COMMIT],cwd=REPO,text=True,timeout=60).splitlines():
  metadata,path=line.split('\t',1);mode,kind,oid,size=metadata.split()
  if path.startswith(ROOTS) or path in TOP:
   assert kind=='blob' and mode in {'100644','100755'}
   result.append({'path':path,'git_blob':oid,'bytes':int(size),'mode':int(mode,8)&0o777})

def stream_package(dest,files):

 assert p.stdout;refs=[];expected={x['path']:x for x in files};reviewed=reports=0
 try:
  with gzip.GzipFile(filename='',mode='wb',fileobj=dest,mtime=0)as gz,tarfile.open(fileobj=gz,mode='w|')as result,tarfile.open(fileobj=p.stdout,mode='r|')as source:
   for member in source:
    if not member.isfile():continue
    assert member.name in expected;inp=source.extractfile(member);assert inp;data=inp.read();assert len(data)==expected[member.name]['bytes']
    if member.name.startswith('initiative/reviews-v2/reviewed/') and member.name.endswith('.jsonl'):
     rows=[json.loads(line)for line in data.splitlines()];assert all(len(x['states'])==12 for x in rows);reviewed+=1;reports+=len(rows)
    info=tarfile.TarInfo(member.name);info.size=len(data);info.mode=expected[member.name]['mode'];info.mtime=0;result.addfile(info,io.BytesIO(data));refs.append({'path':member.name,'bytes':len(data),'sha256':hashlib.sha256(data).hexdigest(),'git_blob':expected[member.name]['git_blob']})
  assert p.wait(timeout=60)==0;assert {x['path']for x in refs}==set(expected);assert len(refs)==len(expected);assert reviewed==80 and reports==3957
  return refs

  if p.poll() is None:p.kill();p.wait()
