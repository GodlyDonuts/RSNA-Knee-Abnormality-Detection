def sha(p):
    return hashlib.sha256(p.read_bytes()).hexdigest()

def ref(p):
    return {"path": str(p.relative_to(W)), "sha256": sha(p)}

def save(name, doc):
    p = G / name
    assert not p.exists(), p
    p.write_text(json.dumps(doc, ensure_ascii=False, indent=2) + "\n")
    return ref(p)
