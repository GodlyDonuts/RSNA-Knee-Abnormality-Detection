    if globals().get("CLS_PATCH_SUMMARY_RECOVERY_PROBE") is True:
        sources, hashes = MEMORY_SOURCES, MEMORY_HASHES
        if set(sources) != set(hashes) or sum(len(v.encode()) for v in sources.values()) > 1024**2:
            raise ValueError("Exact bounded new scientific source required")
        code = working / "cls-patch-summary-source"
        code.mkdir(mode=0o700)
        for name, source in sources.items():
            relative = Path(name)
            if relative.is_absolute() or ".." in relative.parts or hashlib.sha256(source.encode()).hexdigest() != hashes[name]:
                raise ValueError("New source path/hash differs")
            compile(source, name, "exec")
            path = code / relative
            path.parent.mkdir(parents=True, exist_ok=True)
            with path.open("x") as handle:

        remaining = 7200 - (time.monotonic() - notebook_started)
