def progress(epoch, phase, **fields):
    (L/'scheduled-curve-controller-progress.json').write_text(json.dumps({
        'epoch':epoch,'phase':phase,'completed_epochs':finished,
        'elapsed_seconds':time.monotonic()-started,**fields},indent=2)+'\n')

def pause():
    if time.monotonic()-started > 36000:
        raise TimeoutError('Finite deployment deadline; inspect exact originals, no restart')
    time.sleep(60)

def remote_program(epoch, label, program, payload, *, python=MANAGEMENT, memory='4G', minutes=30):
    d=L/f'scheduled-e{epoch:03d}';d.mkdir(exist_ok=True)
    mutating = label not in ('verify','status')
    if mutating:
        with (d/f'{label}.intent.json').open('x') as f:
            json.dump({'epoch':epoch,'phase':label,'program_sha256':hashlib.sha256(program.encode()).hexdigest(),
                       'automatic_retry':False}, f)
    bootstrap='import json,sys;v=json.load(sys.stdin);exec(compile(v["program"],"scheduled_N3_control","exec"),{"__name__":"__main__","PACKAGE":v})'
    package=dict(payload,program=program)
    cmd=['srun','--account=skattel','--partition=normal','--nodes=1','--nodelist=evc33','--ntasks=1',
         '--cpus-per-task=1','--gres=none','--mem='+memory,f'--time=00:{minutes:02d}:00',
         f'--job-name=rsna-compact-e{epoch}-{label}','env','PYTHONDONTWRITEBYTECODE=1',
         'PYTHONNOUSERSITE=1','OMP_NUM_THREADS=1','MKL_NUM_THREADS=1','OPENBLAS_NUM_THREADS=1',python,'-B','-c',bootstrap]
    x=subprocess.run(['ssh','-S',S,'-o','BatchMode=yes','newton',shlex.join(cmd)],
        input=json.dumps(package),capture_output=True,text=True,timeout=minutes*60+100)
    number=len(list(d.glob(label+'-result-*.json')))
    out={'returncode':x.returncode,'stdout':x.stdout,'stderr':x.stderr}
    (d/f'{label}-result-{number}.json').write_text(json.dumps(out,indent=2)+'\n')
    if x.returncode:
        raise RuntimeError(f'Epoch{epoch} {label} stopped; inspect original intent/result, no mutation retry')
    return json.loads(x.stdout)

def credential():
    path=Path('.env');assert path.stat().st_mode&0o077==0
    values={}
    for line in path.read_text().splitlines():
        k,sep,value=line.partition('=')
        if sep:values[k.strip()]=value.strip().strip('\"\'')
    return values['KAGGLE_API_TOKEN']

try:
    for epoch in (8,12,16):
        # A mere snapshot/intent is insufficient: require complete private-download proof.
        pub=L/f'publication/epoch-{epoch:03d}-result.json'
        while not pub.exists():
            progress(epoch,'waiting_for_original_epoch_private_preservation');pause()
        value=json.loads(pub.read_text());assert value['returncode']==0
        private=json.loads(value['stdout']);assert private['status']=='private_epoch_fully_verified' and private['epoch']==epoch and private['files_verified_by_download']==8

        progress(epoch,'CPU_reference')
        prepared=remote_program(epoch,'component',component_program,{'epoch':epoch,'private_checkpoint_sha256':private['checkpoint_sha256']},python=PYTHON,memory='8G',minutes=15)

        d=L/f'scheduled-e{epoch:03d}'

            return remote_program(epoch,name,cloud_program,{'phase':name,'prepared':prepared,

        # Every cloud mutation has its own exclusive local+canonical intent.
        if r.account()['num_allowed_now']<1:

            try:
                state=r.call('datasets.DatasetApiService','GetDatasetStatus',{'ownerSlug':owner,'datasetSlug':name}).get('status')
            except Exception as exc:
                # Immediate post-create read403 was observed for E4. This is a READ,
                # never permission to repeat the create RPC.
                state='read_unavailable:'+type(exc).__name__
            progress(epoch,'waiting_original_Dataset_ingestion',dataset_status=state)
            if state=='READY':ready=True;break
            if state in ('ERROR','FAILED'):raise RuntimeError('Original Dataset ingestion failed')

        assert ready,'Original Dataset not ready; reconcile its one create read-only'

        # Defer public allocation while two cluster GPU nodes are actually active.
        while True:

            q=r.call('kernels.KernelsApiService','GetAcceleratorQuotaStatistics',{})['gpuQuota'];seconds=lambda s:float(s.removesuffix('s'))
            available=min(seconds(q['minimumTimeAllowed']),seconds(q['totalTimeAllowed']))-seconds(q['timeUsed'])-seconds(q['timeReserved'])
            if len(nodes)<=1 and available>=64800:break
            progress(epoch,'waiting_actual_GPU_capacity',active_cluster_nodes=len(nodes),available_Kaggle_seconds=available);pause()
        phase('push')
        public_started=time.monotonic()
        while time.monotonic()-public_started<10800:
            status=r.kernel('GetKernelSessionStatus',notebook['kernel_slug'],1).get('status')
            progress(epoch,'exact_public_execution',public_status=status)
            if status in ('COMPLETE','ERROR','CANCELLED'):
                phase('status')
                assert status=='COMPLETE','Original public version did not pass; no submission'
                accepted=r.output_json(notebook['kernel_slug'],1,'compact-production-accepted.json')
