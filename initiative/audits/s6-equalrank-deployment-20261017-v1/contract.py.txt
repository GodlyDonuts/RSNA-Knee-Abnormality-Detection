def source_contract(original, derived):
    if hashlib.sha256(original.encode()).hexdigest() != ORIGINAL or hashlib.sha256(derived.encode()).hexdigest() != DERIVED:
        raise ValueError("Exact original and already prepared fixed derivative bytes required")
    left, right = ast.parse(original), ast.parse(derived)
    def special(n):
        return isinstance(n, ast.Assign) and len(n.targets) == 1 and (
            isinstance(n.targets[0], ast.Name) and n.targets[0].id == "K1_BUILD_ID" or
            isinstance(n.targets[0], ast.Tuple) and [getattr(x, "id", None) for x in n.targets[0].elts] == ["W_A", "W_B"])
    if sum(special(n) for n in left.body) != 2 or sum(special(n) for n in right.body) != 2:
        raise ValueError("Exactly two intended top-level assignments required")
    l = ast.Module(body=[n for n in left.body if not special(n)], type_ignores=[])
    r = ast.Module(body=[n for n in right.body if not special(n)], type_ignores=[])
    if ast.dump(l) != ast.dump(r):
        raise ValueError("Original inference, geometry, model, TTA, fallback and guards must stay identical")
    assignment = next(n for n in right.body if special(n) and isinstance(n.targets[0], ast.Tuple))
    if ast.literal_eval(assignment.value) != (0.5, 0.5):
        raise ValueError("Predeclared equal weights only")
    fn = next(n for n in left.body if isinstance(n, ast.FunctionDef) and n.name == "build_id_of")
    ns = {"re": re, "hashlib": hashlib}
    exec(compile(ast.Module(body=[fn], type_ignores=[]), "original_build_id_function_only", "exec"), ns)
    declared = ast.literal_eval(next(n.value for n in right.body if special(n) and isinstance(n.targets[0], ast.Name)))
    if ns["build_id_of"](derived) != declared:
        raise ValueError("Actual derived source build ID must match")
    compile(derived, "fixed_equalrank_source", "exec")
    return {"source_sha256": DERIVED, "original_source_sha256": ORIGINAL, "build_id": declared, "weights": WEIGHTS, "only_two_intended_AST_assignments_changed": True}


def compute_budget(quota, *, unfinished_main, unfinished_sampling, known_reserved_main, scout_unfinished):
    def seconds(x):
        if not isinstance(x, str) or not x.endswith("s"):
            raise ValueError("Actual finite seconds required")
        v = float(x[:-1])
        if not math.isfinite(v) or v < 0:
            raise ValueError("Actual finite nonnegative quota required")
        return v
    total, minimum, used, reserved = [seconds(quota[k]) for k in ("totalTimeAllowed", "minimumTimeAllowed", "timeUsed", "timeReserved")]
    if not total >= minimum > 0 or any(type(x) is not int or x < 0 for x in (unfinished_main, unfinished_sampling, known_reserved_main)) or type(scout_unfinished) is not bool:
        raise ValueError("Fresh real grant and all unfinished obligations required")
    limit = total if quota.get("isPayToScaleEnabled") is True else min(total, minimum)
    available = limit - used - reserved + min(reserved, known_reserved_main)
    required = unfinished_main + unfinished_sampling + 3600 + (1800 if scout_unfinished else 0) + 32400
    return {"funded_VM_seconds": available, "required_VM_seconds": required, "own_session_seconds": 32400, "serving_reserve_seconds": 3600, "scout_reserve_seconds": 1800 if scout_unfinished else 0, "allowed": available >= required, "quota_limit_basis": "Already-granted active allowance; no billing/settings changes", "fresh_GPU_quota": quota}
