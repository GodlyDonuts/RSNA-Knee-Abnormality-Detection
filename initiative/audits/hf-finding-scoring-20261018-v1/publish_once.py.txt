"""Distinct small source/proof delta under original cumulative8MiB/shared lock."""
import hashlib
import json
from pathlib import Path
import shlex
import subprocess

D = Path(__file__).resolve().parent
L = Path('artifacts/hf-finding-scoring-20261018-v1')
L.mkdir(mode=0o700, parents=True, exist_ok=True)
assert not (L / 'intent.json').exists()
git = subprocess.check_output(['git', 'rev-parse', 'HEAD'], text=True).strip()
base = 'dfe89e4f23965c55a78b71a8d902b88ddbe8d4a3'
def content(name):
    return subprocess.check_output(['git', 'show', git + ':' + name], text=True)
names = subprocess.check_output(['git', 'diff', '--name-only', base, git], text=True).splitlines()
bindings = {}
for name in names:
    raw = content(name).encode()
    bindings[name] = {'bytes': len(raw), 'sha256': hashlib.sha256(raw).hexdigest(),
                      'git_blob': subprocess.check_output(['git', 'rev-parse', git + ':' + name], text=True).strip()}
files = {'public-Git-source-bindings.json': json.dumps({'source_git': git, 'previous_verified_source': base,
          'scope': 'Exact public source hashes; unchanged old source and large guides are not recopied.',
          'members': bindings}, indent=2)+'\n'}
prefixes = (
 'initiative/experiments/competitive-v1/finding-final-scoring-20261018-v1/',
 'initiative/experiments/competitive-v1/family-final-scoring-20261018-v1/',
 'initiative/experiments/competitive-v1/finding-final-serving-20261018-v1/',
 'initiative/research/connected-witness-paths-20261017-v1/',
 'initiative/research/finding-stratified-bags-20261018-v1/',
 'initiative/research/path-recovery-evc26-20261018-v2/',
 'initiative/research/path-recovery-evc48-20261018-v3/',
 'initiative/experiments/competitive-v1/multisource-supervision-v1/',
 'initiative/audits/label-source-lineage-20261018-v1/',
 'initiative/experiments/competitive-v1/multisource-influence-v1/',
 'initiative/audits/exact-R6-entry-20261018-v1/',
 'initiative/audits/whole94-final-qualification-20261018-v2/',
 'initiative/audits/whole94-final-public-20261018-v2/',
 'initiative/audits/matched-final-scoring-owner-recovery-20261018-v3/',
)
for name in names:
    if name.startswith(prefixes):
        files[name] = content(name)
name='initiative/audits/hf-final-serving-followthrough-20261018-v1/verified.json'
files[name]=content(name)
for guide in ('AGENTS.md','docs/current-state.md','docs/codex-continuation.md'):
    files[guide]=content(guide)
assert files and sum(len(s.encode()) for s in files.values()) < 768 * 2**10
program = content(D.relative_to(Path.cwd()).as_posix() + '/publication_program.py.txt')
compile(program, 'distinct_variable_paths_dense_and_final_private_milestone', 'exec')
E = Path('.env')
assert E.stat().st_mode & 0o077 == 0
values = {k.strip(): val.strip().strip('"\'') for line in E.read_text().splitlines()
          for k, sep, val in [line.partition('=')] if sep and not line.lstrip().startswith('#')}
with (L / 'intent.json').open('x') as f:
    json.dump({'source_git': git, 'previous_source': base, 'program_sha256': hashlib.sha256(program.encode()).hexdigest(),
               'files': len(files), 'bytes': sum(len(s.encode()) for s in files.values()), 'automatic_retry': False}, f, indent=2)
args = ['srun', '--account=skattel', '--partition=normal', '--qos=normal', '--nodelist=evc33',
        '--ntasks=1', '--cpus-per-task=1', '--gres=none', '--mem=2G', '--time=00:15:00',
        '--job-name=rsna-variable-path-dense-final-private-proof', 'env', 'PYTHONNOUSERSITE=1',
        'PYTHONDONTWRITEBYTECODE=1', '/home/sa305415/rsna-knee/hf-epoch-publisher-v1/env-cp312/bin/python', '-B', '-c', program]
x = subprocess.run(['ssh', '-S', '/tmp/rsna-research-20261001-stable.sock', '-o', 'BatchMode=yes',
                    'newton', shlex.join(args)], input=json.dumps({'source_git': git, 'files': files, 'token': values['HF_TOKEN']}),
                   capture_output=True, text=True, timeout=950)
(L / 'result.json').write_text(json.dumps({'returncode': x.returncode, 'stdout': x.stdout, 'stderr': x.stderr}, indent=2)+'\n')
assert x.returncode == 0, 'Reconcile existing original private intent read-only; no automatic retry'
out = json.loads(x.stdout)
(D / 'verified.json').write_text(json.dumps(out, indent=2)+'\n')
print(json.dumps(out))
