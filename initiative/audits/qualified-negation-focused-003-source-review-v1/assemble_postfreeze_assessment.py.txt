def proposed(r,reason,kind):
    uid,target=r['study_uid'],r['target'];key=(uid,target)
    assert key not in registered and r['original_state']==r['effective_state'] and r['original_evidence']==r['effective_evidence']
    state,evidence=r['first_assessor_state'],r['first_assessor_evidence']
    assert evidence=='' if state=='not_mentioned' else evidence.strip() and evidence in source_by_uid[uid]['report']
    before=int(r['effective_state'] in ['negative','positive']);after=int(state in ['negative','positive'])
    return dict(study_uid=uid,target=target,batch=r['batch'],index=r['index'],packet_index=r['packet_index'],
          source_path=r['source_path'],source_sha256=r['source_sha256'],artifact_path=r['original_review_path'],artifact_sha256=r['original_review_sha256'],
          before_state=r['original_state'],before_evidence=r['original_evidence'],proposed_state=state,source_quote=evidence,reason=reason,kind=kind,
          report_weak_mask_before=before,report_weak_mask_after=after,
          mask_impact='observed_to_masked' if before and not after else 'masked_to_masked' if not before and not after else 'masked_to_observed' if not before and after else 'observed_unchanged',
