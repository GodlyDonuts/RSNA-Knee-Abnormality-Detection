def sha(p):
 h=hashlib.sha256()
 with p.open('rb') as f:
  for b in iter(lambda:f.read(4*1024*1024),b''):h.update(b)
 return h.hexdigest()
def save(path,value):

def download(url,path,expected,size):
 with urllib.request.urlopen(url,timeout=30) as r,path.open('xb') as out:
  total=0
  for b in iter(lambda:r.read(1024*1024),b''):
   total+=len(b)
   if total>size:raise ValueError('download exceeds bound')
   out.write(b)
  out.flush();os.fsync(out.fileno())
 assert total==size and sha(path)==expected,'download hash/size mismatch'
