def read(path):
    return json.loads(path.read_text())

def sha(path):
    h = hashlib.sha256()
    with path.open('rb') as f:
        for block in iter(lambda: f.read(1048576), b''):
            h.update(block)
    return h.hexdigest()

def ref(path):
    return {'path': str(path.relative_to(W)), 'sha256': sha(path), 'bytes': path.stat().st_size}

checked = {}
def bindings(value):
    if isinstance(value, dict):
        if {'path', 'sha256', 'bytes'} <= value.keys() and 'sop_uid' not in value and 'storage' not in value:
            path = W / value['path']
            assert path.resolve().is_relative_to(W.resolve())
            assert path.stat().st_size == value['bytes']
            if str(path) not in checked:
                checked[str(path)] = sha(path)
            assert checked[str(path)] == value['sha256'], value['path']
        for child in value.values():
            bindings(child)
    elif isinstance(value, list):
        for child in value:
            bindings(child)
