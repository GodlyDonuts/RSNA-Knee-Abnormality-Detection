def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()


def load(name, p):
    s = importlib.util.spec_from_file_location(name, p)
    m = importlib.util.module_from_spec(s)
    s.loader.exec_module(m)
    return m


def request_for(candidate, mode, text):
    assert candidate in ROLES and mode in ('qualification', 'competition')
    slug = 'godlydonuts/rsna-cached-' + candidate + '-e3-' + ('qual-' if mode == 'qualification' else 'offline-') + hashlib.sha256(text.encode()).hexdigest()[:12]
    return {'slug': slug, 'newTitle': slug.split('/')[1].replace('-', ' '),
            'text': text, 'language': 'python', 'kernelType': 'script',
            'isPrivate': True, 'enableGpu': True, 'machineShape': 'NvidiaTeslaT4',
            'enableInternet': False, 'datasetDataSources': DATASETS,
            'competitionDataSources': [] if mode == 'qualification' else ['rsna-knee-abnormality-detection'],

def remaining_budget(original, mode):
    assert mode in ('qualification', 'competition')
    assert original['original_main_seconds'] == original['original_sampling_seconds'] == 0
    # Only after all original label epochs and all three matched public MRI
    # validations actually complete: retire their recovery, fit and serving.
    # Retain the unresolved reference, old whole94 serving and common margin.
    retired = 1800 + 18000 + 3600
    current = 34200 if mode == 'qualification' else 32400
    required = original['required_all_future_VM_seconds'] - retired + current
    assert required >= current
    return {**original, 'retired_actual_completed_label_recovery_fit_serving_seconds': retired,
            'current_candidate_remaining_caps_seconds': current,
            'required_current_candidate_and_all_original_future_seconds': required,
            'later_head_candidates_not_admitted_by_this_phase': True,
            'allowed': original['available_VM_seconds'] >= required}


def bounds(extra=0):
    assert os.uname().nodename == 'evc33' and not os.environ.get('SLURM_JOB_GPUS')
    assert sum(p.stat().st_size for p in D.rglob('*') if p.is_file()) + extra + 2**20 < 32 * 2**20
    assert sum(p.stat().st_size for p in ROOT.rglob('*') if p.is_file()) + extra + 16 * 2**20 < 256 * 2**20
    assert shutil.disk_usage('/tmp').free > 612022943803
    q = subprocess.check_output(['lfs', 'quota', '-u', 'sa305415', '/home/sa305415'], text=True)
    row = next(x.split() for x in q.splitlines() if x.strip().startswith('/home/sa305415'))
    assert (int(row[2]) - int(row[1])) * 1024 > 20 * 2**30 + 4 * 2**20
    assert int(row[6]) - int(row[5]) > 5008


def record(db, key):
    found = db.execute('SELECT value FROM meta WHERE key=?', (key,)).fetchone()
    assert found is not None, 'Original canonical identity required'
    return json.loads(found[0])


def originals_ready():
    p = PREVIOUS / 'single-scoring-coordinator-v1/program.py.txt'
    assert sha(p) == '38cccb2a1a7d97c834852783a364be90ee94b054ebc9bcea8061f43eabde6915'
    ready = {}
    for role in ('uniform', 'label', 'influence'):
        p = PREVIOUS / role / 'competition-actual-accepted.json'
        if not p.exists():
            return {'ready': False, 'waiting_original_matched_candidate': role}
        actual = json.loads(p.read_text())
        receipt = actual['receipt']
        assert actual['status'] == 'COMPLETE' and actual['mode'] == 'competition'
        assert receipt['status'] == 'actual_matched_IID16_parent_epoch_MRI_test_roster_CSV_complete'
        assert receipt['epoch'] == 3 and receipt['serving_windows'] == 94
        assert receipt['studies'] == 3 and receipt['columns'] == 13
        assert receipt['prediction_counts'] == dict(started=6, completed=6, failed=0, decode_errors=0, empty_views=0)
        assert actual['CSV_verified_in_RAM']['CSV_sha256'] == receipt['CSV_sha256']
        ready[role] = {'kernel_id': actual['kernel_id'], 'source_sha256': actual['source_sha256']}
    return {'ready': True, 'all_three_exact_public_MRI_validations': ready}


def validate_receipt(receipt, candidate, mode):
    role, windows = ROLES[candidate]
    assert receipt['role'] == role and receipt['training_windows'] == windows
    assert receipt['head_sha256'] == FINAL[candidate]
    assert receipt['parent_sha256'] == '7e5315dad125b99fc65b340b3de41de628e9be51ff5835355dd61c86472244ef'
    assert receipt['model_source_git'] == MODEL_SOURCE and receipt['serving_windows'] == 94
    assert receipt['epoch'] == 3 and receipt['completed_training_updates'] == 1458
    assert receipt['encoder_changed'] is False and receipt['independent_internal_validation'] is False
    if mode == 'qualification':
        assert receipt['status'] == 'actual_cached_MIL_bothT4_original32_MRI_training_graph_qualified'
        assert receipt['qualified_studies'] == 32 and receipt['qualification_cannot_submit'] is True
        assert len(receipt['devices']) == 2 and {x['index'] for x in receipt['devices']} == {0, 1}
        assert all('T4' in x['GPU'] and x['MRI_studies'] == 32
                   and x['both_original94_views_bit_equal'] is True
                   and x['backbone_tensor_values_unchanged'] is True
                   and x['all_eight_learned_head_tensors_exact'] is True
                   and x['head_precision'] == 'FP32_outside_encoder_autocast'
                   and 0 <= x['maximum_original_training_graph_probability_error'] <= 1e-6
                   and 0 < x['peak_allocated_bytes'] < 16 * 2**30 for x in receipt['devices'])
    else:
        assert mode == 'competition' and receipt['status'] == 'actual_cached_MIL_MRI_test_roster_CSV_complete'
        assert receipt['studies'] == 3 and receipt['columns'] == 13
        assert receipt['prediction_counts'] == dict(started=6, completed=6, failed=0, decode_errors=0, empty_views=0)
