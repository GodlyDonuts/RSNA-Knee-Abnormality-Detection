"""Private original fitting pixels; credential stdin only, no model training."""
import hashlib,json,os,shutil,sys,zipfile
from pathlib import Path
from urllib.parse import quote
import requests
import pydicom
def main():
 R=Path(sys.argv[1]);v=json.load(sys.stdin);token=v.pop('token');plan=json.loads((R/'plan-admitted.json').read_text());private=json.loads((R/'private-original-roster-and-files.json').read_text());assert os.uname().nodename=='evc31' and not os.environ.get('SLURM_JOB_GPUS')
 assert hashlib.sha256((R/'private-original-roster-and-files.json').read_bytes()).hexdigest()==plan['private_inventory_sha256']
 assert not (R/'acquisition-intent.json').exists() and shutil.disk_usage('/tmp').free>140*2**30+624*2**20
 with (R/'acquisition-intent.json').open('x') as f:json.dump({'files':plan['raw_source_files'],'bytes':plan['raw_source_bytes'],'source_revision':plan['raw_source_revision'],'source_manifest_sha256':plan['private_inventory_sha256'],'automatic_retry':False,'private':True},f)
 raw=R/'original';raw.mkdir(mode=0o700)
 for row in private['files']:
  parts=Path(row['path']).parts;start=parts.index('train_series');relative=Path(*parts[start:]);target=raw/relative;target.parent.mkdir(mode=0o700,parents=True,exist_ok=True)
  assert not target.exists()
  try:
   response=requests.get('https://huggingface.co/datasets/GodlyDonuts/RSNA-Data/resolve/'+private['source_revision']+'/'+quote(row['path'],safe='/'),headers={'Authorization':'Bearer '+token},timeout=(10,60),stream=True)
  except requests.RequestException:raise RuntimeError('Original source transport failed') from None
  with response:
   if response.status_code!=200:raise RuntimeError('Original source HTTP '+str(response.status_code))
   h=hashlib.sha256();count=0
   with target.open('xb') as f:
    try:
     for block in response.iter_content(65536):
      count+=len(block)
      if count>int(row['bytes']):raise RuntimeError('Original source size exceeded')
      h.update(block);f.write(block)
    except requests.RequestException:raise RuntimeError('Original source stream failed') from None
  assert count==int(row['bytes']) and h.hexdigest()==row['sha256']
  ds=pydicom.dcmread(target,stop_before_pixels=True);assert str(ds.StudyInstanceUID)==str(row['study_uid']) and str(ds.SeriesInstanceUID)==str(row['series_uid']) and str(ds.SOPInstanceUID)==str(row['sop_uid'])
 assert sum(p.stat().st_size for p in raw.rglob('*.dcm'))==plan['raw_source_bytes']
 with zipfile.ZipFile('/tmp/rsna-public-solution-models-20261007-v1/pilkwang--rsna-knee-weights--v1.zip') as z:
  data=z.read('m_e5427d6c21.pt');assert hashlib.sha256(data).hexdigest()=='7223cd50ae5e7bdc8d3ffe888ab207bdbb2256d06e33f96e7a709eca5d86ab33'
  with (R/'checkpoint.pt').open('xb') as f:f.write(data)
 del data,token
 # Extract only the verified abi3 headless wheel into this private comparison root.
 # No shared training environment is installed into or changed.
 wheel_member='opencv-python-headless-4120088-x86/opencv_python_headless-4.12.0.88-cp37-abi3-manylinux2014_x86_64.manylinux_2_17_x86_64.whl'
 with zipfile.ZipFile('/tmp/rsna-public-solution-models-20261007-v1/tonylica--rsna-knee-bend-dinov3-0917-repro-assets--v5.zip') as archive:
  wheel=archive.read(wheel_member)
  assert hashlib.sha256(wheel).hexdigest()=='236c8df54a90f4d02076e6f9c1cc763d794542e886c576a6fee46ec8ff75a7a9'
 import io
 extra=R/'private-opencv';extra.mkdir(mode=0o700)
 with zipfile.ZipFile(io.BytesIO(wheel)) as archive:
  for member in archive.infolist():
   relative=Path(member.filename);assert not relative.is_absolute() and '..' not in relative.parts
   if member.is_dir():continue
   target=extra/relative;target.parent.mkdir(mode=0o700,parents=True,exist_ok=True)
   with target.open('xb') as f:f.write(archive.read(member))
 del wheel
 sys.path.insert(0,str(extra))
 import cv2
 assert cv2.__version__=='4.12.0'
 assert sum(p.stat().st_size for p in R.rglob('*') if p.is_file() and not p.is_relative_to(raw))<512*2**20
 from scripts.newton.compare_public_dinov2_inputs import run
 from argparse import Namespace
 inp=json.loads(Path('/tmp/rsna-public-label-inputs-20261006-v1/CPU-accepted.json').read_text())['trainer_inputs']
 args=Namespace(source=str(R/'code/initiative/audits/public-dinov2-source-handoff-20261013-v1/source.notebook.json'),member_manifest=str(R/'private-member-manifest.json'),checkpoint=str(R/'checkpoint.pt'),studies=inp['studies'],artifact_root=inp['artifact-root'],folds=inp['folds'],series=inp['series'],cache=inp['cache'],dicom_root=str(raw),roster_size=8,out=None)
 report=run(args)
 report.update(original_source_files_verified=plan['raw_source_files'],original_source_bytes_verified=plan['raw_source_bytes'],CPU_job=os.environ['SLURM_JOB_ID'],source_revision=plan['raw_source_revision'],clinical_accuracy_claimed=False,model_or_optimizer_executed=False)
 with (R/'actual-input-comparison.json').open('x') as f:json.dump(report,f,indent=2,allow_nan=False)
 print(json.dumps(report,allow_nan=False))

if __name__=='__main__':
 try:main()
 except Exception as error:
  print(json.dumps({'status':'private_original_comparison_failed','error_type':type(error).__name__,'raw_details_exported':False}))
  sys.exit(1)
