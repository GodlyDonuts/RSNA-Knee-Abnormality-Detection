"""Bounded offline validation only; never calls preservation execute()."""
from datetime import datetime, timezone
import importlib.util
import json
from pathlib import Path
import shutil
import stat
import sys
import tempfile

P = Path(__file__).resolve().parent
W = P.parents[2]
R = W / "work/recovered"
AGENT = "/root/duplicate_consistency_review"
spec = importlib.util.spec_from_file_location("preservation491_validation", P / "preserve_verified491_publication.py")
helper = importlib.util.module_from_spec(spec)
sys.modules[spec.name] = helper
spec.loader.exec_module(helper)


def binding(path):
    return {"path": str(path.relative_to(W)), "sha256": helper.sha(path), "bytes": path.stat().st_size}


def write_once(name, obj):
    raw = helper.serialized(obj)
    plan["scanner"].Scanner(plan["secrets"]).feed(raw)
    with (P / name).open("xb") as out:
        out.write(raw)


def target_snapshot():
    result = {}
    for name in ("qualified003-verified491-publication-v1", "qualified003-hf-publication-preparation-v1"):
        folder = R / "initiative/audits" / name
        result[name] = None if not folder.exists() else {
            str(p.relative_to(folder)): (helper.sha(p), p.stat().st_size, stat.S_IMODE(p.stat().st_mode))
            for p in folder.rglob("*") if p.is_file()
        }
    return result


before = target_snapshot()
free_before = shutil.disk_usage(W).free
plan = helper.plan(W, P / "publication-input-pins.json")
checks = []
with tempfile.TemporaryDirectory(prefix="bounded-guards-", dir=P) as folder:
    tmp = Path(folder)
    source = tmp / "source.bin"
    source.write_bytes(b"original preservation bytes")
    source.chmod(0o640)
    item = {"sha256": helper.sha(source), "bytes": source.stat().st_size, "mode": 0o640}
    helper.verify(source, item)
    checks.append("exact input bytes and mode accepted")

    def rejected(label, function, code):
        try:
            function()
        except helper.GuardError as exc:
            assert str(exc) == code
            checks.append(label)
        else:
            raise AssertionError(label)

    rejected("changed hash rejected", lambda: helper.verify(source, dict(item, sha256="0" * 64)), "exact_input_mismatch")
    rejected("changed mode rejected", lambda: helper.verify(source, dict(item, mode=0o600)), "source_mode_mismatch")
    rejected("relative traversal rejected", lambda: helper.safe(tmp, "../escape"), "unsafe_relative_path")
    rejected("absolute destination rejected", lambda: helper.safe(tmp, str(source)), "unsafe_relative_path")
    assert helper.existing(source, source.read_bytes(), 0o640)
    checks.append("existing equal bytes and mode reusable")
    rejected("existing unequal bytes rejected without overwrite", lambda: helper.existing(source, b"different", 0o640), "existing_destination_differs")
    alias = tmp / "alias.bin"
    alias.symlink_to(source)
    rejected("symlink destination rejected", lambda: helper.safe(tmp, "alias.bin"), "symlink_input_or_output")
    assert source.read_bytes() == b"original preservation bytes"

assert target_snapshot() == before
free_after = shutil.disk_usage(W).free
assert free_after > helper.FLOOR
