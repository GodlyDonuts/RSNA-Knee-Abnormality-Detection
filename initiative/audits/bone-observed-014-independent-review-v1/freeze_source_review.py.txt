def uid_strings(x):
    if isinstance(x,str): return {x} if x.startswith('1.2.826.') else set()
    if isinstance(x,dict): return set().union(*(uid_strings(v) for v in x.values()))
    if isinstance(x,list): return set().union(*(uid_strings(v) for v in x)) if x else set()
    return set()
