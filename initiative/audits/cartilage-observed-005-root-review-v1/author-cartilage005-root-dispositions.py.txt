"""Preserve the root's decisions after six personally read complete sources."""
from pathlib import Path
from datetime import datetime, timezone
from collections import Counter
import hashlib, json

W = Path(__file__).resolve().parent.parent
R = W / 'work/recovered'
A = W / 'work/annotation-staging'
G = A / 'cartilage-observed-005-root-review-v1'
S = A / 'cartilage-observed-005-independent-postfreeze-v1'
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
read = lambda p: json.loads(p.read_text())
ref = lambda p: dict(path=str(p.relative_to(W)), sha256=sha(p))
def save(name, data):
    p = G / name
    with p.open('x') as f:
        f.write(json.dumps(data, ensure_ascii=False, indent=2) + '\n')
    return p

freeze = G / 'complete-source-assessment-frozen.json'
comp = G / 'root-postfreeze-comparison.json'
reg = R / 'initiative/reviews-v2/adjudications/registry.json'
raw = S / 'unintegrated-proposals.json'
first = A / 'cartilage-observed-005-postfreeze-v1/proposed-errata.json'
assert sha(freeze) == '7b921db82e9937b90ed8650174004e84371342f2e002a566ad9a078e8d1b7eb3'
assert sha(comp) == '2b96729fb67c3db02ba83119009b6f284da048ca3f2751103499ff805e99b473'
assert sha(reg) == '761ba526baf0c761548e2d5ff41c52ce9ba24ac75e9ac43993a716240ce37693'
assert sha(raw) == 'af93940bc9e152bcbc17e93cd7eb5c9903bdf1e6e1601c0b8601d9bb7e1dd3f9'

# These four choices were individually assessed after reading the full frozen
# source judgments, both readers' findings and all proposed exact quotations.
choices = {
    ('085:13', 'PF OA'): 'Accept negative to uncertain. The source only suspects a patellar contour finding and says rule out an unnamed grade 2. Neither this provisional grading nor trochlear bone sclerosis establishes a whole-PF below-threshold cartilage assessment. My frozen uncertainty is unchanged.',
    ('087:07', 'Lateral OA'): 'Retain positive and accept the longer exact evidence. The lateral loss is expressly diffuse across femoral and tibial surfaces; the same report immediately defines its low/high terminology as below/above half. The expanded quote binds that internal linguistic inference. The detached fragment remains source context and is not needed to establish cartilage size; its dimensions do not transfer to PF. My frozen positive is unchanged.',
    ('087:33', 'Lateral OA'): 'Retain negative and replace the unanchored numerical-grade quotation with the source body describing slight diffuse cartilage thinning in the lateral compartment. This is a direct mild-loss morphology inference with compartmental scope, independent of an assumed grade table. The second reader\'s original stricter uncertainty and their subsequent reconsideration remain preserved. My frozen negative is unchanged.',
    ('088:04', 'PF OA'): 'Accept negative to uncertain. Patchy patellar signal changes plus an unnamed grade I do not themselves supply direct loss fraction or a whole-PF cartilage assessment; trochlear cartilage is not described. My prior detailed source/pixel-duplicate exposure remains disclosed, and this annotation decision does not change that leakage grouping. My frozen uncertainty is unchanged.',
}
def ck(x):
    b, i = x['case'].removeprefix('batch-').split(':')
    return b + ':' + i.zfill(2), x['target']

proposals = read(raw)['records']
assert len(proposals) == 4 and {ck(p) for p in proposals} == set(choices)
selected = {(p['study_uid'], p['target']): p for p in proposals}
base = dict(schema_version=1, agent_id='/root', session_id='/root', at_utc=datetime.now(timezone.utc).isoformat(), rubric_sha256='138c5cc33b92901d57dbf5fbe7e3a9878b73d25b0af6934908d0052cd6786014', source_reports_sha256='5c18838a09e9eec6522966555694fca83a9bae67e4d7c5c68592207ee4b60ed3', source_freeze=ref(freeze), comparison=ref(comp), registry_at_comparison=ref(reg), prior_source_and_decision_exposure_disclosed=True, original_artifacts_preserved=True, gold_values_inspected=False, registry_changed=False, training_ready=False)
records = []
supported = []
for x in read(comp)['records']:
    root = x['root_frozen']
    if root is None:
        continue
    key = x['study_uid'], x['target']
    rkey = ck(x)
    adopted = selected[key]['after_proposed'] if key in selected else x['current358']
    assert adopted['state'] == root['state'] == x['first_postfreeze']['state'] == x['second_postfreeze']['state']
    source = read(R / x['source_path'])[x['source_index_zero_based']]
    assert source['study_uid'] == x['study_uid'] and sha(R / x['source_path']) == x['source_sha256']
    assert (adopted['state'] == 'not_mentioned' and not adopted['evidence']) or (adopted['state'] != 'not_mentioned' and adopted['evidence'] and adopted['evidence'] in source['report'])
    reason = choices.get(rkey, root['reason'])
    if rkey == ('088:00', 'Medial OA'):
        reason = 'Retain positive on the explicitly broad full-thickness medial femoral loss. I clarify my initial reason: the nearby 9 by 19 mm tibial defect has sufficient size but its tibial side is not explicit in that sentence. It is unnecessary to assign it to the medial compartment. The medial femoral description independently supports this unchanged positive; the earlier reason remains preserved.'
    item = dict(case=rkey[0], study_uid=x['study_uid'], target=x['target'], source_path=x['source_path'], source_sha256=x['source_sha256'], current358=x['current358'], root_frozen_state=root['state'], root_frozen_evidence=root['evidence'], root_state_reconsidered=False, root_postfreeze_state=adopted['state'], root_postfreeze_evidence=adopted['evidence'], first_postfreeze_state=x['first_postfreeze']['state'], second_postfreeze_state=x['second_postfreeze']['state'], reason=reason, disposition='support_exact_second_proposal_pending_runtime_integration' if key in selected else 'retain_current_annotation', accepted=False)
    records.append(item)
    if key in selected:
        supported.append(dict(case=rkey[0], study_uid=x['study_uid'], target=x['target'], source_path=x['source_path'], source_sha256=x['source_sha256'], root_frozen_state=root['state'], root_state_reconsidered=False, independent_source_state=adopted['state'], source_evidence=adopted['evidence'], reason=reason))
assert len(records) == 18 and len(supported) == 4
first_support = []
for p in read(first)['proposed_errata']:
    key = p['study_uid'], p['target']
    assert key in selected and selected[key]['after_proposed']['state'] == p['proposed_state']
    first_support.append(dict(case=p['case'], study_uid=p['study_uid'], target=p['target'], state_supported=True, accepted_via_second_proposal=True, first_quote_equals_adopted_second_quote=p['source_quote'] == selected[key]['after_proposed']['evidence'], original_first_proposal_preserved=True))
assert len(first_support) == 3
transitions = dict(Counter(p['before_current']['state'] + '->' + p['after_proposed']['state'] for p in proposals))
save('root-explicit-reconsiderations.json', {**base, 'stage': 'root_postfreeze_reason_clarification_without_state_change', 'state_reconsiderations': [], 'reason_clarifications': [dict(case='088:00', target='Medial OA', clarification='The measured tibial defect is not explicitly sided in its sentence. The accepted positive rests independently on explicit broad full-thickness medial femoral loss.', original_reason_preserved=True)]})
disp = save('root-postfreeze-dispositions.json', {**base, 'stage': 'root_source_based_dispositions_pending_runtime_integration', 'counts': dict(root_reports=6, root_cells=18, own_state_reconsiderations=0, second_proposals_supported=4, second_proposals_rejected=0, first_proposal_states_supported_via_second=3, additional_root_proposals=0), 'records': records, 'rejected_proposals': [], 'first_proposal_state_support': first_support, 'transition_counts': transitions, 'authorship': 'I personally read all six complete neutral attention reports and froze 18 judgments before current005 reader case findings. After reading both comparisons and all four exact proposals, I support all four and retain my frozen states. One anatomical rationale is explicitly clarified. Agreement alone does not determine these decisions.'})
assess = save('second-proposals-independent-assessment.json', {**base, 'stage': 'independent_source_assessment', 'proposer_agent_id': '/root/last_twelve_visual_review', 'dispositions': ref(disp), 'original_proposals': ref(raw), 'first_proposals': ref(first), 'records': supported, 'rejected_proposals': [], 'n_changes_supported': 4, 'n_changes_rejected': 0, 'n_changes_held': 0, 'authorship': 'Actual root source assessment and postfreeze adjudication. Runtime integration and actual-author source-context normalization remain separate steps.'})
for p in [Path(__file__), W / 'work/compare-cartilage005-root-v1.py']:
    with (G / (p.name + '.txt')).open('xb') as f:
        f.write(p.read_bytes())
print(json.dumps(dict(dispositions_sha256=sha(disp), assessment_sha256=sha(assess), root_cells=18, supported=4, rejected=0, state_reconsiderations=0, transitions=transitions, training_ready=False)))
