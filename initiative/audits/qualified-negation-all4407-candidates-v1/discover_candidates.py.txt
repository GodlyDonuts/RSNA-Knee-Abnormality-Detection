def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def write(name, value):
    with (OUT / name).open("x") as f:
        json.dump(value, f, ensure_ascii=False, indent=2)
        f.write("\n")

def jsonl(name, records):
    with (OUT / name).open("x") as f:
        for r in records:
            f.write(json.dumps(r, ensure_ascii=False) + "\n")

def normalize(text):
    # Preserve original character offsets through per-character normalization.
    result, positions = [], []
    for i, c in enumerate(text):
        for folded in unicodedata.normalize("NFKD", c.casefold()):
            if not unicodedata.combining(folded):
                result.append(folded)
                positions.append(i)
    return "".join(result), positions

def original_span(positions, start, end):
    return positions[start], positions[end - 1] + 1

def context_span(text, start, end):
    left, right = max(0, start - 180), min(len(text), end + 180)
    before = list(re.finditer(r"[.;!?\n]", text[left:start]))
    after = re.search(r"[.;!?\n]", text[end:right])
    if before:
        left += before[-1].end()
    if after:
        right = end + after.end()
    return left, right

def all_occurrences(text, quote):
    result = []
    if not quote:
        return result
    start = 0
    while True:
        offset = text.find(quote, start)
        if offset < 0:
            return result
        result.append((offset, offset + len(quote)))
        start = offset + 1

def overlaps(a, b):
    return a[0] < b[1] and b[0] < a[1]
