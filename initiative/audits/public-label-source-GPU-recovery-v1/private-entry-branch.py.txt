    if globals().get("PUBLIC_LABEL_SOURCE_RECOVERY_PROBE") is True:
        sources, hashes = MEMORY_SOURCES, MEMORY_HASHES
        if set(sources) != set(hashes) or sum(len(v.encode()) for v in sources.values()) > 1024**2:

        for name, source in sources.items():
            relative = Path(name)

            compile(source, name, "exec")
            path = code / relative

            with path.open("x") as handle:
