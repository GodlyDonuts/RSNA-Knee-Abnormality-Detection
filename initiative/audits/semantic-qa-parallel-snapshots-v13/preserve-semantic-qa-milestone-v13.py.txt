"""Archive fixed completed-stage cutoffs; preserve all original bytes and old baselines."""
from pathlib import Path
from datetime import datetime,timezone
from collections import defaultdict
import hashlib,json,shutil
from dotenv import dotenv_values
W=Path(__file__).resolve().parent.parent;T=W/'work';S=T/'annotation-staging';R=T/'recovered'
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
config=T/'semantic-qa-v13-snapshot-pins.json'
assert sha(config)=='3f2a27ee5d2355a848696d2442815cd5ff274873f6d2d4e5189f913ab59d8e07'
snapshots=json.loads(config.read_text())['snapshots']
peer=S/'cartilage003-integration-independent-verification-v1';pm=peer/'completion-manifest.json'
assert sha(pm)=='b94625a6250550ae94b254f6fbc5ae050589ddcdeab65857aca7cafd8dba64f2'
for ref in json.loads(pm.read_text())['files']:
    p=W/ref['path'];assert sha(p)==ref['sha256']
snapshots.append({'directory':peer.name,'completion_anchors':[{'path':pm.name,'sha256':sha(pm)}],'files':[{'path':str(p.relative_to(peer)),'sha256':sha(p),'bytes':p.stat().st_size} for p in sorted(peer.rglob('*')) if p.is_file() and '__pycache__' not in p.parts]})
entries={};anchors=[]
def verify_local_refs(value,folder):
    if isinstance(value,dict):
        if isinstance(value.get('path'),str) and isinstance(value.get('sha256'),str):
            rel=value['path'];q=W/rel if rel.startswith('work/') else folder/rel
            if q.is_relative_to(folder) and q.is_file():assert sha(q)==value['sha256'],q
        for v in value.values():verify_local_refs(v,folder)
    elif isinstance(value,list):
        for v in value:verify_local_refs(v,folder)
for item in snapshots:
    folder=S/item['directory'];assert not (R/'initiative/audits'/folder.name).exists()
    for anchor in item['completion_anchors']:
        p=folder/anchor['path'];assert sha(p)==anchor['sha256'];verify_local_refs(json.loads(p.read_text()),folder);anchors.append({'path':str(p.relative_to(W)),'sha256':sha(p)})
    for row in item['files']:
        p=folder/row['path'];assert p.is_file() and not p.is_symlink() and sha(p)==row['sha256'] and p.stat().st_size==row['bytes'];entries[p]=row
secrets=[v.encode() for k,v in dotenv_values(W/'.env').items() if k in {'HF_TOKEN','KAGGLE_API_TOKEN'} and v];assert len(secrets)==2
total=sum(x['bytes'] for x in entries.values());assert shutil.disk_usage(W).free-total>20*2**30+32*2**20
receipts=defaultdict(list)
for p,row in sorted(entries.items()):
    rel=p.relative_to(S);dest=R/'initiative/audits'/rel
    if dest.suffix=='.py':dest=dest.with_suffix('.py.txt')
    raw=p.read_bytes();assert hashlib.sha256(raw).hexdigest()==row['sha256'] and all(s not in raw for s in secrets)
    dest.parent.mkdir(parents=True,exist_ok=True)
    with dest.open('xb') as f:f.write(raw)
    assert sha(dest)==sha(p)==row['sha256']
    receipts[rel.parts[0]].append({'original_path':str(p.relative_to(W)),'archived_path':str(dest.relative_to(R)),'sha256':sha(dest),'bytes':len(raw)})
for name,files in receipts.items():
    out=R/'initiative/audits'/name/'root-snapshot-preservation-v13.json'
    with out.open('x') as f:json.dump({'agent_id':'/root','at_utc':datetime.now(timezone.utc).isoformat(),'files':files,'all_original_bytes_preserved':True,'root_opened_clinical_findings_for_copy':False,'scope':'Completed partial review evidence and technical proofs, not new clinical acceptance or global QA closure. Original286 and older baselines remain historical; future integration must rebase against actual333 or newer. Copied batch-label mistakes and their explicit correction artifacts both remain preserved.','training_ready':False},f,indent=2);f.write('\n')
out=R/'initiative/audits/semantic-qa-parallel-snapshots-v13';out.mkdir()
(out/'preserve-semantic-qa-milestone-v13.py.txt').write_bytes(Path(__file__).read_bytes());(out/config.name).write_bytes(config.read_bytes())
proof={'agent_id':'/root','at_utc':datetime.now(timezone.utc).isoformat(),'directories':{n:len(v) for n,v in sorted(receipts.items())},'completion_anchors':anchors,'source_files':len(entries),'bytes':total,'training_ready':False}
(out/'snapshots.json').write_text(json.dumps(proof,indent=2)+'\n')
print(json.dumps({'directories':len(receipts),'files':len(entries),'bytes':total,'snapshot_sha256':sha(out/'snapshots.json')}))
