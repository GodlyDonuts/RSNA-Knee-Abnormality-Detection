def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


entries = {}


def add(path, expected=None, size=None):
    path = path.resolve()
    assert path.is_relative_to(W) and path.is_file() and not path.is_symlink()
    digest = sha(path)
    assert expected is None or digest == expected
    assert size is None or size == path.stat().st_size
    dest = None
    if path.is_relative_to(B):
        dest_path = Path('initiative/audits') / path.relative_to(B)
        if dest_path.suffix == '.py':
            dest_path = dest_path.with_suffix('.py.txt')
        dest = str(dest_path)
    elif path == W / 'work/qualified004-root-decisions-v1.json':
        assert digest == '5204569b3e14329b69e5136b32132bdf04ee24f6b0b0308b12b0044468ed7bad'
        dest = 'initiative/audits/qualified-negation-focused-004-root-adjudication-v1/qualified004-root-decisions-v1.json'
    else:
        assert path.is_relative_to(R)
    item = {'path': str(path.relative_to(W)), 'sha256': digest, 'bytes': path.stat().st_size, 'mode': stat.S_IMODE(path.stat().st_mode), 'archive_path': dest}
    assert entries.setdefault(item['path'], item) == item
    return item
