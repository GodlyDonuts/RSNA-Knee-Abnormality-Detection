"""Preserve experimental Effusion epochs privately with the original locked uploader."""

from __future__ import annotations

import argparse
import importlib.util
import json
from pathlib import Path

from scripts.newton.run_effusion_screen import CONTROL, OVERLAY_SHA, RECOVERY, sha

ORIGINAL = Path(
    "/home/sa305415/rsna-knee/production-launch-native384-a32-c9-v1/publish_native384_epoch.py"
)
ORIGINAL_SHA = "4e327e271816e4a4055b4f3ad28a8dab59fc650a1367fd13bbd7fe06d9a5efde"
PREFIX = "experiments/dinov3-v1/training-epochs/effusion-reviewed31-fold0-665c0d5-seed20261004-screen3-v1"


def validate_binding(binding, record, expected, role, source):
    enabled = role == "candidate"
    policy = {"kind": "effusion_review_v1", "enabled": enabled, "sha256": OVERLAY_SHA}
    code = {
        name.removeprefix("rsna/"): row["sha256"]
        for name, row in source["files"].items()
        if name.startswith("rsna/") and name.endswith(".py")
    }
    expected["fitting_supervision"] = policy
    config = binding["config"]
    # Additional optional mechanisms must remain inactive, as in the accepted recovery.
    config = {k: v for k, v in config.items() if k not in ("relation", "witness") or v is not None}
    supervision = binding.get("experimental_fitting_supervision", {})
    if (
        config != expected
        or binding.get("code_sha256") != code
        or (record.get("training_studies"), record.get("outer_studies")) != (2247, 1156)
        or supervision.get("overlay_sha256") != OVERLAY_SHA
        or supervision.get("enabled") is not enabled
        or supervision.get("applied_cells") != (31 if enabled else 0)
        or supervision.get("evaluation_masks_changed") is not False
        or supervision.get("experimental_only") is not True
    ):
        raise ValueError(
            "Exact admitted numerical recipe, roster and isolated Effusion arm required"
        )


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("mode", choices=("publish", "reconcile"))
    parser.add_argument("role", choices=("control", "candidate"))
    parser.add_argument("epoch", type=int, choices=(1, 2, 3))
    args = parser.parse_args()
    if sha(ORIGINAL) != ORIGINAL_SHA:
        raise ValueError("Original atomic private uploader changed")
    spec = importlib.util.spec_from_file_location("original_effusion_locked_upload", ORIGINAL)
    original = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(original)
    original.PREFIX = PREFIX + "/" + args.role
    original.TRAINING_GIT = "665c0d515410df24f1f37ed5d9f8cb7a0f0fc480"

    def source_files(snapshot, run, license_file, code_root):
        meta = json.loads(snapshot.read_text())
        record = json.loads((run / "run.json").read_text())
        binding = record["binding"]
        expected = original.expected_training_config()
        expected["epochs"] = 3
        expected["seed"] = 20261004
        source = json.loads((RECOVERY / "source.json").read_text())
        validate_binding(binding, record, expected, args.role, source)
        if (
            {k: binding.get(k) for k in original.EXPECTED_BINDING_FIELDS}
            != original.EXPECTED_BINDING_FIELDS
            or binding["release"]["binding_sha256"] != original.RELEASE_SHA
            or original.sha(license_file) != original.LICENSE_SHA
            or meta.get("status") != "scheduled_epoch_snapshot"
            or meta.get("epoch") != args.epoch
            or meta.get("seed") != 20261004
            or meta.get("optimizer_steps") != args.epoch * 281
            or meta.get("binding_sha256")
            != __import__("hashlib").sha256(original.canonical(binding).encode()).hexdigest()
            or meta.get("checkpoint_file") != f"epoch-{args.epoch:03d}.pt"
        ):
            raise ValueError(
                "Exact immutable completed epoch and original fold/base/license required"
            )
        checkpoint = snapshot.with_suffix(".pt")
        if (
            checkpoint.is_symlink()
            or checkpoint.stat().st_size != meta.get("checkpoint_bytes")
            or checkpoint.stat().st_size >= 128 * 2**20
            or sha(checkpoint) != meta.get("checkpoint_sha256")
        ):
            raise ValueError("Original immutable epoch bytes differ")
        # No original proposal, case-level reviews or raw identifiers enter HF.
        proof = CONTROL / "source-provenance.json"
        files = {
            "checkpoint.pt": checkpoint,
            "snapshot.json": snapshot,
            "source-provenance.json": proof,
            "DINOv3-LICENSE.md": license_file,
        }
        manifest = {
            "schema_version": 1,
            "status": "completed_training_epoch",
            "epoch": args.epoch,
            "role": args.role,
            "optimizer_steps": meta["optimizer_steps"],
            "checkpoint_sha256": meta["checkpoint_sha256"],
            "checkpoint_bytes": meta["checkpoint_bytes"],
            "training_git_commit": original.TRAINING_GIT,
            "binding_sha256": meta["binding_sha256"],
            "release_binding_sha256": original.RELEASE_SHA,
            "encoder": binding["config"]["encoder"],
            "config": binding["config"],
            "license_sha256": original.LICENSE_SHA,
            "license_source": original.LICENSE_SOURCE,
            "experimental_fitting_supervision": True,
            "competition_eligible": False,
            "protected_evaluation_used": False,
            "clinical_accuracy_claimed": False,
            "files": {n: {"bytes": p.stat().st_size, "sha256": sha(p)} for n, p in files.items()},
        }
        return manifest, files

    original.source_files = source_files
    run = CONTROL / args.role
    options = argparse.Namespace(
        mode=args.mode,
        snapshot=run / "epoch-checkpoints" / f"epoch-{args.epoch:03d}.json",
        run=run,
        journal=CONTROL / "epoch-publication" / args.role,
        code_root=CONTROL / "code",
        license_file=Path(
            "/home/sa305415/rsna-knee/encoder-stage/configs/competitive/weights/dinov3-large.LICENSE.md"
        ),
    )
    print(json.dumps(original.run(options), sort_keys=True), flush=True)


if __name__ == "__main__":
    main()
