checked = {}

def digest(data):
    return hashlib.sha256(data).hexdigest()

def sha(path):
    h = hashlib.sha256()
    with path.open('rb') as stream:
        for chunk in iter(lambda: stream.read(2**20), b''):
            h.update(chunk)
    return h.hexdigest()

def bind(path):
    return {'path': str(path.relative_to(W)), 'sha256': sha(path), 'bytes': path.stat().st_size}

def read(path, expected=None):
    if expected is not None:
        assert sha(path) == expected, ('hash', str(path.relative_to(W)))
    checked[str(path.relative_to(W))] = bind(path)
    return json.loads(path.read_text())

def locate(name):
    path = Path(name)
    if path.is_absolute():
        assert path.is_relative_to(W)
        return path
    return (W if name.startswith('work/') else R) / path

def refcheck(ref):
    path = locate(ref['path'])
    assert path.is_file() and not path.is_symlink(), ('file', ref['path'])
    assert sha(path) == ref['sha256'], ('refhash', ref['path'])
    if 'bytes' in ref:
        assert path.stat().st_size == ref['bytes'], ('size', ref['path'])
    checked[str(path.relative_to(W))] = bind(path)
    return path

def archive_peer(path, expected):
    staging = W / 'work/annotation-staging'
    if path.is_relative_to(staging):
        peer = A / path.relative_to(staging)
        if peer.suffix == '.py':
            peer = peer.with_suffix('.py.txt')
        assert peer.is_file() and sha(peer) == expected, ('archived_peer', str(peer.relative_to(W)))
        checked[str(peer.relative_to(W))] = bind(peer)

def key(record):
    return record['study_uid'], record['target']

def keyed(rows):
    result = {key(row): row for row in rows}
    assert len(result) == len(rows)
    return result

def save(name, obj):
    with (S / name).open('x') as stream:
        stream.write(json.dumps(obj, ensure_ascii=False, indent=2) + '\n')
