"""Package explicit personally authored judgments; no automated clinical inference."""
import hashlib
import json
import runpy
from collections import Counter
from datetime import datetime, timezone
from pathlib import Path

S = Path(__file__).resolve().parent
W = S.parents[2]
R = W / 'work/recovered'
AGENT = '/root/last_twelve_visual_review'
TARGETS = ['Medial OA', 'Lateral OA', 'PF OA']


def obj(p):
    return json.loads(p.read_bytes())


def ref(p):
    b = p.read_bytes()
    return {'path': str(p.relative_to(W)), 'sha256': hashlib.sha256(b).hexdigest(), 'bytes': len(b)}


def write(name, data):
    p = S / name
    with p.open('x') as f:
        json.dump(data, f, ensure_ascii=False, indent=2)
        f.write('\n')
    return p


packet = obj(S / 'neutral-source-packet.json')
binding = obj(S / 'source-bindings-and-exposure.json')
assert ref(S / 'neutral-source-packet.json')['sha256'] == 'e859dec3115721fedaf8888e30487d9fe685a76e05abe73c01fc17434f1cf523'
assert ref(R / 'protocols/report_weak_v2.md')['sha256'] == '138c5cc33b92901d57dbf5fbe7e3a9878b73d25b0af6934908d0052cd6786014'
assert not binding['exact_neutral_packet_overlaps']
rows = {}
helpers = sorted(S.glob('decisions_*.py'))
for p in helpers:
    d = runpy.run_path(str(p))['ROWS']
    assert not rows.keys() & d.keys()
    rows.update(d)
assert set(rows) == set(range(50))
out = []
checks = []
for i, p in enumerate(packet):
    assert len(rows[i]) == 3
    source = R / p['source_path']
    assert ref(source)['sha256'] == p['source_sha256']
    assert obj(source)[p['index']] == {'study_uid': p['study_uid'], 'report': p['report']}
    b = binding['bindings'][i]
    assert b['study_uid'] == p['study_uid'] and b['priority_position_zero_based'] == 150 + i
    assert b['report_utf8_sha256'] == hashlib.sha256(p['report'].encode()).hexdigest()
    for t, (state, quote, reason) in enumerate(rows[i]):
        assert state in ['positive', 'negative', 'uncertain', 'not_mentioned']
        assert bool(quote) == (state != 'not_mentioned') and quote in p['report']
        assert isinstance(reason, str) and reason.strip() and 'PLACEHOLDER' not in reason
        checks.append({'packet_index': i, 'study_uid': p['study_uid'], 'target': TARGETS[t], 'quote_start_character': p['report'].find(quote) if quote else None, 'quote_characters': len(quote), 'quote_utf8_sha256': hashlib.sha256(quote.encode()).hexdigest() if quote else None, 'exact_contiguous_or_missing_empty': True})
    out.append({'agent_id': AGENT, 'packet_index': i, 'priority_position_zero_based': 150 + i, 'case': f"{p['batch']}:{p['index']}", 'study_uid': p['study_uid'], 'source_path': p['source_path'], 'source_sha256': p['source_sha256'], 'source_index_zero_based': p['index'], 'report_utf8_sha256': b['report_utf8_sha256'], 'target_order': TARGETS, 'states': [x[0] for x in rows[i]], 'evidence': [x[1] for x in rows[i]], 'target_reasoning': [x[2] for x in rows[i]], 'prior_exact_packet_overlap': [], 'notes': 'Personally authored full-source OA-only pass with historical context disclosed. No first cartilage003 findings or fresh current003 annotations opened before freeze. Unintegrated source assessment only.'})
assert len(out) == 50 and len({x['study_uid'] for x in out}) == 50 and len(checks) == 150
assessment = S / 'independent-assessment.jsonl'
with assessment.open('x') as f:
    for r in out:
        f.write(json.dumps(r, ensure_ascii=False) + '\n')

    'exact_overlap_scope': 'Zero exact UID overlap with the five explicitly checked prior neutral source packets: focused001, focused002, cartilage000, cartilage001 and cartilage002. Those five packets are individually bound in preparation. This does not claim every earlier technical inventory or historical contextual exposure is exhaustively absent.',
    'access_in_this_pass': 'Only neutral cartilage003 source content, neutral priority150–199 source packets, prior neutral UID membership, own earlier mechanical packaging helpers and own newly authored decisions were read for this pass. manifest = write('final-file-list.json', {'agent_id': AGENT, 'files': [ref(p) for p in sorted(S.rglob('*')) if p.is_file()], 'self_excluded': True})

assert {x['path'] for x in items} == {str(p.relative_to(W)) for p in S.rglob('*') if p.is_file() and p != manifest}

    assert ref(W / item['path']) == item
