"""Allocated remote exact-commit pin; errors preserve checkout for reconciliation."""
import hashlib
import json
import os
import signal
import subprocess
from datetime import datetime, timezone
from pathlib import Path, PurePosixPath

control = Path(__file__).resolve().parent
intent = json.loads((control / 'intent.json').read_text())
proof = json.loads((control / 'previous-checkout-proof.json').read_text())
auth = control / 'auth.token'
target = Path(intent['target'])
receipt = {
    'schema_version': 1, 'operation_id': intent['operation_id'],
    'actual_executor': '/root/image_pipeline_audit', 'job_id': os.environ.get('SLURM_JOB_ID'),
    'requested_cpus': 2, 'requested_gpus': 0,
    'allocated_gpu_ids_present': bool(os.environ.get('SLURM_JOB_GPUS', '')),
    'started_at_utc': datetime.now(timezone.utc).isoformat(), 'status': 'started',
    'previous_checkout_receipt_sha256': proof['sha256'],
    'intent_sha256': hashlib.sha256((control / 'intent.json').read_bytes()).hexdigest(),
    'runner_sha256': hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
}
code = 1
signals = (signal.SIGTERM, signal.SIGINT, signal.SIGUSR1)

def interrupted(signum, frame):
    raise InterruptedError('source pin interrupted')

for sig in signals:
    signal.signal(sig, interrupted)

def run_owned(command, env, timeout):
    process = subprocess.Popen(command, env=env, stdout=subprocess.DEVNULL,
                               stderr=subprocess.DEVNULL, start_new_session=True)
    try:
        return process.wait(timeout=timeout)
    finally:
        for sig in signals:
            signal.signal(sig, signal.SIG_IGN)
        try:
            os.killpg(process.pid, signal.SIGTERM)
        except ProcessLookupError:
            pass
        try:
            process.wait(timeout=10)
        except subprocess.TimeoutExpired:
            pass
        try:
            os.killpg(process.pid, signal.SIGKILL)
        except ProcessLookupError:
            pass
        process.wait()
        for sig in signals:
            signal.signal(sig, interrupted)

try:
    if receipt['allocated_gpu_ids_present']:
        raise ValueError('unexpected_gpu_allocation')
    if not str(receipt['job_id']).isdigit() or int(os.environ.get('SLURM_CPUS_PER_TASK', '0')) != 2:
        raise ValueError('expected_cpu_allocation_required')
    old_receipt = Path(proof['path'])
    prior = old_receipt.read_bytes()
    if len(prior) != proof['bytes'] or hashlib.sha256(prior).hexdigest() != proof['sha256']:
        raise ValueError('changed_previous_checkout_receipt')
    if (old_receipt.parent / 'auth.token').exists():
        raise ValueError('old_authorization_not_removed')
    parsed = json.loads(prior)
    if parsed['status'] != 'verified_exact_detached_checkout' or parsed['commit'] != intent['expected_previous_commit'] or not parsed['clean'] or not parsed['auth_file_removed']:
        raise ValueError('previous_checkout_not_verified')
    for p in (target, target.parent, control):
        if p.is_symlink() or not p.is_dir() or p.stat().st_mode & 0o777 != 0o700:
            raise ValueError('private_regular_directory_required')
    if auth.is_symlink() or not auth.is_file() or auth.stat().st_mode & 0o777 != 0o600:
        raise ValueError('private_auth_file_required')
    env = dict(os.environ, GIT_TERMINAL_PROMPT='0', GIT_ASKPASS=str(control / 'askpass.py'),
               RSNA_GIT_AUTH_FILE=str(auth), GIT_CONFIG_NOSYSTEM='1', GIT_CONFIG_GLOBAL='/dev/null',
               GIT_TRACE='0', GIT_CURL_VERBOSE='0', GIT_TRACE_CURL='0')
    def git(*args):
        return subprocess.check_output(['git', '-c', 'core.preloadindex=false', '-c', 'index.threads=1', '-C', str(target), *args], env=env,
                                       stderr=subprocess.DEVNULL, timeout=60)
    if git('remote', 'get-url', 'origin').decode().strip() != intent['source_url']:
        raise ValueError('unexpected_git_origin')
    if git('status', '--porcelain=v1', '--untracked-files=all'):
        raise ValueError('unexpected_dirty_checkout')
    before = git('rev-parse', 'HEAD').decode().strip()
    receipt['previous_commit'] = before
    if before not in (intent['expected_previous_commit'], intent['requested_commit']):
        raise ValueError('unexpected_previous_commit')
    if before != intent['requested_commit']:
        result = run_owned(['git', '-c', 'core.preloadindex=false', '-c', 'index.threads=1', '-C', str(target), '-c', 'pack.threads=2', '-c', 'index.threads=2',
                            'fetch', '--quiet', '--no-tags', '--depth', '1', 'origin', intent['requested_commit']], env, 600)
        receipt['fetch_returncode'] = result
        if result:
            raise RuntimeError('fetch_failed')
    result = run_owned(['git', '-c', 'core.preloadindex=false', '-c', 'index.threads=1', '-C', str(target), 'checkout', '--quiet', '--detach', intent['requested_commit']], env, 120)
    receipt['checkout_returncode'] = result
    if result:
        raise RuntimeError('checkout_failed')
    head = git('rev-parse', 'HEAD').decode().strip()
    tree = git('rev-parse', 'HEAD^{tree}').decode().strip()
    if head != intent['requested_commit'] or tree != intent['requested_tree']:
        raise ValueError('commit_or_tree_mismatch')
    symbolic = subprocess.run(['git', '-c', 'core.preloadindex=false', '-c', 'index.threads=1', '-C', str(target), 'symbolic-ref', '-q', 'HEAD'], env=env,
                              capture_output=True, timeout=30)
    if symbolic.returncode != 1 or symbolic.stdout:
        raise ValueError('detached_checkout_required')
    if git('status', '--porcelain=v1', '--untracked-files=all'):
        raise ValueError('checkout_not_clean')
    if (target / '.env').exists() or git('ls-files', '--', '.env'):
        raise ValueError('unexpected_environment_secret_file')
    rows = [row for row in git('ls-tree', '-r', '-l', '-z', 'HEAD').split(b'\0') if row]
    if any(row.split(b'\t', 1)[0].split()[1] != b'blob' for row in rows):
        raise ValueError('unexpected_nonblob_tree_entry')
    size = sum(int(row.split(b'\t', 1)[0].split()[3]) for row in rows)
    if len(rows) != intent['tracked_files'] or size != intent['tracked_blob_bytes']:
        raise ValueError('full_tree_inventory_mismatch')
    seen = set()
    verified = []
    for expected in intent['files']:
        rel = PurePosixPath(expected['path'])
        if rel.is_absolute() or '..' in rel.parts or str(rel) in seen:
            raise ValueError('unsafe_or_duplicate_source_path')
        seen.add(str(rel))
        p = target / str(rel)
        if p.is_symlink() or not p.is_file() or p.resolve().parent != (target / rel.parent).resolve():
            raise ValueError('regular_source_file_required')
        if not p.resolve().is_relative_to(target.resolve()):
            raise ValueError('source_path_escaped')
        data = p.read_bytes()
        got = {'path': str(rel), 'bytes': len(data), 'sha256': hashlib.sha256(data).hexdigest()}
        if got != expected:
            raise ValueError('source_content_mismatch')
        verified.append(got)
    if len(verified) != intent['source_hash_count']:
        raise ValueError('source_count_mismatch')
    count = total = 0
    for parent, dirs, names in os.walk(target):
        if Path(parent) == target:
            dirs[:] = [name for name in dirs if name != '.git']
        for name in names:
            count += 1
            total += (Path(parent) / name).lstat().st_size
    if count != len(rows) or total != size:
        raise ValueError('worktree_count_or_size_mismatch')
    receipt.update(status='verified_exact_detached_checkout', commit=head, tree_sha=tree,
                   detached=True, clean=True, tracked_files=len(rows), tracked_blob_bytes=size,
                   worktree_files=count, worktree_bytes=total, verified_source_hash_count=len(verified),
                   verified_sources=verified, no_env_copied=True)
    code = 0
except BaseException as error:
    receipt.update(status='failed_checkout_preserved', error_type=type(error).__name__)
finally:
    for sig in signals:
        signal.signal(sig, signal.SIG_IGN)
    auth.unlink(missing_ok=True)
    receipt.update(auth_file_removed=not auth.exists(), completed_at_utc=datetime.now(timezone.utc).isoformat())
    with (control / 'receipt.json').open('x') as handle:
        json.dump(receipt, handle, indent=2, sort_keys=True)
        handle.write('\n')
    print(json.dumps({k: v for k, v in receipt.items() if k != 'verified_sources'}), flush=True)
raise SystemExit(code)
