"""Three source-bound cached representative views for gross body-region review."""

from __future__ import annotations

import hashlib
import json
from pathlib import Path

import numpy as np
from collect_technical import ROOT, WORK, ref, require, save, sha
from PIL import Image, ImageDraw, ImageFont


def main() -> None:
    output = ROOT / "cache-panels"
    require(not output.exists(), "Panel output must be new")
    summary_path = ROOT / "technical-evidence/summary.json"
    require(
        sha(summary_path) == "acc7c23d21f7eb43889d9756aa7cf6ca70b7096a7792bcf408aefb6b9dba4003",
        "Technical evidence changed",
    )
    summary = json.loads(summary_path.read_text())
    for item in summary["cache_metadata_bindings"] + summary["evidence_bindings"]:
        require(sha(WORK / item["path"]) == item["sha256"], "Cache/source evidence changed")
    case = json.loads((ROOT / "technical-evidence/case-052-20.json").read_text())
    inventory = json.loads((ROOT / "technical-evidence/inventory-one-cases.json").read_text())
    source_by_sop = {r["sop_uid"]: r for r in inventory}
    selected_numbers = [201, 401, 601]
    canvas = Image.new("L", (832, 338), 16)
    draw = ImageDraw.Draw(canvas)
    font = ImageFont.load_default(size=14)
    draw.text(
        (12, 8),
        "052:20 | verified cached mid-slices | body-region review only",
        font=font,
        fill=255,
    )
    panels = []
    for column, number in enumerate(selected_numbers):
        series = next(
            r
            for r in case["series"]
            if r["header_endpoint_values"]["SeriesNumber"] == [str(number)]
        )
        metadata_path = WORK / series["cache_metadata"]["path"]
        meta = json.loads(metadata_path.read_text())
        path = metadata_path.with_suffix(".npz")
        require(
            sha(path) == meta["cache_sha256"] and path.stat().st_size == meta["cache_bytes"],
            "Full NPZ bytes differ",
        )
        with np.load(path, allow_pickle=False) as archive:
            pixels = archive["volume"]
        require(
            pixels.dtype == np.uint16 and list(pixels.shape) == [meta["n_slices"], 256, 256],
            "Cache array contract differs",
        )
        index = meta["n_slices"] // 2
        display = np.rint(pixels[index].astype(float) / 65535 * 255).astype(np.uint8)
        x = 12 + column * 276
        canvas.paste(Image.fromarray(display), (x, 36))
        draw.text(
            (x, 296), f"Series {number} | index {index}/{meta['n_slices'] - 1}", font=font, fill=255
        )
        draw.text((x, 315), "Cache 256x256, unchanged placement", font=font, fill=220)
        panels.append(
            {
                "series_uid": meta["series_uid"],
                "study_uid": meta["study_uid"],
                "series_number": number,
                "source": source_by_sop[meta["sop_uids"][index]],
                "cache": ref(path),
                "cache_metadata": ref(metadata_path),
                "ordered_index": index,
                "cached_slice_bytes_sha256": hashlib.sha256(pixels[index].tobytes()).hexdigest(),
                "native_pixel_sha256_from_bound_cache_metadata": meta["pixel_sha256"][index],
                "native_shape": meta["original_shape"][1:],
                "cache_shape": [256, 256],
                "source_position": meta["positions"][index],
                "source_orientation": meta["orientation"],
                "source_pixel_spacing": meta["pixel_spacing"],
                "display": "cached normalized/resized uint16 mapped linearly to uint8; no additional resize, crop, rotation or flip",
            }
        )
    output.mkdir()
    image = output / "three-verified-cache-mid-slices.png"
    canvas.save(image)
    save(
        output / "panel-manifest.json",
        {
            "schema_version": 1,
            "stage": "bound_cached_representative_body_region_panels",
            "agent_id": "/root/image_pipeline_audit",
            "case": "052:20",
            "script": ref(Path(__file__).resolve()),
            "technical_summary": ref(summary_path),
            "image": ref(image),
            "panels": panels,
            "scope": "Three selected middle slices from three fully hash-verified NPZ files. Images are existing256px normalized cache, not original native resolution. All source UID/hash/geometry references preserved from strict full decode; no new source download or native DICOM decode. Body-region review only, no clinical label inference.",
        },
    )
    print(
        json.dumps({"manifest": ref(output / "panel-manifest.json"), "image": ref(image)}, indent=2)
    )


if __name__ == "__main__":
    main()
