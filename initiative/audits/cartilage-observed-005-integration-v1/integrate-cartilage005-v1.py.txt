class GuardError(Exception):
    pass

def require(value, code):
    if not value:
        raise GuardError(code)

def sha(path):
    digest = hashlib.sha256()
    with path.open('rb') as stream:
        for chunk in iter(lambda: stream.read(2**20), b''):
            digest.update(chunk)
    return digest.hexdigest()

def encoded(document):
    return (json.dumps(document, ensure_ascii=False, indent=2) + '\n').encode()

def read(path, pin=None):
    if pin is not None:
        require(re.fullmatch('[0-9a-f]{64}', pin) is not None, 'invalid_required_pin')
        require(sha(path) == pin, 'input_hash_mismatch')
    return json.loads(path.read_text())

def write_new(path, data):
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open('xb') as stream:
        stream.write(data)
        stream.flush()
        os.fsync(stream.fileno())

def archive_state(path, data):
    require(not path.is_symlink(), 'archive_symlink')
    if path.exists():
        require(path.is_file() and path.read_bytes() == data, 'archive_existing_bytes_differ')
        return 'reused_existing_equal'
    return 'new_file'

def preserve_bytes(path, data):
    state = archive_state(path, data)
    if state == 'new_file':
        write_new(path, data)
    require(path.read_bytes() == data, 'archive_write_verification')
    return state

def unique(rows):
    result = {}
    for row in rows:
        key = row['study_uid'], row['target']
        require(key not in result, 'duplicate_target_key')
        result[key] = row
    return result

def pin_path(workspace, reference):
    require(isinstance(reference, dict) and {'path', 'sha256'} <= reference.keys(), 'missing_reference')
    name = reference['path']
    require(isinstance(name, str) and not Path(name).is_absolute(), 'absolute_reference')
    path = (workspace / name).resolve()
    require(path.is_relative_to(workspace), 'reference_escapes_workspace')
    require(sha(path) == reference['sha256'], 'reference_hash_mismatch')
    size = reference.get('bytes', reference.get('size_bytes'))
    if size is not None:
        require(path.stat().st_size == size, 'reference_size_mismatch')
    return path
