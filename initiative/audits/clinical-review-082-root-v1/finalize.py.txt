def sha(p):
    return hashlib.sha256(p.read_bytes()).hexdigest()

def save(p, obj):
    assert not p.exists(), p
    p.write_text(json.dumps(obj, ensure_ascii=False, indent=2) + "\n")

state_names = {'u': 'uncertain', 'n': 'negative', 'x': 'not_mentioned'}

def choose(i, t, state, reason, quote_from='pre'):
    assert (i, t) not in choices
    assert pre[i]['states'][t] != draft[i]['states'][t]
    name = state_names[state]
    quote = '' if state == 'x' else (pre if quote_from == 'pre' else draft)[i]['evidence'][t]
    assert state == 'x' or quote and quote in source[i]['report']
    review[i]['states'][t] = name
    review[i]['evidence'][t] = quote
    choices[i, t] = {'source_index': i, 'study_uid': source[i]['study_uid'],
        'target_index': t, 'target': TARGETS[t], 'preread_state': pre[i]['states'][t],
        'draft_state': draft[i]['states'][t], 'final_state': name,
        'preread_evidence': pre[i]['evidence'][t], 'draft_evidence': draft[i]['evidence'][t],
        'final_evidence': quote, 'reason': reason}
