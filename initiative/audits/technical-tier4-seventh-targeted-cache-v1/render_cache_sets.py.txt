"""Source-bound representative cache sets for literal-side tag consistency review.

No network or fresh native decoding. No spatial display transform is applied.
"""

from __future__ import annotations

import hashlib
import json
import shutil
from pathlib import Path

import numpy as np
from collect_technical import ROOT, WORK, ref, require, save, sha
from PIL import Image, ImageDraw, ImageFont


def main() -> None:
    require(shutil.disk_usage(ROOT).free > 20 * 2**30, "20 GiB reserve unavailable")
    output = ROOT / "cache-panels"
    require(not output.exists(), "Panel output must be new")
    summary_path = ROOT / "technical-evidence/summary.json"
    require(
        sha(summary_path)
        == "5d7511a52934d5b259ffc11dab47f7c04590510d5edba2d39a8499236927fb8a",
        "Frozen technical summary changed",
    )
    summary = json.loads(summary_path.read_text())
    require(len(summary["cases"]) == 4, "Exactly four case summaries required")
    for item in summary["cache_metadata_bindings"] + summary["evidence_bindings"]:
        path = WORK / item["path"]
        require(
            sha(path) == item["sha256"] and path.stat().st_size == item["bytes"],
            "Bound cache/source evidence changed",
        )
    inventory_path = ROOT / "technical-evidence/inventory-four-cases.json"
    inventory = json.loads(inventory_path.read_text())
    require(
        len({r["sop_uid"] for r in inventory}) == len(inventory), "Duplicate source SOP"
    )
    source_by_sop = {r["sop_uid"]: r for r in inventory}
    all_case_records = []
    font = ImageFont.load_default(size=13)
    output.mkdir()
    plan_path = ROOT / "selection-plan.json"
    require(sha(plan_path) == "3930dc8d16f72bd395347640b3ae7f403d402283a69eddb99d5feb18a46d5501", "Selection plan changed")
    plan = json.loads(plan_path.read_text())
    require(len(plan["cases"]) == 4 and plan["selected_series"] == 16, "Selection scope differs")
    for choice in plan["cases"]:
        case_name = choice["case"].replace(":", "-")
        case_path = ROOT / f"technical-evidence/case-{case_name}.json"
        case = json.loads(case_path.read_text())
        case_output = output / f"case-{case_name}"
        case_output.mkdir()
        require(case["study_uid"] == choice["study_uid"] and case["series_count"] == choice["whole_study_series"] and case["slices"] == choice["whole_study_source_memberships"], "Whole-study selection identity differs")
        selected_ids = {r["series_uid"] for r in choice["selected_series"]}
        require(len(selected_ids) == len(choice["selected_series"]), "Duplicate selected series")
        selected_series = [r for r in case["series"] if r["series_uid"] in selected_ids]
        require({r["series_uid"] for r in selected_series} == selected_ids, "Selected series missing")
        for item in choice["selected_series"]:
            actual = next(r for r in selected_series if r["series_uid"] == item["series_uid"])
            require(int(actual["header_endpoint_values"]["SeriesNumber"][0]) == item["series_number"] and actual["cache_original_shape"][0] == item["slices"], "Selected number/count differs")
        ordered_series = sorted(
            selected_series,
            key=lambda r: int(r["header_endpoint_values"]["SeriesNumber"][0]),
        )
        panels = []
        images = []
        canvases = []
        for series_index, series in enumerate(ordered_series):
            metadata_path = WORK / series["cache_metadata"]["path"]
            require(
                ref(metadata_path) == series["cache_metadata"], "Case metadata differs"
            )
            meta = json.loads(metadata_path.read_text())
            require(
                meta["study_uid"] == case["study_uid"]
                and meta["series_uid"] == series["series_uid"]
                and meta["source_revision"]
                == "0b5e410288232f570c69ba4f44723356e6ae6ab7"
                and meta["cache_version"] == "series-u16-v3",
                "Cache identity/version differs",
            )
            source_rows = [
                r for r in inventory if r["series_uid"] == meta["series_uid"]
            ]
            require(
                {r["study_uid"] for r in source_rows} == {meta["study_uid"]}
                and dict(zip(meta["sop_uids"], meta["slice_sha256"], strict=True))
                == {r["sop_uid"]: r["sha256"] for r in source_rows}
                and len(source_rows) == len(meta["sop_uids"]) == meta["n_slices"],
                "Complete source membership differs",
            )
            path = metadata_path.with_suffix(".npz")
            require(
                sha(path) == meta["cache_sha256"]
                and path.stat().st_size == meta["cache_bytes"],
                "Full cache NPZ hash/size differs",
            )
            with np.load(path, allow_pickle=False) as archive:
                require(archive.files == ["volume"], "Unexpected cache array keys")
                pixels = archive["volume"]
            require(
                pixels.dtype == np.uint16
                and list(pixels.shape) == [meta["n_slices"], 256, 256],
                "Cache array contract differs",
            )
            number = int(series["header_endpoint_values"]["SeriesNumber"][0])
            side = "/".join(series["header_endpoint_values"]["Laterality"])
            description = "/".join(
                series["header_endpoint_values"]["SeriesDescription"]
            )
            if series_index % 3 == 0:
                rows = min(3, len(ordered_series) - series_index)
                canvas = Image.new("L", (832, 44 + 308 * rows), 16)
                draw = ImageDraw.Draw(canvas)
                draw.text(
                    (12, 8),
                    f"{case_name.replace('-', ':')} | verified 256px cache | 25/50/75% ordered slices",
                    font=font,
                    fill=255,
                )
                canvases.append(canvas)
            canvas = canvases[-1]
            draw = ImageDraw.Draw(canvas)
            row = series_index % 3
            y = 44 + row * 308
            draw.text(
                (12, y),
                f"Series {number} | literal side {side} | {description}",
                font=font,
                fill=255,
            )
            indices = [
                int((meta["n_slices"] - 1) * fraction) for fraction in [0.25, 0.5, 0.75]
            ]
            require(len(set(indices)) == 3, "Three distinct sample indices required")
            for column, index in enumerate(indices):
                display = np.rint(pixels[index].astype(float) / 65535 * 255).astype(
                    np.uint8
                )
                x = 12 + column * 276
                canvas.paste(Image.fromarray(display), (x, y + 22))
                draw.text(
                    (x, y + 282),
                    f"index {index}/{meta['n_slices'] - 1}",
                    font=font,
                    fill=230,
                )
                panels.append(
                    {
                        "case": case_name.replace("-", ":"),
                        "study_uid": meta["study_uid"],
                        "series_uid": meta["series_uid"],
                        "series_number": number,
                        "literal_laterality": side,
                        "series_description": description,
                        "official_plane": series["official_series_features"][
                            "Anatomical_Plane"
                        ],
                        "source": source_by_sop[meta["sop_uids"][index]],
                        "cache": ref(path),
                        "cache_metadata": ref(metadata_path),
                        "ordered_index": index,
                        "cached_slice_bytes_sha256": hashlib.sha256(
                            pixels[index].tobytes()
                        ).hexdigest(),
                        "display_pixel_bytes_sha256": hashlib.sha256(
                            display.tobytes()
                        ).hexdigest(),
                        "source_native_pixel_sha256_from_bound_metadata": meta[
                            "pixel_sha256"
                        ][index],
                        "original_native_shape": meta["original_shape"][1:],
                        "cached_shape": [256, 256],
                        "source_position": meta["positions"][index],
                        "source_orientation": meta["orientation"],
                        "source_pixel_spacing": meta["pixel_spacing"],
                        "page": len(canvases) - 1,
                        "display_box_xywh": [x, y + 22, 256, 256],
                    }
                )
        for i, canvas in enumerate(canvases):
            image = case_output / f"cache-series-page-{i:02}.png"
            canvas.save(image)
            images.append(ref(image))
        manifest = case_output / "panel-manifest.json"
        save(
            manifest,
            {
                "schema_version": 1,
                "stage": "bound_cached_representative_side_consistency_panels",
                "agent_id": "/root/image_pipeline_audit",
                "case": case_name.replace("-", ":"),
                "study_uid": case["study_uid"],
                "script": ref(Path(__file__).resolve()),
                "source_metadata_collector": ref(ROOT / "collect_technical.py"),
                "technical_summary": ref(summary_path),
                "case_metadata": ref(case_path),
                "source_inventory_subset": ref(inventory_path),
                "images": images,
                "series_sampled": len(ordered_series),
                "cached_panels": len(panels),
                "selection_plan": ref(plan_path),
                "whole_study_series_metadata_verified": case["series_count"],
                "whole_study_source_membership_verified_from_metadata": case["slices"],
                "selected_series_uids": sorted(selected_ids),
                "selected_source_membership_verified": sum(r["slices"] for r in choice["selected_series"]),
                "unselected_series_arrays_opened": 0,
                "panels": panels,
                "display": "Existing normalized/resized cache uint16 values are linearly mapped to uint8 with rounding. Each 256x256 display matrix is pasted unchanged; no additional resize, crop, rotation or flip.",
                "scope": "Three ordered representative cache slices per selected series. Full NPZ hashes, arrays and complete source identities were verified only for selected series; complete whole-study metadata is bound separately. Unselected series arrays were not read or viewed. This is not native-resolution viewing, fresh native DICOM decoding, or a new official-source comparison. No clinical target states or laterality inferred by renderer.",
            },
        )
        all_case_records.append({"manifest": ref(manifest), "images": images})
        print(
            json.dumps({"case": case_name, "manifest": ref(manifest), "images": images})
        )
    save(output / "index.json", {"schema_version": 1, "cases": all_case_records})


if __name__ == "__main__":
    main()
