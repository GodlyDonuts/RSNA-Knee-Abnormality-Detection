"""Independent metadata field and slice-origin extent checks for triage groups."""
from pathlib import Path

import numpy as np
P=Path(__file__).resolve().parent; W=P.parent

ref=lambda p:{'path':str(p.relative_to(W)),'sha256':sha(p),'bytes':p.stat().st_size}
for name,n in [('technical-tier4-third-six-v1',42),('technical-tier4-fourth-six-v1',44)]:
 G=P/name;out=G/'root-field-extent-verification.json';assert not out.exists()

  height,width=m['original_shape'][1:];spacing=m['pixel_spacing'];field=[height*spacing[0],width*spacing[1]];assert field==r['field_height_width_mm']

No images, labels or source bytes changed.','training_ready':False}
 out.write_text(json.dumps(result,indent=2)+'\n');print(name,sha(out))
