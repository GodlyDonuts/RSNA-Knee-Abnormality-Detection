"""Integrate the actual distinct assessor's ten accepted additional corrections."""
from pathlib import Path
from datetime import datetime, timezone
from collections import Counter
import copy
import hashlib
import json
import sys
from dotenv import dotenv_values

T = Path(__file__).resolve().parent
W, R = T.parent, T / "recovered"
sys.path.insert(0, str(R))
from rsna.adjudications import inspect_adjudications
from rsna.schema import TARGETS

def sha(p):
    return hashlib.sha256(p.read_bytes()).hexdigest()

def bind(p):
    return {"path": str(p.relative_to(R)), "sha256": sha(p)}

def read(p, expected=None):
    if expected:
        assert sha(p) == expected, p
    return json.loads(p.read_text())

def save(p, doc):
    p.parent.mkdir(parents=True, exist_ok=True)
    with p.open("x") as f:
        f.write(json.dumps(doc, ensure_ascii=False, indent=2) + "\n")
    return bind(p)

registry_path = R / "initiative/reviews-v2/adjudications/registry.json"
registry = read(registry_path, "9a9c90dce97a245f903ef1e9079e604e0efcfb52952d7421d1420550196383ff")
reports = {r["study_uid"]: r["report"] for p in sorted((R / "initiative/reviews/inputs").glob("batch-*.json")) for r in read(p)}
before = inspect_adjudications(reports, R / "initiative/reviews-v2/reviewed", R / "protocols/report_weak_v2.md", R)
assert before["n_changes"] == 108 and len(before["pending_semantic_qa"]) == 3
before_map = {(c["study_uid"], c["target"]): c for c in before["changes"]}
primary_validation = read(R / "initiative/audits/qualified-negation-focused-000-integration-v1/primary-integration-validation.json", "fcc27c57868bfe955f9e332a836e803c04202c8055196906ed2e9901b7f7a55f")

G = T / "annotation-staging/qualified-negation-focused-000-additional-independent-v1"
A = R / "initiative/audits/qualified-negation-focused-000-additional-independent-v1"
assert not A.exists()
manifest_path = G / "completion-manifest.json"
manifest = read(manifest_path, "1ffd6955c1c6a977e11e9c2a4554c67d4cf50e008c4768b246d0761b5b09944b")
assessment = read(G / "independent-source-assessment.json", "1193e667db83458dc9c1f9c121cbc001bee602c046ff51e2149463baa5364624")
decision = read(G / "accepted-decision.json", "9962b544d69084bb3b9d2c91b4e74c004d12c6911815962ca8fe10eb51a76915")
proposed_case = read(G / "proposed-runtime-case.json", "694339bcbf60f7ce65463a4ad1d0a1e352d5c5d1b94436dd413f1f8ac5700720")
assert assessment["agent_id"] == decision["adjudicator_agent_id"] == "/root/quality_temporal_review"
assert decision["proposer_agent_id"] == "/root" and decision["n_changes"] == 10
assert all(r["disposition"] == "accepted" for r in assessment["records"])
assert len(assessment["records"]) == 10 and sum(r["state_reconsidered_from_own_freeze"] for r in assessment["records"]) == 6
secrets = [v.encode() for k, v in dotenv_values(W / ".env").items() if k in {"HF_TOKEN", "KAGGLE_API_TOKEN"} and v]
assert len(secrets) == 2
assert isinstance(manifest["files"], dict)
entries = {G / name: digest for name, digest in manifest["files"].items()}
entries[manifest_path] = sha(manifest_path)
preserved = []
for p, digest in sorted(entries.items()):
    if isinstance(digest, dict):
        digest = digest["sha256"]
    data = p.read_bytes()
    assert hashlib.sha256(data).hexdigest() == digest and all(s not in data for s in secrets)
    dest = A / p.relative_to(G)
    if dest.suffix == ".py":
        dest = dest.with_suffix(".py.txt")
    dest.parent.mkdir(parents=True, exist_ok=True)
    with dest.open("xb") as f:
        f.write(data)
    assert sha(dest) == sha(p)
    preserved.append({"original_path": str(p.relative_to(W)), **bind(dest), "bytes": len(data)})
save(A / "preservation.json", {"stage": "byte_preserved_actual_independent_additional_ten_decision", "agent_id": "/root", "at_utc": datetime.now(timezone.utc).isoformat(), "files": preserved, "all_originals_unchanged": True})

# Only the runtime wrapper's archive locations change. All assessor artifacts,
# including its originally suggested case wrapper, remain byte-identical.
case = copy.deepcopy(proposed_case)
relocations = []
for reference in [case["proposal"], *case["supporting_evidence"].values()]:
    old = reference["path"]
    if old.startswith("initiative/audits/qualified-negation-focused-000-root-review-v1/"):
        reference["path"] = old.replace("qualified-negation-focused-000-root-review-v1/", "qualified-negation-focused-000-root-complete-v1/")
        assert sha(R / reference["path"]) == reference["sha256"]
        relocations.append({"suggested_path": old, "actual_preserved_path": reference["path"], "same_sha256": reference["sha256"]})
assert len(relocations) == 2
new_keys = {(c["study_uid"], c["target"]) for c in case["changes"]}
assert len(new_keys) == 10 and not new_keys & set(before_map)
case_path = R / "initiative/reviews-v2/adjudications/cases/qualified-negation-focused-000-additional-ten.json"
save(case_path, case)
audit = R / "initiative/audits/qualified-negation-focused-000-integration-v1"
save(audit / "registry-before-additional.json", registry)
registry_after = {**registry, "cases": [*registry["cases"], bind(case_path)]}
registry_path.write_text(json.dumps(registry_after, ensure_ascii=False, indent=2) + "\n")
after = inspect_adjudications(reports, R / "initiative/reviews-v2/reviewed", R / "protocols/report_weak_v2.md", R)
after_map = {(c["study_uid"], c["target"]): c for c in after["changes"]}
assert after["n_changes"] == 118 and after["pending_semantic_qa"] == before["pending_semantic_qa"]
assert all(after_map[k] == v for k, v in before_map.items())
assert set(after_map) - set(before_map) == new_keys
assert all(sha(R / ref["path"]) == ref["sha256"] for ref in primary_validation["protected_artifacts"])

# Reconcile all576 final root dispositions with the registered states. These are
# inspection copies, not a label export; pending semantic QA still blocks export.
root_dispositions = read(R / "initiative/audits/qualified-negation-focused-000-root-complete-v1/root-postfreeze-dispositions.json", "303869c3d1addb04f6eacce7960da6a14e959df8b18e4b3cdbd01375b63ddf91")
original = {r["study_uid"]: r for p in sorted((R / "initiative/reviews-v2/reviewed").glob("*-reviewed.jsonl")) for r in [json.loads(line) for line in p.read_text().splitlines()]}
assert len(original) == 4407
coverage = {target: Counter() for target in TARGETS}
for uid, row in original.items():
    for j, target in enumerate(TARGETS):
        key = uid, target
        state = after_map[key]["after"]["state"] if key in after_map else row["states"][j]
        coverage[target][state] += 1
for d in root_dispositions["records"]:
    key = d["study_uid"], d["target"]
    j = TARGETS.index(d["target"])
    current = after_map[key]["after"] if key in after_map else {"state": original[d["study_uid"]]["states"][j], "evidence": original[d["study_uid"]]["evidence"][j]}
    assert current["state"] == d["root_postfreeze_state"]
    if d["action"] != "retain_effective_annotation":
        assert current["evidence"] == d["root_postfreeze_evidence"]
all_new = [c for c in after["changes"] if c["case_id"].startswith("qualified-negation-focused-000-")]
assert len(all_new) == 84
transition = Counter(c["before"]["state"] + "->" + c["after"]["state"] for c in all_new)
assert transition == {"negative->uncertain": 78, "positive->uncertain": 1, "uncertain->not_mentioned": 2, "negative->negative": 3}
receipt = {"schema_version": 1, "stage": "root_verified_complete_qualified000_registration", "agent_id": "/root", "at_utc": datetime.now(timezone.utc).isoformat(), "registry": bind(registry_path), "additional_case": bind(case_path), "actual_assessor_decision": bind(A / "accepted-decision.json"), "runtime_wrapper_path_relocations_only": relocations, "prior108_changes_preserved": True, "original34_and_all298_protected_artifacts_unchanged": True, "new_additional": 10, "new_primary_and_additional": 84, "total_registered": 118, "new_transition_counts": dict(transition), "root576_final_states_match_registry_inspection": True, "root84_changed_annotations_match_exact_evidence": True, "all4407_inspection_only_state_counts": {k: dict(v) for k, v in coverage.items()}, "pending_semantic_qa": after["pending_semantic_qa"], "label_export_performed": False, "training_ready": False}
save(audit / "complete118-integration-validation.json", receipt)
save(audit / "complete118-inspection.json", after)
(audit / "integrate-qualified000-additional-v1.py.txt").write_bytes(Path(__file__).read_bytes())
print(json.dumps({"receipt": bind(audit / "complete118-integration-validation.json"), "registry": bind(registry_path), "total_registered": 118, "new_registered": 84, "new_transition_counts": dict(transition), "root576_states_reconciled": True, "training_ready": False}, indent=2))
