def choose(i, t, code, reason, quote_from='pre'):
    assert (i, t) not in choices
    assert pre[i]['states'][t] != draft[i]['states'][t]
    quote = '' if code == 'x' else (pre if quote_from == 'pre' else draft)[i]['evidence'][t]
    assert code == 'x' or quote and quote in source[i]['report']
    review[i]['states'][t] = codes[code]
    review[i]['evidence'][t] = quote
    choices[i, t] = dict(source_index=i, study_uid=source[i]['study_uid'], target_index=t, target=TARGETS[t],
        preread_state=pre[i]['states'][t], draft_state=draft[i]['states'][t], final_state=codes[code],
        preread_evidence=pre[i]['evidence'][t], draft_evidence=draft[i]['evidence'][t], final_evidence=quote, reason=reason)
