from pathlib import Path
ROOT=Path(__file__).resolve().parent
BASE='https://download.rockylinux.org/pub/rocky/8.10/AppStream/x86_64/os/'
RPM='python3.12-devel-3.12.13-3.el8_10.x86_64.rpm'
RPM_SHA='0b2b188bf7c920e92ff0f3952ad43100dce29d565dc2c27adea7b4d0dda34dd2'
PRIMARY='68a865f0e7f6eaefef33b6089f15784fc6b2c83f8345f8cad90bebf8154a4c88'

def sha(p):
 h=hashlib.sha256()
 with p.open('rb') as f:
  for b in iter(lambda:f.read(4*1024*1024),b''):h.update(b)
 return h.hexdigest()
def save(path,value):
 with path.open('x') as f:json.dump(value,f,indent=2,sort_keys=True);f.write('\n');f.flush();os.fsync(f.fileno())
def download(url,path,expected,size):
 with urllib.request.urlopen(url,timeout=30) as r,path.open('xb') as out:
  total=0
  for b in iter(lambda:r.read(1024*1024),b''):
   total+=len(b)
   if total>size:raise ValueError('download exceeds bound')
   out.write(b)
  out.flush();os.fsync(out.fileno())
 assert total==size and sha(path)==expected,'download hash/size mismatch'
def command(args,**kw):
 p=subprocess.run(args,capture_output=True,text=True,timeout=120,**kw)
 return {'command':args,'returncode':p.returncode,'stdout':p.stdout,'stderr':p.stderr}
def stage():
 assert platform.python_version()=='3.12.13' and sysconfig.get_config_var('SOABI')=='cpython-312-x86_64-linux-gnu'
 pkg=ROOT/RPM
 download(BASE+'Packages/p/'+RPM,pkg,RPM_SHA,299268)
 signature=command(['rpmkeys','--checksig','--verbose',str(pkg)])
 save(ROOT/'rpm-signature.json',signature)
 text=signature['stdout']+signature['stderr']
 assert signature['returncode']==0 and 'Signature' in text and ': OK' in text and not any(x in text.upper() for x in ['NOKEY','NOT OK','NOTTRUSTED','BAD SIGNATURE'])
 identity=command(['rpm','-qp','--qf','%{NAME}|%{VERSION}|%{RELEASE}|%{ARCH}|%{EPOCHNUM}\n',str(pkg)])
 assert identity['returncode']==0 and identity['stdout'].strip()=='python3.12-devel|3.12.13|3.el8_10|x86_64|0'
 installed=command(['rpm','-q','--qf','%{NAME}|%{VERSION}|%{RELEASE}|%{ARCH}\n','python3.12','python3.12-libs'])
 assert installed['returncode']==0 and set(installed['stdout'].splitlines())=={'python3.12|3.12.13|3.el8_10|x86_64','python3.12-libs|3.12.13|3.el8_10|x86_64'}
 save(ROOT/'package-identities.json',{'development':identity,'installed_runtime':installed})
 primary=ROOT/'primary.xml.gz';download(BASE+'repodata/'+PRIMARY+'-primary.xml.gz',primary,PRIMARY,3969099)
 raw=gzip.decompress(primary.read_bytes());assert len(raw)==53852511 and hashlib.sha256(raw).hexdigest()=='7b4991eea6dc713ad2cb57e43c718d6c502279e82954636763ddadb11eaa7097'
 # The preserved repomd chain was independently verified; primary is content addressed.
 xml=ET.fromstring(raw);ns={'c':'http://linux.duke.edu/metadata/common'};matches=[]
 for item in xml.findall('c:package',ns):
  ver=item.find('c:version',ns)
  if item.findtext('c:name',namespaces=ns)=='python3.12-devel' and item.findtext('c:arch',namespaces=ns)=='x86_64' and ver.attrib=={'epoch':'0','ver':'3.12.13','rel':'3.el8_10'}:matches.append(item)
 assert len(matches)==1
 item=matches[0];assert item.find('c:checksum',ns).attrib['type']=='sha256' and item.findtext('c:checksum',namespaces=ns)==RPM_SHA
 assert item.find('c:location',ns).attrib['href']=='Packages/p/'+RPM and item.find('c:size',ns).attrib['package']=='299268'
 (ROOT/'verified-package-element.xml').write_bytes(ET.tostring(item))
 listing=command(['rpm','-qpl',str(pkg)]);assert listing['returncode']==0
 expected={p for p in listing['stdout'].splitlines() if p.startswith('/usr/include/python3.12/')}
 assert '/usr/include/python3.12/Python.h' in expected and '/usr/include/python3.12/pyconfig.h' in expected
 extract=ROOT/'extracted';extract.mkdir(mode=0o700)
 first=subprocess.Popen(['rpm2cpio',str(pkg)],stdout=subprocess.PIPE,stderr=subprocess.PIPE)
 second=subprocess.run(['cpio','--extract','--make-directories','--no-absolute-filenames','--no-preserve-owner','--quiet','./usr/include/python3.12/*'],stdin=first.stdout,cwd=extract,capture_output=True,timeout=60)
 first.stdout.close();first_error=first.stderr.read();first_rc=first.wait(timeout=30)
 save(ROOT/'header-extraction.json',{'rpm2cpio_returncode':first_rc,'cpio_returncode':second.returncode,'rpm2cpio_stderr':first_error.decode(),'cpio_stderr':second.stderr.decode(),'selection':'only ./usr/include/python3.12/*'})
 assert first_rc==0 and second.returncode==0
 headers=extract/'usr/include/python3.12';assert (headers/'Python.h').is_file() and (headers/'pyconfig.h').is_file()
 files=[]
 for p in sorted(extract.rglob('*')):
  assert not p.is_symlink(),'Unexpected header symlink'
  if p.is_file():
   rel=str(p.relative_to(extract));assert '/'+rel in expected and rel.startswith('usr/include/python3.12/')
   files.append({'path':rel,'bytes':p.stat().st_size,'sha256':sha(p)})
 assert files
 canonical=json.dumps(files,sort_keys=True,separators=(',',':')).encode();tree=hashlib.sha256(canonical).hexdigest()
 save(ROOT/'header-tree.json',{'files':files,'canonical_tree_sha256':tree})
 with (ROOT/'headers.tar').open('xb') as out:
  with tarfile.open(fileobj=out,mode='w',format=tarfile.PAX_FORMAT) as tar:
   for f in files:tar.add(extract/f['path'],arcname='headers/'+str(Path(f['path']).relative_to('usr/include/python3.12')),recursive=False)
  out.flush();os.fsync(out.fileno())
 save(ROOT/'staging-receipt.json',{'status':'private_headers_verified','rpm_sha256':RPM_SHA,'rpm_signature_verified_with_system_trusted_keys':True,'rpm_installed':False,'system_mutations':False,'header_tree_sha256':tree,'header_files':len(files),'header_bytes':sum(x['bytes'] for x in files),'header_archive_sha256':sha(ROOT/'headers.tar'),'header_archive_bytes':(ROOT/'headers.tar').stat().st_size,'cpath':str(headers),'runtime_python':'3.12.13','SOABI':sysconfig.get_config_var('SOABI'),'cpu_only':True,'job_id':os.environ['SLURM_JOB_ID']})

def proof():

 assert os.environ.get('CPATH')==str(headers)
 sources=['triton/backends/nvidia/driver.c','triton/backends/nvidia/driver.py','triton/runtime/build.py','torch/_native/registry.py','torch/_native/ops/bmm_outer_product/triton_impl.py','torch/_native/ops/bmm_outer_product/triton_kernels.py']

 driver=backend/'driver.c';data=driver.read_text();init=data.split('PyMODINIT_FUNC PyInit_cuda_utils(void)',1)[1]

 save(ROOT/'cpu-proof.json',{'status':'actual_triton_helper_compile_and_import_passed','job_id':os.environ['SLURM_JOB_ID'],'requested_gpus':0,'gpu_work_performed':False,'original_environment_sha256':'89597ed4d974e5874c7bb2ffb42545e6c0bb6a9ac796a5839a4b1902917ef338','runtime_python':platform.python_version(),'SOABI':sysconfig.get_config_var('SOABI'),'Py_DEBUG':sysconfig.get_config_var('Py_DEBUG'),'ABIFLAGS':sysconfig.get_config_var('ABIFLAGS'),'header_tree_sha256':json.loads((ROOT/'header-tree.json').read_text())['canonical_tree_sha256'],'rpm_sha256':RPM_SHA,'CPATH':str(headers),'actual_driver_source_sha256':sha(driver),'compiled_extension_sha256':sha(output),'CPU_signature_metadata_hex':result.hex(),'abi_compiled_values':abi_result['stdout'].strip(),'installed_sources':sourceproof,'system_mutations':False,'backend_disable_or_replacement':False})
 print(json.dumps({'status':'CPU header and actual Triton compiler/import proof passed','job_id':os.environ['SLURM_JOB_ID']}),flush=True)

if __name__=='__main__':
