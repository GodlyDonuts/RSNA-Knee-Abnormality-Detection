R=W/'work/recovered'
def sha(p): return hashlib.sha256(p.read_bytes()).hexdigest()
def ref(p): return {'path':str(p.relative_to(W)),'sha256':sha(p),'size_bytes':p.stat().st_size}
def write(name,data):
    p=S/name
    assert not p.exists(),f'Will not replace {name}'
    p.write_text(json.dumps(data,ensure_ascii=False,indent=2)+'\n')
    return ref(p)
ns={'__file__':str(S/'author_source_assessment.py'),'__name__':'source_authoring'}
for name in ['author_source_assessment.py','author_part2.py','author_part3.py']:
    exec(compile((S/name).read_text(),str(S/name),'exec'),ns)

assert len(records)==50 and len({r['study_uid'] for r in records})==50
rubric=R/'protocols/report_weak_v2.md'
assert sha(rubric)=='138c5cc33b92901d57dbf5fbe7e3a9878b73d25b0af6934908d0052cd6786014'
now=datetime.now(timezone.utc).isoformat()
prior=[W/f'work/annotation-staging/qualified-negation-focused-{i:03}-source-review-v1/source-assessment-frozen.json' for i in range(5)]

def uid_strings(x):
    if isinstance(x,str): return {x} if x.startswith('1.2.826.') else set()
    if isinstance(x,dict): return set().union(*(uid_strings(v) for v in x.values()))
    if isinstance(x,list): return set().union(*(uid_strings(v) for v in x)) if x else set()
    return set()
overlaps=[]; all_overlap=set()
for p in prior:
    found=uid_strings(json.loads(p.read_text()))
    selected=[{k:r[k] for k in ('packet_ordinal','study_uid','source_path','source_index_zero_based','report_sha256')} for r in records if r['study_uid'] in found]
    overlaps.append({'prior_artifact':ref(p),'exact_overlap_count':len(selected),'exact_overlaps':selected})
    all_overlap.update(r['study_uid'] for r in selected)
