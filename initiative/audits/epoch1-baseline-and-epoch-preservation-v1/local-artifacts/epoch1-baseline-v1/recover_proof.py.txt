"""Certify the completed epoch-1 checkpoint after expected torchrun exit-75 propagation."""
from __future__ import annotations
import hashlib
import json
import os
import subprocess
from pathlib import Path
import torch

ROOT = Path('/home/sa305415/rsna-knee')
RUN = ROOT / 'runs/dinov3-l16-epoch1-baseline-b740-v1'
PRIMARY = ROOT / 'runs/dinov3-l16-production-fold0-b74060b-v1'
LOGS = ROOT / 'logs'
SOURCE = ROOT / 'code-stopfix-1f486765'
SOURCE_SHA = 'b50ae5591b7b5b02f36423676f1f7fcbedacd69c8d848172bee459e4ae8d044a'

def digest(path: Path) -> str:
    h=hashlib.sha256()
    with path.open('rb') as f:
        for part in iter(lambda:f.read(4*1024*1024),b''):h.update(part)
    return h.hexdigest()

def events(path: Path) -> dict[int, dict]:
    found={}
    for line in path.read_text(errors='replace').splitlines():
        if not line.startswith('{"epoch": '):continue
        row=json.loads(line)
        if row.get('epoch')==0 and type(row.get('update')) is int:
            found[row['update']]={k:row[k] for k in ('epoch','update','optimizer_steps','loss','gradient_norm','observed_cells')}
    return found

def main() -> None:
    os.umask(0o077)
    proof_path=RUN/'baseline-proof.json'
    if proof_path.exists() or proof_path.is_symlink():raise ValueError('Proof already attempted')
    sacct=subprocess.run(['sacct','-j','841126','--format=JobID,State,ExitCode','-P','--noheader'],capture_output=True,text=True,check=True)
    if not any(line.startswith('841126|FAILED|1:0') for line in sacct.stdout.splitlines()):raise ValueError('Replay job state changed')
    invocation=json.loads((RUN/'invocation.json').read_text())
    run=json.loads((RUN/'run.json').read_text())
    checkpoint=RUN/'checkpoint.pt'
    sha=digest(checkpoint)
    if invocation.get('status')!='checkpointed_for_resume' or invocation.get('exit_code')!=75 or invocation.get('checkpoint_sha256')!=sha:raise ValueError('Trainer stop receipt changed')
    state=torch.load(checkpoint,map_location='cpu',weights_only=True)
    binding=state['binding']
    if (state.get('schema_version')!=1 or state.get('optimizer_boundary') is not True
        or (state.get('epoch'),state.get('next_update'),state.get('step'))!=(1,0,281)
        or state.get('scheduled_epoch') is not False or binding!=run['binding']
        or run.get('training_studies')!=2247 or binding['world_size']!=2
        or binding['config']['diagnostic_study_limit'] is not None
        or binding['config']['submission_every_epochs']!=2
        or binding['config']['fold']!=0 or binding['config']['seed']!=20260922):raise ValueError('Completed full-data checkpoint marker changed')
    code={str(p.relative_to(SOURCE/'rsna')):digest(p) for p in sorted((SOURCE/'rsna').rglob('*.py'))}
    if binding['code_sha256']!=code:raise ValueError('Training source hash changed')
    git=subprocess.check_output(['git','-C',str(SOURCE),'rev-parse','HEAD']).decode().strip()
    if git!='b74060b98f01bff1b924070ccba5ce44dd002f40':raise ValueError('Training Git changed')
    original=PRIMARY/'pilot-checkpoints/update-026.pt'
    if digest(original)!=SOURCE_SHA:raise ValueError('Replay source checkpoint changed')
    expected=events(LOGS/'rsna-dinov3-fold0-b740-resume-841107.out')
    actual=events(LOGS/'rsna-dinov3-e1-replay-841126.out')
    if set(expected)!=set(range(27,282)) or set(actual)!=set(range(27,282)) or expected!=actual:raise ValueError('Replay optimizer events differ from primary')
    record={'schema_version':1,'status':'completed_epoch1_replay','job_id':'841126','proof_job_id':os.environ['SLURM_JOB_ID'],
            'trainer_exit_code':75,'wrapper_slurm_exit_code':'1:0','checkpoint_sha256':sha,
            'checkpoint_bytes':checkpoint.stat().st_size,'epoch':1,'next_update':0,'optimizer_steps':281,
            'scheduled_epoch':False,'source_checkpoint_sha256':SOURCE_SHA,
            'matching_primary_updates':len(actual),'training_git_commit':git,
            'binding_sha256':hashlib.sha256(json.dumps(binding,sort_keys=True,separators=(',',':'),allow_nan=False).encode()).hexdigest()}
    with proof_path.open('x') as f:json.dump(record,f,sort_keys=True,indent=2);f.write('\n');f.flush();os.fsync(f.fileno())
    print(json.dumps(record,sort_keys=True),flush=True)

if __name__=='__main__':main()
