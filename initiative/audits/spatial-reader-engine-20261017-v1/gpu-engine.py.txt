def identical(first, second, path='root'):
    import numpy as np
    import torch
    if isinstance(first, torch.Tensor):
        assert isinstance(second, torch.Tensor) and first.dtype == second.dtype and torch.equal(first, second), path
    elif isinstance(first, np.ndarray):
        assert isinstance(second, np.ndarray) and first.dtype == second.dtype and np.array_equal(first, second), path
    elif isinstance(first, dict):
        assert isinstance(second, dict) and first.keys() == second.keys(), path
        for key in first: identical(first[key], second[key], path + '.' + str(key))
    elif isinstance(first, (tuple, list)):
        assert type(first) is type(second) and len(first) == len(second), path
        for i, (a, b) in enumerate(zip(first, second)): identical(a, b, path + '.' + str(i))
    else:
        assert type(first) is type(second) and first == second, path


def parent_digest(parent):
    value = hashlib.sha256()
    for name, tensor in parent.state_dict().items():
        array = tensor.detach().cpu().contiguous().numpy()
        value.update(name.encode()); value.update(str(array.dtype).encode())
        value.update(json.dumps(list(array.shape)).encode()); value.update(array.tobytes())
    return value.hexdigest()


def inputs(admission, mode):
    import numpy as np
    import pandas as pd
    import torch
    from rsna.competitive.data import CompetitiveStudyDataset
    from rsna.competitive.spatial_data import CachedSpatialData, OnlineSpatialData
    from rsna.competitive.compact_infer import load_model
    from rsna.schema import TARGETS
    rows = json.loads((R / 'pilot-recovery-v2-accepted.json').read_text())['private_records']
    original = json.loads((R / 'admitted.json').read_text())
    membership = json.loads((R / 'membership.private.json').read_text())
    fit_ids = membership['fitting']; outer_ids = membership['outer']
    assert len(fit_ids) == 2247 and len(outer_ids) == 1156 and not (set(fit_ids) & set(outer_ids))
    assert all(hashlib.sha256(json.dumps(ids).encode()).hexdigest() == original[role + '_uids_sha256'] for role, ids in membership.items())
    by_role = {role: [x for x in rows if x['role'] == role] for role in ('fitting', 'outer')}
    assert all([x['study_uid'] for x in by_role[role]] == membership[role][:16] for role in by_role)
    label_path = A / 'initiative/releases/report-supervision-v1/studies.parquet'
    parent_binding = json.loads((Path(original['checkpoint']).parents[1] / 'run.json').read_text())['binding']
    assert sha(label_path) == parent_binding['release']['manifest_sha256']
    ids = [x['study_uid'] for x in by_role['fitting']]
    columns = ['study_uid', *TARGETS, *[x + '__mask' for x in TARGETS]]
    labels = pd.read_parquet(label_path, columns=columns, filters=[('study_uid', 'in', ids)])
    assert len(labels) == 16 and not labels.study_uid.duplicated().any()
    labels = labels.set_index('study_uid').loc[ids].reset_index()
    lab = labels[list(TARGETS)].to_numpy(dtype=np.float32)
    mask = labels[[x + '__mask' for x in TARGETS]].to_numpy(dtype=np.float32)
    assert np.isin(mask, [0, 1]).all() and np.isin(lab[mask.astype(bool)], [0, 1]).all()
    lab = np.where(mask, lab, 0).astype(np.float32)
    data_binding = {'original_private_index_sha256': INDEX_SHA, 'parent_checkpoint_sha256': PARENT_SHA,
                    'original_source_labels_sha256': sha(label_path), 'fitting16_labels_masks_sha256': hashlib.sha256(lab.tobytes() + mask.tobytes()).hexdigest(),
                    'original_source2247_fitting_roster_sha256': original['fitting_uids_sha256'], 'original_source1156_outer_roster_sha256': original['outer_uids_sha256'],
                    'fixed_input': {'side':384,'slices':16,'max_series':6,'precision':'float32','chunk_size':12},
                    'diagnostic_only': True, 'seed':20261017}
    # Clinical values are loaded for the 16 fitting cases only. The engineering
    # outer provider has no target columns and is never evaluated for accuracy.
    source_frames = {'fitting':labels, 'outer':pd.DataFrame({'study_uid':[x['study_uid'] for x in by_role['outer']]})}
    cached = {role: CachedSpatialData(R/'features', by_role[role], lab if role == 'fitting' else np.zeros((16,12),np.float32), mask if role == 'fitting' else np.zeros((16,12),np.float32), {**data_binding,'role':role}, seed=20261017) for role in by_role}
    if mode == 'cached': return cached, None, None
    assert mode == 'online'
    parent, policy = load_model(Path(original['checkpoint']), PARENT_SHA, precision='float32', device=torch.device('cuda'))
    assert policy['epoch'] == 16 and policy['optimizer_steps'] == 4496 and policy['competition_eligible']
    before = parent_digest(parent)
    series = pd.read_csv(original['series_path'])
    assert sha(original['series_path']) == parent_binding['release']['series_sha256']
    online = {role: OnlineSpatialData(CompetitiveStudyDataset(source_frames[role], series, Path(original['native_cache']), side=384, slices=16, max_series=6, augment=False, soft_targets=False), parent, cached[role].binding, seed=20261017, chunk_size=12) for role in by_role}
    # This is the provider substitution being tested; exact equality is required
    # before fitting, including full maps, geometry operators, logits and masks.
    start = time.monotonic()
    maximum = 0.0
    for role in by_role:
        for i in range(16):
            actual, expected = online[role][i], cached[role][i]
            assert actual['study_uid'] == expected['study_uid']
            for key in ('maps', 'base_logits', 'labels', 'label_mask'):
                a,b=actual[key].cpu(),expected[key].cpu()
                maximum=max(maximum,float((a.float()-b.float()).abs().max()))
                assert torch.equal(a,b), 'original cached/online MRI tensor differs: ' + key
            assert actual['plan'].metadata_sha256 == expected['plan'].metadata_sha256
            assert len(actual['plan'].groups) == len(expected['plan'].groups)
            for a,b in zip(actual['plan'].groups,expected['plan'].groups):
                assert a.kind == b.kind and np.array_equal(a.first,b.first) and np.array_equal(a.second,b.second)
                assert torch.equal(a.aligned,b.aligned) and torch.equal(a.control,b.control)
    assert parent_digest(parent) == before
    (T/'online-inputs-exact.json').write_text(json.dumps({'status':'actual32_online_full_map_base_input_and_operator_exact','GPU_job':os.environ['SLURM_JOB_ID'],'cases':32,'maximum_absolute_tensor_difference':maximum,'elapsed_seconds':time.monotonic()-start,'parent_all_parameter_buffer_fingerprint':before,'outer_clinical_targets_read':False})+'\n')
    return online, parent, before


def worker(admission, mode, arm, phase):
    import torch
    from rsna.competitive.spatial_train import SpatialRecipe, SpatialTrainer
    data,parent,before = inputs(admission, mode)
    recipe = SpatialRecipe(epochs=2)
    binding = {'diagnostic_only': True, 'source_git':admission['source_git'], 'source_manifest_sha256':admission['source_manifest_sha256'], 'original_base_sha256':PARENT_SHA, 'production_eligible':False}
    engine = SpatialTrainer(data['fitting'], recipe, arm=arm, device=torch.device('cuda'), binding=binding)
    output=T/(mode+'-'+arm+('-replay' if phase in {'stop','resume'} else '-full'))
    initial_reader=parent_digest(engine.model)
    result=engine.fit(output, resume=phase=='resume', stop_after=1 if phase=='stop' else None)
    (output/'initial-reader.json').write_text(json.dumps({'state_sha256':initial_reader})+'\n')
    if phase!='stop':
        assert result['completed']==4 and len(engine.history)==4
        assert engine.history[0]['projection_gradient_norm']==0 and engine.history[-1]['projection_gradient_norm']>0
        frame=engine.predictions(data['outer']);assert frame.shape==(16,13)
        frame.to_csv(output/'predictions-private.csv',index=False)
    if parent is not None:
        assert parent_digest(parent)==before and not parent.training and not any(p.requires_grad for p in parent.parameters())
        (output/'frozen-parent-exact.json').write_text(json.dumps({'all_parameters_buffers_exact':True,'fingerprint':before})+'\n')
    cap()
    print(json.dumps({'phase':mode+'-'+arm+'-'+phase,'status':result['status'],'actual_finite_updates':engine.completed,'GPU_job':os.environ['SLURM_JOB_ID']}),flush=True)
