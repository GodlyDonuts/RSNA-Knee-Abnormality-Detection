def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def load(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module

def existing_record(db, key):
    found = db.execute('SELECT value FROM meta WHERE key=?', (key,)).fetchone()
    assert found is not None, 'Exact original canonical record required'
    return json.loads(found[0])


def request_for(candidate, mode, base, source):
    h = hashlib.sha256(source.encode()).hexdigest()

    slug = 'godlydonuts/rsna-final-' + candidate + '-e3-' + ('qual-' if mode == 'qualification' else 'offline-') + h[:12]
    return {
        'slug': slug, 'newTitle': slug.split('/')[1].replace('-', ' '), 'text': source,

def entry_eligibility(w, db, account, submissions, description, candidate):
    slots = w.allowance(db, account)
    active = [x['ref'] for x in submissions if w.submission_outcome(x) in ('pending_or_unknown', 'complete_unscored')]
    assert not any(x.get('description') == description for x in submissions), 'Original exact description exists; reconcile without retry'
    # Preserve ordered scientific comparisons; actual previous outcome is required.
    previous = CANDIDATES.index(candidate) - 1
    previous_outcome = 'not_required'
    if previous >= 0:
        D = R / CANDIDATES[previous]
        proof = json.loads((D / 'prepared.json').read_text())
        rec = existing_record(db, 'licensed-parent-full-epoch-serving:' + proof['model_sha256'])
        found = [x for x in submissions if x['ref'] == rec.get('submission_id')]
        previous_outcome = w.submission_outcome(found[0]) if len(found) == 1 else 'unsubmitted_or_unknown'
        assert previous_outcome != 'failed', 'Previous actual hidden failure needs inspection before another entry'
    return {'eligible': slots > 0 and not active and previous_outcome in ('scored', 'not_required'),
            'effective_allowance': slots, 'active_unresolved_submission_ids': active,
            'previous_actual_outcome': previous_outcome, 'account': account}
