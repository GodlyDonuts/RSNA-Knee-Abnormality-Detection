def new(path, value):
    with path.open('x') as f:
        json.dump(value, f, indent=2, allow_nan=False)
        f.write('\n')
        f.flush()
        os.fsync(f.fileno())


def record(value, terminal=False):
    out = {'PID': os.getpid(), 'elapsed_seconds': time.monotonic() - START, **value}
    path = L / ('owner.result.json' if terminal else 'owner.progress.json')
    tmp = path.with_suffix('.tmp')
    tmp.write_text(json.dumps(out, indent=2, allow_nan=False) + '\n')
    tmp.replace(path)
    print(json.dumps(out, allow_nan=False), flush=True)


def ssh(args, value=None, timeout=120):
    result = subprocess.run(
        ['ssh', '-S', SOCKET, '-o', 'BatchMode=yes', '-o', 'ConnectTimeout=8',
         'newton', shlex.join(args)],
        input=None if value is None else json.dumps(value), text=True,
        capture_output=True, timeout=timeout,
    )
    return result


def cpu(node, program, timeout=120):
    x = ssh(['srun', '--account=skattel', '--partition=normal', '--qos=normal',
             '--nodelist=' + node, '--nodes=1', '--ntasks=1', '--cpus-per-task=1',
             '--gres=none', '--mem=1G', '--time=00:02:00', MANAGEMENT, '-B',
             '-c', program], timeout=timeout)
    if x.returncode:
        raise RuntimeError('Read-only original-state observation failed; no restart or mutation')
    return json.loads(x.stdout)


def credential():
    path = ROOT / '.env'
    assert path.stat().st_mode & 0o077 == 0
    values = {}
    for line in path.read_text().splitlines():
        if '=' in line and not line.lstrip().startswith('#'):
            k, v = line.split('=', 1)
            values[k.strip()] = v.strip().strip('"').strip("'")
    token = values['HF_TOKEN']
    assert token.startswith('hf_')
    return token
