import hashlib
import importlib.util
import json
import os
import shutil
import tempfile
from pathlib import Path

v = json.load(__import__("sys").stdin)
assert os.uname().nodename == "evc33" and not os.environ.get("SLURM_JOB_GPUS")
os.umask(0o077)
D = Path("/tmp/rsna-predictive-cached-engine-milestone-20261020-v4")
assert not D.exists()
raw = Path("/tmp/rsna-predictive-full-raw-cache-20261020-v1")
context = Path("/tmp/rsna-predictive-full-context-cache-20261020-v1")
paid_raw = json.loads(Path("/tmp/rsna-predictive-raw16-cache-20261020-v1/actual.json").read_text())[
    "cache_bytes"
] + sum(p.stat().st_size for p in (raw / "cache").glob("*.npz"))
paid_context = json.loads(
    Path("/tmp/rsna-predictive-context16-cache-20261020-v1/actual.json").read_text()
)["cache_bytes"] + sum(p.stat().st_size for p in context.glob("context-*.npz"))
source_bytes = sum(len(s.encode()) for s in v["files"].values())
assert source_bytes < 2**20 and 3 * source_bytes + 128 * 2**10 < 4 * 2**20
assert (
    shutil.disk_usage("/tmp").free
    > (20 + 8 + 12 + 4 + 24) * 2**30
    + max(0, 320 * 2**30 - paid_raw)
    + max(0, 96 * 2**30 - paid_context)
    + (166 + 64 + 4 + 16 + 256 + 8 + 32 + 4 + 4) * 2**20
)
q = __import__("subprocess").check_output(
    ["lfs", "quota", "-u", "sa305415", "/home/sa305415"], text=True
)
a = next(x.split() for x in q.splitlines() if x.strip().startswith("/home/sa305415"))
assert (int(a[2]) - int(a[1])) * 1024 > 20 * 2**30 + 76 * 2**20
assert int(a[6]) - int(a[5]) > 5000 + 128 + 128 + 64 + 24 + 6 + 3 + 3
D.mkdir(mode=0o700)
files = {}
for n, text in v["files"].items():
    p = D / "source" / n
    assert p.resolve().is_relative_to((D / "source").resolve())
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text(text)
    files[n] = p
assert sum(p.stat().st_size for p in files.values()) < 2**20
core = Path(
    "/home/sa305415/rsna-knee/production-launch-native384-a32-c9-v1/publish_native384_epoch.py"
)
assert (
    hashlib.sha256(core.read_bytes()).hexdigest()
    == "4e327e271816e4a4055b4f3ad28a8dab59fc650a1367fd13bbd7fe06d9a5efde"
)
os.environ.update(
    HF_HUB_DISABLE_XET="1",
    HF_HUB_DISABLE_PROGRESS_BARS="1",
    HF_HOME=str(D / "hf"),
    HF_HUB_CACHE=str(D / "hf-cache"),
    HF_XET_CACHE=str(D / "xet"),
)
s = importlib.util.spec_from_file_location("original_canonical_publisher", core)
pub = importlib.util.module_from_spec(s)
s.loader.exec_module(pub)
from huggingface_hub import CommitOperationAdd, HfApi

prefix = "research-milestones/predictive-cached-engine-20261020-v4-node33-" + v["source_git"][:12]
table = {n: {"bytes": p.stat().st_size, "sha256": pub.sha(p)} for n, p in files.items()}
with pub.shared_publication_lock():
    api = HfApi(token=v["token"])
    info = api.repo_info("GodlyDonuts/RSNA-Data", repo_type="dataset")
    assert info.private is True
    assert not list(
        api.get_paths_info(
            "GodlyDonuts/RSNA-Data", paths=[prefix + "/" + n for n in files], repo_type="dataset"
        )
    )
    pub.atomic_new(
        D / "publication-intent.json",
        {
            "source_git": v["source_git"],
            "parent": info.sha,
            "prefix": prefix,
            "files": table,
            "automatic_retry": False,
        },
    )
    commit = api.create_commit(
        "GodlyDonuts/RSNA-Data",
        repo_type="dataset",
        parent_commit=info.sha,
        commit_message="Preserve owner predictive-evidence proposal, new masked-input/density/model/trainer sources and actual original-MRI admission receipts; no accuracy claim",
        operations=[
            CommitOperationAdd(path_in_repo=prefix + "/" + n, path_or_fileobj=p)
            for n, p in files.items()
        ],
    )
    pub.atomic_new(D / "publication-commit.json", {"revision": commit.oid})
    for n, item in table.items():
        with tempfile.TemporaryDirectory(prefix="mission-source-byte-check-", dir="/tmp") as tmp:
            p = Path(
                api.hf_hub_download(
                    "GodlyDonuts/RSNA-Data",
                    prefix + "/" + n,
                    repo_type="dataset",
                    revision=commit.oid,
                    cache_dir=str(D / "hf-cache"),
                    local_dir=tmp,
                    force_download=True,
                )
            )
            assert p.stat().st_size == item["bytes"] and pub.sha(p) == item["sha256"]
    out = {
        "status": "selected_sanitized_mission_sources_all_objects_private_download_size_SHA_verified",
        "CPU_job": os.environ["SLURM_JOB_ID"],
        "source_git": v["source_git"],
        "revision": commit.oid,
        "prefix": prefix,
        "objects": len(table),
        "bytes": sum(r["bytes"] for r in table.values()),
        "canonical_shared_publication_lock_used": True,
        "MRI_reports_UIDs_per_case_predictions_external_weights_included": False,
        "old_mirror_or_checkpoint_repeated": False,
    }
    pub.atomic_new(D / "verified.json", out)
    assert (
        sum(p.stat().st_size for p in D.rglob("*") if p.is_file() and not p.is_symlink())
        < 4 * 2**20
    )
    print(json.dumps(out))
