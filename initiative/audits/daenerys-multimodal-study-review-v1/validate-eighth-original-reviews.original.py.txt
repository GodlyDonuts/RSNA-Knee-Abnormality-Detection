"""Preserve and source-bind new candidate reviews, without changing labels."""
import collections
import hashlib
import json
from pathlib import Path

D=Path(__file__).resolve().parent
packet=json.loads((D/'packet.json').read_text())
hosted=json.loads((D/'hosted-confirmed.json').read_text())
assert len(hosted['images'])==len(packet['images'])==26
index={r['reference']:(r,s) for r,s in zip(hosted['images'],packet['images'],strict=True)}
manifest_sha=hashlib.sha256((D/'image-only-manifest.json').read_bytes()).hexdigest()
rows=[];originals=[]
for name,expected,size in [('provisional-001.original.json','977068664b6f7bcf73d77bb6d1cb62f1f305424d73cbbc4f8e6fe0dc977d971d',71837),('report-informed-002.original.json','ef69d5d171b0d8eeef691285ad03deda7bd12c60388b20bb5393dfe632c36b56',82227)]:
 raw=(D/name).read_bytes();assert len(raw)==size and hashlib.sha256(raw).hexdigest()==expected
 v=json.loads(raw);originals.append(v);assert v['packet_id']==packet['packet_id'] and v['source']['study_uid']==packet['study_uid'] and v['source']['source_revision']==packet['source_revision'] and v['source']['manifest_sha256']==manifest_sha and v['candidate_annotation_only'] is True and v['accepted_labels_changed'] is False
 c=v['coverage'];assert c['expected_references']==c['personally_inspected_references']==26 and c['unique_original_slices']==152 and c['inaccessible_references']==[] and len(c['access_log'])==26
 assert len({a['reference'] for a in c['access_log']})==26
 for a in c['access_log']:
  r,s=index[a['reference']];assert a['sha256']==r['sha256']==s['sha256'] and a['byte_count']==r['byte_size']==s['size'] and (a['series'],a['slice_start'],a['slice_end'])==(s['series_index'],s['slice_start'],s['slice_end']) and tuple(a['sheet_dimensions'])==(s['width'],s['height'])
  series=packet['series'][s['series_index']-1];sops=series['ordered_sop_uids'][s['slice_start']-1:s['slice_end']];assert a['source_sop_uids']==sops and a['source_hashes_from_manifest']==series['source_sha256'][s['slice_start']-1:s['slice_end']]
 from rsna.schema import TARGETS
 assert [t['target'] for t in v['targets']]==list(TARGETS)
 quotes=0;evidence=0
 for t in v['targets']:
  assert t['status'] in ('positive','negative','uncertain','not_assessable') and t['confidence'] in ('low','medium','high') and t['confidence_is_calibrated'] is False
  for q in t['literal_report_quotes']:assert q in packet['report'];quotes+=1
  for e in t['supporting_image_evidence']+t['contrary_image_evidence']:
   si=e['series'];assert type(si) is int and 1<=si<=len(packet['series']);assert all(type(n) is int and 1<=n<=packet['series'][si-1]['slice_count'] for n in e['slices']);evidence+=1
 rows.append({'file':name,'bytes':size,'sha256':expected,'coverage_records':26,'targets':12,'literal_report_quotes_matched':quotes,'evidence_records_with_valid_series_slice_ranges':evidence,'states':dict(collections.Counter(t['status'] for t in v['targets']))})
a,b=originals;assert a['reading_order']['report_opened'] is False and a['reading_order']['early_report_access'] is False and not any(t['literal_report_quotes'] for t in a['targets'])
assert b['preserved_provisional']['sha256']==rows[0]['sha256'] and b['preserved_provisional']['byte_count']==rows[0]['bytes'] and b['preserved_provisional']['unchanged'] is True
assert b['report']['original_utf8_text']==packet['report'] and b['report']['sha256']==packet['report_sha256']
changes=[x['target'] for x,y in zip(a['targets'],b['targets'],strict=True) if x['status']!=y['status']]
assert changes==[]
out={'status':'original_image_only_and_report_informed_candidate_bytes_and_source_coverage_validated','originals':rows,'status_changes_after_report':changes,'image_only_reading_order_reviewer_reported':True,'provisional_timestamp_error_reported':False,'independent_underlying_image_or_report_tool_call_trace_collected':False,'clinical_gold_or_model_accuracy_claim':False,'accepted_labels_changed':False,'alternate_original_windows_requested':14}
p=D/'original-reviews-validated.json';assert not p.exists();p.write_text(json.dumps(out,indent=2)+'\n');print(json.dumps(out))
