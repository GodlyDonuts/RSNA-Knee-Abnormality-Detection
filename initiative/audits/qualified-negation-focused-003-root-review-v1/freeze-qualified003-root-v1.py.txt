"""Bind actual root-authored complete sources before current reader findings."""
from pathlib import Path
from datetime import datetime, timezone
from collections import Counter
import hashlib
import json

W = Path(__file__).resolve().parent.parent
R = W / 'work/recovered'
G = W / 'work/annotation-staging/qualified-negation-focused-003-root-review-v1'
N = W / 'work/annotation-staging/qualified-negation-focused-003-post-freeze-comparison-v1'
SHA = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
read = lambda p: json.loads(p.read_text())
pin = lambda p: dict(path=str(p.relative_to(W)), sha256=SHA(p), bytes=p.stat().st_size)
PARTS = [
    ('first-five', 'c893e5a460862f11bee215d7d8ebafb77e41b8865424c444d87d7aae3bcd0503'),
    ('next-five', '52a0c27789e33f20cdccd24d78d97f6e37a97a38c63cacb67c8deffb290eeef6'),
    ('third-five', '4abf2a6b83701e0e3c5aaac8c3c557de8061716b54defdab5058633e9566ba29'),
    ('fourth-five', 'e82010464b0879afb246bebd36f9fbdb26e948a431fbbedf0c2ec175a13a053a'),
    ('fifth-five', '197ca7fe40d8df5d7de64f5a5dda93496224b3fabfbbd7f6b3a4b4a900dedc8a'),
    ('sixth-five', 'ea6cd3476937f65007ad54765279fa46cfa2509245eacc47737d4d5a1e467947'),
    ('seventh-five', 'bce1de7534ecc254482103518fd7c4efe9ae6b80bacf5c33dbbb11a428f893f9'),
    ('eighth-five', 'bd9f1e3302f980cfb887aacfeab4c0b34d023882074998259abaccac74b03218'),
    ('final-five', '578c9c1804c5f7870e8307a068450076a3cb2dde756832dbb1b660c18d0e71d9'),
]
index = N / 'neutral-root-packet-index.json'
assert SHA(index) == '7b63e8b77f14faab7ae77cfd9476873ca3bc63dd3c16a3571b9e7d9411b8d78e'
rubric = R / 'protocols/report_weak_v2.md'
assert SHA(rubric) == '138c5cc33b92901d57dbf5fbe7e3a9878b73d25b0af6934908d0052cd6786014'
assert SHA(G / 'partial-ten-source-freeze.json') == '87711d936d33f71a010118229e6ec28f5d5222df946b5f2bb600ba84a3248564'
assert SHA(G / 'partial-twenty-source-freeze.json') == '98dd485e975028d7eb909036c567420a2bf60e52ac22b29d7cd3eb2c1d57c5e1'
before = {p: SHA(p) for p in G.iterdir() if p.is_file()}
sources = []
for p in read(index)['packets']:
    path = W / p['path']
    assert SHA(path) == p['sha256'] and path.stat().st_size == p['bytes']
    rows = read(path)
    assert len(rows) == p['reports']
    sources.extend(rows)
assert len(sources) == len({r['study_uid'] for r in sources}) == 45
records = []
for name, digest in PARTS:
    p = G / (name + '-source-assessment.json')
    assert SHA(p) == digest
    d = read(p)
    assert d['agent_id'] == d['session_id'] == '/root'
    assert d['current003_reader_case_findings_opened'] is False
    assert d['source_index_uid_report_hash_and_exact_quotes_verified']
    assert d['full_reports'] * 12 == d['target_judgments']
    assert d['rubric_sha256'] == SHA(rubric)
    assert d['validation_helper_sha256'] == SHA(G / 'qualified003_source_tools.py.txt')
    records.extend(d['records'])
assert len(records) == 45
assert [r['study_uid'] for r in records] == [r['study_uid'] for r in sources]
assert len({r['study_uid'] for r in records}) == 45
targets = ['ACL', 'MCL', 'Medial Meniscus', 'Lateral Meniscus', 'Medial OA', 'Lateral OA', 'PF OA', 'Effusion', 'Synovitis', "Baker's", 'Contusion', 'Fracture']
counts = Counter()
for rec, src in zip(records, sources):
    p = R / src['source_path']
    assert SHA(p) == src['source_sha256'] == rec['source_sha256']
    assert rec['source_path'] == src['source_path']
    assert rec['source_index_zero_based'] == src['index']
    assert read(p)[src['index']] == dict(study_uid=src['study_uid'], report=src['report'])
    assert hashlib.sha256(src['report'].encode()).hexdigest() == rec['report_sha256']
    assert rec['target_order'] == targets and len(rec['states']) == len(rec['evidence']) == 12
    for state, quote in zip(rec['states'], rec['evidence']):
        assert state in {'positive', 'negative', 'uncertain', 'not_mentioned'}
        assert (state == 'not_mentioned' and quote == '') or (state != 'not_mentioned' and quote and quote in src['report'])
        counts[state] += 1
all_sources = [r for p in sorted((R / 'initiative/reviews/inputs').glob('batch-*.json')) for r in read(p)]
assert len(all_sources) == len({r['study_uid'] for r in all_sources}) == 4407
all_sources.sort(key=lambda r: r['study_uid'])
fingerprint = hashlib.sha256(json.dumps(all_sources, ensure_ascii=False, sort_keys=True, separators=(',', ':')).encode()).hexdigest()
assert fingerprint == '5c18838a09e9eec6522966555694fca83a9bae67e4d7c5c68592207ee4b60ed3'
for p in before:
    assert SHA(p) == before[p]
now = datetime.now(timezone.utc).isoformat()
full = dict(stage='complete_qualified003_root_source_assessment_before_current_reader_findings', agent_id='/root', session_id='/root', at_utc=now, full_reports=45, target_judgments=540, target_order=targets, records=records, notes='These are actual personally authored source-first judgments. They are not final adjudications or proof of clinical diagnosis. Initial originals, including interpretations later reconsidered, must remain unchanged.', current003_reader_case_findings_opened=False, training_ready=False)
with (G / 'full-source-assessment.json').open('x') as f:
    f.write(json.dumps(full, ensure_ascii=False, indent=2) + '\n')
helper = G / 'freeze-qualified003-root-v1.py.txt'
with helper.open('xb') as f:
    f.write(Path(__file__).read_bytes())
freeze = dict(stage='complete_root_qualified003_source_freeze_v1', agent_id='/root', session_id='/root', at_utc=now, reports=45, target_judgments=540, nonmissing_exact_quotes=540-counts['not_mentioned'], empty_not_mentioned_quotes=counts['not_mentioned'], state_counts=dict(counts), full_source_assessment=pin(G / 'full-source-assessment.json'), neutral_index=pin(index), neutral_packets=read(index)['packets'], rubric=pin(rubric), canonical_source_sha256=fingerprint, canonical_sources_recomputed=4407, partial_assessments=[pin(G / (name + '-source-assessment.json')) for name, _ in PARTS], existing_files_preserved=[pin(p) for p in sorted(before)], helper=pin(helper), history='Prior root full4407 work, wording000/001/002, cartilage000-005 and technical/image work include possible historical overlaps. This is source-before-current-wording003 comparison, not first-ever blindness.', personally_read_full_sources_before_current_case_findings=True, current003_reader_case_findings_opened=False, current003_selected_registry_rows_opened=False, gold_values_opened=False, states_or_evidence_changed=False, registry_changed=False, qa_gate_closed=False, training_ready=False)
with (G / 'source-freeze.json').open('x') as f:
    f.write(json.dumps(freeze, ensure_ascii=False, indent=2) + '\n')
for p in before:
    assert SHA(p) == before[p]
print(json.dumps(dict(status='complete_source_freeze', reports=45, target_judgments=540, full_source_sha256=SHA(G/'full-source-assessment.json'), source_freeze_sha256=SHA(G/'source-freeze.json'), state_counts=dict(counts), current_reader_findings_opened=False, training_ready=False)))
