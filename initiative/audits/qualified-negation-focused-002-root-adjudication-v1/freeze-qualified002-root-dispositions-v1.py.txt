def sha(p):
    return hashlib.sha256(p.read_bytes()).hexdigest()

def ref(p):
    return {'path': str(p.relative_to(W)), 'sha256': sha(p), 'bytes': p.stat().st_size}

def read(p, pin=None):
    if pin:
        assert sha(p) == pin, p
    return json.loads(p.read_text())

def write(name, doc):
    p = D / name
    with p.open('x') as f:
        f.write(json.dumps(doc, ensure_ascii=False, indent=2) + '\n')
    return ref(p)
