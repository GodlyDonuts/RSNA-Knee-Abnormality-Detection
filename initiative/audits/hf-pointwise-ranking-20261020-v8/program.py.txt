def paid_unique_npz(roots):
    identities = {}
    for root in roots:
        for p in Path(root).rglob("*.npz"):
            try:
                st = p.lstat()
            except FileNotFoundError:
                continue  # A missing transient file grants no paid-byte credit.
            assert stat.S_ISREG(st.st_mode) and not p.is_symlink() and st.st_uid == os.getuid()
            identities[(st.st_dev, st.st_ino)] = st.st_size
    return sum(identities.values())

@contextlib.contextmanager
def serialized_milestone_locks(pub):
    # Reuse the existing eight-inode canonical account transient envelope.
    # Holding its claim prevents an account phase from consuming that scope.
    claim = Path("/home/sa305415/rsna-knee/kaggle/coordinator-claim")
    owner = {"kind": "private_HF_source_milestone_v8", "CPU_job": os.environ["SLURM_JOB_ID"],
             "pid": os.getpid(), "nonce": uuid.uuid4().hex}
    claim.mkdir(mode=0o700)  # Existing claim: stop without a publication intent.
    marker = claim / "owner.json"
    try:
        pub.atomic_new(marker, owner)
        runner = pub.load_pinned(pub.LOCK_RUNNER, pub.LOCK_RUNNER_SHA, "v8_pinned_shared_lock")
        with runner.publication_lock(pub.PUBLICATION_LOCK, timeout=30, poll=1):
            owned = [claim, marker, pub.PUBLICATION_LOCK, pub.PUBLICATION_LOCK / "owner.json"]
            stats = [p.lstat() for p in owned]
            assert all(not p.is_symlink() and st.st_uid == os.getuid() for p, st in zip(owned, stats))
            assert len({(st.st_dev, st.st_ino) for st in stats}) == 4
            hf_owner = json.loads(owned[-1].read_text())
            assert hf_owner["pid"] == os.getpid() and hf_owner["job_id"] == os.environ["SLURM_JOB_ID"]
            ledger = claim.parent / "account-ledger.sqlite3"
            assert not any(Path(str(ledger) + suffix).exists() for suffix in ("-journal", "-wal", "-shm"))
            q = subprocess.check_output(["lfs", "quota", "-u", "sa305415", "/home/sa305415"], text=True)
            a = next(line.split() for line in q.splitlines() if line.strip().startswith("/home/sa305415"))
            free_files = int(a[6]) - int(a[5])
            account = shared_file_accounting(free_files, len(owned))
            assert (int(a[2]) - int(a[1])) * 1024 > 20 * 2**30 + 123 * 2**20
            yield account
    finally:
        if marker.exists():
            assert not claim.is_symlink() and not marker.is_symlink() and json.loads(marker.read_text()) == owner
            marker.unlink()
        claim.rmdir()


def shared_file_accounting(free_files, owned_current_files):
    assert type(free_files) is int and type(owned_current_files) is int
    assert owned_current_files == 4 and owned_current_files <= 8
    remaining = 376 - owned_current_files
    assert free_files >= 5000 + remaining, "Shared inode reserve after paid owned locks is insufficient"
    return {"standing_future_shared_files": 376, "existing_canonical_transient_envelope_files": 8,
            "actually_owned_transient_files_already_charged": owned_current_files,
            "remaining_future_shared_files": remaining, "actual_free_files_with_locks": free_files,
            "files_after_all_remaining_liabilities": free_files - remaining,
            "new_permanent_shared_files": 0, "new_shared_byte_reservation": 0,
            "canonical_account_claim_held_through_HF_lock": True}
