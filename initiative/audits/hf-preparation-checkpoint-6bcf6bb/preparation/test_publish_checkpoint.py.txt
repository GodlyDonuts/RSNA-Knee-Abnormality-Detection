def ref(name, data=b"abc"):
    return {
        "path": name,
        "sha256": hashlib.sha256(data).hexdigest(),
        "bytes": len(data),
    }

def test_same_size_changed_bytes_and_symlink_rejected(tmp_path):
    p = tmp_path / "file"
    p.write_bytes(b"abc")
    m.verify_file(p, ref("file"))
    p.write_bytes(b"abd")
    with pytest.raises(m.GuardError, match="sha256"):
        m.verify_file(p, ref("file"))
    link = tmp_path / "link"
    link.symlink_to(p)
    with pytest.raises(m.GuardError, match="symlink"):
        m.verify_file(link, ref("file", b"abd"))


def test_duplicate_or_malformed_operation_refs_rejected():
    with pytest.raises(m.GuardError, match="duplicate"):
        m.validate_refs([ref("x"), ref("x")])
    with pytest.raises(m.GuardError, match="ref_size"):
        m.validate_refs([{**ref("x"), "bytes": True}])

        m.validate_refs([ref("x")], 3)

    request = httpx.Request("GET", "https://example.invalid")

        "write": 999,

    m.bounded_request(request)
    assert request.extensions["timeout"] == {
        "connect": 15,
        "read": 120,
        "write": 120,
        "pool": 4,
    }


def test_hub_negotiates_http_only_in_fresh_process(tmp_path):
    # This process imports no network fixtures. The child installs a MockTransport
    # to demonstrate timeout=None is clamped on actual httpx request construction.
    code = r"""
import importlib.util,sys,tempfile
from pathlib import Path
spec=importlib.util.spec_from_file_location('recovery',sys.argv[1]);m=importlib.util.module_from_spec(spec);spec.loader.exec_module(m)
hub=m.configure_hub()
from huggingface_hub import _commit_api as c
from huggingface_hub.utils._runtime import is_xet_available
assert is_xet_available() is False
offered=[]
def batch(**kw):
 offered.append(kw['transfers']);return [],[],"basic"
c.post_lfs_batch_info=batch
def forbidden(**kw):raise AssertionError('xet must not run')
c._upload_xet_files=forbidden
p=Path(sys.argv[2]);p.write_bytes(b'unchanged fixture')
c._upload_files(additions=[hub.CommitOperationAdd(path_in_repo='fixture',path_or_fileobj=p)],repo_type='dataset',repo_id='test/repo',headers={})
assert offered==[['basic','multipart']]
import httpx
seen=[]
def transport(req):seen.append(req.extensions['timeout']);return httpx.Response(200,json={})
with httpx.Client(transport=httpx.MockTransport(transport),event_hooks={'request':[m.bounded_request]}) as client:
 client.get('https://example.invalid',timeout=None)
assert seen==[m.HTTP_TIMEOUTS]
print('http_only_negotiation_and_finite_timeout_verified')
"""
    result = subprocess.run(
        [sys.executable, "-c", code, str(SCRIPT), str(tmp_path / "fixture")],
        capture_output=True,
        text=True,
        timeout=30,
    )
    assert result.returncode == 0, result.stderr
    assert "http_only_negotiation_and_finite_timeout_verified" in result.stdout


def test_source_tree_and_five_csv_identity_all_required(monkeypatch):
    ids = {
        name: NS(
            path=name,
            **{("tree_id" if name in m.SOURCE_TREES else "blob_id"): "a" * 40},
        )
        for name in m.SOURCE_TREES + m.SOURCE_CSVS
    }
    api = NS(list_repo_tree=lambda *_args, **_kwargs: list(ids.values()))
    assert len(m.verify_source_unchanged(api, "c" * 40)) == 7

    def changed(_repo, revision, **_kwargs):
        rows = list(ids.values())
        if revision == "c" * 40:
            rows[-1] = NS(path="sample_submission.csv", blob_id="b" * 40)
        return rows

    api.list_repo_tree = changed
    with pytest.raises(m.GuardError, match="publication_changed_source"):
        m.verify_source_unchanged(api, "c" * 40)
    api.list_repo_tree = lambda *_args, **_kwargs: list(ids.values())[:-1]
    with pytest.raises(m.GuardError, match="source_entries_missing"):
        m.source_identity(api, m.PARENT)


@pytest.mark.parametrize(
    "damage", [None, "missing", "duplicate", "size", "oid", "bytes"]
)
def test_remote_complete_membership_lfs_and_ordinary_hashes(monkeypatch, damage):
    from huggingface_hub.utils import _http

    refs = [ref("large.bin"), ref("README.md", b"document")]
    entries = [
        NS(path="large.bin", size=3, lfs=NS(sha256=refs[0]["sha256"])),
        NS(path="README.md", size=8, lfs=None),
    ]
    if damage == "missing":
        entries.pop()
    if damage == "duplicate":
        entries.append(entries[0])
    if damage == "size":
        entries[0].size = 4
    if damage == "oid":
        entries[0].lfs.sha256 = "0" * 64
    payload = b"changed!" if damage == "bytes" else b"document"
    client = httpx.Client(
        transport=httpx.MockTransport(lambda _req: httpx.Response(200, content=payload))
    )
    monkeypatch.setattr(_http, "get_session", lambda: client)
    api = NS(get_paths_info=lambda *_args, **_kwargs: entries)
    hub = NS(hf_hub_url=lambda *_args, **_kwargs: "https://example.invalid/file")
    if damage is None:
        result = m.verify_remote(api, hub, "fixture-token", "c" * 40, refs)
        assert [r["verification"] for r in result] == [
            "hub_lfs_sha256_and_size",
            "full_http_bytes_sha256_and_size",
        ]
    else:
        with pytest.raises(m.GuardError):
            m.verify_remote(api, hub, "fixture-token", "c" * 40, refs)
    client.close()

@pytest.mark.parametrize(
    "failure", ["ci", "private", "head_race", "upload_info", "remote_mismatch"]
)
def test_guards_fail_without_commit_or_success_receipt(
    monkeypatch, publication_fixture, failure
):
    api, hub, operation, journal, out = publication_fixture
    if failure == "ci":

        def bad_ci():
            raise m.GuardError("exact_commit_ci_not_success")

        monkeypatch.setattr(m, "ci_guard", bad_ci)
    elif failure == "private":
        api.repo_info = lambda *_args, **_kwargs: NS(private=False, sha=api.head)
    elif failure == "head_race":
        sequence = iter([m.PARENT, "d" * 40])
        api.repo_info = lambda *_args, **_kwargs: NS(private=True, sha=next(sequence))
    elif failure == "upload_info":
        hub.CommitOperationAdd = lambda **_kw: NS(
            upload_info=NS(size=3, sha256=b"wrong")
        )
    else:
        api.head = "c" * 40

        def mismatch(*_args):
            raise m.GuardError("remote_sha256_mismatch")

        monkeypatch.setattr(m, "verify_remote", mismatch)
    with pytest.raises(m.GuardError):
        m.publish(api, hub, "fixture", operation, journal, out)
    assert api.commit_calls == []
    assert not (out / "publication-receipt.json").exists()

@pytest.mark.parametrize("free_delta", [-1, 0, 1])
def test_space_preflight_preserves_floor_and_reserves(monkeypatch, free_delta):
    package_size = 1_203_620_292
    needed = m.FLOOR_BYTES + m.OVERHEAD_BYTES + m.ROOT_RESERVE_BYTES + package_size
    monkeypatch.setattr(m.shutil, "disk_usage", lambda _: NS(free=needed + free_delta))
    if free_delta < 0:
        with pytest.raises(m.GuardError, match="disk_floor"):
            m.disk_guard(package_size)
    else:
        assert m.disk_guard(package_size)["additional_bytes"] == package_size

@pytest.mark.parametrize(
    "damage",
    [None, "changed", "missing", "extra", "duplicate", "git_blob", "mode", "secret"],
)
def test_tar_exact_pinned_blobs_and_membership(monkeypatch, tmp_path, damage):
    monkeypatch.setattr(m, "EXPECTED_BATCHES", 0)
    monkeypatch.setattr(m, "EXPECTED_REVIEWED", 0)
    payloads = [("rsna/a", b"abc"), ("rsna/b", b"def")]
    refs = [ref(n, d) for n, d in payloads]
    pinned = [
        {
            "path": n,
            "git_blob": hashlib.sha1(
                b"blob " + str(len(d)).encode() + b"\0" + d
            ).hexdigest(),
            "bytes": len(d),
            "mode": 0o644,
        }
        for n, d in payloads
    ]
    if damage == "changed":
        payloads[0] = ("rsna/a", b"abd")
    elif damage == "missing":
        payloads.pop()
    elif damage == "extra":
        payloads.append(("rsna/c", b"ghi"))
    elif damage == "duplicate":
        payloads.append(payloads[0])
    elif damage == "git_blob":
        pinned[0]["git_blob"] = "0" * 40
    p = tmp_path / "package.tar.gz"
    with tarfile.open(p, "w:gz") as tar:
        for name, data in payloads:
            member = tarfile.TarInfo(name)
            member.size = len(data)
            member.mode = 0o755 if damage == "mode" else 0o644
            tar.addfile(member, io.BytesIO(data))
    if damage is None:
        assert m.validate_tar(p, refs, pinned)["all_member_git_blobs_verified"] is True
    else:
        with pytest.raises(m.GuardError):
            m.validate_tar(p, refs, pinned, [b"abc"] if damage == "secret" else [])

def test_builder_stops_before_creating_package_if_disk_insufficient(
    monkeypatch, tmp_path
):
    spec = importlib.util.spec_from_file_location(
        "checkpoint_builder", SCRIPT.with_name("build_package.py")
    )
    builder = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(builder)
    monkeypatch.setattr(builder, "STAGE", tmp_path)
    monkeypatch.setattr(builder.m, "PACKAGE", tmp_path)
    monkeypatch.setattr(builder.m, "load_plan", lambda _: {})

    (tmp_path / "pinned-package-measurement.json").write_text(
        json.dumps({"files": [], "package": {"bytes": 100}})
    )

        raise builder.m.GuardError("disk_floor_would_be_breached")

    monkeypatch.setattr(builder.m, "disk_guard", blocked)

        builder.subprocess,

        builder.build("a" * 64)
    assert not (tmp_path / "checkpoint.tar.gz.partial").exists()


def test_guarded_writer_checks_floor_and_measured_size_before_write(monkeypatch):
    spec = importlib.util.spec_from_file_location(
        "checkpoint_builder", SCRIPT.with_name("build_package.py")
    )
    builder = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(builder)
    calls = []
    monkeypatch.setattr(builder.m, "disk_guard", lambda n: calls.append(n))
    stream = io.BytesIO()
    writer = builder.GuardedWriter(stream, 3)
    assert writer.write(b"abc") == 3
    with pytest.raises(builder.m.GuardError, match="compressed_size"):
        writer.write(b"x")
    assert stream.getvalue() == b"abc" and calls == [3]
