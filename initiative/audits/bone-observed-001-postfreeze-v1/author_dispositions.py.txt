"""Assemble personally authored bone001 decisions; no clinical rule engine."""
import hashlib
import json
from datetime import datetime, timezone
from pathlib import Path

P = Path(__file__).resolve().parent
W = P.parents[2]
S = P.parent / 'bone-observed-001-source-review-v1'
AGENT = '/root/quality_temporal_review'

def read(p):
    return json.loads(p.read_text())

def sha(p):
    return hashlib.sha256(p.read_bytes()).hexdigest()

def bind(p):
    return {'path': str(p.relative_to(W)), 'sha256': sha(p), 'bytes': p.stat().st_size}

def write(name, obj):
    path = P / name
    path.parent.mkdir(exist_ok=True)
    with path.open('x') as f:
        f.write(json.dumps(obj, ensure_ascii=False, indent=2) + '\n')

def main():
    assert sha(S / 'source-freeze.json') == 'a0229dfab840558ab485f60b8af038532c5cf61ae8fb6c23a93e3cde7cefab0c'
    c = read(P / 'all100-bound-comparison.json')
    assert c['registered_changes'] == 205
    sources = read(S / 'neutral-source-packet.json')['reports']
    protected = [bind(p) for p in sorted(S.iterdir()) if p.is_file()]
    prior = P.parent / 'bone-observed-000-independent-postfreeze-v1/completion-manifest.json'
    protected.append(bind(prior))
    write('protected-prior-bindings.json', {'schema_version': 1, 'agent_id': AGENT, 'files': protected, 'prior_stages_must_remain_unchanged': True})
    rows, proposals, reconsiderations = [], [], []
    for old in c['records']:
        key = old['ordinal'], old['target']
        own = old['own_source_frozen']
        state, quote = own['state'], own['evidence']
        reconsidered = key == (26, 'Fracture')
        if reconsidered:
            state = 'uncertain'
            quote = 'Lateral femoral kondilin santral kesimindeki subkortikal hafif kemik iliği ödemi gerilemekte olan kontüzyonu temsil edebilir. Eklem kıkırdakları ve kemikler bunun dışında normal.'
        reason = REASONS.get(key, 'I retain my personally authored source state after comparing the complete source, original review and selected current annotation. ' + own['rationale'])
        if old['current_accepted_correction']:
            reason += ' The existing accepted correction already supports this state and remains untouched; my source quotation is preserved as assessment evidence, not an additional evidence-change proposal.'
        source = sources[old['ordinal']]
        assert (state == 'not_mentioned' and quote == '') or (state != 'not_mentioned' and quote and quote in source['report'])
        state_change = state != old['current205_effective']['state']
        evidence_only = key == (44, 'Contusion')
        action = 'propose_state_change_pending_distinct_review' if state_change else 'propose_evidence_expansion_pending_distinct_review' if evidence_only else 'retain_current_mask' if state in {'uncertain', 'not_mentioned'} else 'retain_current_observed_state'
        row = {**old, 'first_postfreeze_judgment': {'state': state, 'evidence': quote, 'rationale': reason}, 'own_state_reconsidered': reconsidered, 'own_evidence_reconsidered': reconsidered, 'state_differs_from_current': state_change, 'evidence_only_proposal': evidence_only, 'action': action, 'current_report_observation_mask': int(old['current205_effective']['state'] in {'positive', 'negative'}), 'proposed_report_observation_mask': int(state in {'positive', 'negative'}), 'distinct_bone001_reader_comparison_done': False, 'acceptance_or_registration': False}
        rows.append(row)
        if reconsidered:
            reconsiderations.append(row)
        if state_change or evidence_only:
            assert old['current_accepted_correction'] is None
            proposals.append({'ordinal': old['ordinal'], 'study_uid': old['study_uid'], 'target': old['target'], 'source_path': old['source_path'], 'source_sha256': old['source_sha256'], 'source_report_sha256': old['report_sha256'], 'index': old['source_index_zero_based'], 'artifact_path': old['original_review_path'], 'artifact_sha256': old['original_review_sha256'], 'original_state': old['original']['state'], 'original_evidence': old['original']['evidence'], 'before_state': old['current205_effective']['state'], 'before_evidence': old['current205_effective']['evidence'], 'proposed_state': state, 'source_quote': quote, 'reason': reason, 'kind': 'state_correction' if state_change else 'evidence_correction', 'proposer_agent_id': AGENT, 'source_freeze': bind(S / 'source-freeze.json'), 'existing_accepted_key_overlap': False, 'status': 'pending_distinct_source_review_and_root_adjudication', 'accepted': False})
    common = {'schema_version': 1, 'agent_id': AGENT, 'session_id': AGENT, 'identity_kind': 'actual_codex_agent_task', 'completed_at_utc': datetime.now(timezone.utc).isoformat(), 'rubric_version': 'report-weak-v2.0', 'rubric_sha256': '138c5cc33b92901d57dbf5fbe7e3a9878b73d25b0af6934908d0052cd6786014', 'source_reports_sha256': c['source_reports_sha256'], 'source_freeze': bind(S / 'source-freeze.json'), 'source_assessment': bind(S / 'source-assessment-frozen.jsonl'), 'comparison': bind(P / 'all100-bound-comparison.json'), 'actual_registry': bind(P / 'registry-at-comparison.json'), 'actual_registered_changes': 205, 'historical_exposure_binding': bind(S / 'start.json'), 'exposure': 'The entire50-source100-bone assessment was personally frozen before selected current bone001 annotation access. This separate post-freeze stage personally compared all100 original/current cells, with no other bone001 reader case findings opened. It is not a newly blinded pass or MRI diagnostic reread.', 'other_bone001_reader_findings_opened': False, 'accepted_changes': [], 'registry_changed': False, 'gold_accessed': False, 'global_gate_closed': False}

        packets.append({**bind(P / name), 'n_reports': len(neutral[offset:offset + 10])})

    print(json.dumps({'counts': counts, 'freeze_sha256': sha(P / 'postfreeze-disposition-freeze.json'), 'neutral_manifest_sha256': sha(P / 'neutral-root-packet-manifest.json')}, indent=2))
