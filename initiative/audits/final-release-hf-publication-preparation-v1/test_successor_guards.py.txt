    def test_all_unknown_pins_explicitly_unset(self):
        for name in self.pins():
            self.assertIsNone(getattr(b, name), name)

    def test_all_operational_entrypoints_stop_before_helper_or_credentials(self):
        common = ['--target-commit','1'*40,'--target-tree','2'*40,'--expected-file-count','99',
                  '--package','unused','--manifest-sha256','a'*64]
        cases = [(builder, ['--actual-executor',b.PEER]),
                 (peer, common+['--actual-executor',b.PEER]),
                 (root, common+['--actual-executor',b.ROOT,'--peer-proof-sha256','a'*64,'--peer-build-receipt-sha256','b'*64]),
                 (publisher, ['prepare',*common,'--actual-executor',b.ROOT,'--ci-run','123',
                              '--peer-proof-sha256','a'*64,'--root-proof-sha256','b'*64,'--root-verifier-sha256','c'*64]),
                 (publisher, ['publish',*common,'--actual-executor',b.ROOT,'--ci-run','123',
                              '--peer-proof-sha256','a'*64,'--root-proof-sha256','b'*64,'--root-verifier-sha256','c'*64])]
        for module, args in cases:
            with self.subTest(module=module.__name__, args=args[0]), patch.object(sys,'argv',['synthetic',*args]), patch.object(b,'helper') as shared:
                with self.assertRaisesRegex(b.GuardError,'unset_or_historical_target_git'): module.main()
                shared.assert_not_called()

    def test_every_missing_pin_fails_closed(self):
        with patch.multiple(b, **self.pins()):
            b.future_pins_guard()
            for name in self.pins():
                with self.subTest(name=name), patch.object(b, name, None):
                    with self.assertRaises(b.GuardError): b.future_pins_guard()

    def test_bool_or_historical_or_escaping_pins_fail(self):
        with patch.multiple(b, **self.pins()):
            for name,value in [('TARGET_FILE_COUNT',True),('TARGET_CI_RUN',True),
                               ('TARGET_GIT',b.PREVIOUS_GIT),('ACCEPTANCE_PATH','initiative/audits/../escape/acceptance.json')]:
                with self.subTest(name=name),patch.object(b,name,value):
                    with self.assertRaises(b.GuardError):b.future_pins_guard()

    def test_exact_synthetic_release_eligible_without_changing_transport_scope(self):
        with patch.multiple(b, **self.pins()):
            fixture=self.fixture();result=e.evaluate(*fixture[:5])
            self.assertEqual(result['preserved_accepted_corrections'],569)
            self.assertFalse(result['competitive_accuracy_claim'])
            self.assertFalse(result['training_strategy_complete'])
            self.assertNotIn('training_ready',result)

            for mutate in [lambda a:a.update(status='running'),lambda a:a.update(status='failed'),
                           lambda a:a.update(actual_executor=b.PEER),lambda a:a.update(competitive_performance_claim=True),

                           lambda a:a['commands'].pop(),lambda a:a['commands'][0].update(exit_code=1),
                           lambda a:a['training_checks'].update(full_loaded_checkpoint_equal=False),
                           lambda a:a['training_checks'].update(completed_epochs=1),
                           lambda a:a['training_checks']['release_binding'].update(manifest_sha256='5'*64),
                           lambda a:a['training_checks']['run_config'].update(diagnostic_study_limit=None)]:
                f=self.fixture();mutate(f[4])
                with self.assertRaises(b.GuardError):e.evaluate(*f[:5])

        with patch.multiple(b, **self.pins()):
            f=self.fixture();f[2]['cases']=[]
            with self.assertRaisesRegex(b.GuardError,'prior_correction_cases_changed'):e.evaluate(*f[:5])
            f=self.fixture();doc=f[5]['case.json'];doc['changes'].pop()
            f[1]['cases'][0]['sha256']=f[2]['cases'][0]['sha256']=e.digest(doc)
            with self.assertRaisesRegex(b.GuardError,'preserved569'):e.evaluate(*f[:5])

    def test_fresh_ci_requires_actual_exact_commit_success(self):
        args=argparse.Namespace(ci_run=123,target_commit='1'*40)
        with patch.object(publisher,'TARGET_CI_RUN',123):
            for result in [{'status':'queued','headSha':args.target_commit,'conclusion':None},
                           {'status':'completed','headSha':b.PREVIOUS_GIT,'conclusion':'success'}]:
                with patch.object(publisher.subprocess,'check_output',return_value=json.dumps(result).encode()):
                    with self.assertRaises(b.GuardError):publisher.ci(args)
            result={'status':'completed','headSha':args.target_commit,'conclusion':'success'}
            with patch.object(publisher.subprocess,'check_output',return_value=json.dumps(result).encode()):
                self.assertEqual(publisher.ci(args),result)

    def test_restoration_must_be_actual_distinct_executor(self):
        args=argparse.Namespace(target_commit='1'*40,target_tree='2'*40,manifest_sha256='a'*64,expected_file_count=99)
        manifest={'delta_archive':{'sha256':'b'*64,'bytes':1},'secret_scan_policy':{},'files':[],
                  'n_delta_files':2,'n_base_archive_files':3,'deleted_paths':[]}
        proof={'verifier_agent_id':b.PEER,'actual_executor':b.PEER,'status':'every_target_git_file_restorable_identically',
               'target_git_commit':args.target_commit,'target_git_tree':args.target_tree,'manifest':{'sha256':'a'*64},
               'n_result_files_verified':99,'delta_archive':manifest['delta_archive'],'base_git_commit':b.BASE_GIT,
               'base_hf_revision':b.BASE_HF,'base_archive_sha256':b.BASE_ARCHIVE_SHA,'base_manifest_sha256':b.BASE_MANIFEST_SHA,
               'every_file_sha256_git_blob_size_mode_verified':True,'membership_exact':True,'full_tree_extracted':False,
               'training_ready':False,'helper_sha256':b.HELPER_SHA,'secret_scan_policy':{},'verified_secret_scan_exception_paths':[],
               'n_delta_members_verified':2,'n_base_archive_members_verified':3,'n_deletions_applied':0}
        b.proof_guard(proof,args,manifest,b.PEER)
        with self.assertRaisesRegex(b.GuardError,'actual_executor_mismatch'):b.proof_guard(proof,args,manifest,b.ROOT)
        proof['training_ready']=True
        with self.assertRaisesRegex(b.GuardError,'proof_scope'):b.proof_guard(proof,args,manifest,b.PEER)
