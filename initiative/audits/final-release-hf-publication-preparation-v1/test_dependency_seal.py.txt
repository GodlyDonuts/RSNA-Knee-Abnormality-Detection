    def test_actual_read_capture_binds_nested_input_and_blocks_mutation(self):
        with tempfile.TemporaryDirectory(dir=b.WORK)as tmp:
            root=Path(tmp);p=root/'proof.json';p.write_text('{"current":true}')
            capture=s.InputCapture(root);sys.addaudithook(capture.hook)
            try:
                capture.active=True;p.read_text();self.assertIn('proof.json',capture.files)
                with self.assertRaisesRegex(b.GuardError,'runtime_attempted_repository_write'):p.write_text('changed')
            finally:capture.active=False
            p.write_text('changed')
            with self.assertRaisesRegex(b.GuardError,'seal_input_changed_during_validation'):capture.unchanged()
    def test_capture_rejects_escape_and_symlink(self):
        with tempfile.TemporaryDirectory(dir=b.WORK)as tmp:
            root=Path(tmp);p=root/'inside';p.write_text('value');link=root/'link';link.symlink_to(p)
            capture=s.InputCapture(root)
            with self.assertRaisesRegex(b.GuardError,'seal_input_scope_or_symlink'):capture.add(link)
            with self.assertRaisesRegex(b.GuardError,'seal_input_scope_or_symlink'):capture.add(root/'..'/'no-file')
