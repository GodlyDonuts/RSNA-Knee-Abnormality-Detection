def ref(name, data=b"abc"):
    return {
        "path": name,
        "sha256": hashlib.sha256(data).hexdigest(),
        "bytes": len(data),
    }

def test_exact_mapping_preserves_package_and_freezes_only_three_policies(tmp_path):
    package, repo, frozen = [tmp_path / n for n in ("package", "repo", "frozen")]
    assert (
        m.local_path(m.PREFIX + "checkpoint.tar.gz", package, repo, frozen)
        == package / "checkpoint.tar.gz"
    )
    assert (
        m.local_path("README.md", package, repo, frozen) == package / "dataset-card.md"
    )
    assert (
        m.local_path("audits/case/proof.json", package, repo, frozen)
        == repo / "initiative/audits/case/proof.json"
    )
    for path in m.FROZEN_POLICIES:
        assert m.local_path(path, package, repo, frozen) == frozen / path
    for path in ("train.csv", "train_series/x.dcm", m.PREFIX + "other.txt"):
        with pytest.raises(m.GuardError, match="outside_frozen_mapping"):
            m.local_path(path, package, repo, frozen)


def test_same_size_changed_bytes_and_symlink_rejected(tmp_path):
    p = tmp_path / "file"
    p.write_bytes(b"abc")
    m.verify_file(p, ref("file"))
    p.write_bytes(b"abd")
    with pytest.raises(m.GuardError, match="sha256"):
        m.verify_file(p, ref("file"))
    link = tmp_path / "link"
    link.symlink_to(p)
    with pytest.raises(m.GuardError, match="symlink"):
        m.verify_file(link, ref("file", b"abd"))

    with pytest.raises(m.GuardError, match="ref_size"):

        m.validate_refs([ref("x")], 490)


@pytest.mark.parametrize("mode", ["valid", "changed", "missing", "duplicate", "extra"])
def test_tar_integrity_exact_membership_and_payloads(tmp_path, mode):
    p = tmp_path / "frozen.tar.gz"
    members = [("a", b"abc"), ("b", b"def")]
    expected = [ref(name, data) for name, data in members]
    if mode == "changed":
        members[0] = ("a", b"abd")
    if mode == "missing":
        members.pop()
    if mode == "duplicate":
        members.append(members[0])
    if mode == "extra":
        members.append(("c", b"ghi"))
    with tarfile.open(p, "w:gz") as archive:
        for name, data in members:
            member = tarfile.TarInfo(name)
            member.size = len(data)
            archive.addfile(member, io.BytesIO(data))
    before = p.read_bytes()
    if mode == "valid":
        assert m.validate_tar(p, expected) == 2
    else:
        with pytest.raises(m.GuardError):
            m.validate_tar(p, expected)
    assert p.read_bytes() == before

def test_hub_negotiates_http_only_in_fresh_process(tmp_path):
    # This process imports no network fixtures. The child installs a MockTransport
    # to demonstrate timeout=None is clamped on actual httpx request construction.
    code = r"""

from pathlib import Path
spec=importlib.util.spec_from_file_location('recovery',sys.argv[1]);m=importlib.util.module_from_spec(spec);spec.loader.exec_module(m)

from huggingface_hub import _commit_api as c

offered=[]
def batch(**kw):
 offered.append(kw['transfers']);return [],[],"basic"
c.post_lfs_batch_info=batch
def forbidden(**kw):raise AssertionError('xet must not run')
c._upload_xet_files=forbidden
p=Path(sys.argv[2]);p.write_bytes(b'unchanged fixture')
c._upload_files(additions=[hub.CommitOperationAdd(path_in_repo='fixture',path_or_fileobj=p)],repo_type='dataset',repo_id='test/repo',headers={})
assert offered==[['basic','multipart']]
import httpx
seen=[]
def transport(req):seen.append(req.extensions['timeout']);return httpx.Response(200,json={})
with httpx.Client(transport=httpx.MockTransport(transport),event_hooks={'request':[m.bounded_request]}) as client:
 client.get('https://example.invalid',timeout=None)

    result = subprocess.run(
        [sys.executable, "-c", code, str(SCRIPT), str(tmp_path / "fixture")],
        capture_output=True,
        text=True,
        timeout=30,
    )
    assert result.returncode == 0, result.stderr
    assert "http_only_negotiation_and_finite_timeout_verified" in result.stdout


def test_source_tree_and_five_csv_identity_all_required(monkeypatch):
    ids = {
        name: NS(
            path=name,
            **{("tree_id" if name in m.SOURCE_TREES else "blob_id"): "a" * 40},
        )
        for name in m.SOURCE_TREES + m.SOURCE_CSVS
    }
    api = NS(list_repo_tree=lambda *_args, **_kwargs: list(ids.values()))
    assert len(m.verify_source_unchanged(api, "c" * 40)) == 7

    def changed(_repo, revision, **_kwargs):
        rows = list(ids.values())
        if revision == "c" * 40:
            rows[-1] = NS(path="sample_submission.csv", blob_id="b" * 40)
        return rows

    api.list_repo_tree = changed
    with pytest.raises(m.GuardError, match="publication_changed_source"):
        m.verify_source_unchanged(api, "c" * 40)
    api.list_repo_tree = lambda *_args, **_kwargs: list(ids.values())[:-1]
    with pytest.raises(m.GuardError, match="source_entries_missing"):
        m.source_identity(api, m.PARENT)


@pytest.mark.parametrize(
    "damage", [None, "missing", "duplicate", "size", "oid", "bytes"]
)
def test_remote_complete_membership_lfs_and_ordinary_hashes(monkeypatch, damage):
    from huggingface_hub.utils import _http

    refs = [ref("large.bin"), ref("README.md", b"document")]
    entries = [
        NS(path="large.bin", size=3, lfs=NS(sha256=refs[0]["sha256"])),
        NS(path="README.md", size=8, lfs=None),
    ]
    if damage == "missing":
        entries.pop()
    if damage == "duplicate":
        entries.append(entries[0])
    if damage == "size":
        entries[0].size = 4
    if damage == "oid":
        entries[0].lfs.sha256 = "0" * 64
    payload = b"changed!" if damage == "bytes" else b"document"
    client = httpx.Client(
        transport=httpx.MockTransport(lambda _req: httpx.Response(200, content=payload))
    )
    monkeypatch.setattr(_http, "get_session", lambda: client)
    api = NS(get_paths_info=lambda *_args, **_kwargs: entries)
    hub = NS(hf_hub_url=lambda *_args, **_kwargs: "https://example.invalid/file")
    if damage is None:
        result = m.verify_remote(api, hub, "fixture-token", "c" * 40, refs)
        assert [r["verification"] for r in result] == [
            "hub_lfs_sha256_and_size",
            "full_http_bytes_sha256_and_size",
        ]
    else:
        with pytest.raises(m.GuardError):
            m.verify_remote(api, hub, "fixture-token", "c" * 40, refs)
    client.close()

@pytest.mark.parametrize(
    "failure", ["ci", "private", "head_race", "upload_info", "remote_mismatch"]
)
def test_guards_fail_without_commit_or_success_receipt(
    monkeypatch, recovery_fixture, failure
):
    api, hub, operation, journal, out = recovery_fixture
    if failure == "ci":

        def bad_ci():
            raise m.GuardError("exact_commit_ci_not_success")

        monkeypatch.setattr(m, "ci_guard", bad_ci)
    elif failure == "private":
        api.repo_info = lambda *_args, **_kwargs: NS(private=False, sha=api.head)
    elif failure == "head_race":
        sequence = iter([m.PARENT, "d" * 40])
        api.repo_info = lambda *_args, **_kwargs: NS(private=True, sha=next(sequence))
    elif failure == "upload_info":
        hub.CommitOperationAdd = lambda **_kw: NS(
            upload_info=NS(size=3, sha256=b"wrong")
        )
    else:
        api.head = "c" * 40

        def mismatch(*_args):
            raise m.GuardError("remote_sha256_mismatch")

        monkeypatch.setattr(m, "verify_remote", mismatch)
    with pytest.raises(m.GuardError):
        m.recover(api, hub, "fixture", operation, journal, out)
    assert api.commit_calls == []
    assert not (out / "publication-recovery-receipt.json").exists()

def test_main_suppresses_sensitive_external_exception(monkeypatch, tmp_path, capsys):

        sys,
