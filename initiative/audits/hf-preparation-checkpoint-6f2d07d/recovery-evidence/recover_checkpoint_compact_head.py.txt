class GuardError(Exception):
    """Only a fixed, nonsensitive code may be used as the exception argument."""


def require(value, code):
    if not value:
        raise GuardError(code)


def now():
    return dt.datetime.now(dt.timezone.utc).isoformat()


def digest_stream(stream):
    h = hashlib.sha256()
    size = 0
    for block in iter(lambda: stream.read(1024 * 1024), b""):
        h.update(block)
        size += len(block)
    return h.hexdigest(), size


def file_ref(path):
    with path.open("rb") as stream:
        digest, size = digest_stream(stream)
    return {"path": str(path), "sha256": digest, "bytes": size}


def verify_file(path, expected):
    require(path.is_file() and not path.is_symlink(), "local_missing_or_symlink")
    ref = file_ref(path)
    require(ref["sha256"] == expected["sha256"], "local_sha256_mismatch")
    if "bytes" in expected:
        require(ref["bytes"] == expected["bytes"], "local_size_mismatch")
    return ref


def safe_remote_path(value):
    require(isinstance(value, str), "remote_path_type")
    p = PurePosixPath(value)
    require(
        not p.is_absolute()
        and str(p) == value
        and all(part not in {".", ".."} for part in p.parts)
        and "\\" not in value,
        "remote_path_unsafe",
    )
    return p


def local_path(remote, package=PACKAGE, repo=REPO, frozen=STAGE / "frozen-mutable"):
    """Exact inverse of the original publisher's frozen files_to_upload mapping."""
    safe_remote_path(remote)
    if remote in {
        PREFIX + n for n in ("checkpoint.tar.gz", "manifest.json", "README.md")
    }:
        return package / remote.rsplit("/", 1)[1]
    if remote == "README.md":
        return package / "dataset-card.md"
    if remote in FROZEN_POLICIES:
        return frozen / remote
    if remote.startswith("audits/"):
        path = repo / "initiative" / remote
        require(
            path.resolve().is_relative_to((repo / "initiative/audits").resolve()),
            "audit_path_escape",
        )
        return path
    raise GuardError("remote_path_outside_frozen_mapping")


def validate_refs(refs, expected_count=None):
    require(isinstance(refs, list) and bool(refs), "refs_not_nonempty_list")
    paths = []
    for ref in refs:
        require(
            isinstance(ref, dict) and set(ref) == {"path", "sha256", "bytes"},
            "ref_schema",
        )
        safe_remote_path(ref["path"])
        require(
            isinstance(ref["sha256"], str) and SHA256.fullmatch(ref["sha256"]),
            "ref_digest",
        )
        require(type(ref["bytes"]) is int and ref["bytes"] >= 0, "ref_size")
        paths.append(ref["path"])
    require(len(paths) == len(set(paths)), "duplicate_remote_path")
    if expected_count is not None:
        require(len(paths) == expected_count, "remote_object_count")


def validate_tar(path, refs):
    validate_refs(refs)
    expected = {r["path"]: r for r in refs}
    seen = set()
    with tarfile.open(path, "r|gz") as archive:
        for member in archive:
            require(
                member.isfile() and member.name in expected, "unexpected_tar_member"
            )
            require(member.name not in seen, "duplicate_tar_member")
            seen.add(member.name)
            stream = archive.extractfile(member)
            require(stream is not None, "missing_tar_payload")
            with stream:
                sha, size = digest_stream(stream)
            require(
                size == member.size == expected[member.name]["bytes"], "tar_member_size"
            )
            require(sha == expected[member.name]["sha256"], "tar_member_sha256")
    require(seen == set(expected), "missing_tar_members")
    return len(seen)

def original_stopped():
    try:
        os.kill(ORIGINAL_PID, 0)
    except ProcessLookupError:
        pass
    else:
        raise GuardError("original_pid_still_exists")
    # The fixed PID is necessary but also detect a separately restarted publisher.
    lines = subprocess.check_output(
        ["ps", "-axo", "pid=,command="],
        text=True,
        timeout=15,
        stderr=subprocess.DEVNULL,
    ).splitlines()
    for line in lines:
        parts = line.strip().split(maxsplit=1)
        if len(parts) == 2 and parts[0] != str(os.getpid()):
            require(
                "publish_review3607_checkpoint.py" not in parts[1],
                "another_original_publisher_exists",
            )


def bounded_request(request):
    """Clamp even explicit timeout=None passed by repo_info and redirects."""
    prior = request.extensions.get("timeout", {})
    bounded = {}
    for key, cap in HTTP_TIMEOUTS.items():
        value = prior.get(key)
        bounded[key] = (
            min(float(value), cap)
            if isinstance(value, (int, float)) and math.isfinite(value) and value > 0
            else cap
        )
    request.extensions["timeout"] = bounded


def configure_hub():
    # No Hub imports are allowed before these environment values are fixed.
    require("huggingface_hub.constants" not in sys.modules, "hub_already_imported")
    os.environ.update(
        {
            "HF_HUB_DISABLE_XET": "1",
            "HF_HUB_DISABLE_PROGRESS_BARS": "1",
            "HF_HUB_DISABLE_TELEMETRY": "1",
            "HF_DEBUG": "0",
            "HF_HUB_DOWNLOAD_TIMEOUT": "120",
            "HF_HUB_ETAG_TIMEOUT": "30",
        }
    )
    for name in ("huggingface_hub", "httpx", "httpcore", "hf_xet"):
        logging.getLogger(name).setLevel(logging.CRITICAL)
    spec = importlib.util.find_spec("huggingface_hub")
    require(spec is not None and spec.origin is not None, "hub_missing")
    hub_root = Path(spec.origin).parent
    for name, sha in HUB_FILES.items():
        verify_file(hub_root / name, {"sha256": sha})
    import httpx
    import huggingface_hub as hub
    from huggingface_hub import constants
    from huggingface_hub.utils._http import hf_request_event_hook
    from huggingface_hub.utils._runtime import is_xet_available

    require(
        hub.__version__ == "1.18.0"
        and constants.HF_HUB_DISABLE_XET
        and not is_xet_available()
        and constants.ENDPOINT == ENDPOINT,
        "unreviewed_hub_transport_configuration",
    )

    def factory():
        return httpx.Client(
            follow_redirects=True,
            timeout=httpx.Timeout(**HTTP_TIMEOUTS),
            limits=httpx.Limits(max_connections=4, max_keepalive_connections=2),
            event_hooks={"request": [bounded_request, hf_request_event_hook]},
        )

    hub.set_client_factory(factory)
    return hub

def private_head(api):
    info = api.repo_info(HF_REPO, repo_type="dataset", revision="main", timeout=30, expand=["sha", "private"])
    require(
        info.private is True and REVISION.fullmatch(info.sha), "repository_not_private"
    )
    return info.sha


def source_identity(api, revision):
    wanted = set(SOURCE_TREES + SOURCE_CSVS)
    entries = {}
    for entry in api.list_repo_tree(HF_REPO, repo_type="dataset", revision=revision):
        if entry.path in wanted:
            require(entry.path not in entries, "duplicate_source_entry")
            attr = "tree_id" if entry.path in SOURCE_TREES else "blob_id"
            value = getattr(entry, attr, None)
            require(
                isinstance(value, str) and REVISION.fullmatch(value),
                "source_entry_identity",
            )
            entries[entry.path] = {attr: value}
    require(set(entries) == wanted, "source_entries_missing")
    return entries


def verify_source_unchanged(api, revision):
    before = source_identity(api, SOURCE)
    require(source_identity(api, PARENT) == before, "original_parent_changed_source")
    require(source_identity(api, revision) == before, "publication_changed_source")
    return before


def verify_remote(api, hub, token, revision, refs, progress=lambda *_, **__: None):
    """Every LFS OID+size; every ordinary file's full streamed SHA-256+size."""
    from huggingface_hub.utils._http import get_session

    entries = {}
    for start in range(0, len(refs), 100):
        chunk = refs[start : start + 100]
        found = api.get_paths_info(
            HF_REPO,
            paths=[r["path"] for r in chunk],
            repo_type="dataset",
            revision=revision,
        )
        for entry in found:
            require(entry.path not in entries, "duplicate_remote_entry")
            entries[entry.path] = entry
    require(set(entries) == {r["path"] for r in refs}, "remote_membership_mismatch")
    verified = []
    for index, ref in enumerate(refs):
        entry = entries[ref["path"]]
        require(getattr(entry, "size", None) == ref["bytes"], "remote_size_mismatch")
        if getattr(entry, "lfs", None):
            require(entry.lfs.sha256 == ref["sha256"], "remote_lfs_oid_mismatch")
            method = "hub_lfs_sha256_and_size"
        else:
            url = hub.hf_hub_url(
                HF_REPO, ref["path"], repo_type="dataset", revision=revision
            )
            h = hashlib.sha256()
            size = 0
            with get_session().stream(
                "GET", url, headers={"Authorization": "Bearer " + token}
            ) as response:
                response.raise_for_status()
                for chunk in response.iter_bytes(1024 * 1024):
                    size += len(chunk)
                    require(size <= ref["bytes"], "remote_stream_exceeds_size")
                    h.update(chunk)
            require(
                size == ref["bytes"] and h.hexdigest() == ref["sha256"],
                "remote_full_sha256_mismatch",
            )
            method = "full_http_bytes_sha256_and_size"
        verified.append({**ref, "verification": method})
        if (index + 1) % 25 == 0 or index + 1 == len(refs):
            progress(
                "remote_verification_progress", verified=index + 1, total=len(refs)
            )
    return verified


def choose_action(head, verify_existing):
    if head == PARENT:
        return "commit_from_original_parent", None
    require(REVISION.fullmatch(head), "invalid_head")
    # A different HEAD never authorizes a new commit, even if only one file differs.
    # Exact current object/source equality proves existing publication content;
    # it does not by itself prove which process authored it or its Git parent.
    return "verify_existing_exact_operation", verify_existing(head)


def write_new_json(path, value):
    with path.open("x", encoding="utf-8") as stream:
        json.dump(value, stream, indent=2)
        stream.write("\n")
        stream.flush()
        os.fsync(stream.fileno())


class RunJournal:
    def __init__(self, output):
        self.path = output / "events.jsonl"
        self.lock = threading.Lock()
        self.stage = "starting"
        self.revision = None

    def record(self, stage, **fields):
        with self.lock:
            self.stage = stage
            event = {"at_utc": now(), "stage": stage, **fields}
            with self.path.open("a", encoding="utf-8") as stream:
                stream.write(json.dumps(event) + "\n")
                stream.flush()
                os.fsync(stream.fileno())
            print(json.dumps({"stage": stage, **fields}), flush=True)


def recover(api, hub, token, operation, journal, output):
    refs = operation["uploaded_files"]
    ci = ci_guard()
    head = private_head(api)
    source_ids = verify_source_unchanged(api, head)

    def existing(revision):
        return verify_remote(api, hub, token, revision, refs, journal.record)

    action, verification = choose_action(head, existing)
    journal.record("remote_preflight_passed", head=head, action=action)
    if action == "commit_from_original_parent":
        original_stopped()
        ci = ci_guard()
        require(private_head(api) == PARENT, "head_changed_before_commit")
        # CommitOperationAdd recomputes UploadInfo. Check all of it before any
        # create_commit preupload or commit request; never build replacement bytes.
        operations = []
        for ref in refs:
            path = local_path(ref["path"])
            verify_file(path, ref)
            add = hub.CommitOperationAdd(path_in_repo=ref["path"], path_or_fileobj=path)
            require(
                add.upload_info.size == ref["bytes"]
                and add.upload_info.sha256.hex() == ref["sha256"],
                "upload_info_mismatch",
            )
            operations.append(add)
        original_stopped()
        require(private_head(api) == PARENT, "head_changed_after_upload_preparation")
        journal.record(
            "commit_call_starting", parent_commit=PARENT, objects=len(operations)
        )
        result = api.create_commit(
            repo_id=HF_REPO,
            repo_type="dataset",
            revision="main",
            parent_commit=PARENT,
            operations=operations,
            commit_message=MESSAGE,
            commit_description=DESCRIPTION,
            num_threads=2,
        )
        revision = result.oid
        require(
            REVISION.fullmatch(revision) and revision != PARENT,
            "commit_result_revision",
        )
        journal.revision = revision
        journal.record("committed_remote_verification_pending", revision=revision)
        verification = existing(revision)
    else:
        revision = head
        journal.revision = revision
    source_ids = verify_source_unchanged(api, revision)
    require(private_head(api) == revision, "head_changed_during_verification")
    for ref in refs:
        verify_file(local_path(ref["path"]), ref)
    verify_file(PACKAGE / "publication-operation.json", {"sha256": OPERATION_SHA})
    require(len(verification) == 490, "incomplete_remote_verification")
    receipt = {
        "schema_version": 1,
        "status": "verified_private_preparation_checkpoint",
        "recovery_action": action,
        "completed_at_utc": now(),
        "repo_id": HF_REPO,
        "private": True,
        "github_commit": COMMIT,
        "github_ci_run": CI_RUN,
        "github_ci_result": ci,
        "hf_publication_revision": revision,
        "hf_parent_revision": PARENT
        if action == "commit_from_original_parent"
        else None,
        "expected_hf_parent_revision": PARENT,
        "parent_binding_scope": "expected parent for a new commit; existing publication content is verified without inferring its author or parent",
        "effective_source_revision": SOURCE,
        "hf_commit_url": ENDPOINT + "/datasets/" + HF_REPO + "/commit/" + revision,
        "operation": file_ref(PACKAGE / "publication-operation.json"),
        "manifest": file_ref(PACKAGE / "manifest.json"),
        "recovery_script": file_ref(Path(__file__)),
        "original_publisher_stop_record": file_ref(
            WORK / "hf-review3607-stalled-attempt-stop.json"
        ),
        "uploaded_files": refs,
        "remote_verifications": verification,
        "remote_objects_verified": True,
        "source_tree_and_official_metadata_unchanged": True,
        "source_tree_and_official_metadata_identities": source_ids,
        "xet_enabled": False,
        "http_timeouts_seconds": HTTP_TIMEOUTS,
        "upload_threads": 2,
        "package_rebuilt": False,
        "original_package_receipt_modified": False,
        "training_ready": False,
    }
    write_new_json(output / "publication-recovery-receipt.json", receipt)
    journal.record("verified", revision=revision, objects=490, training_ready=False)


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument("--check-local", action="store_true")
    mode.add_argument("--execute", action="store_true")
    parser.add_argument(
        "--output",
        type=Path,
        required=True,
        help="New directory under this staging package's runs/.",
    )
    parser.add_argument("--max-runtime-seconds", type=int, default=1800)
    args = parser.parse_args()
    journal = None
    done = threading.Event()
    lock = None
    try:
        require(60 <= args.max_runtime_seconds <= 3600, "runtime_bound_invalid")
        output = args.output.resolve()
        require(
            output.is_relative_to(STAGE / "runs") and not output.exists(),
            "output_not_new_run",
        )
        if args.execute:
            original_stopped()
            verify_file(
                WORK / "hf-review3607-stalled-attempt-stop.json", {"sha256": STOP_SHA}
            )
        operation, proof = local_preflight()
        if args.execute:
            original_stopped()
        # Local originals are all verified before even the recovery's own writes.
        output.parent.mkdir(parents=True, exist_ok=True)
        lock = (STAGE / "recovery.lock").open("a")
        fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
        output.mkdir(exist_ok=False)
        write_new_json(output / "local-preflight.json", proof)
        journal = RunJournal(output)
        if args.check_local:
            journal.record("offline_local_check_passed", objects=490, tar_members=3610)
            return 0

        def deadline():
            if not done.wait(args.max_runtime_seconds):
                # Kill this recovery only, including blocked upload workers. The
                # next run must reconcile HEAD; it must not assume commit failure.
                try:
                    journal.record(
                        "deadline_exceeded_remote_outcome_may_be_uncertain",
                        last_known_revision=journal.revision,
                    )
                finally:
                    os._exit(124)

        threading.Thread(target=deadline, daemon=True).start()
        hub = configure_hub()
        from dotenv import dotenv_values

        token = dotenv_values(WORK.parent / ".env").get("HF_TOKEN")
        require(isinstance(token, str) and bool(token), "hf_token_missing")
        api = hub.HfApi(token=token, endpoint=ENDPOINT)
        journal.record(
            "guarded_recovery_started", max_runtime_seconds=args.max_runtime_seconds
        )
        recover(api, hub, token, operation, journal, output)
        return 0
    except BaseException as error:
        # Never print str(error), traceback, subprocess output, auth, or URLs.
        code = (
            error.args[0]
            if type(error) is GuardError
            else "external_or_unexpected_error"
        )
        info = {
            "error_code": code,
            "error_type": type(error).__name__,
            "remote_outcome_may_be_uncertain": bool(
                journal and journal.stage != "starting"
            ),
        }
        if journal is not None:
            journal.record("stopped_without_success_claim", **info)
        else:
            print(json.dumps({"stage": "preflight_blocked", **info}), flush=True)
        return 1
    finally:
        done.set()
        if lock is not None:
            lock.close()


if __name__ == "__main__":
    sys.exit(main())
