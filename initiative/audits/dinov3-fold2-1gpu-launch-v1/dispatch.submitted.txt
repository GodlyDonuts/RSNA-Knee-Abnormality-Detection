"""One-shot Newton dispatch for the bounded fold-2 single-H100 pilot."""

from __future__ import annotations

import datetime
import hashlib
import json
import os
import subprocess
from pathlib import Path


ROOT = Path("/home/sa305415/rsna-knee/production-launch-b740-fold2-1gpu-v1")
NAME = "rsna-dinov3-fold2-1gpu-pilot"


def sha(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


def write_once(path: Path, record: dict) -> None:
    with path.open("x") as stream:
        json.dump(record, stream, indent=2, sort_keys=True)
        stream.write("\n")
        stream.flush()
        os.fsync(stream.fileno())


def main() -> None:
    intent = json.loads((ROOT / "dispatch-intent.json").read_text())
    assert sha(ROOT / "pilot.sbatch") == intent["script_sha256"]
    assert sha(ROOT / "pilot-config.json") == intent["config_sha256"]
    assert not Path(intent["output"]).exists()
    assert not (ROOT / "dispatch-attempt-started.json").exists()
    assert not (ROOT / "dispatch-result.json").exists()
    jobs = subprocess.run(
        ["squeue", "-h", "-u", "sa305415", "-o", "%j"],
        capture_output=True,
        text=True,
        check=True,
        timeout=15,
    )
    assert NAME not in jobs.stdout.splitlines()
    write_once(
        ROOT / "dispatch-attempt-started.json",
        {
            "status": "one_attempt_started",
            "started_at_utc": datetime.datetime.now(datetime.timezone.utc).isoformat(),
            "script_sha256": intent["script_sha256"],
        },
    )
    result = subprocess.run(
        ["sbatch", "--parsable", str(ROOT / "pilot.sbatch")],
        capture_output=True,
        text=True,
        timeout=60,
    )
    receipt = {
        "finished_at_utc": datetime.datetime.now(datetime.timezone.utc).isoformat(),
        "returncode": result.returncode,
        "stdout": result.stdout.strip(),
        "stderr": result.stderr.strip(),
        "script_sha256": intent["script_sha256"],
    }
    write_once(ROOT / "dispatch-result.json", receipt)
    print(json.dumps(receipt, sort_keys=True))
    if result.returncode != 0:
        raise SystemExit(2)


if __name__ == "__main__":
    main()
