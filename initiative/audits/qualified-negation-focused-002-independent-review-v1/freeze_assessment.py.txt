"""Seal explicitly authored source decisions; no automated clinical classifier."""
from pathlib import Path
from datetime import datetime, timezone
import hashlib
import json
import runpy

S=Path(__file__).resolve().parent
W=S.parents[2]
R=W/'work/recovered'
AGENT='/root/last_twelve_visual_review'
TARGETS=['ACL','MCL','Medial Meniscus','Lateral Meniscus','Medial OA','Lateral OA','PF OA','Effusion','Synovitis',"Baker's",'Contusion','Fracture']
PACKET=W/'work/annotation-staging/qualified-negation-all4407-candidates-v1/focused-neutral-batches/batch-002.json'
PACKET_SHA='8491396b5d7dee79b95e019391c78d18996f264827eeb5a898df1a0cba5b0aee'
RUBRIC=R/'protocols/report_weak_v2.md'
RUBRIC_SHA='138c5cc33b92901d57dbf5fbe7e3a9878b73d25b0af6934908d0052cd6786014'
# These observed below-threshold negatives were individually selected by the
# author. Every other negative is grounded in an explicit normal/absence clause.
BELOW={
 (0,0),(1,1),(1,6),(5,2),(5,4),(7,2),(11,2),(11,4),(11,6),(15,2),
 (16,1),(16,7),(19,5),(20,0),(20,3),(21,4),(22,2),(23,2),(24,1),
 (25,6),(27,4),(27,5),(28,7),(31,7),(32,1),(32,2),(32,3),(32,7),(32,9),
 (34,0),(36,6),(38,0),(38,3),(39,0),(39,2),(39,3),(41,7),(42,7),(43,1),(48,2),
}

def digest(raw):return hashlib.sha256(raw).hexdigest()
def ref(p):
    b=p.read_bytes()
    return {'path':str(p.relative_to(W)),'sha256':digest(b),'bytes':len(b)}
def obj(p):return json.loads(p.read_bytes())
def write(name,data):
    p=S/name
    with p.open('x',encoding='utf-8') as f:f.write(json.dumps(data,ensure_ascii=False,indent=2)+'\n')
    return p

assert ref(PACKET)['sha256']==PACKET_SHA
assert ref(S/'neutral-source-packet.json')['sha256']==PACKET_SHA
assert ref(RUBRIC)['sha256']==RUBRIC_SHA
packet=obj(PACKET)
bindings=obj(S/'source-bindings-and-exposure.json')
assert bindings['neutral_packet']==ref(PACKET)
assert bindings['rubric']['sha256']==RUBRIC_SHA
assert len(packet)==len(bindings['source_records'])==50
assert len({x['study_uid'] for x in packet})==50
partial=obj(S/'partial-source-checkpoint-030.json')
for entry in partial['artifacts']:
    path=W/entry['path']
    assert ref(path)['sha256']==entry['sha256']
authored_files=sorted(S.glob('decisions_*.py'))
assert len(authored_files)==10
authored={}
for path in authored_files:
    data=runpy.run_path(str(path))['ROWS']
    assert not authored.keys() & data.keys()
    authored.update(data)
assert set(authored)==set(range(50))
reconsiderations=obj(S/'prefreeze-reconsiderations.json')
overrides={}
for change in reconsiderations['changes']:
    i,t=change['packet_index_zero_based'],change['target_index_zero_based']
    before=authored[i][t]
    assert change['before']==dict(zip(['state','evidence','reason'],before))
    assert change['target']==TARGETS[t]
    assert (i,t) not in overrides
    overrides[i,t]=change['after']
assert len(overrides)==2
overlap=obj(S/'recognized-correspondence-overlap.json')['overlap']
assert len(overlap)==1
rows=[]
checks=[]
for i,source in enumerate(packet):
    p=R/source['source_path']
    assert ref(p)['sha256']==source['source_sha256']
    original=obj(p)[source['index']]
    assert original=={'study_uid':source['study_uid'],'report':source['report']}
    bound=bindings['source_records'][i]
    assert bound['packet_index_zero_based']==i
    assert bound['study_uid']==source['study_uid']
    assert bound['source_index_zero_based']==source['index']
    assert bound['source_path']==source['source_path']
    assert bound['source_sha256']==source['source_sha256']
    assert bound['report_utf8_sha256']==digest(source['report'].encode('utf-8'))
    assert bound['case']==f"{source['batch'].removeprefix('batch-')}:{source['index']:02d}"
    assert len(authored[i])==12
    decisions=[]
    offsets=[]
    negative_basis=[]
    for t,initial in enumerate(authored[i]):
        d=overrides.get((i,t),dict(zip(['state','evidence','reason'],initial)))
        state,quote,reason=d['state'],d['evidence'],d['reason']
        assert state in {'negative','positive','uncertain','not_mentioned'}
        assert (quote=='')==(state=='not_mentioned')
        assert quote in source['report'] and reason.strip(),(i,t)
        if quote:
            start=source['report'].index(quote)
            end=start+len(quote)
            assert source['report'][start:end]==quote
            offsets.append({'start_unicode_codepoint':start,'end_exclusive_unicode_codepoint':end})
        else:offsets.append(None)
        if (i,t) in BELOW:assert state=='negative'
        negative_basis.append(('below_threshold' if (i,t) in BELOW else 'explicit_absence') if state=='negative' else None)
        decisions.append(d)
    recognized=[x for x in overlap if x['study_uid']==source['study_uid']]
    if recognized:assert recognized[0]['report_utf8_sha256']==bound['report_utf8_sha256']
    rows.append({**bound,'reader_id':AGENT,'target_order':TARGETS,
                 'rubric_version':'report-weak-v2.0','rubric_sha256':RUBRIC_SHA,
                 'states':[x['state'] for x in decisions],
                 'evidence':[x['evidence'] for x in decisions],
                 'target_reasoning':[x['reason'] for x in decisions],
                 'negative_basis':negative_basis,'evidence_offsets':offsets,
                 'prefreeze_reconsidered_target_indices':[t for j,t in overrides if j==i],
                 'recognized_personal_source_overlap':recognized,
                 'reading_phase':'personally read/authored before intervening cartilage comparison; any subsequent reconsideration separately recorded' if i<30 else 'personally read/authored after intervening disjoint cartilage comparison',
                 'notes':'All12 decisions personally authored after full-source reading. First focused002 clinical findings, current focused002 state/proposal tables and gold were not opened before this freeze. Historical source/technical/clinical and intervening conceptual exposure are separately bound; no first-ever or entirely pre-feedback blinding is claimed. Exact quotations prove traceability rather than correctness.'})
    checks.append({**bound,'source_index_uid_report_hash_match':True,'all12_states_and_individual_reasons_present':True,'all_nonmissing_evidence_exact_contiguous':True,'not_mentioned_evidence_empty':True})
assessment=S/'independent-assessment.jsonl'
with assessment.open('x',encoding='utf-8') as f:
    for row in rows:f.write(json.dumps(row,ensure_ascii=False)+'\n')
assert [json.loads(line) for line in assessment.read_text().splitlines()]==rows
exposure_files=[S/'source-bindings-and-exposure.json',S/'recognized-correspondence-overlap.json',S/'intervening-cartilage-comparison-exposure.json',S/'partial-source-checkpoint-030.json',S/'prefreeze-reconsiderations.json']
validation=write('validation.json',{
 'stage':'mechanical_full_source_binding_evidence_and_completeness_validation',
 'agent_id':AGENT,'at_utc':datetime.now(timezone.utc).isoformat(),'result':'pass',
 'reports':50,'unique_uids':50,'targets_per_report':12,'judgments':600,
 'assessment':ref(assessment),'neutral_packet':ref(PACKET),'rubric':ref(RUBRIC),
 'all_exact_source_index_uid_report_hash_bindings_verified':True,
 'all600_states_and_individual_reasons_present':True,
 'all_nonmissing_quotes_exact_contiguous_substrings':True,
 'all_not_mentioned_quotes_empty':True,'precomparison30_report_checkpoint_bytes_preserved':True,
 'per_source_checks':checks,
 'limitation':'Mechanical checks do not establish clinical correctness or independently prove historical access order.',
})
freeze=write('independent-freeze.json',{
 'stage':'qualified_negation_focused002_second_source_50_full_reports_600_targets_frozen_v1',
 'agent_id':AGENT,'session_id':AGENT,'at_utc':datetime.now(timezone.utc).isoformat(),
 'rubric_version':'report-weak-v2.0','rubric':ref(RUBRIC),'target_order':TARGETS,
 'reports':50,'judgments':600,'neutral_packet':ref(PACKET),
 'preserved_neutral_packet':ref(S/'neutral-source-packet.json'),
 'assessment':ref(assessment),'validation':ref(validation),
 'exposure_and_prefreeze_lineage':[ref(x) for x in exposure_files],
 'authored_decision_helpers':[ref(x) for x in authored_files],
 'packaging_validation_helper':ref(Path(__file__).resolve()),
 'personal_reading_attestation':'I, /root/last_twelve_visual_review, personally read all50 complete original focused002 reports and authored all600 target judgments with individual reasoning. Code only packages explicit decisions, checks source/evidence bindings, and applies my two separately recorded pre-freeze changes; it does not infer labels.',
 'historical_exposure_attestation':'This is not a first-ever blind reader. Prior focused001 full-source/post-comparison work, cartilage source review, technical correspondence/source/header and policy evidence, technical crosswalks and series-code review are known. Exact neutral-source overlap checks find zero focused001/cartilage50/last12 overlap and one recognized prior correspondence report, individually bound in its addendum. Original disclosure records remain unchanged.',
 'intervening_exposure_attestation':'Reports0–29 and360 judgments were authored before the authorized disjoint cartilage150 postfreeze comparison. Reports30–49 were read afterward. That comparison exposed grading/depth/extent concepts and the other cartilage reader\'s reasoning, while clinical state inspection was restricted to those disjoint cartilage150 keys. One earlier PF state was explicitly reconsidered before this final freeze and one later quotation was expanded; both original authored versions remain preserved. No first002/current002 case findings were opened. This is not claimed to be a wholly pre-feedback source pass.',
 'chronology_limit':'Timestamps and attestations record the workflow but cannot independently prove past access order or erase remembered context.',
 'no_original_rubric_gold_runtime_registry_or_shared_repository_changes':True,
 'no_label_registration_or_global_qa_closure':True,
 'case_findings_withheld_pending_root_authorization':True,
})
with (S/'summary.md').open('x',encoding='utf-8') as f:
    f.write('Completed all50 full-source readings and600 personally authored target judgments under the unchanged report-weak-v2.0 rubric. Original source/index/UID/hash and all exact contiguous evidence checks pass.\n\nHistorical and intervening exposure is disclosed, including one prior correspondence-source overlap and the disjoint cartilage comparison between the first30 and final20 reports. One pre-freeze state reconsideration and one evidence expansion are preserved separately from their original authored versions. No first002/current002 case findings or gold were opened.\n\nNo labels were registered, no global QA gate was closed, and no repository/original/runtime files changed. This freeze and all prior records must remain unchanged; later comparisons belong in a separate stage.\n')
files=[ref(p) for p in sorted(S.iterdir()) if p.is_file()]
manifest=write('final-file-list.json',{'agent_id':AGENT,'at_utc':datetime.now(timezone.utc).isoformat(),'files':files,'self_excluded':True,'reports':50,'judgments':600,'source_freeze':ref(freeze)})
for entry in files:assert ref(W/entry['path'])==entry
for entry in partial['artifacts']:assert ref(W/entry['path'])['sha256']==entry['sha256']
print(json.dumps({'result':'pass','reports':50,'judgments':600,'assessment':ref(assessment),'freeze':ref(freeze),'validation':ref(validation),'manifest':ref(manifest)},indent=2))
