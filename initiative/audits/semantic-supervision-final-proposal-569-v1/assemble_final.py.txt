def need(ok,message):
    if not ok: raise ValueError(message)
def sha(path):
    h=hashlib.sha256()
    with path.open('rb') as f:
        for b in iter(lambda:f.read(2**20),b''):h.update(b)
    return h.hexdigest()
def jb(obj):return json.dumps(obj,ensure_ascii=False,sort_keys=True,separators=(',',':'),allow_nan=False).encode()
def digest(obj):return hashlib.sha256(jb(obj)).hexdigest()
def read(path):
    with (gzip.open(path,'rt') if path.name.endswith('.gz') else path.open()) as f:return json.load(f)
def lines(path):
    with gzip.open(path,'rt') as f:
        for line in f:yield json.loads(line)
def put(name,obj):
    path=D/name
    with path.open('xb') as f:f.write(jb(obj)+b'\n')
    return path
def ref(path):return {'path':str(path.relative_to(R)),'sha256':sha(path)}
def wref(path):return {'path':str(path.relative_to(W)),'sha256':sha(path)}
def finalref(name):return {'path':PREFIX+name,'sha256':sha(D/name)}
def archive_path(p):

def write_array_doc(name,fields,array_name,rows):
    path=D/name
    opener=path.open('xb')
    with opener as raw:
        with gzip.GzipFile(filename='',fileobj=raw,mode='wb',mtime=0) as out:
            out.write(b'{')
            for key in sorted(fields):out.write(jb(key)+b':'+jb(fields[key])+b',')
            out.write(jb(array_name)+b':[')
            for i,row in enumerate(rows):
                if i:out.write(b',')
                out.write(jb(row))
            out.write(b']}\n')
    # Streaming decompression is an independent byte proof, not another full file.
    h=hashlib.sha256();size=0
    with gzip.open(path,'rb') as f:
        for block in iter(lambda:f.read(2**20),b''):h.update(block);size+=len(block)
    return {'path':str(path.relative_to(W)),'sha256':sha(path),'bytes':path.stat().st_size,'decompressed_sha256':h.hexdigest(),'decompressed_bytes':size}
