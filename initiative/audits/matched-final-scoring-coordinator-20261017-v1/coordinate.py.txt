"""Execute one original phase with a frozen program and stdin credentials."""
import hashlib
import json
import shlex
import subprocess
import sys
from pathlib import Path

L = Path(__file__).resolve().parent
ROOT = L.parents[1]
phase = sys.argv[1]
candidate = sys.argv[2] if len(sys.argv) > 2 else 'uniform'
mode = sys.argv[3] if len(sys.argv) > 3 else 'qualification'
assert phase in ('stage', 'prepare_label', 'push', 'status', 'eligibility', 'submit', 'outcome')
program = (L / 'program.py.txt').read_text()
compile(program, 'matched_final_canonical_scoring_phase', 'exec')
v = {'phase': phase, 'candidate': candidate, 'mode': mode,
     'program_sha256': hashlib.sha256(program.encode()).hexdigest(),
     'code_source': '86cfc7b673b11ca9440db312a90507b0097b1be1'}
if phase == 'stage':
    assert not (L / 'stage.intent.json').exists()
    with (L / 'stage.intent.json').open('x') as f:
        json.dump({'program_sha256': v['program_sha256'], 'code_source': v['code_source'],
                   'CPU': 1, 'GPU': 0, 'automatic_retry': False}, f, indent=2)
    v['program'] = program
else:
    accepted = json.loads((L / 'staged.json').read_text())
    assert accepted['program_sha256'] == v['program_sha256']
    env = ROOT / '.env'
    assert env.stat().st_mode & 0o077 == 0
    values = {k.strip(): t.strip().strip('"\'') for line in env.read_text().splitlines()
              for k, sep, t in [line.partition('=')] if sep}
    v['token'] = values['KAGGLE_API_TOKEN']
    if phase == 'prepare_label':
        previous = ROOT / 'artifacts/s6-label-fixed3-followthrough-20261017-v2'
        v['actual'] = json.loads((previous / 'epoch-3-actual-accepted.json').read_text())
        v['publication'] = json.loads((previous / 'epoch-3-private-verified.json').read_text())
cmd = ['srun', '--account=skattel', '--partition=preemptable', '--qos=preemptable',
       '--nodelist=evc33', '--ntasks=1', '--cpus-per-task=1', '--gres=none',
       '--mem=4G', '--time=00:10:00', '--job-name=rsna-final-score-' + phase,
       'env', 'PYTHONDONTWRITEBYTECODE=1', 'PYTHONNOUSERSITE=1', 'CUDA_VISIBLE_DEVICES=',
       'OMP_NUM_THREADS=1', 'MKL_NUM_THREADS=1', 'OPENBLAS_NUM_THREADS=1',
       '/tmp/rsna-environment-907632-7sr9a7i6/environment/bin/python', '-B', '-c', program]
p = subprocess.run(['ssh', '-S', '/tmp/rsna-research-20261001-stable.sock', '-o', 'BatchMode=yes',
                    'newton', shlex.join(cmd)], input=json.dumps(v), capture_output=True, text=True, timeout=700)
result = {'returncode': p.returncode, 'stdout': p.stdout, 'stderr': p.stderr}
import time
name = phase + '-' + candidate + '-' + mode + '-' + str(time.time_ns()) + '.json'
with (L / name).open('x') as f: json.dump(result, f, indent=2)
assert p.returncode == 0, p.stderr[-2000:]
out = json.loads(p.stdout)
if phase == 'stage': (L / 'staged.json').write_text(json.dumps(out, indent=2) + '\n')
print(json.dumps(out), flush=True)
