@pytest.mark.parametrize('prior',['failed','pending','scored'])
def test_next_model_waits_for_actual_previous_score(control,prior):
    v=control.accepted_entry('label');v['phase']='eligibility';control.state['account'].update(num_today=0,num_allowed_now=5)
    key,row=control.get();row['submission_id']=42;control.save('uniform',row)
    old={'ref':42,'status':'COMPLETE' if prior!='pending' else 'RUNNING'}
    if prior=='scored':old['publicScore']=.95
    if prior=='failed':old['errorDescription']='fixture hidden error'
    control.state['submissions']=[old]
    if prior=='failed':
        with pytest.raises(AssertionError):control.ns['execute'](v)
    else:assert control.ns['execute'](v)['fresh']['eligible']==(prior=='scored')
    assert mutations(control)==[]
