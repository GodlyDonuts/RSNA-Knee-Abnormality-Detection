"""Package600 personally authored target decisions and verify immutable source evidence."""
import hashlib
import json
import runpy
from collections import Counter
from datetime import datetime, timezone
from pathlib import Path

S = Path(__file__).resolve().parent
W = S.parents[2]
R = W / 'work/recovered'
AGENT = '/root/last_twelve_visual_review'
TARGETS = ['ACL', 'MCL', 'Medial Meniscus', 'Lateral Meniscus', 'Medial OA', 'Lateral OA', 'PF OA', 'Effusion', 'Synovitis', "Baker's", 'Contusion', 'Fracture']


def obj(p):
    return json.loads(p.read_bytes())


def ref(p):
    b = p.read_bytes()
    return {'path': str(p.relative_to(W)), 'sha256': hashlib.sha256(b).hexdigest(), 'bytes': len(b)}


def write(name, value):
    p = S / name
    with p.open('x') as f:
        json.dump(value, f, ensure_ascii=False, indent=2)
        f.write('\n')
    return p


packet = obj(S / 'neutral-source-packet.json')
binding = obj(S / 'source-bindings-and-exposure.json')
rubric = R / 'protocols/report_weak_v2.md'
assert ref(S / 'neutral-source-packet.json')['sha256'] == '2958d5d58e76befd3e0974d3fca513a1e8478b94ff0dbb3b4b25db3a7b81df50'
assert ref(rubric)['sha256'] == '138c5cc33b92901d57dbf5fbe7e3a9878b73d25b0af6934908d0052cd6786014'
assert (S / 'neutral-source-packet.json').read_bytes() == (W / binding['source_packet']['path']).read_bytes()
for prior in binding['prior_neutral_packet_bindings']:
    assert ref(W / prior['path']) == prior
rows = {}
helpers = sorted(S.glob('decisions_*.py'))
assert len(helpers) == 10
for h in helpers:
    authored = runpy.run_path(str(h))['ROWS']
    assert not rows.keys() & authored.keys()
    rows.update(authored)
assert len(packet) == 50 and set(rows) == set(range(50))
checks, out = [], []
for i, p in enumerate(packet):
    b = binding['bindings'][i]
    sp = R / p['source_path']
    assert ref(sp)['sha256'] == p['source_sha256']
    assert obj(sp)[p['index']] == {'study_uid': p['study_uid'], 'report': p['report']}
    assert b['study_uid'] == p['study_uid'] and b['source_index_zero_based'] == p['index']
    assert b['report_utf8_sha256'] == hashlib.sha256(p['report'].encode()).hexdigest()
    assert len(rows[i]) == 12
    for target, (state, quote, reason) in zip(TARGETS, rows[i]):
        assert state in {'positive', 'negative', 'uncertain', 'not_mentioned'}
        assert bool(quote) == (state != 'not_mentioned') and quote in p['report']
        assert reason.strip() and 'PLACEHOLDER' not in reason
        checks.append({'packet_index': i, 'study_uid': p['study_uid'], 'target': target,
                       'quote_start_character': p['report'].find(quote) if quote else None,
                       'quote_characters': len(quote),
                       'quote_utf8_sha256': hashlib.sha256(quote.encode()).hexdigest() if quote else None,
                       'exact_contiguous_or_missing_empty': True})
    out.append({'agent_id': AGENT, 'session_id': AGENT, 'packet_index': i,
                'case': f"{p['batch'].removeprefix('batch-')}:{p['index']}",
                'study_uid': p['study_uid'], 'source_path': p['source_path'],
                'source_sha256': p['source_sha256'], 'source_index_zero_based': p['index'],
                'report_utf8_sha256': b['report_utf8_sha256'], 'target_order': TARGETS,
                'states': [x[0] for x in rows[i]], 'evidence': [x[1] for x in rows[i]],
                'target_reasoning': [x[2] for x in rows[i]],
                'prior_exact_neutral_overlaps': [x for x in binding['exact_prior_packet_overlaps'] if x['packet_index'] == i],
                'notes': 'Personally authored full-source wording005 first pass. Historical exact source/clinical overlap and broader contextual exposure disclosed. No other wording005 findings or selected current005 annotations opened before freeze. Decisions remain unintegrated.'})
assert len(out) == len({x['study_uid'] for x in out}) == 50
assert len(checks) == len({(x['study_uid'], x['target']) for x in checks}) == 600
assessment = S / 'source-assessment-frozen.jsonl'
with assessment.open('x') as f:
    for row in out:
        f.write(json.dumps(row, ensure_ascii=False) + '\n')
revision = write('pre-freeze-own-reconsiderations.json', {
    'agent_id': AGENT, 'at_utc': datetime.now(timezone.utc).isoformat(),
    'scope': 'Own source-only review before freezing, without other005/current005 case findings.',
    'changes': [],
    'mechanical_validation_history': 'The initial first300 evidence checks passed. The first600 check found two typed ordinary spaces replacing literal nonbreaking spaces at packet38. Those two quotations were corrected to exact source bytes, with a separate correction record; no clinical state or reason changed and original source text was untouched.',
})
exposure = write('completed-source-exposure-and-chronology.json', {
    'agent_id': AGENT, 'session_id': AGENT, 'at_utc': datetime.now(timezone.utc).isoformat(),
    'preparation': ref(S / 'source-bindings-and-exposure.json'),
    'full_source_reading': 'All50 complete neutral original reports were personally read in ten sequential groups of five. All12 judgments per report were authored explicitly with individual reasoning and exact contiguous evidence. Code only assembles these decisions and checks provenance/evidence.',
    'rubric_use': 'The full unchanged rubric and AGENTS instructions were reread before this wording005 source pass; source-first scope takes precedence over opening current case findings. The rubric exact bytes/hash were rechecked. No rubric revision was made.',
    'prior_exposure': binding['historical_exposure'],
    'exact_overlap_scope': 'Six exact UID overlaps across twelve prior neutral packets, all in cartilage003. In addition packet12 was personally recognized from earlier repeated-temporal source/policy work and is explicitly disclosed separately. This is not an exhaustive reconstruction of every prior viewed byte.',
    'additional_recognized_prior_exposure': ref(S/'recognized-additional-source-exposure.json'),
    'mechanical_quote_corrections': ref(S/'pre-freeze-mechanical-quote-corrections.json'),
    'exact_overlaps': binding['exact_prior_packet_overlaps'],
    'workflow_notifications': 'Root assigned this as first focused005 reader with no other first005 clinical work assigned elsewhere and actual254 registry. No005 case-specific findings or current selected annotations were opened before this freeze.',
    'historical_process_access': 'Previous comparison helpers mechanically accessed actual254 registry/case bytes. Prior source, OA, wording, technical, visual and legacy contexts are disclosed in preparation. This is not first-ever blinding or a claim of no historical process access to current-state bytes.',
    'postfreeze_boundary': 'All600 source judgments are frozen before the planned original/current254 comparison. Any subsequent reconsideration must be separate.',
    'chronology_limit': 'Timestamps and workflow attestations do not independently prove past access order.',
})
validation = write('validation.json', {
    'agent_id': AGENT, 'at_utc': datetime.now(timezone.utc).isoformat(),
    'reports': 50, 'judgments': 600, 'unique_study_uids': 50, 'unique_study_target_keys': 600,
    'target_order': TARGETS,
    'all_original_file_hash_index_uid_and_full_report_bindings_match': True,
    'all_report_utf8_hashes_match': True,
    'all_nonmissing_quotes_exact_contiguous_and_missing_quotes_empty': True,
    'all600_decisions_have_individually_authored_reasons': True,
    'rubric': ref(rubric), 'prior_exact_overlap_reports': 6,'additional_recognized_report_exposures':1,
    'checks': checks, 'clinical_decisions_authored_by_agent_not_inferred_by_script': True,
    'no_gold_or_other005_reader_or_selected_current_annotation_access_in_this_pass': True,
})
freeze = write('source-freeze.json', {
    'stage': 'qualified_negation_focused005_first_reader_full_source_50_reports_600_targets_v1',
    'agent_id': AGENT, 'session_id': AGENT,
    'identity_note': 'Actual canonical collaboration agent/session identity; no separate UUID supplied or invented.',
    'at_utc': datetime.now(timezone.utc).isoformat(),
    'rubric_version': 'report-weak-v2.0', 'rubric': ref(rubric),
    'reports': 50, 'judgments': 600, 'target_order': TARGETS,
    'neutral_source_packet': ref(S / 'neutral-source-packet.json'),
    'source_and_exposure_bindings': ref(S / 'source-bindings-and-exposure.json'),
    'completed_exposure_and_chronology': ref(exposure),
    'pre_freeze_own_reconsiderations': ref(revision),
    'assessment': ref(assessment), 'validation': ref(validation),
    'authored_helpers': [ref(p) for p in helpers], 'packaging_helper': ref(Path(__file__)),
    'state_counts': dict(Counter(x[0] for r in rows.values() for x in r)),
    'personal_reading_attestation': 'I, /root/last_twelve_visual_review, personally read all50 complete original reports and personally authored all600 targets under the unchanged rubric. Quotes are literal original-language substrings. Mechanical code validates bindings and packages explicit choices; it does not generate clinical decisions.',
    'independence_and_exposure': 'Other wording005 case findings and selected current005 annotations were not opened before this first source freeze. Historical exact source/clinical overlap and prior registry/case-process exposure are disclosed. This is not a first-ever blindness claim or a feedback-free conceptual reading.',
    'chronology_limit': 'Recorded timestamps and attestations do not independently prove past access order.',
    'no_repository_registry_runtime_gold_or_global_QA_changes': True,
    'unintegrated_source_opinions_only': True,
    'case_findings_withheld_pending_root_source_assessment': True,
})
manifest = write('final-file-list.json', {'agent_id': AGENT,
    'files': [ref(p) for p in sorted(S.rglob('*')) if p.is_file()], 'self_excluded': True})
files = obj(manifest)['files']
assert {x['path'] for x in files} == {str(p.relative_to(W)) for p in S.rglob('*') if p.is_file() and p != manifest}
for item in files:
    assert ref(W / item['path']) == item
print(json.dumps({'reports': 50, 'judgments': 600, 'freeze': ref(freeze), 'assessment': ref(assessment),
                  'validation': ref(validation), 'manifest': ref(manifest), 'bound_files': len(files)}, indent=2))
