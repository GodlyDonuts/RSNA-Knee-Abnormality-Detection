def ci(args):
    require(type(args.ci_run) is int and args.ci_run == TARGET_CI_RUN, 'exact491_ci_run_required')
    raw = subprocess.check_output(['gh', 'run', 'view', str(args.ci_run), '--repo',
        'GodlyDonuts/RSNA-Knee-Abnormality-Detection', '--json', 'conclusion,headSha,status'],
        cwd=R, stderr=subprocess.DEVNULL, timeout=60)
    value = json.loads(raw)
    require(value == {'conclusion': 'success', 'headSha': args.target_commit, 'status': 'completed'}, 'exact_future_commit_ci_not_success')
    return value

def proof_inputs(args, package, manifest):
    for value in [args.peer_proof_sha256, args.root_proof_sha256, args.root_verifier_sha256]:
        require(SHA.fullmatch(value), 'future_proof_hash_required')
    peer_path = package / 'peer-restoration-proof.json'
    root_path = package / 'root-restoration-proof.json'
    require(sha(peer_path) == args.peer_proof_sha256 and sha(root_path) == args.root_proof_sha256, 'proof_hash_changed')
    peer, root = read(peer_path), read(root_path)
    proof_guard(peer, args, manifest, PEER)
    proof_guard(root, args, manifest, ROOT)
    require(peer['peer_runner'] in [ref(STAGE / n) for n in ['verify_qualified003_peer.py', 'build_qualified003_peer.py']], 'peer_runner_changed')
    require(peer['protected_source576_inventory']['sha256'] == SOURCE_INVENTORY_SHA and peer['source576_exact_target_git_identities_verified'] == 576, 'peer_source576_proof')
    require(root['independent_peer_proof_sha256'] == args.peer_proof_sha256 and root['peer_artifact_unchanged'] is True, 'root_peer_binding')
    require(root['root_runner_sha256'] == args.root_verifier_sha256, 'actual_root_runner_binding')
    require(root['actual_executor'] != peer['actual_executor'], 'same_executor_not_independent')
    return peer, root

def remote_guard(api, transport, revision, *, check_card=True):
    require(transport.source_identity(api, revision) == SOURCE_IDENTITIES, 'remote_source_tree_or_csv_changed')
    if not check_card:
        return None
    root_card = {x.path: x.blob_id for x in api.list_repo_tree(HF_REPO, repo_type='dataset', revision=revision) if x.path == 'README.md'}
    require(root_card == ROOT_CARD, 'root_dataset_card_changed')
    return root_card

def source576_metadata(api, rows):
    """LFS OID/size verification only; never download source payload bytes."""
    seen = {}
    for start in range(0, len(rows), 100):
        chunk = rows[start:start + 100]
        expected = {x['path']: x for x in chunk}
        entries = api.get_paths_info(HF_REPO, paths=list(expected), repo_type='dataset', revision=PARENT_HF)
        found = set()
        for entry in entries:
            require(entry.path in expected and entry.path not in seen, 'source576_metadata_membership')
            row = expected[entry.path]
            lfs = getattr(entry, 'lfs', None)
            require(lfs is not None, 'source576_lfs_metadata_required_no_download_fallback')
            require(entry.size == row['bytes'] and lfs.sha256 == row['sha256'], 'source576_metadata_identity')
            found.add(entry.path)
            seen[entry.path] = dict(row, verification='hub_lfs_sha256_and_size_no_source_download')
        require(found == set(expected), 'source576_metadata_missing')
    require(len(seen) == 576, 'source576_metadata_count')
    return [seen[x['path']] for x in rows]
