"""Replay four source/header identities against the corrected inventory and cache."""
from pathlib import Path
from datetime import datetime, timezone
import hashlib
import json
import pandas as pd

W = Path(__file__).resolve().parent
G = W / "annotation-staging/correspondence28-root-source-01-v1"
P = W / "annotation-staging/correspondence28-neutral-source-packets-v1"
CASES = {"021:44", "042:14", "042:41", "069:10"}


def sha(path):
    h = hashlib.sha256()
    with path.open("rb") as stream:
        for data in iter(lambda: stream.read(1024 * 1024), b""):
            h.update(data)
    return h.hexdigest()


def ref(path):
    return {"path": str(path.relative_to(W)), "sha256": sha(path), "bytes": path.stat().st_size}


def main():
    packet_paths = [P / f"source-packet-{i:02}.json" for i in (1, 3, 4)]
    cases = {r["case"]: r for p in packet_paths for r in json.loads(p.read_text())["cases"] if r["case"] in CASES}
    assert set(cases) == CASES
    inv_path = W / "cache256-v3-official-recovery-v1/inventory_effective.parquet"
    assert sha(inv_path) == "68d202eb951babc9365eddd2f7a1219201cd7b0905f04afdace785ad0e079537"
    index_path = W / "recovered/initiative/audits/full-image-recovery-v1/audit/series_cache_index.parquet"
    assert sha(index_path) == "0d1e3b69fd7c527cf3fd980012751a68a13d38439bd71bc029b2c641a9f46a81"
    uids = [r["study_uid"] for r in cases.values()]
    inv = pd.read_parquet(inv_path, filters=[("study_uid", "in", uids)])
    index = pd.read_parquet(index_path, filters=[("study_uid", "in", uids)])
    assert len(inv) == 699 and len(index) == 24
    endpoints_path = W / "header-endpoint-full-v1/endpoint-headers.jsonl"
    assert sha(endpoints_path) == "d5569c33e453bb1ccdf4600c39e2093821af7ffc1e7e9691f3a5d0ff97e258f3"
    endpoints = []
    with endpoints_path.open() as stream:
        for line in stream:
            row = json.loads(line)
            if row["study_uid"] in uids:
                endpoints.append(row)
    assert len(endpoints) == 48
    out = []
    for case, source in cases.items():
        uid = source["study_uid"]
        batch_path = W / source["source_batch"]["path"]
        assert sha(batch_path) == source["source_batch"]["sha256"]
        row = json.loads(batch_path.read_text())[source["source_batch_index_zero_based"]]
        assert row == {"study_uid": uid, "report": source["full_report_text"]}
        assert hashlib.sha256(row["report"].encode()).hexdigest() == source["report_utf8_sha256"]
        chosen = next(r for r in source["technical_evidence_references"] if r["kind"] in {"case_specific_source_membership_and_endpoint_metadata", "series_endpoint_metadata"})
        technical_path = W / chosen["path"]
        assert sha(technical_path) == chosen["sha256"]
        technical = json.loads(technical_path.read_text())
        series = technical["series"] if isinstance(technical, dict) else technical
        inv_case = inv.loc[inv.study_uid == uid]
        assert {s["series_uid"] for s in series} == set(inv_case.series_uid)
        series_out = []
        for s in series:
            sid = s["series_uid"]
            members = inv_case.loc[inv_case.series_uid == sid].set_index("sop_uid")
            ep = [e for e in endpoints if e["study_uid"] == uid and e["series_uid"] == sid]
            assert len(ep) == 2
            expected_headers = s.get("header_endpoint_values", s.get("header_fields"))
            for field, values in expected_headers.items():
                assert sorted(set(e["header_fields"][field] for e in ep)) == sorted(values), (case, sid, field)
            for e in ep:
                v = members.loc[e["sop_uid"]]
                assert e["source_sha256"] == v.sha256 and e["source_bytes"] == int(v.bytes)
                assert e["source_path"] == v.path
            idx = index.loc[index.series_uid == sid].iloc[0]
            npz = W / "cache256-v3-official-recovery-v1/cache" / idx.cache_path
            meta_path = npz.with_suffix(".json")
            meta = json.loads(meta_path.read_text())
            assert meta["study_uid"] == uid and meta["series_uid"] == sid
            assert meta["source_revision"] == "0b5e410288232f570c69ba4f44723356e6ae6ab7"
            assert meta["n_slices"] == len(members) == int(idx.n_slices)
            assert dict(zip(meta["sop_uids"], meta["slice_sha256"])) == members.sha256.to_dict()
            assert sha(npz) == meta["cache_sha256"] == idx.cache_sha256
            assert npz.stat().st_size == meta["cache_bytes"] == int(idx.cache_bytes)
            assert meta["warnings"] == []
            series_out.append({"series_uid": sid, "n_sources": len(members), "header_endpoints": 2,
                               "literal_fields": {k: expected_headers[k] for k in ("SeriesNumber", "SeriesDescription", "Laterality", "ImageLaterality", "FrameLaterality", "SeriesTime", "AcquisitionTime")},
                               "cache_metadata": ref(meta_path), "cache": ref(npz)})
        out.append({"case": case, "study_uid": uid, "source_batch": source["source_batch"], "source_index_zero_based": source["source_batch_index_zero_based"],
                    "report_utf8_sha256": source["report_utf8_sha256"], "technical_evidence": ref(technical_path), "series": series_out,
                    "n_sources": len(inv_case), "n_series": len(series), "n_endpoints": 2 * len(series)})
    proof = {"stage": "root_four_case_corrected_source_header_cache_identity_replay", "agent_id": "/root", "status": "passed",
             "at_utc": datetime.now(timezone.utc).isoformat(), "script": ref(Path(__file__)), "packets": [ref(p) for p in packet_paths],
             "corrected_inventory": ref(inv_path), "cache_index": ref(index_path), "full_endpoint_archive": ref(endpoints_path),
             "root_source_freezes": [ref(G / n) for n in ("root-source-freeze.json", "root-three-observed-source-freeze.json")],
             "cases": out, "verified_sources": 699, "verified_series": 24, "verified_header_endpoints": 48,
             "limits": "Archived endpoint values and all selected cache membership/file identities rechecked. No fresh native fetch, new pixel viewing, diagnostic assessment, or proof of named anatomical side. Prior source/visual evidence retains its own scope.",
             "gold_opened": False, "originals_modified": False, "training_ready": False}
    target = G / "root-four-case-metadata-replay.json"
    with target.open("x") as stream:
        json.dump(proof, stream, indent=2)
        stream.write("\n")
    print(json.dumps({"status": "passed", "cases": 4, "series": 24, "sources": 699, "endpoints": 48, "sha256": sha(target)}))


if __name__ == "__main__":
    main()
