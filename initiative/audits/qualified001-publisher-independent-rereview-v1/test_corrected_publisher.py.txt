def reverse_patch(new_text,patch_text):
    source=new_text.splitlines(True);diff=patch_text.splitlines(True)
    result=[];cursor=0;i=0
    while i<len(diff):
        if not diff[i].startswith('@@ '):i+=1;continue
        match=re.match(r'@@ -(\d+)(?:,(\d+))? \+(\d+)(?:,(\d+))? @@',diff[i]);assert match
        start=int(match.group(3))-1
        result.extend(source[cursor:start]);cursor=start;i+=1
        while i<len(diff) and not diff[i].startswith('@@ '):
            line=diff[i]
            if line.startswith(' '):
                assert source[cursor]==line[1:];result.append(line[1:]);cursor+=1
            elif line.startswith('+'):
                assert source[cursor]==line[1:];cursor+=1
            elif line.startswith('-'):result.append(line[1:])
            else:raise AssertionError('Unexpected patch line')
            i+=1
    result.extend(source[cursor:]);return ''.join(result)

    def tree(*args,**kwargs):

        return [SimpleNamespace(path='README.md',blob_id='5'*40)]
    def create_commit(**kwargs):
