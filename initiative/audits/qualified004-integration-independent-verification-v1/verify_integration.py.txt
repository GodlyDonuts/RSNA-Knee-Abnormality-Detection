def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def pin(path, expected=None):
    path = path.resolve()
    assert path.is_file() and not path.is_symlink() and path.is_relative_to(W)
    row = {'path': str(path.relative_to(W)), 'sha256': sha(path), 'bytes': path.stat().st_size}
    assert expected is None or row['sha256'] == expected, ('hash_mismatch', row['path'])
    assert bound.setdefault(row['path'], row) == row
    return row


def read(path, expected=None):
    pin(path, expected)
    return json.loads(path.read_text())


def idx(rows):
    result = {(row['study_uid'], row['target']): row for row in rows}
    assert len(result) == len(rows)
    return result


def write(name, value):
    raw = (json.dumps(value, ensure_ascii=False, indent=2) + '\n').encode()
    scanner.Scanner(secrets).feed(raw)
    with (P / name).open('xb') as out:
        out.write(raw)
    return pin(P / name)
