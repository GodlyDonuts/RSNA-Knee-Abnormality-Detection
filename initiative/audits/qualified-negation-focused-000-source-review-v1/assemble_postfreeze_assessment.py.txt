def read(path):
    return json.loads(path.read_text())

def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def write(name, value):
    path = HERE / name
    assert not path.exists(), f'Refusing overwrite: {name}'
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n')

def proposal_record(r, reason, kind):
    uid, target = r['study_uid'], r['target']
    assert (uid, target) not in registered
    assert r['original_state'] == r['effective_state']
    assert r['original_evidence'] == r['effective_evidence']
    state, quote = r['first_assessor_state'], r['first_assessor_evidence']
    assert quote in source_by_uid[uid]['report'] if state != 'not_mentioned' else quote == ''
    before_mask = int(r['effective_state'] in ('negative', 'positive'))
    after_mask = int(state in ('negative', 'positive'))
    return dict(study_uid=uid, target=target, batch=r['batch'], index=r['index'],
                packet_index=r['packet_index'], source_path=r['source_path'],
                source_sha256=r['source_sha256'], artifact_path=r['original_review_path'],
                artifact_sha256=r['original_review_sha256'],
                before_state=r['original_state'], before_evidence=r['original_evidence'],
                proposed_state=state, source_quote=quote, reason=reason,
                kind=kind, report_weak_mask_before=before_mask,
                report_weak_mask_after=after_mask,
                mask_impact=('observed_to_masked' if before_mask and not after_mask else
                             'masked_to_masked' if not before_mask and not after_mask else
                             'observed_unchanged'),
                gold_value_or_mask_changes='none; evaluation labels were not accessed',
                status='proposed_pending_distinct_assessment_and_adjudication')

                  effective_annotation_changed=False)
    disposition_records.append(record)
    if disposition == 'propose':
        proposals.append(proposal_record(r, reason, 'state_correction'))

# Hash-bind all registered cases and their evidence without clinical re-reading.
def collect_references(value):
    if isinstance(value, dict):
        if set(value) >= {'path', 'sha256'} and isinstance(value['path'], str):
            path = REPO / value['path']
            if path.is_file():
                assert digest(path) == value['sha256'], value['path']
                paths.add(value['path'])
        for sub in value.values():
            collect_references(sub)
    elif isinstance(value, list):
        for sub in value:
            collect_references(sub)
collect_references(inspection)
archive_manifest = []
for relative in sorted(paths):
    origin = REPO / relative
    target = HERE / 'bound-originals' / relative
    assert not target.exists()
    target.parent.mkdir(parents=True, exist_ok=True)
    shutil.copyfile(origin, target)
    assert origin.read_bytes() == target.read_bytes()
    archive_manifest.append(dict(original_path=relative, archived_path=str(target.relative_to(HERE)),
                                 sha256=digest(origin), bytes=origin.stat().st_size))
archive_by_path = {r['original_path']: r for r in archive_manifest}
