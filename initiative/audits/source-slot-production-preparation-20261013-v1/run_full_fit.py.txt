def sha(p):return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def save(p,v):
 with Path(p).open('x') as f:json.dump(v,f,indent=2);f.flush();os.fsync(f.fileno())
assert os.uname().nodename=='evc31' and os.environ.get('SLURM_CPUS_PER_TASK')=='2'
assert R.is_dir() and not(R/'production').exists() and not(R/'production-admission.json').exists()

# Import precisely the separately admitted numerical tree, never the current checkout.

from scripts.newton.probe_source_slot_recovery import prerequisites,CPU_STATUS
from scripts.newton.probe_public_input_recipe_recovery import trainer_command,check_placement
from rsna.competitive.train import TrainConfig
check_placement();code,hashes,original,values=prerequisites(A)

wall=min(14*3600,remaining-600)
inputs=original['trainer_inputs'];output=R/'production'

env=os.environ.copy();env.update(PYTHONPATH=str(code),PYTHONNOUSERSITE='1',PYTHONDONTWRITEBYTECODE='1',OMP_NUM_THREADS='2',MKL_NUM_THREADS='2',OPENBLAS_NUM_THREADS='2',WORLD_SIZE='1',RANK='0',LOCAL_RANK='0',CUBLAS_WORKSPACE_CONFIG=':4096:8')

 x=subprocess.run(command,env=env,cwd=code,stdout=out,stderr=err,timeout=wall)
save(R/'trainer-terminal.json',{'returncode':x.returncode,'source_git':cpu['source_git'],'accuracy_requires_independent_final_replay':True});assert x.returncode==0
print(json.dumps({'status':'source_slot_fixed10_trainer_terminal_requires_final_verification','returncode':x.returncode}),flush=True)
