def read(p):
    return json.loads(p.read_text())

def sha(p):
    return hashlib.sha256(p.read_bytes()).hexdigest()

def bind(p):
    return {'path': str(p.relative_to(W)), 'sha256': sha(p)}

def runtime_bind(p):
    if p.is_relative_to(R):
        name = str(p.relative_to(R))
    else:
        name = 'initiative/audits/' + str(p.relative_to(P.parent))
    return {'path': name, 'sha256': sha(p)}

def write(name, obj):
    with (P / name).open('x') as f:
        f.write(json.dumps(obj, ensure_ascii=False, indent=2) + '\n')
