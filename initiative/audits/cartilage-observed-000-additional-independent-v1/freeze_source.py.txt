import json

W = P.parents[2]
R = W / 'work/recovered'
AGENT = '/root/quality_temporal_review'
TARGETS = ['Medial OA', 'Lateral OA', 'PF OA']

def read(p):
    return json.loads(p.read_text())

def sha(p):
    return hashlib.sha256(p.read_bytes()).hexdigest()

def bind(p):
    return {'path': str(p.relative_to(W)), 'sha256': sha(p), 'bytes': p.stat().st_size}

def write(name, obj):
    with (P / name).open('x') as f:
        f.write(json.dumps(obj, ensure_ascii=False, indent=2) + '\n')

def span(source, start, end):
    a = source.index(start)
    b = source.index(end, a) + len(end)
    return source[a:b]
