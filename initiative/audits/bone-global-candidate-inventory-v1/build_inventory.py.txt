def read(p): return json.loads(p.read_text())


def sha(p):
    h = hashlib.sha256()
    with p.open('rb') as f:
        for chunk in iter(lambda: f.read(1024 * 1024), b''): h.update(chunk)
    return h.hexdigest()


def dump(name, obj, lines=False):
    p = HERE / name
    p.parent.mkdir(parents=True, exist_ok=True)
    if lines:
        data = ''.join(json.dumps(x, ensure_ascii=False, sort_keys=True, separators=(',', ':')) + '\n' for x in obj)
    else:
        data = json.dumps(obj, ensure_ascii=False, indent=2, sort_keys=True) + '\n'
    if p.exists(): assert p.read_text() == data, 'existing output differs'
    else: p.write_text(data)
    return {'path': name, 'bytes': p.stat().st_size, 'sha256': sha(p)}


def normalized(text):
    chars, offsets = [], []
    for i, c in enumerate(text):
        for folded in unicodedata.normalize('NFKD', c.casefold()).replace('ł', 'l').replace('ı', 'i'):
            if not unicodedata.combining(folded):
                chars.append(folded); offsets.append(i)
    return ''.join(chars), offsets


def hits(text):
    norm, offsets = normalized(text)
    found = {}
    for family, pattern in PATTERNS.items():
        rows = []
        for m in re.finditer(pattern, norm):
            start, end = offsets[m.start()], offsets[m.end() - 1] + 1
            rows.append(dict(family=family, start=start, end=end, quote=text[start:end],
                             normalized_start=m.start(), normalized_end=m.end()))
        found[family] = rows
    return found


def nearby(a, b, distance=220):
    return [(x, y) for x in a for y in b if max(x['normalized_start'], y['normalized_start'])
            - min(x['normalized_end'], y['normalized_end']) <= distance]


def excerpts(text, cue_rows, evidence):
    points = sorted({(x['start'], x['end']) for x in cue_rows})
    spans = []
    for start, end in points:
        lo, hi = max(0, start - 180), min(len(text), end + 220)
        if spans and lo <= spans[-1][1]: spans[-1][1] = max(hi, spans[-1][1])
        else: spans.append([lo, hi])
    # Do not truncate the list of cue windows. Every selection trigger remains traceable.
    result = [dict(start=a, end=b, quote=text[a:b], kind='source_cue_context') for a, b in spans]
    if evidence:
        a = text.index(evidence); result.append(dict(start=a, end=a + len(evidence), quote=evidence, kind='current_effective_evidence'))
    return result


def discover_cell(target, state, evidence, source_hits, all_states, prior_member):
    eh = hits(evidence)
    bone = source_hits['bone'] + source_hits['contusion'] + source_hits['fracture']
    bone_edema = [x for x, y in nearby(source_hits['edema'], source_hits['bone'] + source_hits['marrow'])]
    triggers = {}
    def add(rule, cues): triggers[rule] = cues
    if state == 'positive': add('P01_observed_positive_criterion_review', [])
    if source_hits['microtrabecular']: add('P02_microtrabecular_fracture_morphology', source_hits['microtrabecular'])
    cross = nearby(source_hits['contusion'] + bone_edema, source_hits['fracture'], 350)
    if cross or all(all_states[t] == 'positive' for t in BONE_TARGETS):
        add('P03_contusion_fracture_nearby_context', [x for pair in cross for x in pair])
    for rule, family in [('P04_infarct_necrosis_destructive_scope','infarct_necrosis_destructive'),
                         ('P07_historical_or_chronic_bone_context','history_chronic'),
                         ('P14_equivocal_bone_context','equivocal'),
                         ('P15_fracture_line_cortical_impaction_context','line_cortex_impaction')]:
        linked = [x for x, y in nearby(source_hits[family], bone)]
        if linked or (eh[family] and any(eh[x] for x in ['bone','marrow','contusion','fracture','edema'])):
            add(rule, linked)
    if source_hits['subchondral_osteochondral']:
        add('P05_osteochondral_subchondral_context', source_hits['subchondral_osteochondral'])
    if bone_edema: add('P06_bone_edema_causality', bone_edema)
    qlinked = [x for x,y in nearby(source_hits['qualifier'], bone)]
    qdenial = [x for x,y in nearby(qlinked, source_hits['denial_normal'], 180)]
    if qdenial or (eh['qualifier'] and eh['denial_normal']): add('P08_qualified_bone_denial', qdenial)
    if target == 'Contusion' and state == 'positive' and not(source_hits['trauma'] or source_hits['contusion']):
        add('P09_contusion_positive_no_impact_cue', bone_edema)
    if target == 'Fracture' and state == 'positive' and not source_hits['acute']:
        add('P10_fracture_positive_no_acute_cue', source_hits['fracture'])
    if state == 'negative' and not any(eh[x] for x in ['bone','marrow','contusion','fracture','edema']):
        add('P11_negative_evidence_bone_scope_not_lexically_detected', [])
    if target == 'Contusion' and state == 'negative' and eh['fracture'] and not any(eh[x] for x in ['bone','marrow','contusion','edema']):
        add('P12_contusion_negative_fracture_only_quote_proxy', [])
    if prior_member: add('P13_preexisting_source_bound_bone_inventory', [])
    alt = nearby(bone_edema, source_hits['alternative_cause'], 350)
    if alt: add('P16_bone_edema_with_alternative_cause', [x for pair in alt for x in pair])
    cue_rows = {(x['family'],x['start'],x['end']):x for group in triggers.values() for x in group}
    return sorted(triggers), sorted(cue_rows.values(), key=lambda x:(x['start'],x['end'],x['family']))
