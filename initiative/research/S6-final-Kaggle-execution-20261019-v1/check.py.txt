def load(name, path):
    spec = importlib.util.spec_from_loader(name, SourceFileLoader(name, str(path)))
    m = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(m)
    return m

def state_trial(phase, fail_save=False):
    with tempfile.TemporaryDirectory() as tmp:
        root = Path(tmp)
        F = root / "control"
        models = root / "models"
        P = root / "production"
        (F / "code").mkdir(parents=True)
        (models / candidate["role"]).mkdir(parents=True)
        (models / candidate["role"] / ("qualification.private.py")).write_text(source)
        (models / candidate["role"] / ("competition.private.py")).write_text(source)
        (models / candidate["role"] / "prepared.private.json").write_text(json.dumps(candidate))
        (models / "verified.json").write_text(
            json.dumps(
                {
                    "status": "actual_both_original_fixed_final_H100_models_packaged_for_separate_T4_qualification",
                    "candidates": {candidate["role"]: candidate},
                }
            )
        )
        (F / "staged.json").write_text(json.dumps({"source_git": "f" * 40, "source_sha256": {}}))
        target = F / candidate["role"]
        target.mkdir()
        mode = "qualification" if phase == "push" else "competition"
        row = {
            "candidate": candidate,
            "role": candidate["role"],
            "model_sha256": candidate["model_sha256"],
            "state": "full_parent_epoch_dataset_verified"
            if phase == "push"
            else "full_parent_epoch_competition_accepted",
            "dataset": dataset,
        }
        if phase == "submit":
            row["competition"] = {"source_sha256": source_sha, "kernel_id": 321, "receipt": public}
            (target / "competition-actual-accepted.json").write_text(
                json.dumps(
                    {
                        "receipt": public,
                        "CSV_verified_in_RAM": {"test_roster_sha256": policy.ROSTER},
                    }
                )
            )
        key = "licensed-parent-full-epoch-serving:" + candidate["model_sha256"]
        dbpath = root / "ledger.sqlite3"
        with workflow.ledger(dbpath, "godlydonuts") as db:
            db.execute("INSERT INTO meta VALUES (?,?)", (key, json.dumps(row)))
            db.commit()
        calls = []
        request = policy.request(candidate["role"], mode, source, source_sha, dataset)

        class Remote:
            def call(self, service, method, body):
                calls.append(method)
                if method == "IntrospectToken":
                    return {"active": True, "username": "godlydonuts"}
                if method == "GetAcceleratorQuotaStatistics":
                    return {"gpuQuota": {"totalTimeAllowed": "999999s"}}
                if method == "SaveKernel":
                    if fail_save:
                        raise RuntimeError("simulated uncertain RPC after intent")
                    return {"ref": "/code/" + request["slug"], "versionNumber": 1, "kernelId": 321}
                if method == "CreateCodeSubmission":
                    return {"ref": 123456}
                raise AssertionError(method)

            def kernel(self, method, slug, version):
                if method == "GetKernelSessionStatus":
                    return {"status": "COMPLETE"}
                return {"metadata": {**request, "id": 321}, "blob": {"source": source}}

            def require_absent_kernel(self, slug):
                return {"synthetic_owned_absence": True}

            def account(self):
                return {
                    "entered": True,
                    "checked_at": workflow.utcnow(),
                    "num_today": 0,
                    "num_allowed_now": 5,
                }

            def submissions(self):
                return []

            def competition(self, method, body):
                return self.call("competition", method, body)

        fake_w = SimpleNamespace(
            Remote=Remote,
            SAVE_KERNEL_INVALID=workflow.SAVE_KERNEL_INVALID,
            ledger=lambda path, account: workflow.ledger(dbpath, account),
            verify_account=workflow.verify_account,
            new_json=workflow.new_json,
            object_hash=workflow.object_hash,
            utcnow=workflow.utcnow,
            allowance=workflow.allowance,
        )
        publication = SimpleNamespace(metadata=lambda remote, slug: dataset)
        capacity = SimpleNamespace(admission=lambda *a, **k: {"available": True})
        modules = {
            "workflow": fake_w,
            "publication": publication,
            "policy": policy,
            "gpu_capacity": capacity,
        }
        program.F, program.D, program.P = F, models, P
        program.bounds = lambda: None
        program.load = lambda name: modules[name]
        original_path = Path
        program.Path = lambda value: (
            root / "claim"
            if str(value) == "/home/sa305415/rsna-knee/kaggle/coordinator-claim"
            else original_path(value)
        )
        value = {
            "phase": phase,
            "role": candidate["role"],
            "mode": mode,
            "source_git": "f" * 40,
            "source_sha256": {},
            "token": "synthetic-not-a-token",
        }
        if fail_save:
            with pytest.raises(RuntimeError):
                program.execute(value)
            assert (root / "claim/owner.json").exists() and (
                target / "qualification-push-intent.json"
            ).exists()
            with workflow.ledger(dbpath, "godlydonuts") as db:
                saved = json.loads(
                    db.execute("SELECT value FROM meta WHERE key=?", (key,)).fetchone()[0]
                )
            assert saved["state"] == "full_parent_epoch_qualification_push_intent"
            assert (
                program.execute(value)["status"] == "waiting_existing_canonical_claim_no_mutation"
            )
            assert calls.count("SaveKernel") == 1
        else:
            out = program.execute(value)
            assert (
                out["status"] == "one_exact_final_code_submission_accepted"
                and out["public_example_CSV_uploaded"] is False
            )
            assert not (root / "claim").exists()
            with pytest.raises(AssertionError):
                program.execute(value)
            assert calls.count("CreateCodeSubmission") == 1
            submitted = json.loads((target / "entry-intent.json").read_text())
            assert submitted["fileName"] == "submission.csv" and submitted["version"] == 1
        program.Path = Path
