# Initial observations are source/input diagnostics, not clinical endpoints.
def drift():
    errors = []
    score_max = vote_max = 0.0
    with torch.inference_mode():
        for i in range(4):
            positions, first, logq = sample(data.uids[i], recipe, 0, data.proposal(i))
            for mirror in (first, not first):
                case = data.case(i, positions, mirror)
                images = case["images"].to(device)[None]
                reference = {n: t.to(device)[None] for n, t in case["attention_reference"].items()}
                with torch.autocast("cuda", dtype=torch.float16):
                    output = model(images, logq.to(device)[None], reference).float()
                    encoded = model.encode(images)
                with torch.autocast("cuda", enabled=False):
                    h = model.norm(encoded.float())
                    s = model.att(h)
                    v = torch.einsum("bkd,td->bkt", h, model.clsW)
                idx = reference["indices"][..., None].expand_as(s)
                score_max = max(
                    score_max, float((s - reference["scores"].gather(1, idx)).abs().max())
                )
                vote_max = max(vote_max, float((v - reference["votes"].gather(1, idx)).abs().max()))
                errors.append(
                    (output.sigmoid()[0] - case["parent_logits"].to(device).sigmoid())
                    .detach()
                    .cpu()
                )
    e = torch.stack(errors)
    return {
        "studies": 4,
        "views": 8,
        "max_probability_gap_to_frozen_full_teacher": float(e.abs().max()),
        "RMS_probability_gap_to_frozen_full_teacher": float(e.square().mean().sqrt()),
        "max_selected_score_change": score_max,
        "max_selected_vote_change": vote_max,
        "actual_encoder_forward": True,
        "clinical_accuracy_claim": False,
    }
