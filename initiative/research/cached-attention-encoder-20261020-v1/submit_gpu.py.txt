"""Submit only the newly admitted real recovery; never restart any original run."""

import hashlib
import json
import os
import shlex
import shutil
import subprocess
import sys
from pathlib import Path

v = json.load(sys.stdin)
assert os.uname().nodename == "evc48" and not os.environ.get("SLURM_JOB_GPUS")
D = Path("/tmp/rsna-cached-attention-encoder-20261020-v1")
s = json.loads((D / "GPU-prepared.json").read_text())
assert s == v["prepared"]
assert (
    s["source_git"] == v["source_git"]
    and s["training_engineering_studies"] == 17
    and s["maximum_updates_each"] == 6
)
assert not (D / "GPU-dispatch-intent.json").exists() and not (D / "GPU-run-intent.json").exists()
for name in ("worker", "driver"):
    assert (
        hashlib.sha256((D / ("gpu_" + name + ".py")).read_bytes()).hexdigest()
        == s[name + "_sha256"]
    )
for name, h in s["modules"].items():
    assert hashlib.sha256((D / "code" / name).read_bytes()).hexdigest() == h
assert (
    shutil.disk_usage("/tmp").free > (20 + 87 + 12 + 24 + 4 + 12) * 2**30 + 128 * 2**20 + 8 * 2**20
)
node = subprocess.check_output(["scontrol", "show", "node", "evc48", "-o"], text=True)
assert "gres/gpu=2" in next(x for x in node.split() if x.startswith("CfgTRES="))
include = json.loads(
    Path("/tmp/rsna-medgemma15-runtime-headers-20261019-v1/verified.json").read_text()
)["include_path"]
env = [
    "env",
    "PYTHONDONTWRITEBYTECODE=1",
    "PYTHONNOUSERSITE=1",
    "OMP_NUM_THREADS=1",
    "MKL_NUM_THREADS=1",
    "OPENBLAS_NUM_THREADS=1",
    "HF_HUB_OFFLINE=1",
    "TRANSFORMERS_OFFLINE=1",
    "CUBLAS_WORKSPACE_CONFIG=:4096:8",
    "C_INCLUDE_PATH=" + include,
    "CPLUS_INCLUDE_PATH=" + include,
    "CUDA_CACHE_MAXSIZE=" + str(128 * 2**20),
]
for key in ("XDG_CACHE_HOME", "TORCHINDUCTOR_CACHE_DIR", "TRITON_CACHE_DIR", "CUDA_CACHE_PATH"):
    env.append(key + "=" + str(D / "runtime-cache"))
args = [
    "sbatch",
    "--parsable",
    "--account=skattel",
    "--partition=normal",
    "--qos=normal",
    "--nodelist=evc48",
    "--nodes=1",
    "--ntasks=1",
    "--cpus-per-task=2",
    "--gres=gpu:nvidia_h100_pcie:1",
    "--mem=32G",
    "--time=00:30:00",
    "--job-name=rsna-cached-attention-encoder-recovery",
    "--chdir=" + str(D),
    "--output=" + str(D / "GPU-stdout.private.log"),
    "--error=" + str(D / "GPU-stderr.private.log"),
    "--wrap=" + shlex.join(env + [sys.executable, "-B", str(D / "gpu_driver.py")]),
]
with (D / "GPU-dispatch-intent.json").open("x") as f:
    json.dump(
        {
            "source_git": v["source_git"],
            "maximum_new_GPU_jobs": 1,
            "diagnostic_only": True,
            "all_original_live_jobs_untouched": True,
            "automatic_retry": False,
        },
        f,
    )
x = subprocess.run(args, capture_output=True, text=True)
(D / "GPU-dispatch-response.json").write_text(
    json.dumps({"returncode": x.returncode, "stdout": x.stdout, "stderr": x.stderr}) + "\n"
)
assert x.returncode == 0, "Reconcile uncertain original dispatch read-only"
job = x.stdout.strip().split(";")[0]
assert job.isdigit()
out = {
    "status": "actual_new_cached_attention_real_encoder_recovery_submitted_once",
    "CPU_job": os.environ["SLURM_JOB_ID"],
    "GPU_job": job,
    "source_git": v["source_git"],
    "node": "evc48",
    "physical_node_GPUs": 2,
    "requested_GPU": 1,
    "requested_CPU": 2,
    "engineering_training_studies": 17,
    "engineering_prediction_studies": 15,
    "maximum_updates_each": 6,
    "variants": ["reference", "linear"],
    "production_fitting_or_competition_entry": False,
    "actual_GPU_progress_or_acceptance": False,
    "original_live_jobs_restarted_or_cancelled": False,
}
(D / "GPU-submitted.json").write_text(json.dumps(out, sort_keys=True, indent=2) + "\n")
print(json.dumps(out), flush=True)
