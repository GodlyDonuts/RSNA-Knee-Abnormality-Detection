@contextlib.contextmanager
def range_opener(row, offset):
    headers = {"Authorization": "Bearer " + token, "Accept-Encoding": "identity"}
    if offset:
        headers["Range"] = f"bytes={offset}-"
    url = (
        "https://huggingface.co/datasets/GodlyDonuts/RSNA-Data/resolve/"
        + revision
        + "/"
        + quote(row["path"], safe="/")
    )
    with requests.get(url, headers=headers, timeout=(10, 90), stream=True) as response:
        response.raw.decode_content = False
        yield source_resume.SourceResponse(
            response.raw, response.status_code, response.headers, revision, row["path"]
        )


@contextlib.contextmanager
def opener(row):
    with range_opener(row, 0) as response:
        assert response.status == 200
        assert response.headers.get("Content-Encoding", "identity").lower() == "identity"
        yield response.stream

def output_path(index):
    old = inv["orphan_caches"].get(index)
    if old is None:
        return D / "cache" / f"case-{index:04d}.private.npz"
    try:
        with np.load(old, allow_pickle=False) as z:
            # Force all ZIP member CRC/decompression checks; partial bytes are
            # retained and replaced only in the distinct successor directory.
            for name in z.files:
                z[name]
            manifest = json.loads(z["manifest"].tobytes())
    except (ValueError, OSError, EOFError, zipfile.BadZipFile):
        return D / "cache" / f"case-{index:04d}.private.npz"
    assert manifest["version"] == VERSION
    assert manifest["source"] == recon.source_provenance(records[index], revision)
    before = signature(old)
    with old.open("rb") as f:
        sha = hashlib.file_digest(f, "sha256").hexdigest()
    assert signature(old) == before
    orphan_proofs[str(old)] = {
        "version": VERSION,
        "bytes": before[1],
        "sha256": sha,
        "stat": before,
        "groups": manifest["groups"],
        "regions": manifest["regions"],
    }
    return old


def write(path, sources, provenance, *, cap_bytes):
    if str(path) in orphan_proofs:
        proof = orphan_proofs[str(path)]
        restored, actual_provenance = load(path, proof)
        assert actual_provenance == provenance
        assert len(restored) == len(sources)
        for source, admitted in zip(sources, restored):
            assert torch.equal(normalized(source), admitted.frames)
            assert all(
                torch.equal(getattr(source, key), getattr(admitted, key))
                for key in ("aliases", "acquisition", "ordinal")
            )
        return proof
    return original_write(path, sources, provenance, cap_bytes=cap_bytes)
