"""One CPU replay dispatch, credential only through SSH stdin."""

import hashlib
import importlib.util
import json
import shlex
import subprocess
from pathlib import Path

R = Path.cwd()
D = Path(__file__).resolve().parent
L = R / "artifacts/S6-Kaggle-existing-output-replay-20261019-v1"
L.mkdir(exist_ok=True)
assert not (L / "intent.json").exists()
assert not (
    R
    / "artifacts/S6-complete-target-execution-v2-20261019/production-owner-v2/production-CPU-dispatch-intent.json"
).exists()
actual = json.loads(
    (
        R / "initiative/research/S6-Kaggle-target-recovery-20261019-v1/actual-terminal.json"
    ).read_text()
)
assert actual["kernel_id"] == 139695847 and actual["actual_status"] == "ERROR"
git = subprocess.check_output(["git", "rev-parse", "HEAD"], text=True).strip()
for name in ("replay.py.txt", "run_once.py.txt"):
    p = D / name
    assert p.read_bytes() == subprocess.check_output(
        ["git", "show", git + ":" + str(p.relative_to(R))]
    )
with (L / "intent.json").open("x") as f:
    json.dump(
        {
            "source_git": git,
            "program_sha256": hashlib.sha256((D / "replay.py.txt").read_bytes()).hexdigest(),
            "kernel_id": 139695847,
            "version": 1,
            "new_GPU_jobs": 0,
            "maximum_held_original_unstarted_jobs": 1,
            "automatic_retry": False,
        },
        f,
    )
env = R / ".env"
assert env.stat().st_mode & 0o077 == 0
values = {
    k.strip(): z.strip().strip("\"'")
    for line in env.read_text().splitlines()
    for k, sep, z in [line.partition("=")]
    if sep and not line.lstrip().startswith("#")
}
s = importlib.util.spec_from_file_location(
    "t", R / "initiative/research/paired-logit-head-gradient-20261018-v1/stdin_transport.py"
)
t = importlib.util.module_from_spec(s)
s.loader.exec_module(t)
args = [
    "srun",
    "--account=skattel",
    "--partition=normal",
    "--qos=normal",
    "--nodelist=evc48",
    "--ntasks=1",
    "--cpus-per-task=1",
    "--gres=none",
    "--mem=16G",
    "--time=00:30:00",
    "--job-name=rsna-S6-existing-output-CPU-replay",
    "env",
    "CUDA_VISIBLE_DEVICES=",
    "PYTHONDONTWRITEBYTECODE=1",
    "PYTHONNOUSERSITE=1",
    "OMP_NUM_THREADS=1",
    "/tmp/rsna-environment-971427-pyk8rld8/environment/bin/python",
    "-c",
    t.COMMAND,
]
x = subprocess.run(
    ["ssh", "-S", "/tmp/rsna-research-20261001-stable.sock", "newton", shlex.join(args)],
    input=t.envelope(
        (D / "replay.py.txt").read_text(), {"token": values["KAGGLE_API_TOKEN"], "source_git": git}
    ),
    capture_output=True,
    text=True,
    timeout=2100,
)
(L / "transport.json").write_text(
    json.dumps({"returncode": x.returncode, "stdout": x.stdout, "stderr": x.stderr}, indent=2)
    + "\n"
)
assert x.returncode == 0, "Inspect the exact original CPU/transfer before any further mutation"
z = json.loads(x.stdout)
(D / "actual-accepted.json").write_text(json.dumps(z, indent=2) + "\n")
print(json.dumps(z), flush=True)
