"""New CPU-only retained-input bridge for an H100-specific gradient experiment."""
import ast
import base64
import hashlib
import json
import os
import sys
from pathlib import Path

v = json.load(sys.stdin)
assert os.uname().nodename == 'evc33' and not os.environ.get('SLURM_JOB_GPUS')
source = Path('/tmp/rsna-licensed-parent-whole94-recovery-20261016-v1/recovery.private.py').read_text()
literals = {}
for node in ast.parse(source).body:
    if isinstance(node, ast.Assign) and len(node.targets) == 1 and isinstance(node.targets[0], ast.Name):
        if node.targets[0].id in ('TARGETS', 'WORKER'):
            literals[node.targets[0].id] = ast.literal_eval(node.value)
raw = base64.b64decode(literals['TARGETS'], validate=True)
assert hashlib.sha256(raw).hexdigest() == '87cbd096202eb3a0512f918b181ecd765c893817beed4678e4eea2b47032c834'
targets = json.loads(raw)
worker = base64.b64decode(literals['WORKER'], validate=True).decode()
main = [n for n in ast.parse(worker).body if isinstance(n, ast.FunctionDef) and n.name == 'main']
assert len(main) == 1
prefix = ''.join(worker.splitlines(keepends=True)[:main[0].lineno - 1])
# Input-only execution: omit trainer imports, never construct an encoder/optimizer.
tree = ast.parse(prefix)
tree.body = [n for n in tree.body if not (isinstance(n, ast.ImportFrom) and (n.module or '').startswith('rsna.'))]
ns = {'__name__': 'new_H100_input_bridge', '__file__': 'new_H100_input_bridge.py'}
exec(compile(ast.fix_missing_locations(tree), 'unchanged_original_renderer_and_PixelCases', 'exec'), ns)
import torch
assert torch.cuda.device_count() == 0
assert targets['eligible_fitting_only'] is True
torch.set_num_threads(1)
torch.set_num_interop_threads(1)
pixel = Path('/tmp/rsna-licensed-parent-pixels32-20261016-v1/collected/index.private.json')
feature = Path('/tmp/rsna-released-context-head-fit-20261016-v1/features32/index.private.json')
assert hashlib.sha256(pixel.read_bytes()).hexdigest() == '3a1567989bf65779aeb19785dafdedef7386fb6bedba5ec45da6d3a67a9a79e3'
assert hashlib.sha256(feature.read_bytes()).hexdigest() == 'a1bba9e618444f16ddd453efa1bb8650245f8259dcd9a00348b17efd87c1e100'
def exact_index(size, sha):
    assert pixel.stat().st_size == size and hashlib.sha256(pixel.read_bytes()).hexdigest() == sha
    return pixel
def original32():
    return feature, json.loads(feature.read_text())['records']
ns['exact_index'] = exact_index
ns['_pixel_original32'] = original32
cases = ns['PixelCases'](targets, list(range(4)))
assert len(cases) == 4 and not set(cases.uids) & set(targets['uids'][16:])
checks = []
for i in range(4):
    for mirror in (False, True):
        case = cases.case(i, list(range(94)), mirror)
        assert tuple(case['images'].shape) == (94, 3, 320, 320)
        assert case['images'].dtype == torch.float32 and torch.isfinite(case['images']).all()
        assert torch.equal(case['labels'], torch.tensor(targets['labels'][i], dtype=torch.float32))
        assert torch.equal(case['mask'], torch.tensor(targets['masks'][i], dtype=torch.bool))
        checks.append({'case_index': i, 'mirror': mirror, 'tensor_sha256': ns['_pixel_tensor_sha'](case['images'])})
        del case
print(json.dumps({'status': 'new_H100_location_bridge_four_original_fitting_inputs_equal', 'CPU_job': os.environ['SLURM_JOB_ID'], 'source_git': v['source_git'], 'program_sha256': hashlib.sha256(v['program'].encode()).hexdigest(), 'original_worker_sha256': hashlib.sha256(worker.encode()).hexdigest(), 'inputs': checks, 'original_renderer_and_decode_numerical_bodies_unchanged': True, 'new_model_transfers_GPU_optimizer_updates_entries': 0, 'model_runtime_gradient_or_accuracy_acceptance': False}), flush=True)
