source acquisition, labels, target teacher, model, optimizer or GPU operation.

assert selection["source_revision"] == "52f995f4568e36bfa5fbbaf69a89af41226b4f5d"

        [

B = Path("/tmp/rsna-S6-Kaggle-production-route-20261019-v1")
paid = sum(
    p.stat().st_size
    for p in B.rglob("*")
    if p.is_file() and not p.is_symlink() and "inputs" not in p.relative_to(B).parts
)
assert shutil.disk_usage("/tmp").free > (20 + 8) * 2**30 + max(0, 8 * 2**20 - paid) + 34 * 2**20
memory = {
    x.split(":")[0]: int(x.split()[1]) * 1024
    for x in Path("/proc/meminfo").read_text().splitlines()
    if ":" in x
}
assert memory["MemAvailable"] > 24 * 2**30

source = v["input_source"]
assert hashlib.sha256(source.encode()).hexdigest() == v["input_source_sha256"]
D.mkdir(mode=0o700)
(D / "inputs.py").write_text(source)
(D / "intent.json").write_text(
    json.dumps(
        {
            "CPU_job": os.environ["SLURM_JOB_ID"],
            "source_git": v["source_git"],
            "new_source_downloads": 0,

            "controls_cap_bytes": 2 * 2**20,
            "job_RSS_cap_bytes": 4 * 2**30,
            "new_shared_bytes_files": 0,

        }
    )
    + "\n"
)
spec = importlib.util.spec_from_file_location("new_raw_masked_predictive_inputs", D / "inputs.py")
m = importlib.util.module_from_spec(spec)
sys.modules[spec.name] = m
spec.loader.exec_module(m)
start = time.monotonic()
groups = {}
series = sorted({r["series_uid"] for r in rows})
alias_table = {}
bits = {}
file_bytes = 0
for row in rows:
    assert row["study_uid"] == uid and row["split"] == "train_series"
    path = ram / "data/train_series" / uid / row["series_uid"] / (row["sop_uid"] + ".dcm")
    raw = path.read_bytes()
    assert len(raw) == row["bytes"] and hashlib.sha256(raw).hexdigest() == row["sha256"]
    file_bytes += len(raw)
    ds = pydicom.dcmread(path)
    assert (str(ds.StudyInstanceUID), str(ds.SeriesInstanceUID), str(ds.SOPInstanceUID)) == (
        uid,
        row["series_uid"],
        row["sop_uid"],
    )
    pixel = ds.pixel_array
    assert pixel.ndim == 2
    slope = float(ds.get("RescaleSlope", 1))
    intercept = float(ds.get("RescaleIntercept", 0))
    assert np.isfinite([slope, intercept]).all() and slope > 0
    b = int(ds.BitsStored)
    signed = int(ds.PixelRepresentation)
    assert signed in (0, 1)
    key = hashlib.sha256(
        np.array(pixel.shape, dtype="<i8").tobytes() + pixel.astype("<f8").tobytes()
    ).hexdigest()
    if key not in alias_table:
        alias_table[key] = len(alias_table)
    frame = torch.from_numpy(pixel.astype(np.int32))
    norm = m.fixed_metadata_scale(
        frame, bits_stored=b, signed=bool(signed), photometric=str(ds.PhotometricInterpretation)
    )
    group = groups.setdefault(tuple(pixel.shape), [])
    # Actual geometry ordering stays within each source acquisition. No registration claim.
    iop = np.array(ds.get("ImageOrientationPatient", []), dtype=float)
    ipp = np.array(ds.get("ImagePositionPatient", []), dtype=float)
    position = (
        float(np.dot(ipp, np.cross(iop[:3], iop[3:])))
        if len(iop) == 6 and len(ipp) == 3
        else float(ds.InstanceNumber)
    )
    group.append(
        (
            series.index(row["series_uid"]),
            position,
            int(ds.InstanceNumber),
            frame,
            norm,
            alias_table[key],
            b,
            bool(signed),
            str(ds.PhotometricInterpretation),
        )
    )
    bits[str(b)] = bits.get(str(b), 0) + 1
assert file_bytes == 25131360
roster = 0
observed = [0, 0]
raw_intervention = True
render_intervention = True
alias_closed = True
descriptor_finite = True
visible = []
ranges = []
global_before = torch.get_rng_state().clone()
for items in groups.values():
    items.sort(key=lambda x: (x[0], x[1], x[2]))
    frames = torch.stack([x[4] for x in items])
    aliases = torch.tensor([x[5] for x in items])
    acq = torch.tensor([x[0] for x in items])
    ordinal = torch.zeros(len(items), dtype=torch.int64)
    for a in acq.unique():
        ordinal[acq == a] = torch.arange(int((acq == a).sum()))
    outputs = [m.masked_study(frames, aliases, acq, ordinal, pattern=k, halo=8) for k in (0, 1)]
    assert torch.equal(outputs[0].observed, ~outputs[1].observed)
    roster += len(outputs[0].descriptors)
    centers = torch.arange(len(items))
    neighbors = []
    for i in range(len(items)):
        same = (acq == acq[i]).nonzero().flatten()
        j = int((same == i).nonzero()[0])
        neighbors.append([int(same[max(0, j - 1)]), i, int(same[min(len(same) - 1, j + 1)])])
    neighbors = torch.tensor(neighbors, dtype=torch.int64)
    for pattern, study in enumerate(outputs):
        altered = []
        for i, item in enumerate(items):
            value = item[3].clone()
            low = -(2 ** (item[6] - 1)) if item[7] else 0
            high = low + 2 ** item[6] - 1
            value[study.raw_hidden[i]] = low + high - value[study.raw_hidden[i]]
            altered.append(
                m.fixed_metadata_scale(
                    value, bits_stored=item[6], signed=item[7], photometric=item[8]
                )
            )
        changed = m.masked_study(
            torch.stack(altered), aliases, acq, ordinal, pattern=pattern, halo=8
        )
        raw_intervention &= torch.equal(study.context_frames, changed.context_frames)
        assert not torch.equal(
            study.descriptors[study.observed], changed.descriptors[changed.observed]
        )
        # Stream at most eight triplets to avoid a whole-exam rendered tensor tree.
        for first in range(0, len(items), 8):
            left = m.triplets(
                study, centers[first : first + 8], neighbors[first : first + 8], side=320
            )
            right = m.triplets(
                changed, centers[first : first + 8], neighbors[first : first + 8], side=320
            )
            render_intervention &= torch.equal(left, right)
        for alias in aliases.unique():
            peers = (aliases == alias).nonzero().flatten()
            alias_closed &= all(
                torch.equal(study.raw_hidden[int(peers[0])], study.raw_hidden[int(i)])
                for i in peers
            )
        descriptor_finite &= bool(torch.isfinite(study.descriptors).all())
        observed[pattern] += int(study.observed.sum())
        visible.append(float((~study.raw_hidden).float().mean()))
    ranges.append(
        {
            "metadata_scaled_min": float(frames.min()),
            "metadata_scaled_max": float(frames.max()),
            "metadata_scaled_mean": float(frames.mean()),
        }
    )
assert (
    raw_intervention

    and alias_closed
    and descriptor_finite
    and torch.equal(global_before, torch.get_rng_state())
)
