paths = {}
for alias, root in roots.items():
    assert root.exists() and not root.is_symlink() and root.stat().st_uid == os.getuid()
    candidates = [root] if root.is_file() else sorted(root.rglob("*"))
    for p in candidates:
        assert not p.is_symlink(), "Explicit original source may not escape via symlinks"
        if not p.is_file() or "__pycache__" in p.parts:
            continue
        rel = p.name if root.is_file() else p.relative_to(root).as_posix()
        if alias == "source" and not (
            rel.startswith("code/")
            or rel
            in (
                "admitted.json",
                "parent_source.private.py",
                "engineering-targets.private.json",
                "source-permission.txt",
            )
        ):
            continue
        if alias == "headers" and p.suffix != ".h" and rel != "verified.json":
            continue
        paths[alias + "/" + rel] = p
# Only the two missing classifier packages and their binary dependencies, no Torch copy.
for root in sorted(S.iterdir()):
    if not (
        root.name in ("torchvision", "torchvision.libs")
        or root.name.startswith(("torchvision-",))
        and root.name.endswith(".dist-info")
    ):
        continue
    assert root.is_dir() and not root.is_symlink()
    for p in sorted(root.rglob("*")):
        assert not p.is_symlink()
        if p.is_file() and "__pycache__" not in p.parts:
            paths["extensions/" + p.relative_to(S).as_posix()] = p
assert paths and len(paths) <= 4000
assert any(n.startswith("extensions/torchvision/") for n in paths)
assert any(n.startswith("overlay/timm/") for n in paths)
stats, table = {}, {}
for name, p in paths.items():
    stat = p.stat()
    stats[name] = (stat.st_ino, stat.st_size, stat.st_mtime_ns)
    with p.open("rb") as f:
        sha = hashlib.file_digest(f, "sha256").hexdigest()
    assert (p.stat().st_ino, p.stat().st_size, p.stat().st_mtime_ns) == stats[name]
    table[name] = {"bytes": stat.st_size, "sha256": sha}
valid_table(table)

        assert len(raw) <= 2**21
        output.write(AUX_MAGIC + len(raw).to_bytes(8, "big") + raw)
