def write(name, value):
    with (L / name).open("x") as f:
        json.dump(value, f, indent=2)
        f.write("\n")


def remote(args):
    return subprocess.run([*SSH, shlex.join(args)], capture_output=True, text=True, timeout=60)


def state(job):
    x = remote(["sacct", "-X", "-n", "-P", "-j", job, "--format=JobIDRaw,State,ExitCode"])
    # Access loss is not a job failure. Retain the same handle and re-observe.
    if x.returncode != 0:
        return None
    rows = [r.split("|") for r in x.stdout.splitlines() if r.split("|")[0] == job]
    return rows[0][1:3] if rows else None


def cpu(program, payload, name):
    args = [
        "srun",
        "--account=skattel",
        "--partition=preemptable",
        "--qos=preemptable",
        "--nodelist=evc48",
        "--ntasks=1",
        "--cpus-per-task=1",
        "--gres=none",
        "--mem=6G",
        "--time=00:15:00",
        "--job-name=" + name,
        "env",
        "CUDA_VISIBLE_DEVICES=",
        "PYTHONDONTWRITEBYTECODE=1",
        "PYTHONNOUSERSITE=1",
        "OMP_NUM_THREADS=1",
        "/tmp/rsna-environment-971427-pyk8rld8/environment/bin/python",
        "-B",
        "-c",
        t.COMMAND,
    ]
    return subprocess.run(
        [*SSH, shlex.join(args)],
        input=t.envelope(program, payload),
        capture_output=True,
        text=True,
        timeout=7200,
    )

        "COMPLETED",
        "FAILED",
        "CANCELLED",
        "TIMEOUT",
        "OUT_OF_MEMORY",
        "NODE_FAIL",
        "PREEMPTED",
        "BOOT_FAIL",
        "DEADLINE",
    )
