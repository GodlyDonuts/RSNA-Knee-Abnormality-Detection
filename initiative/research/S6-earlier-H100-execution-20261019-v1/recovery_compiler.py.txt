"""New one-H100 recovery route; retain every original numerical trainer module."""

import ast
import base64
import hashlib
from pathlib import PurePosixPath

ROOT = "/tmp/rsna-S6-Newton-student-recovery-20261019-v1"


def sha(s):
    return hashlib.sha256(s.encode()).hexdigest()


def values(source):
    return {
        n.targets[0].id: ast.literal_eval(n.value)
        for n in ast.parse(source).body
        if isinstance(n, ast.Assign)
        and len(n.targets) == 1
        and isinstance(n.targets[0], ast.Name)
        and n.targets[0].id in ("WORKER", "FILES", "TARGETS", "PERMISSION")
    }


def once(source, old, new):
    if source.count(old) != 1:
        raise ValueError("Exact original source site required, never a broad device substitution")
    return source.replace(old, new)


def build(source, metadata):
    if (
        sha(source) != metadata.get("source_sha256")
        or metadata.get("actual_GPU_recovery") is not False
        or metadata.get("full_production_fit_dispatched") is not False
        or metadata.get("roles") != ["parent_only", "cross_lineage_equal_rank"]
    ):
        raise ValueError(
            "Original complete-target recovery compilation and preparation scope required"
        )
    old = values(source)
    if set(old) != {"WORKER", "FILES", "TARGETS", "PERMISSION"} or {
        n: sha(s) for n, s in old["FILES"].items()
    } != metadata.get("own_module_sha256"):
        raise ValueError("Exact original numerical module closure required")
    worker = base64.b64decode(old["WORKER"], validate=True).decode()
    if sha(worker) != metadata.get("worker_sha256"):
        raise ValueError("Exact original compiled worker required")
    worker = once(
        worker, "assert int(os.environ['WORLD_SIZE'])==2", "assert int(os.environ['WORLD_SIZE'])==1"
    )
    worker = once(
        worker,
        "assert torch.cuda.get_device_name(device)=='Tesla T4'",
        "assert torch.cuda.device_count()==1 and torch.cuda.get_device_name(device).startswith('NVIDIA H100')",
    )
    worker = once(worker, "for i in range(rank,16,2):", "for i in range(rank,16,1):")
    worker = once(
        worker,
        "gathered=[None,None];dist.all_gather_object(gathered,pred)",
        "gathered=[None];dist.all_gather_object(gathered,pred)",
    )
    # These are only storage mounts. No reader/renderer/target calculation changes.
    assert worker.count("/kaggle/input") >= 3 and worker.count("/kaggle/working") == 1
    worker = worker.replace("/kaggle/input", ROOT + "/input").replace(
        "/kaggle/working", ROOT + "/work"
    )
    new = once(
        source,
        "WORKER=" + repr(old["WORKER"]),
        "WORKER=" + repr(base64.b64encode(worker.encode()).decode()),
    )
    new = once(new, "'--nproc_per_node=2'", "'--nproc_per_node=1'")
    new = once(
        new,
        "env={**os.environ,'PYTHONPATH':str(code)}",
        "env={**os.environ,'PYTHONPATH':str(code)+os.pathsep+'/tmp/rsna-compact-native-n3-assets-20261016-v1/runtime-overlay'}",
    )
    new = once(new, "full['binding']['world_size']==2", "full['binding']['world_size']==1")
    new = once(new, "for rank in (0,1)", "for rank in (0,)")
    new = once(new, "x['GPU']=='Tesla T4'", "x['GPU'].startswith('NVIDIA H100')")
    new = once(new, "'world_size':2", "'world_size':1")
    new = once(
        new,
        "actual_complete_S6_target_change_full_encoder_twoT4_MRI_recovery_accepted",
        "actual_complete_S6_target_change_full_encoder_oneH100_MRI_recovery_accepted",
    )
    assert new.count("/kaggle/input") == 1 and new.count("/kaggle/working") == 4
    new = new.replace("/kaggle/input", ROOT + "/input").replace("/kaggle/working", ROOT + "/work")
    assert "/kaggle/" not in worker + new
    updated = values(new)
    assert (
        updated["FILES"] == old["FILES"]
        and updated["TARGETS"] == old["TARGETS"]
        and updated["PERMISSION"] == old["PERMISSION"]
    )
    for name in updated["FILES"]:
        assert not PurePosixPath(name).is_absolute() and ".." not in PurePosixPath(name).parts
    compile(worker, "new_oneH100_complete_teacher_worker", "exec")
    compile(new, "new_oneH100_complete_teacher_recovery", "exec")
    return new, {
        "status": "new_oneH100_complete_target_recovery_source_prepared_only",
        "original_twoT4_source_sha256": sha(source),
        "source_sha256": sha(new),
        "worker_sha256": sha(worker),
        "own_module_sha256": metadata["own_module_sha256"],
        "all_original_numerical_modules_unchanged": True,
        "changes": [
            "one H100 assigned CUDA rank",
            "one-rank engineering collection and acceptance",
            "private node-local storage mounts",
            "explicit new hardware outcome scope",
            "reuse existing admitted timm/torchvision runtime overlay",
        ],
        "both_roles_effective_study_batch": 8,
        "both_roles_sampled_windows": 16,
        "both_roles_FP16_forward_FP32_loss_dynamic_scaler": True,
        "fit_studies": 16,
        "disjoint_engineering_studies": 16,
        "full_94_two_view_engineering_inference": True,
        "complete_state_and_prediction_equality_requirements_retained": True,
        "actual_GPU_recovery": False,
        "T4_to_H100_numerical_or_accuracy_equivalence_claim": False,
        "full_production_or_competition_admission": False,
        "maximum_private_recovery_output_bytes": 12 * 2**30,
    }
