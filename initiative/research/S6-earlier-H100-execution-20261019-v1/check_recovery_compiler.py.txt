"""Test actual emitted one-rank collection and strict source/device refusals."""

import ast
import base64
import contextlib
import copy
import importlib.util
import io
import json
import runpy
import tempfile
from importlib.machinery import SourceFileLoader
from pathlib import Path
from types import SimpleNamespace

import numpy as np
import pytest
import torch

R = Path.cwd()
D = Path(__file__).resolve().parent
with contextlib.redirect_stdout(io.StringIO()):
    old = runpy.run_path(
        str(R / "initiative/research/S6-target-real-MRI-recovery-20261019-v1/check.py.txt")
    )
spec = importlib.util.spec_from_loader(
    "oneH100_compiler", SourceFileLoader("oneH100_compiler", str(D / "recovery_compiler.py.txt"))
)
m = importlib.util.module_from_spec(spec)
spec.loader.exec_module(m)
source, meta = m.build(old["source"], old["meta"])
values = m.values(source)
worker = base64.b64decode(values["WORKER"]).decode()
assert values["FILES"] == m.values(old["source"])["FILES"]
main = next(
    n for n in ast.parse(worker).body if isinstance(n, ast.FunctionDef) and n.name == "main"
)
collection = next(
    n for n in main.body if isinstance(n, ast.If) and ast.unparse(n.test) == "a.mode != 'stop'"
)


class Cases:
    def case(self, i, positions, mirror):
        assert positions == list(range(94))
        return {"images": torch.full((94, 3, 2, 2), i / 16 + int(mirror) / 8)}


class Model:
    def eval(self):
        return self

    def __call__(self, images):
        return images.mean().expand(1, 12)


class Dist:
    @staticmethod
    def all_gather_object(gathered, pred):
        assert len(gathered) == 1 and [i for i, _ in pred] == list(range(16))
        gathered[0] = pred


with tempfile.TemporaryDirectory() as tmp:
    output = Path(tmp)
    namespace = dict(
        torch=torch,
        np=np,
        a=SimpleNamespace(mode="full"),
        rank=0,
        pred=[],
        engineering=Cases(),
        model=Model(),
        device=torch.device("cpu"),
        dist=Dist,
        output=output,
    )
    with pytest.warns(UserWarning, match="CUDA is not available"):
        exec(
            compile(
                ast.Module(body=[collection], type_ignores=[]),
                "actual_emitted_one_rank_engineering_collection",
                "exec",
            ),
            namespace,
        )
    actual = np.load(output / "engineering-predictions.private.npy", allow_pickle=False)
    assert actual.shape == (16, 12)
    expected = np.array(
        [
            ((torch.tensor(i / 16).sigmoid() + torch.tensor(i / 16 + 1 / 8).sigmoid()) / 2).item()
            for i in range(16)
        ]
    )
    np.testing.assert_allclose(actual, expected[:, None].repeat(12, axis=1), atol=1e-7, rtol=0)
    first = (output / "engineering-predictions.private.npy").read_bytes()
    namespace.update(pred=[], a=SimpleNamespace(mode="resume"))
    with pytest.warns(UserWarning, match="CUDA is not available"):
        exec(
            compile(
                ast.Module(body=[collection], type_ignores=[]),
                "actual_emitted_one_rank_engineering_collection",
                "exec",
            ),
            namespace,
        )
    assert (output / "engineering-predictions.private.npy").read_bytes() == first

checks = 0
for bad in (
    {**old["meta"], "source_sha256": "0" * 64},
    {**old["meta"], "actual_GPU_recovery": True},
    {**old["meta"], "own_module_sha256": {}},
    {**old["meta"], "roles": ["influence", "uniform"]},
):
    with pytest.raises(ValueError):
        m.build(old["source"], bad)
    checks += 1
bad = old["source"].replace("'--nproc_per_node=2'", "'--nproc_per_node=3'")
metadata = copy.deepcopy(old["meta"])
metadata["source_sha256"] = m.sha(bad)
with pytest.raises(ValueError):
    m.build(bad, metadata)
checks += 1
assert (
    meta["actual_GPU_recovery"] is False
    and meta["T4_to_H100_numerical_or_accuracy_equivalence_claim"] is False
)
print(
    json.dumps(
        {
            "status": "oneH100_recovery_compiler_synthetic_CPU_contracts_passed",
            "actual_emitted_one_rank_collection_all16_two_views_executed": True,
            "full_resume_synthetic_prediction_bytes_equal": True,
            "original_numerical_modules_unchanged": True,
            "source_refusals": checks,
            "actual_MRI_GPU_or_optimizer_acceptance": False,
        }
    )
)
