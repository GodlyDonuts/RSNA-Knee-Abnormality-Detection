"""Single RAM-only CPU input audit using committed sources; no automatic retry."""
import hashlib
import json
from pathlib import Path
import shlex
import subprocess

D = Path(__file__).resolve().parent
L = Path('artifacts/connected-witness-dense-inputs-20261017-v1')
L.mkdir(mode=0o700, parents=True, exist_ok=True)
assert not (L / 'intent.json').exists()
git = subprocess.check_output(['git', 'rev-parse', 'HEAD'], text=True).strip()
def source(name):
    return subprocess.check_output(['git', 'show', git + ':' + name], text=True)
relative = D.relative_to(Path.cwd()).as_posix() + '/'
program = source(relative + 'assess_dense_inputs.py.txt')
numerical = json.loads(source('initiative/audits/spatial-reader-engine-20261017-v1/QA-source-frozen.json'))['numerical_source_before']
module = source(relative + 'dense_inputs.py')
payload = {'source_git': git, 'program_sha256': hashlib.sha256(program.encode()).hexdigest(),
           'source': module, 'source_sha256': hashlib.sha256(module.encode()).hexdigest(), 'original118': numerical}
compile(program, 'original32_dense_MRI_CPU_input_assessment', 'exec')
with (L / 'intent.json').open('x') as f:
    json.dump({k: x for k, x in payload.items() if k not in ('source', 'original118')}, f, indent=2)
args = ['srun', '--account=skattel', '--partition=normal', '--qos=normal', '--nodelist=evc40',
        '--ntasks=1', '--cpus-per-task=1', '--gres=none', '--mem=4G', '--time=00:10:00',
        '--job-name=rsna-dense-original32-CPU-input', 'env', 'CUDA_VISIBLE_DEVICES=',
        'PYTHONNOUSERSITE=1', 'PYTHONDONTWRITEBYTECODE=1', 'OMP_NUM_THREADS=1', 'MKL_NUM_THREADS=1',
        'OPENBLAS_NUM_THREADS=1', '/tmp/rsna-environment-907632-7sr9a7i6/environment/bin/python', '-B', '-c', program]
x = subprocess.run(['ssh', '-S', '/tmp/rsna-research-20261001-stable.sock', '-o', 'BatchMode=yes',
                    'newton', shlex.join(args)], input=json.dumps(payload), text=True, capture_output=True, timeout=700)
(L / 'result.json').write_text(json.dumps({'returncode': x.returncode, 'stdout': x.stdout, 'stderr': x.stderr}, indent=2)+'\n')
assert x.returncode == 0, 'Reconcile the original CPU intent/job; no repeated dispatch'
out = json.loads(x.stdout)
(D / 'actual-dense-inputs.json').write_text(json.dumps(out, indent=2)+'\n')
print(json.dumps(out))
