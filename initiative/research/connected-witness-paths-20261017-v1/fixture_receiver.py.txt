"""One private evc26 fixture admission; original hashes, no extraction shortcuts."""
import hashlib
import json
import os
from pathlib import Path, PurePosixPath
import shutil
import subprocess
import sys
import tarfile
import time

assert os.uname().nodename == 'evc26' and not os.environ.get('SLURM_JOB_GPUS')
os.umask(0o077)
source_git, program_sha = sys.argv[1:]
assert hashlib.sha256(sys.orig_argv[sys.orig_argv.index('-c') + 1].encode()).hexdigest() == program_sha
root = Path('/tmp/rsna-connected-witness-evc26-20261017-v1')
assert not root.exists()
cap = 3 * 2**30
assert shutil.disk_usage('/tmp').free > 20 * 2**30 + cap
quota = subprocess.check_output(['lfs', 'quota', '-u', 'sa305415', '/home/sa305415'], text=True)
q = next(x.split() for x in quota.splitlines() if x.strip().startswith('/home/sa305415'))
assert (int(q[2]) - int(q[1])) * 1024 > 20 * 2**30 and int(q[6]) - int(q[5]) > 5000
start = time.monotonic()

def metadata(archive, name, limit):
    member = archive.next()
    assert member is not None and member.isfile() and member.name == name and 0 < member.size <= limit
    source = archive.extractfile(member)
    data = source.read(limit + 1)
    assert len(data) == member.size
    return data

with tarfile.open(fileobj=sys.stdin.buffer, mode='r|') as archive:
    raw_index = metadata(archive, 'pilot-recovery-v2-accepted.json', 16 * 2**20)
    assert hashlib.sha256(raw_index).hexdigest() == 'f058c0969c56d754874ceb3fbb1fb9c3fff2d2a5c9bb1554d94cf68d9a3bd18f'
    raw_membership = metadata(archive, 'membership.private.json', 2 * 2**20)
    raw_parent = metadata(archive, 'parent-run-binding.json', 8 * 2**20)
    rows = json.loads(raw_index)['private_records']
    membership = json.loads(raw_membership)
    assert set(membership) == {'fitting', 'outer'}
    assert len(membership['fitting']) == 2247 and len(membership['outer']) == 1156
    assert not set(membership['fitting']) & set(membership['outer'])
    expected_rosters = {'fitting': '850f170598e5ddd09a80317950acba29c03799f1a620eb2d4d9793d22a2f4e74', 'outer': '2983434d353efee3b3f13af5bb8cb7520e43f8fc23c329cfd0351def93da62cd'}
    for role, digest in expected_rosters.items():
        assert hashlib.sha256(json.dumps(membership[role]).encode()).hexdigest() == digest
        assert [r['study_uid'] for r in rows if r['role'] == role] == membership[role][:16]
    parent = json.loads(raw_parent)['binding']
    assert parent['release']['manifest_sha256'] == '2bf49fd2a8ae0b7c5af7e8c105fc3f2337845a92654d136fe67bc41a1e52fae7'
    assert len(rows) == 32 and sum(r['windows'] for r in rows) == 2345
    expected = {'features/' + r['file']: r for r in rows}
    assert len(expected) == 32
    total = sum(r['bytes'] for r in rows) + len(raw_index) + len(raw_membership) + len(raw_parent)
    assert total + 128 * 2**20 < cap
    root.mkdir(mode=0o700)
    (root / 'intent.json').write_text(json.dumps({'source_git': source_git, 'receiver_program_sha256': program_sha, 'CPU_job': os.environ['SLURM_JOB_ID'], 'node_root_cap_bytes': cap, 'expected_feature_members': 32, 'expected_private_input_bytes': total, 'original_full_rosters_retained': True, 'node_future_reserve_bytes': 20 * 2**30, 'other_existing_node26_assets_retained': True, 'new_shared_bytes_files': 0, 'automatic_retry': False}, indent=2) + '\n')
    (root / 'intent.json').chmod(0o600)
    for name, data in [('pilot-recovery-v2-accepted.json', raw_index), ('membership.private.json', raw_membership), ('parent-run-binding.json', raw_parent)]:
        target = root / name
        with target.open('xb') as f:
            f.write(data)
        target.chmod(0o600)
    seen = set()
    while True:
        member = archive.next()
        if member is None:
            break
        assert member.isfile() and member.name in expected and member.name not in seen
        row = expected[member.name]
        assert member.size == row['bytes'] and 0 < member.size < 128 * 2**20
        relative = PurePosixPath(member.name)
        assert not relative.is_absolute() and '..' not in relative.parts
        target = root / member.name
        assert target.resolve().is_relative_to(root.resolve()) and not target.exists()
        target.parent.mkdir(mode=0o700, parents=True, exist_ok=True)
        partial = target.with_name(target.name + '.partial')
        digest = hashlib.sha256()
        copied = 0
        source = archive.extractfile(member)
        with partial.open('xb') as f:
            while True:
                block = source.read(4 * 2**20)
                if not block:
                    break
                copied += len(block)
                assert copied <= member.size
                digest.update(block)
                f.write(block)
        assert copied == row['bytes'] and digest.hexdigest() == row['sha256']
        partial.chmod(0o600)
        partial.replace(target)
        seen.add(member.name)
    assert seen == set(expected)
paid = sum(p.stat().st_size for p in root.rglob('*') if p.is_file())
assert paid < cap and shutil.disk_usage('/tmp').free > 20 * 2**30 + max(0, cap - paid)
