"""Integrated dense MRI path-trainer recovery; frozen encoder, no accuracy claim."""
import hashlib
import importlib.util
import json
import os
from pathlib import Path
import shutil
import subprocess
import sys
import time

R = Path('/tmp/rsna-criterion-N3-features-20261017-v1')
T = R / 'dense-online-input-fp32-v2/integrated-recovery-v1'

def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()

def cap():
    paid = sum(p.stat().st_size for p in R.rglob('*') if p.is_file())
    assert paid < 3 * 2**30
    family=Path('/tmp/rsna-family-label-epoch-preservation-20261018-v1')
    receipt=family/'completed-fixed3-unspent512MiB-retirement-v1.json'
    assert sha(receipt)=='f210549eb1d2aa76a62140ce8bbc9ae03164f1d68542f191357710ee0a00c54a'
    funding=json.loads(receipt.read_text());assert funding['new_family_cap_bytes']==3758096384
    family_paid=sum(p.stat().st_size for p in family.rglob('*') if p.is_file())
    assert shutil.disk_usage('/tmp').free > 603433009211 + max(0, 3 * 2**30 - paid) + max(0,3758096384-family_paid)
    assert sum(p.stat().st_size for p in T.rglob('*') if p.is_file()) < 16 * 2**20

def prepare():
    assert os.uname().nodename == 'evc33'
    admission = json.loads((T / 'admitted.json').read_text())
    assert sha(__file__) == admission['worker_sha256']
    for name, digest in admission['files_sha256'].items():
        assert sha(T / 'code' / name) == digest
    sys.path[:0] = [str(T / 'code'), str(R / 'reader-engine-recovery-v2/code'), '/tmp/rsna-compact-native-n3-assets-20261016-v1/runtime-overlay']
    import torch
    assert str(torch.__version__) == '2.14.0+cu130'
    assert torch.cuda.is_available() and torch.cuda.device_count() == 1 and 'H100' in torch.cuda.get_device_name(0)
    torch.ones(1, device='cuda'); torch.cuda.synchronize()
    torch.set_num_threads(2)
    torch.use_deterministic_algorithms(True)
    torch.backends.cuda.matmul.allow_tf32 = False
    torch.backends.cudnn.allow_tf32 = True
    torch.backends.cudnn.benchmark = False
    torch.backends.cudnn.deterministic = True
    assert os.environ['CUBLAS_WORKSPACE_CONFIG'] == ':4096:8'
    cap()
    return admission

def helper():
    spec = importlib.util.spec_from_file_location('original_verified_cached_MRI_input_helper', T / 'code/original_input_helper.py')
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module

def worker(admission, arm, phase):
    import torch
    from path_reader import PathCachedData, PathTrainer
    from rsna.competitive.spatial_train import SpatialRecipe
    from integrated_provider import providers
    original, parent, _ = helper().inputs({}, 'cached')
    assert parent is None
    assert original['fitting'].binding['fitting16_labels_masks_sha256'] == 'a2e8e852dd7e5d02efec61cd01a3b1b66ad5cd1f756914568d3d11f3d9ee0dc5'
    data, parent = providers(original, helper(), R)
    before = helper().parent_digest(parent)
    assert before == '8ba6c84e2c7715799dc78c64e4bfdde198cd8d1ef1db307119d70e1b55773c1e'
    engine = PathTrainer(data['fitting'], SpatialRecipe(epochs=2), arm=arm, device=torch.device('cuda'), binding={'diagnostic_only': True, 'path_mechanism_source': admission['mechanism_binding'], 'production_eligible': False, 'source_git': admission['source_git'], 'actual_original_streamed_dense_MRI_inputs': True, 'source_sampling':'dense', 'teacher':'unchanged_sparse16', 'encoder_fitted':False})
    initial = helper().parent_digest(engine.model)
    output = T / (arm + ('-replay' if phase in {'stop', 'resume'} else '-full'))
    result = engine.fit(output, resume=phase == 'resume', stop_after=1 if phase == 'stop' else None)
    if phase != 'resume':
        (output / 'initial.json').write_text(json.dumps({'model': initial}) + '\n')
    if phase != 'stop':
        assert engine.completed == 4 and len(engine.history) == 4
        assert engine.history[0]['projection_gradient_norm'] == 0 and engine.history[-1]['projection_gradient_norm'] > 0
        predictions = engine.predictions(data['outer'])
        assert predictions.shape == (16, 13)
        predictions.to_csv(output / 'predictions-private.csv', index=False)
    stats = {'GPU_job': os.environ['SLURM_JOB_ID'], 'arm': arm, 'phase': phase, 'actual_completed_updates': engine.completed, 'status': result['status'], 'peak_device_allocated_bytes': torch.cuda.max_memory_allocated(), 'frozen_encoder_instantiated':True,'encoder_trained':False}
    assert helper().parent_digest(parent) == before
    stats['frozen_parent_all_parameters_buffers_exact']=True
    (output / ('phase-' + phase + '.json')).write_text(json.dumps(stats) + '\n')
    cap()
    print(json.dumps(stats), flush=True)

def main():
    admission = prepare()
    if len(sys.argv) > 1:
        assert len(sys.argv) == 3
        worker(admission, *sys.argv[1:]); return
    import torch
    assert not (T / 'run-start.json').exists() and not (T / 'accepted.json').exists()
    start = time.monotonic()
    (T / 'run-start.json').write_text(json.dumps({'GPU_job': os.environ['SLURM_JOB_ID'], 'source_git': admission['source_git'], 'GPU': torch.cuda.get_device_name(0), 'torch': str(torch.__version__), 'CUDA_visible_devices': os.environ.get('CUDA_VISIBLE_DEVICES'), 'diagnostic_only': True, 'source_files_sha256': admission['files_sha256']}) + '\n')
    h = helper()
    histories, peaks = {}, []
    for arm in ('aligned', 'null'):
        for phase in ('full', 'stop', 'resume'):
            subprocess.run([sys.executable, '-B', __file__, arm, phase], check=True)
        full, replay = T / (arm + '-full'), T / (arm + '-replay')
        a, b = [torch.load(p / 'checkpoint.pt', map_location='cpu', weights_only=True) for p in (full, replay)]
        h.identical(a, b)
        for epoch in (1, 2):
            h.identical(*[torch.load(p / 'epoch-checkpoints' / ('epoch-%03d.pt' % epoch), map_location='cpu', weights_only=True) for p in (full, replay)])
        assert (full / 'predictions-private.csv').read_bytes() == (replay / 'predictions-private.csv').read_bytes()
        histories[arm] = a['history']
        peaks.extend(json.loads((p / n).read_text())['peak_device_allocated_bytes'] for p, n in ((full, 'phase-full.json'), (replay, 'phase-stop.json'), (replay, 'phase-resume.json')))
    assert (T / 'aligned-full/initial.json').read_bytes() == (T / 'null-full/initial.json').read_bytes()
    receipt = {'status': 'actual_real32_dense_online_MRI_path_trainer_full_stop1_resume_recovery_exact', 'GPU_job': os.environ['SLURM_JOB_ID'], 'node': 'evc33', 'source_git': admission['source_git'], 'files_sha256': admission['files_sha256'], 'worker_sha256': admission['worker_sha256'], 'original_private_fixture_sha256': 'f058c0969c56d754874ceb3fbb1fb9c3fff2d2a5c9bb1554d94cf68d9a3bd18f', 'original_parent_checkpoint_sha256': '5bec977964231fe67009d7106ad63ec71fdbd9cde5ddeb2cc2883ac59b0a7be5', 'fitting_prediction_only_original_cases': [16, 16], 'outer_clinical_targets_read': False, 'epochs': 2, 'updates_per_full': 4, 'complete_learned_AdamW_LR_order_cursor_allRNG_history_each_epoch_and16x12_CSV_recovery_exact': True, 'both_arms_same_initial_model': True, 'channels_width': [1024, 64], 'learned_parameters': 68560, 'peak_child_device_allocated_bytes': max(peaks), 'elapsed_seconds': time.monotonic() - start, 'updates': histories, 'frozen_encoder_instantiated':True,'encoder_trained':False, 'original_native_MRI_streamed_without_full_map_cache':True, 'integrated_dense_online_trainer_recovery_accepted':True, 'clinical_accuracy_full_production_or_competition_admission': False, 'within16MiB_child_of_original32MiB_input_control_node3GiB_budget':True}
    cap()
    (T / 'accepted.json').write_text(json.dumps(receipt, indent=2) + '\n')
    print(json.dumps(receipt), flush=True)

if __name__ == '__main__':
    main()
