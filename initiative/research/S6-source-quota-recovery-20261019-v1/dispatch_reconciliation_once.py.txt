"""One source-only recovery admission. All private values/credentials stay in stdin."""

import importlib.util
import json
import shlex
import subprocess
from pathlib import Path

R = Path.cwd()
D = Path(__file__).resolve().parent
L = R / "artifacts/S6-source-quota-recovery-20261019-v1"
L.mkdir(mode=0o700, exist_ok=True)
assert not (L / "reconcile-intent.json").exists()
sources = {}
git = subprocess.check_output(["git", "rev-parse", "HEAD"], text=True).strip()
for name in ("bulk_transport.py", "worker_bulk.py", "reconcile.py"):
    p = D / (name + ".txt")
    sources[name] = p.read_text()
    assert sources[name] == subprocess.check_output(
        ["git", "show", git + ":" + p.relative_to(R).as_posix()], text=True
    )
env = R / ".env"
assert env.stat().st_mode & 0o077 == 0
values = {
    k.strip(): x.strip().strip("\"'")
    for line in env.read_text().splitlines()
    for k, sep, x in [line.partition("=")]
    if sep
}
s = importlib.util.spec_from_file_location(
    "stdin_transport",
    R / "initiative/research/paired-logit-head-gradient-20261018-v1/stdin_transport.py",
)
t = importlib.util.module_from_spec(s)
s.loader.exec_module(t)
(L / "reconcile-intent.json").write_text(
    json.dumps(
        {
            "source_git": git,
            "failed_job": "968696",
            "preserve_rows": 880,
            "source_only_no_model": True,
            "only_missing_source_bytes": True,
            "automatic_retry": False,
        },
        indent=2,
    )
)
a = [
    "srun",
    "--account=skattel",
    "--partition=preemptable",
    "--qos=preemptable",
    "--nodelist=evc39",
    "--ntasks=1",
    "--cpus-per-task=1",
    "--gres=none",
    "--mem=2G",
    "--time=00:15:00",
    "--job-name=rsna-S6-source-quota-reconcile",
    "env",
    "CUDA_VISIBLE_DEVICES=",
    "PYTHONDONTWRITEBYTECODE=1",
    "PYTHONNOUSERSITE=1",
    "/tmp/rsna-environment-907632-7sr9a7i6/environment/bin/python",
    "-B",
    "-c",
    t.COMMAND,
]
payload = {
    "token": values["HF_TOKEN"],
    "sources": {n: v for n, v in sources.items() if n != "reconcile.py"},
}
x = subprocess.run(
    [
        "ssh",
        "-S",
        "/tmp/rsna-research-20261001-stable.sock",
        "-o",
        "BatchMode=yes",
        "newton",
        shlex.join(a),
    ],
    input=t.envelope(sources["reconcile.py"], payload),
    capture_output=True,
    text=True,
    timeout=1200,
)
(L / "reconcile-transport.json").write_text(
    json.dumps({"returncode": x.returncode, "stdout": x.stdout, "stderr": x.stderr}, indent=2)
)
assert x.returncode == 0, (
    "Recovery stopped/uncertain; inspect retained original data without replay: " + x.stderr[-1200:]
)
(D / "actual-sources-reconciled.json").write_text(json.dumps(json.loads(x.stdout), indent=2) + "\n")
print(x.stdout, flush=True)
