"""Compile only after actual final model preservation; no Kaggle API mutations."""

import hashlib
import importlib.util
import json
import shlex
import subprocess
from importlib.machinery import SourceFileLoader
from pathlib import Path

R = Path.cwd()
D = Path(__file__).resolve().parent
FIT = R / "initiative/research/source-zero-full-T4-20261020-v1"
L = R / "artifacts/source-zero-final-serving-20261020-v1"


def run():
    accepted, publication = (
        FIT / "actual-status-epoch-3.json",
        FIT / "actual-private-verified-epoch-3.json",
    )
    pointer = FIT / "actual-record_publication-epoch-3.json"
    if not accepted.exists() or not publication.exists() or not pointer.exists():
        return {
            "status": "waiting_actual_fixed_final_source_zero_preservation_no_intent",
            "new_GPU_or_API_mutations": 0,
        }
    training, pub = json.loads(accepted.read_text()), json.loads(publication.read_text())
    recorded = json.loads(pointer.read_text())
    assert recorded["status"] == "actual_source_zero_private_epoch_pointer_recorded"
    assert (
        recorded["model_sha256"] == pub["model_sha256"] and recorded["revision"] == pub["revision"]
    )
    loader = SourceFileLoader("new_final_source_zero_prepare", str(D / "prepare.py.txt"))
    m = importlib.util.module_from_spec(importlib.util.spec_from_loader(loader.name, loader))
    loader.exec_module(m)
    old = R / "initiative/research/S6-Kaggle-production-route-20261019-v1/serving_prepare.py.txt"
    program = m.build(old.read_text())
    paths = {
        "compiler.py": D / "compile.py.txt",
        "validate.py": D / "validate.py.txt",
        "legacy.py": R / "scripts/kaggle/licensed_importance_serving.py",
        "bounded.py": R / "rsna/competitive/bounded_preparation.py",
        "codec.py": R / "rsna/competitive/licensed_pixel_cache.py",
        "mixin.py": R / "rsna/competitive/licensed_full_bag.py",
        "importance.py": R / "rsna/competitive/licensed_importance.py",
    }
    files = {n: p.read_text() for n, p in paths.items()}
    h = {"__name__": "local_pre_intent_final_source_zero"}
    exec(compile(files["validate.py"], "local_pre_intent_final_validator", "exec"), h)
    h["provenance"](training, pub, pub["model_sha256"])
    pin = subprocess.check_output(["git", "rev-parse", "HEAD"], text=True).strip()
    for p in (*paths.values(), old, *D.glob("*.py.txt")):
        assert p.read_bytes() == subprocess.check_output(
            ["git", "show", pin + ":" + p.relative_to(R).as_posix()]
        )
    L.mkdir(mode=0o700, exist_ok=True)
    with (L / "CPU-compile-intent.json").open("x") as f:
        json.dump(
            {
                "source_git": pin,
                "epoch": 3,
                "model_sha256": pub["model_sha256"],
                "new_GPU_or_API_mutations": 0,
                "automatic_retry": False,
            },
            f,
        )
    spec = importlib.util.spec_from_file_location(
        "original_transport",
        R / "initiative/research/paired-logit-head-gradient-20261018-v1/stdin_transport.py",
    )
    t = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(t)
    args = [
        "srun",
        "--account=skattel",
        "--partition=normal",
        "--qos=normal",
        "--nodelist=evc33",
        "--ntasks=1",
        "--cpus-per-task=1",
        "--gres=none",
        "--mem=3G",
        "--time=00:08:00",
        "--job-name=rsna-source-zero-final-compile",
        "env",
        "CUDA_VISIBLE_DEVICES=",
        "PYTHONDONTWRITEBYTECODE=1",
        "PYTHONNOUSERSITE=1",
        "OMP_NUM_THREADS=1",
        "/tmp/rsna-environment-907632-7sr9a7i6/environment/bin/python",
        "-B",
        "-c",
        t.COMMAND,
    ]
    x = subprocess.run(
        [
            "ssh",
            "-S",
            "/tmp/rsna-research-20261001-stable.sock",
            "-o",
            "BatchMode=yes",
            "newton",
            shlex.join(args),
        ],
        input=t.envelope(
            program,
            {
                "epoch": 3,
                "source_git": pin,
                "source": files,
                "source_SHA": {n: hashlib.sha256(s.encode()).hexdigest() for n, s in files.items()},
            },
        ),
        capture_output=True,
        text=True,
        timeout=900,
    )
    (L / "CPU-compile-transport.json").write_text(
        json.dumps({"returncode": x.returncode, "stdout": x.stdout, "stderr": x.stderr}, indent=2)
        + "\n"
    )
    assert x.returncode == 0, "Reconcile original compilation; never repeat uncertain work"
    out = json.loads(x.stdout)
    (D / "actual-final-serving-compiled.json").write_text(json.dumps(out, indent=2) + "\n")
    return out


if __name__ == "__main__":
    print(json.dumps(run()), flush=True)
