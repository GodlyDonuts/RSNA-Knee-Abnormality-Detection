"""One original-input availability observation, with no GPU or upload."""

import hashlib
import importlib.util
import json
import shlex
import subprocess
from pathlib import Path

R = Path.cwd()
D = Path(__file__).resolve().parent
L = R / "artifacts/S6-Newton-student-production-20261019-v1"
L.mkdir(mode=0o700, exist_ok=True)
assert not (L / "inventory-intent.json").exists()
pin = subprocess.check_output(["git", "rev-parse", "HEAD"], text=True).strip()
for name in ("inventory.py.txt", "inventory_once.py.txt"):
    assert (
        subprocess.check_output(
            ["git", "show", pin + ":" + (D / name).relative_to(R).as_posix()], text=True
        )
        == (D / name).read_text()
    )
program = (D / "inventory.py.txt").read_text()
spec = importlib.util.spec_from_file_location(
    "transport", R / "initiative/research/paired-logit-head-gradient-20261018-v1/stdin_transport.py"
)
t = importlib.util.module_from_spec(spec)
spec.loader.exec_module(t)
with (L / "inventory-intent.json").open("x") as f:
    json.dump(
        {
            "source_git": pin,
            "program_sha256": hashlib.sha256(program.encode()).hexdigest(),
            "one_CPU_zero_GPU": True,
            "large_rehash_or_copy_or_upload": False,
            "automatic_retry": False,
        },
        f,
        indent=2,
    )
args = [
    "srun",
    "--account=skattel",
    "--partition=preemptable",
    "--qos=preemptable",
    "--nodelist=evc33",
    "--ntasks=1",
    "--cpus-per-task=1",
    "--gres=none",
    "--mem=2G",
    "--time=00:03:00",
    "--job-name=rsna-S6-full-input-availability",
    "env",
    "CUDA_VISIBLE_DEVICES=",
    "PYTHONDONTWRITEBYTECODE=1",
    "PYTHONNOUSERSITE=1",
    "/tmp/rsna-environment-907632-7sr9a7i6/environment/bin/python",
    "-B",
    "-c",
    t.COMMAND,
]
x = subprocess.run(
    [
        "ssh",
        "-S",
        "/tmp/rsna-research-20261001-stable.sock",
        "-o",
        "BatchMode=yes",
        "newton",
        shlex.join(args),
    ],
    input=t.envelope(program, {}),
    capture_output=True,
    text=True,
    timeout=7200,
)
(L / "inventory-transport.json").write_text(
    json.dumps({"returncode": x.returncode, "stdout": x.stdout, "stderr": x.stderr}, indent=2)
    + "\n"
)
assert x.returncode == 0, "Original observer requires read-only reconciliation"
actual = json.loads(x.stdout)
(D / "actual-node33-full-input-availability.json").write_text(json.dumps(actual, indent=2) + "\n")
print(json.dumps(actual), flush=True)
