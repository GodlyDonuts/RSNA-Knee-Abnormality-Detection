def load(name):
    spec = importlib.util.spec_from_file_location(name, S / (name + ".py"))
    m = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(m)
    return m

def check_bounds():
    # Count private aliases/hardlinks once, never claim a directory alias is new space.
    inodes = set()
    used = 0
    for p in P.rglob("*"):
        if p.is_symlink() or "input" in p.relative_to(P).parts:
            continue
        try:
            s = p.stat()
            if p.is_file() and (s.st_dev, s.st_ino) not in inodes:
                inodes.add((s.st_dev, s.st_ino))
                used += s.st_size
        except FileNotFoundError:
            continue
    cap = 64 * 2**30 + 16 * 2**20
    assert used < cap
    assert shutil.disk_usage("/tmp").free > 20 * 2**30 + 32 * 2**20 + max(0, cap - used)
    assert time.monotonic() - begin < 11.8 * 3600


try:
    # Prioritize the original new mechanism; still run its matched control.
    for role in ("cross_lineage_equal_rank", "parent_only"):
        item = meta["roles"][role]
        bundle = json.loads((S / (role + ".private.json")).read_text())
        code = S / ("code-" + role)
        code.mkdir(mode=0o700)
        for name, text in {
            **bundle["FILES"],
            "rsna/__init__.py": "",
            "rsna/competitive/__init__.py": "",
        }.items():
            path = code / name
            assert path.resolve().is_relative_to(code.resolve())
            path.parent.mkdir(mode=0o700, parents=True, exist_ok=True)
            path.write_text(text)
            path.chmod(0o600)
        worker = code / "worker.private.py"
        raw = base64.b64decode(bundle["WORKER"], validate=True)
        assert hashlib.sha256(raw).hexdigest() == item["worker_sha256"]
        worker.write_bytes(raw)
        worker.chmod(0o600)
        # Parent/adaptation permission travels with every role.
        permission = base64.b64decode(bundle["PERMISSION"], validate=True)
        assert (
            hashlib.sha256(permission).hexdigest()
            == "34c44b20e4d7ace4eb523b0423b36d36dec826fe6ae54c66346796bba0190e79"

                "--nproc_per_node=1",
