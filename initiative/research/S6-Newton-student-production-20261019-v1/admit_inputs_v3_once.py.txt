"""One actual complete-input CPU gate after the original private replica succeeds."""

import hashlib
import importlib.util
import json
import os
import shlex
import subprocess
import time
from pathlib import Path

R = Path.cwd()
D = Path(__file__).resolve().parent
L = R / "artifacts/S6-Newton-student-production-20261019-v1"
assert not (L / "input-follower-v3-intent.json").exists()
pin = subprocess.check_output(["git", "rev-parse", "HEAD"], text=True).strip()
program = (D / "admit_inputs.py.txt").read_text()
for name in ("admit_inputs.py.txt", "admit_inputs_v3_once.py.txt"):
    assert (
        subprocess.check_output(
            ["git", "show", pin + ":" + (D / name).relative_to(R).as_posix()], text=True
        )
        == (D / name).read_text()
    )
with (L / "input-follower-v3-intent.json").open("x") as f:
    json.dump(
        {
            "PID": os.getpid(),
            "source_git": pin,
            "program_sha256": hashlib.sha256(program.encode()).hexdigest(),
            "waits_original_replica_PID": 59645,
            "maximum_CPU_gates": 1,
            "maximum_hours": 24,
            "GPU_or_competition_mutations": 0,
            "automatic_retry": False,
        },
        f,
        indent=2,
    )
try:
    started = time.monotonic()
    while not (D / "actual-private-full-input-replica-v3.json").exists():
        assert time.monotonic() - started < 24 * 3600
        if (L / "TLS-replica-transport.json").exists():
            assert json.loads((L / "TLS-replica-transport.json").read_text())["returncode"] == 0, (
                "Original replica failed; no input intent"
            )
        time.sleep(60)
    actual = json.loads((D / "actual-private-full-input-replica-v3.json").read_text())
    assert actual["status"] == "actual_original_pixel_replica_both_CPU_endpoints_completed"
    spec = importlib.util.spec_from_file_location(
        "transport",
        R / "initiative/research/paired-logit-head-gradient-20261018-v1/stdin_transport.py",
    )
    t = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(t)
    args = [
        "srun",
        "--account=skattel",
        "--partition=preemptable",
        "--qos=preemptable",
        "--nodelist=evc33",
        "--ntasks=1",
        "--cpus-per-task=1",
        "--gres=none",
        "--mem=4G",
        "--time=00:05:00",
        "--job-name=rsna-S6-Newton-full-input-gate",
        "env",
        "CUDA_VISIBLE_DEVICES=",
        "PYTHONDONTWRITEBYTECODE=1",
        "PYTHONNOUSERSITE=1",
        "OMP_NUM_THREADS=1",
        "/tmp/rsna-environment-907632-7sr9a7i6/environment/bin/python",
        "-B",
        "-c",
        t.COMMAND,
    ]
    with (L / "input-dispatch-v3-intent.json").open("x") as f:
        json.dump(
            {
                "source_git": pin,
                "copy_CPU": actual["receiver"]["CPU_job"],
                "program_sha256": hashlib.sha256(program.encode()).hexdigest(),
                "automatic_retry": False,
            },
            f,
            indent=2,
        )
    x = subprocess.run(
        [
            "ssh",
            "-S",
            "/tmp/rsna-research-20261001-stable.sock",
            "-o",
            "BatchMode=yes",
            "newton",
            shlex.join(args),
        ],
        input=t.envelope(
            program,
            {"source_git": pin, "copy": actual["receiver"], "sender_CPU": actual["sender_CPU"]},
        ),
        capture_output=True,
        text=True,
        timeout=7200,
    )
    (L / "input-gate-transport-v3.json").write_text(
        json.dumps({"returncode": x.returncode, "stdout": x.stdout, "stderr": x.stderr}, indent=2)
        + "\n"
    )
    assert x.returncode == 0, "Original input gate requires read-only reconciliation, never retry"
    accepted = json.loads(x.stdout)
    assert accepted["status"] == "actual_full3887_original_MRI_population_on_Newton_CPU_admitted"
    (D / "actual-full-inputs-accepted.json").write_text(json.dumps(accepted, indent=2) + "\n")
    (L / "input-follower-v3-result.json").write_text(
        json.dumps(
            {
                "PID": os.getpid(),
                "status": "actual_original_full_input_CPU_admission_finished",
                "actual": accepted,
            },
            indent=2,
        )
        + "\n"
    )
    print(json.dumps(accepted), flush=True)
except BaseException as e:
    (L / "input-follower-v3-result.json").write_text(
        json.dumps(
            {
                "PID": os.getpid(),
                "status": "original_full_input_CPU_follower_stopped",
                "type": type(e).__name__,
                "reason": str(e),
                "automatic_retry": False,
            },
            indent=2,
        )
        + "\n"
    )
    raise
