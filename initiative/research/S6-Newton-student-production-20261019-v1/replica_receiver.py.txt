import json

from pathlib import Path

v = PACKAGE
assert os.uname().nodename == "evc33" and not os.environ.get("SLURM_JOB_GPUS")
os.umask(0o077)
P = Path("/tmp/rsna-S6-Newton-student-production-20261019-v1")
assert not P.exists(), "Existing original transfer requires read-only reconciliation"
# Fund both the input replica and the actual upcoming epoch/optimizer/publication
# obligations, retaining the already admitted recovery/input/MedGemma scopes.
R = Path("/tmp/rsna-S6-Newton-student-recovery-20261019-v1")
S = Path("/tmp/rsna-S6-target-execution-inputs-20261019-v1")


def paid(root):
    return sum(p.stat().st_size for p in root.rglob("*") if p.is_file() and not p.is_symlink())


old_unpaid = (
    max(0, 12 * 2**30 + 512 * 2**20 + 16 * 2**20 - paid(R)) + max(0, 12 * 2**20 - paid(S)) + 2**30
)
new = 40 * 2**30 + 24 * 2**30 + 16 * 2**20
assert shutil.disk_usage("/tmp").free > 20 * 2**30 + 32 * 2**20 + old_unpaid + new
q = subprocess.check_output(["lfs", "quota", "-u", "sa305415", "/home/sa305415"], text=True)
a = next(x.split() for x in q.splitlines() if x.strip().startswith("/home/sa305415"))
assert (int(a[2]) - int(a[1])) * 1024 > 20 * 2**30 + (32 + 16 + 16 + 4) * 2**20
assert int(a[6]) - int(a[5]) > 5000 + 256 + 128 + 256
# Parse the bounded header before creating the root/first intent. receive owns
# the one copy root, preserves prefixes, and rejects an uncertain repeated copy.

    assert old["archive_sha256"] == table[f"shard-{i}/pixels.tar"]["sha256"]
    assert old["index_sha256"] == table[f"shard-{i}/index.private.json"]["sha256"]
