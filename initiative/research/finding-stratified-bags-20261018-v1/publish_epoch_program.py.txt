import requests
from urllib.parse import urljoin, urlparse

session = requests.Session()
session.headers["Authorization"] = "Bearer " + v["kaggle_token"]
API = "https://api.kaggle.com/v1/"
owner, slug = actual["slug"].split("/")


def download(remote, p, size, digest):
    assert not p.exists()
    session.cookies.clear()
    response = session.post(
        API + "kernels.KernelsApiService/DownloadKernelOutput",
        json={"ownerSlug": owner, "kernelSlug": slug, "versionNumber": 1, "filePath": remote},
        timeout=(10, 90),
        allow_redirects=False,
        stream=True,
    )
    for _ in range(6):
        if response.status_code == 200:
            break
        assert response.status_code in (301, 302, 303, 307, 308)
        url = urljoin(response.url, response.headers.get("Location", ""))
        response.close()
        u = urlparse(url)
        assert u.scheme == "https" and (
            u.hostname
            in (
                "storage.googleapis.com",
                "api.kaggle.com",
                "www.kaggle.com",
                "www.kaggleusercontent.com",
            )
            or u.hostname.endswith(".googleusercontent.com")
        )
        response = (session if u.hostname == "api.kaggle.com" else requests).get(
            url, timeout=(10, 90), allow_redirects=False, stream=True
        )
    assert response.status_code == 200
    h = hashlib.sha256()
    n = 0
    try:
        with p.open("xb") as f:
            os.fchmod(f.fileno(), 0o600)
            for block in response.iter_content(2**20):
                n += len(block)
                assert n <= size
                f.write(block)
                h.update(block)
    except requests.RequestException:
        raise RuntimeError(
            "Original immutable epoch stream failed; retain prefix and reconcile read-only"
        ) from None
    finally:
        response.close()
    assert n == size and h.hexdigest() == digest
