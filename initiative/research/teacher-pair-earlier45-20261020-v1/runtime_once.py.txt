"""One CPU-only runtime restoration; all exact source programs retained."""

import hashlib
import importlib.util
import json
import shlex
import subprocess
from pathlib import Path

R = Path.cwd()
D = Path(__file__).resolve().parent
L = R / "artifacts/teacher-pair-earlier45-20261020-v1"
L.mkdir(mode=0o700, exist_ok=True)
assert not (L / "runtime-intent.json").exists()
git = subprocess.check_output(["git", "rev-parse", "HEAD"], text=True).strip()
files = {
    n: (R / "scripts/newton" / n).read_text()
    for n in ("runtime_environment.py", "runtime_archive.py")
}
program = (D / "runtime_stage.py.txt").read_text()
for p, body in [
    (D / "runtime_stage.py.txt", program),
    *((R / "scripts/newton" / n, b) for n, b in files.items()),
]:
    assert body.encode() == subprocess.check_output(
        ["git", "show", git + ":" + str(p.relative_to(R))]
    )
(L / "runtime-intent.json").write_text(
    json.dumps(
        {
            "source_git": git,
            "CPU_only": True,
            "node": "evc45",
            "program_sha256": hashlib.sha256(program.encode()).hexdigest(),
            "maximum_restorations": 1,
            "automatic_retry": False,
        },
        indent=2,
    )
    + "\n"
)
s = importlib.util.spec_from_file_location(
    "transport", R / "initiative/research/paired-logit-head-gradient-20261018-v1/stdin_transport.py"
)
t = importlib.util.module_from_spec(s)
s.loader.exec_module(t)
a = [
    "srun",
    "--account=skattel",
    "--partition=normal",
    "--qos=normal",
    "--nodelist=evc45",
    "--ntasks=1",
    "--cpus-per-task=1",
    "--gres=none",
    "--mem=2G",
    "--time=00:15:00",
    "--job-name=rsna-teacher-pair45-runtime",
    "env",
    "CUDA_VISIBLE_DEVICES=",
    "PYTHONNOUSERSITE=1",
    "PYTHONDONTWRITEBYTECODE=1",
    "OMP_NUM_THREADS=1",
    "/usr/bin/python3.12",
    "-B",
    "-c",
    t.COMMAND,
]
x = subprocess.run(
    ["ssh", "-S", "/tmp/rsna-research-20261001-stable.sock", "newton", shlex.join(a)],
    input=t.envelope(
        program,
        {
            "source_git": git,
            "files": files,
            "hashes": {n: hashlib.sha256(b.encode()).hexdigest() for n, b in files.items()},
        },
    ),
    capture_output=True,
    text=True,
    timeout=1200,
)
(L / "runtime-transport.json").write_text(
    json.dumps({"returncode": x.returncode, "stdout": x.stdout, "stderr": x.stderr}, indent=2)
    + "\n"
)
assert x.returncode == 0, "Reconcile original runtime; never automatically repeat"
(D / "actual-runtime.json").write_text(json.dumps(json.loads(x.stdout), indent=2) + "\n")
print(x.stdout, flush=True)
