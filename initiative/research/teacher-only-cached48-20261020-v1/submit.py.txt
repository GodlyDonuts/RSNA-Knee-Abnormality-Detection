# ruff: noqa: E402
import hashlib
import json
import os
import subprocess
import sys
from pathlib import Path

v = json.load(sys.stdin)
assert os.uname().nodename == "evc48" and not os.environ.get("SLURM_JOB_GPUS")
R = Path("/tmp/rsna-teacher-only-recovery48-20261020-v1")
prepared = json.loads((R / "prepared.json").read_text())
assert prepared == v["prepared"]
staged = json.loads((R / "staged.json").read_text())
assert staged["source_git"] == prepared["source_git"]
for name, key in (("worker.py", "worker_sha256"), ("driver.py", "driver_sha256")):
    assert hashlib.sha256((R / name).read_bytes()).hexdigest() == staged[key]
assert (
    not (R / "cancel-original-intent.json").exists() and not (R / "dispatch-intent.json").exists()
)
held = subprocess.check_output(["scontrol", "show", "job", "975914", "-o"], text=True)
assert "JobState=PENDING" in held and "Reason=JobHeldUser" in held and "RunTime=00:00:00" in held
(R / "cancel-original-intent.json").write_text(
    json.dumps(
        {
            "job": "975914",
            "source_git": v["source_git"],
            "actual_cached48_prepared": prepared,
            "expected_zero_runtime": True,
        }
    )
    + "\n"
)
subprocess.run(["scancel", "975914"], check=True)
terminal = subprocess.check_output(
    ["sacct", "-X", "-n", "-P", "-j", "975914", "--format=JobIDRaw,State,Elapsed,ExitCode"],
    text=True,
)
assert "975914|CANCELLED" in terminal and "|00:00:00|" in terminal
(R / "original-zero-runtime-cancelled.json").write_text(json.dumps({"terminal": terminal}) + "\n")
header = Path("/tmp/rsna-medgemma15-runtime-headers-20261019-v1/verified.json")
assert (
    hashlib.sha256(header.read_bytes()).hexdigest()
    == "fb587efe77486f5fee42f996a4a72c9ee9bec80d71ab8e7cdea91baa55c58bff"
)
include = json.loads(header.read_text())["include_path"]
command = [
    "/tmp/rsna-environment-971427-pyk8rld8/environment/bin/python",
    "-B",
    str(R / "driver.py"),
]
import shlex

job = (
    "#!/bin/bash\nset -eu\n"
    + "exec env PYTHONNOUSERSITE=1 PYTHONDONTWRITEBYTECODE=1 OMP_NUM_THREADS=1 MKL_NUM_THREADS=1 OPENBLAS_NUM_THREADS=1 CUBLAS_WORKSPACE_CONFIG=:4096:8 C_INCLUDE_PATH="
    + shlex.quote(include)
    + " CPLUS_INCLUDE_PATH="
    + shlex.quote(include)
    + " "
    + shlex.join(command)
    + "\n"
)
script = R / "job.sh"
script.write_text(job)
script.chmod(0o700)
(R / "dispatch-intent.json").write_text(
    json.dumps(
        {
            "source_git": v["source_git"],
            "maximum_GPU_jobs": 1,
            "new_production_or_competition_entries": 0,
            "automatic_retry": False,
        }
    )
    + "\n"
)
result = subprocess.run(
    [
        "sbatch",
        "--parsable",
        "--account=skattel",
        "--partition=normal",
        "--qos=normal",
        "--nodelist=evc45",
        "--ntasks=1",
        "--cpus-per-task=2",
        "--gres=gpu:nvidia_h100_pcie:1",
        "--mem=24G",
        "--time=00:10:00",
        "--job-name=rsna-teacher-only-real-recovery",
        "--output=" + str(R / "GPU-stdout.private.log"),
        "--error=" + str(R / "GPU-stderr.private.log"),
        str(script),
    ],
    capture_output=True,
    text=True,
    timeout=30,
    env={k: value for k, value in os.environ.items() if k != "CUDA_VISIBLE_DEVICES"},
)
(R / "dispatch-response.json").write_text(
    json.dumps({"returncode": result.returncode, "stdout": result.stdout, "stderr": result.stderr})
    + "\n"
)
assert result.returncode == 0 and result.stdout.strip().split(";")[0].isdigit(), (
    "Uncertain mutation: inspect Slurm read-only, never repeat"
)
out = {
    "status": "actual_cached48_zero_source_recovery_submitted_after_original_zero_runtime_cancellation",
    "source_git": v["source_git"],
    "CPU_job": os.environ["SLURM_JOB_ID"],
    "GPU_job": result.stdout.strip().split(";")[0],
    "node": "evc48",
    "H100_GPUs": 1,
    "physical_node_GPUs": 2,
    "CPUs": 2,
    "maximum_minutes": 10,
    "source_weight": 0.0,
    "new_RAM_cap_bytes": 6 * 2**30,
    "old_assets_or_RAM_deleted": False,
    "actual_GPU_recovery_or_accuracy_pass": False,
    "new_production_fits_or_Kaggle_submissions": 0,
    "automatic_retry": False,
}
(R / "submitted.json").write_text(json.dumps(out, indent=2) + "\n")
print(json.dumps(out), flush=True)
