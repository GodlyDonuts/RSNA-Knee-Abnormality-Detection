"""One bounded allocated-device investigation after original pre-model failure."""

import hashlib
import json
import shlex
import subprocess
from pathlib import Path

R = Path.cwd()
D = Path(__file__).resolve().parent
L = R / "artifacts/predictive-device47-20261020-v1"
L.mkdir(mode=0o700, exist_ok=True)
assert not (L / "intent.json").exists()
git = subprocess.check_output(["git", "rev-parse", "HEAD"], text=True).strip()
source = (D / "probe.py.txt").read_text()
assert source.encode() == subprocess.check_output(
    ["git", "show", git + ":" + (D / "probe.py.txt").relative_to(R).as_posix()]
)
root = "/tmp/rsna-predictive-real-trainer47-20261020-v1/device-investigation-v1"
python = "/tmp/rsna-predictive-real-trainer47-20261020-v1/runtime/bin/python"
script = "\n".join(
    [
        "#!/bin/bash",
        "#SBATCH --account=skattel",
        "#SBATCH --partition=normal",
        "#SBATCH --qos=normal",
        "#SBATCH --nodelist=evc47",
        "#SBATCH --ntasks=1",
        "#SBATCH --cpus-per-task=1",
        "#SBATCH --gres=gpu:nvidia_h100_pcie:1",
        "#SBATCH --mem=2G",
        "#SBATCH --time=00:03:00",
        "#SBATCH --job-name=rsna-predictive-allocated-device",
        "#SBATCH --output=/tmp/rsna-predictive-real-trainer47-20261020-v1/device-%j.out",
        "#SBATCH --error=/tmp/rsna-predictive-real-trainer47-20261020-v1/device-%j.err",
        "set -euo pipefail",
        "umask 077",
        "mkdir " + shlex.quote(root),
        "cat > " + shlex.quote(root + "/probe.py") + " <<'RSNA_PINNED_DEVICE_SOURCE'",
        source,
        "RSNA_PINNED_DEVICE_SOURCE",
        "export PYTHONNOUSERSITE=1 PYTHONDONTWRITEBYTECODE=1 OMP_NUM_THREADS=1",
        shlex.join([python, "-B", root + "/probe.py", "batch"]),
        shlex.join(
            [
                "srun",
                "--ntasks=1",
                "--cpus-per-task=1",
                "--gres=gpu:nvidia_h100_pcie:1",
                python,
                "-B",
                root + "/probe.py",
                "step",
            ]
        ),
        "",
    ]
)
(L / "intent.json").write_text(
    json.dumps(
        {
            "source_git": git,
            "script_sha256": hashlib.sha256(script.encode()).hexdigest(),
            "maximum_GPU_jobs": 1,
            "time_cap_minutes": 3,
            "MRI_models_or_optimizers": False,
            "automatic_retry": False,
        },
        indent=2,
    )
    + "\n"
)
x = subprocess.run(
    [
        "ssh",
        "-S",
        "/tmp/rsna-research-20261001-stable.sock",
        "-o",
        "BatchMode=yes",
        "newton",
        "sbatch --parsable",
    ],
    input=script,
    text=True,
    capture_output=True,
    timeout=90,
)
(L / "transport.json").write_text(
    json.dumps({"returncode": x.returncode, "stdout": x.stdout, "stderr": x.stderr}, indent=2)
    + "\n"
)
assert x.returncode == 0 and x.stdout.strip().split(";")[0].isdigit(), (
    "Reconcile exact job, never repeat"
)
out = {
    "status": "one_allocated_device_context_investigation_submitted",
    "GPU_job": x.stdout.strip().split(";")[0],
    "source_git": git,
    "original_pre_model_failure_job": "977901",
    "MRI_model_or_optimizer": False,
}
(D / "actual-submit.json").write_text(json.dumps(out, indent=2) + "\n")
print(json.dumps(out), flush=True)
