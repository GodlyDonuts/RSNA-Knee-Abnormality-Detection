"""Allocated batch/step CUDA facts only; no MRI, models or optimizer."""

import json
import os
import subprocess
import sys
from pathlib import Path

import torch

assert os.uname().nodename == "evc47" and os.environ.get("SLURM_JOB_GPUS")
mode = sys.argv[1]
assert mode in ("batch", "step")
facts = {
    "mode": mode,
    "GPU_job": os.environ["SLURM_JOB_ID"],
    "environment": {
        n: os.environ.get(n)
        for n in (
            "CUDA_VISIBLE_DEVICES",
            "NVIDIA_VISIBLE_DEVICES",
            "SLURM_JOB_GPUS",
            "SLURM_STEP_GPUS",
            "SLURM_GPUS_ON_NODE",
        )
    },
    "torch": str(torch.__version__),
    "torch_cuda_build": torch.version.cuda,
    "count": torch.cuda.device_count(),
}
query = subprocess.run(
    ["nvidia-smi", "--query-gpu=index,uuid,name,compute_mode", "--format=csv,noheader"],
    capture_output=True,
    text=True,
    timeout=20,
)
facts["NVML_query_returncode"] = query.returncode
facts["NVML_inventory"] = query.stdout.strip().splitlines()
facts["NVML_error"] = query.stderr[-1000:]
facts["device_files"] = {
    p.name: oct(p.stat().st_mode & 0o777)
    for p in Path("/dev").glob("nvidia*")
    if p.is_char_device()
}
if facts["count"] == 1:
    try:
        facts["name"] = torch.cuda.get_device_name(0)
        facts["assigned_context_sum"] = float(torch.ones(4, device="cuda").sum())
        torch.cuda.synchronize()
    except Exception as error:
        facts["context_error"] = {"type": type(error).__name__, "message": str(error)[:1000]}
else:
    facts["context_attempted"] = False
facts["MRI_model_or_optimizer"] = False
print(json.dumps(facts), flush=True)
