"""Reuse exact qualified/public model and canonical coordinator; metadata-only admission."""

import hashlib
import importlib.util
import json
import shlex
import subprocess
import sys
import time
from pathlib import Path

R = Path.cwd()
D = Path(__file__).resolve().parent
phase = sys.argv[1]
assert phase in ("eligibility", "submit", "outcome")
L = R / "artifacts/reconnected-teacher-scoring-20261019-v1"
L.mkdir(mode=0o700, parents=True, exist_ok=True)
git = subprocess.check_output(["git", "rev-parse", "HEAD"], text=True).strip()


def committed(p):
    value = subprocess.check_output(
        ["git", "show", git + ":" + p.relative_to(R).as_posix()], text=True
    )
    assert value == p.read_text()
    return value


policy = committed(D / "resource_policy.py.txt")
control = R / "initiative/experiments/competitive-v1/paired-final-scoring-budget-20261018-v4"
old = committed(control / "program.py.txt")
h = hashlib.sha256(old.encode()).hexdigest()
assert h == json.loads((control / "actual-staged.json").read_text())["recovery_sha256"]
site = "    g['qualification_training_provenance'] = qualification_training_provenance"
assert old.count(site) == 1
patched = old.replace(site, "    g['bounds'] = scoring_bounds\n" + site)
program = policy + "\n" + patched
v = {
    "phase": phase,
    "mode": "competition",
    "candidate": "paired",
    "control_git": git,
    "program_sha256": "8aaf2cf53beb63c63969b1167a0e8be54d9b3a47e531679968b259498d288b11",
    "recovery_sha256": h,
    "code_source": "456c5d9f2d88db8057a9feddaba9454ea3f3a389",
}
E = R / ".env"
assert E.stat().st_mode & 0o077 == 0
values = {
    k.strip(): z.strip().strip("\"'")
    for line in E.read_text().splitlines()
    for k, sep, z in [line.partition("=")]
    if sep and not line.lstrip().startswith("#")
}
v["token"] = values["KAGGLE_API_TOKEN"]
if phase == "submit":
    assert not (L / "entry-intent.json").exists()
    (L / "entry-intent.json").write_text(
        json.dumps(
            {
                "source_git": git,
                "same_original_source_and_model": True,
                "canonical_ledger_unchanged": True,
                "automatic_retry": False,
            }
        )
    )
s = importlib.util.spec_from_file_location(
    "bounded_stdin",
    R / "initiative/research/paired-logit-head-gradient-20261018-v1/stdin_transport.py",
)
t = importlib.util.module_from_spec(s)
s.loader.exec_module(t)
a = [
    "srun",
    "--account=skattel",
    "--partition=preemptable",
    "--qos=preemptable",
    "--nodelist=evc33",
    "--ntasks=1",
    "--cpus-per-task=1",
    "--gres=none",
    "--mem=4G",
    "--time=00:10:00",
    "--job-name=rsna-current-paired-" + phase,
    "env",
    "PYTHONDONTWRITEBYTECODE=1",
    "PYTHONNOUSERSITE=1",
    "OMP_NUM_THREADS=1",
    "MKL_NUM_THREADS=1",
    "OPENBLAS_NUM_THREADS=1",
    "/tmp/rsna-environment-907632-7sr9a7i6/environment/bin/python",
    "-B",
    "-c",
    t.COMMAND,
]
x = subprocess.run(
    [
        "ssh",
        "-S",
        "/tmp/rsna-research-20261001-stable.sock",
        "-o",
        "BatchMode=yes",
        "newton",
        shlex.join(a),
    ],
    input=t.envelope(program, v),
    capture_output=True,
    text=True,
    timeout=700,
)
(L / (phase + "-" + str(time.time_ns()) + ".json")).write_text(
    json.dumps({"returncode": x.returncode, "stdout": x.stdout, "stderr": x.stderr}, indent=2)
)
assert x.returncode == 0, x.stderr[-1500:]
out = json.loads(x.stdout)
(D / ("actual-" + phase + ".json")).write_text(json.dumps(out, indent=2) + "\n")
print(json.dumps(out), flush=True)
