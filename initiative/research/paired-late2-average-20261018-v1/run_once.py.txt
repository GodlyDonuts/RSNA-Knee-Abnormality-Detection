"""One CPU-only real model construction; no repeated parent training or inference."""

import hashlib
import importlib.util
import json
import shlex
import subprocess
from pathlib import Path

R = Path.cwd()
D = Path(__file__).resolve().parent
L = R / "artifacts/paired-late2-average-20261018-v1"
L.mkdir(mode=0o700, parents=True, exist_ok=True)
assert not (L / "intent.json").exists()
source = subprocess.check_output(["git", "rev-parse", "HEAD"], text=True).strip()


def read(p):
    return subprocess.check_output(
        ["git", "show", source + ":" + p.relative_to(R).as_posix()], text=True
    )


program = read(D / "program.py.txt")
helper = read(D / "average.py.txt")
assert (
    program == (D / "program.py.txt").read_text() and helper == (D / "average.py.txt").read_text()
)
publications = {
    str(e): json.loads(
        (
            R
            / "initiative/research/paired-full-fitting-20261018-v1"
            / ("epoch-" + str(e) + "-private-verified.json")
        ).read_text()
    )
    for e in (2, 3)
}
E = R / ".env"
assert E.stat().st_mode & 0o077 == 0
values = {
    k.strip(): x.strip().strip("\"'")
    for line in E.read_text().splitlines()
    for k, sep, x in [line.partition("=")]
    if sep and not line.lstrip().startswith("#")
}
with (L / "intent.json").open("x") as f:
    json.dump(
        {
            "source_git": source,
            "program_sha256": hashlib.sha256(program.encode()).hexdigest(),
            "helper_sha256": hashlib.sha256(helper.encode()).hexdigest(),
            "fixed_source_epochs": [2, 3],
            "weights": [0.5, 0.5],
            "CPU": 1,
            "GPU": 0,
            "memory_GiB": 16,
            "RAM_cap_GiB": 4,
            "new_permanent_model_bytes": 0,
            "automatic_retry": False,
        },
        f,
        indent=2,
    )
s = importlib.util.spec_from_file_location(
    "stdin", R / "initiative/research/paired-logit-head-gradient-20261018-v1/stdin_transport.py"
)
t = importlib.util.module_from_spec(s)
s.loader.exec_module(t)
a = [
    "srun",
    "--account=skattel",
    "--partition=preemptable",
    "--qos=preemptable",
    "--nodelist=evc33",
    "--ntasks=1",
    "--cpus-per-task=1",
    "--gres=none",
    "--mem=16G",
    "--time=00:20:00",
    "--job-name=rsna-real-paired-late2",
    "env",
    "CUDA_VISIBLE_DEVICES=",
    "PYTHONDONTWRITEBYTECODE=1",
    "PYTHONNOUSERSITE=1",
    "OMP_NUM_THREADS=1",
    "MKL_NUM_THREADS=1",
    "OPENBLAS_NUM_THREADS=1",
    "/tmp/rsna-environment-907632-7sr9a7i6/environment/bin/python",
    "-B",
    "-c",
    t.COMMAND,
]
x = subprocess.run(
    [
        "ssh",
        "-S",
        "/tmp/rsna-research-20261001-stable.sock",
        "-o",
        "BatchMode=yes",
        "newton",
        shlex.join(a),
    ],
    input=t.envelope(
        program,
        {
            "source_git": source,
            "helper": helper,
            "helper_sha256": hashlib.sha256(helper.encode()).hexdigest(),
            "publications": publications,
            "token": values["HF_TOKEN"],
        },
    ),
    capture_output=True,
    text=True,
    timeout=1300,
)
(L / "transport.json").write_text(
    json.dumps({"returncode": x.returncode, "stdout": x.stdout, "stderr": x.stderr}, indent=2)
    + "\n"
)
assert x.returncode == 0, (
    "Original candidate build failed/uncertain; reconcile without retry: " + x.stderr[-1000:]
)
v = json.loads(x.stdout)
(D / "actual-private-verified.json").write_text(json.dumps(v, indent=2) + "\n")
print(json.dumps(v), flush=True)
