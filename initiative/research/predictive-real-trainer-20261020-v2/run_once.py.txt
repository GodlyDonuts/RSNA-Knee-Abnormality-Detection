"""One CPU admission, then at most one distinct real-MRI B/C H100 diagnostic."""

import hashlib
import importlib.util
import json
import shlex
import subprocess
from pathlib import Path

R = Path.cwd()
D = Path(__file__).resolve().parent
ORIGINAL = D.with_name("predictive-real-trainer-20261020-v1")
L = R / "artifacts/predictive-real-trainer-20261020-v2"
L.mkdir(mode=0o700, exist_ok=True)
assert not (L / "intent.json").exists()
git = subprocess.check_output(["git", "rev-parse", "HEAD"], text=True).strip()
sources = {
    p.relative_to(R).as_posix(): p.read_text()
    for p in (R / "rsna/competitive/predictive").glob("*.py")
}
sources.update(
    {
        "worker.py": (ORIGINAL / "worker.py.txt").read_text(),
        "driver.py": (ORIGINAL / "driver.py.txt").read_text(),
        "rsna/__init__.py": "",
        "rsna/competitive/__init__.py": "",
    }
)
for p in [
    *D.glob("*.py.txt"),
    *ORIGINAL.glob("*.py.txt"),
    *(R / "rsna/competitive/predictive").glob("*.py"),
]:
    assert p.read_bytes() == subprocess.check_output(
        ["git", "show", git + ":" + p.relative_to(R).as_posix()]
    )
spec = importlib.util.spec_from_file_location("fixed_label_join", D / "builder.py.txt")
# Explicit source loader because .txt files are intentionally not executable entrypoints.
builder = {}
exec(
    compile((D / "builder.py.txt").read_text(), "fixed_original_label_join_builder", "exec"),
    builder,
)
source = builder["build"]((ORIGINAL / "prepare.py.txt").read_text())
raw = (R / "initiative/research/predictive-raw-input-pilot-20261020-v1/program.py.txt").read_text()
payload = {
    "source_git": git,
    "files": sources,
    "sha256": {n: hashlib.sha256(s.encode()).hexdigest() for n, s in sources.items()},
    "original_raw_decoder": raw,
    "original_raw_decoder_sha256": hashlib.sha256(raw.encode()).hexdigest(),
}
(L / "intent.json").write_text(
    json.dumps(
        {
            "source_git": git,
            "maximum_CPU_admission_jobs": 1,
            "maximum_GPU_diagnostics": 1,
            "diagnostic_studies": 1,
            "arms": ["B", "C"],
            "new_source_downloads": 0,
            "engineering_checkpoint_cap_bytes": 12 * 2**30,
            "private_case_cap_bytes": 128 * 2**20,
            "control_cap_bytes": 4 * 2**20,
            "new_shared_bytes_files": 0,
            "automatic_retry": False,
        },
        indent=2,
    )
    + "\n"
)
s = importlib.util.spec_from_file_location(
    "original_stdin_transport",
    R / "initiative/research/paired-logit-head-gradient-20261018-v1/stdin_transport.py",
)
t = importlib.util.module_from_spec(s)
s.loader.exec_module(t)
program = (
    "import sys\ntry:\n"
    + "".join("    " + line + "\n" for line in source.splitlines())
    + 'except Exception as error:\n    import traceback,json\n    frame=traceback.extract_tb(error.__traceback__)[-1]\n    print(json.dumps({"error_type":type(error).__name__,"line":frame.lineno}),file=sys.stderr)\n    raise SystemExit(1)\n'
)
a = [
    "srun",
    "--account=skattel",
    "--partition=normal",
    "--qos=normal",
    "--nodelist=evc33",
    "--ntasks=1",
    "--cpus-per-task=2",
    "--gres=none",
    "--mem=8G",
    "--time=00:15:00",
    "--job-name=rsna-predictive-trainer-inputs",
    "env",
    "CUDA_VISIBLE_DEVICES=",
    "PYTHONNOUSERSITE=1",
    "PYTHONDONTWRITEBYTECODE=1",
    "OMP_NUM_THREADS=2",
    "/tmp/rsna-environment-907632-7sr9a7i6/environment/bin/python",
    "-B",
    "-c",
    t.COMMAND,
]
ssh = ["ssh", "-S", "/tmp/rsna-research-20261001-stable.sock", "-o", "BatchMode=yes", "newton"]
x = subprocess.run(
    [*ssh, shlex.join(a)],
    input=t.envelope(program, payload),
    text=True,
    capture_output=True,
    timeout=1200,
)
(L / "prepare-transport.json").write_text(
    json.dumps({"returncode": x.returncode, "stdout": x.stdout, "stderr": x.stderr}, indent=2)
    + "\n"
)
assert x.returncode == 0, "Inspect original admission state; no mutation retry"
prepared = json.loads(x.stdout)
assert prepared["status"] == "actual_raw_MRI_parent_BC_trainer_CPU_inputs_and_source_admitted"
(D / "actual-prepare.json").write_text(json.dumps(prepared, indent=2) + "\n")
root = "/tmp/rsna-predictive-real-trainer-20261020-v1"
script = "\n".join(
    [
        "#!/bin/bash",
        "#SBATCH --account=skattel",
        "#SBATCH --partition=normal",
        "#SBATCH --qos=normal",
        "#SBATCH --nodelist=evc33",
        "#SBATCH --ntasks=1",
        "#SBATCH --cpus-per-task=2",
        "#SBATCH --gres=gpu:nvidia_h100_pcie:1",
        "#SBATCH --mem=24G",
        "#SBATCH --time=00:20:00",
        "#SBATCH --job-name=rsna-predictive-BC-real-MRI",
        "#SBATCH --output=" + root + "/GPU-%j.out",
        "#SBATCH --error=" + root + "/GPU-%j.err",
        "set -euo pipefail",
        "umask 077",
        "export PYTHONNOUSERSITE=1 PYTHONDONTWRITEBYTECODE=1 OMP_NUM_THREADS=2 CUBLAS_WORKSPACE_CONFIG=:4096:8",
        "exec /tmp/rsna-environment-907632-7sr9a7i6/environment/bin/python -B "
        + root
        + "/code/driver.py",
        "",
    ]
)
(L / "GPU-submit-intent.json").write_text(
    json.dumps(
        {
            "source_git": git,
            "script_sha256": hashlib.sha256(script.encode()).hexdigest(),
            "CPU_acceptance": prepared,
            "maximum_GPU_jobs": 1,
            "automatic_retry": False,
        },
        indent=2,
    )
    + "\n"
)
x = subprocess.run(
    [*ssh, "sbatch --parsable"], input=script, text=True, capture_output=True, timeout=90
)
(L / "GPU-submit-transport.json").write_text(
    json.dumps({"returncode": x.returncode, "stdout": x.stdout, "stderr": x.stderr}, indent=2)
    + "\n"
)
assert x.returncode == 0 and x.stdout.strip().split(";")[0].isdigit(), (
    "Read-only reconciliation required; never repeat submit"
)
out = {
    "status": "one_original_real_raw_MRI_BC_H100_diagnostic_submitted",
    "GPU_job": x.stdout.strip().split(";")[0],
    "source_git": git,
    "node": "evc33",
    "physical_node_GPUs": 2,
    "requested_GPUs": 1,
    "CPUs": 2,
    "time_cap_minutes": 20,
    "fixed_original_source_case": True,
    "full_cohort_fit_or_actual_GPU_acceptance_or_accuracy": False,
    "automatic_retry": False,
}
(D / "actual-submit.json").write_text(json.dumps(out, indent=2) + "\n")
print(json.dumps(out), flush=True)
