"""Private node-local stream after actual same-runtime CPU admission on26."""

import json
import shlex
import subprocess
from pathlib import Path

R = Path.cwd()
D = Path(__file__).resolve().parent
L = R / "artifacts/multimodal-evidence-teacher-20261019-v1/earlier-slot-evc26-v1"
L.mkdir(mode=0o700, exist_ok=True)
assert not (L / "copy-intent.json").exists()
state = subprocess.check_output(
    [
        "ssh",
        "-S",
        "/tmp/rsna-research-20261001-stable.sock",
        "-o",
        "BatchMode=yes",
        "newton",
        "sacct -X -n -P -j 968374 --format=JobIDRaw,State,ExitCode",
    ],
    text=True,
)
assert "968374|COMPLETED|0:0" in state
sources = {}
for name in ("send.py.txt", "receive.py.txt"):
    p = D / name
    sources[name] = p.read_text()
    assert sources[name] == subprocess.check_output(
        ["git", "show", "HEAD:" + p.relative_to(R).as_posix()], text=True
    )
(L / "copy-intent.json").write_text(
    json.dumps(
        {
            "original_model_CPU_job": "968270",
            "earlier_node_runtime_CPU_job": "968374",
            "new_node_copy_cap_bytes": 10 * 2**30,
            "originals_preserved": True,
            "automatic_retry": False,
        }
    )
)


def endpoint(node, python, source):
    return [
        "srun",
        "--account=skattel",
        "--partition=preemptable",
        "--qos=preemptable",
        "--nodelist=" + node,
        "--ntasks=1",
        "--cpus-per-task=1",
        "--gres=none",
        "--mem=2G",
        "--time=00:20:00",
        "--job-name=rsna-medgemma-earlier-copy-" + node,
        "env",
        "CUDA_VISIBLE_DEVICES=",
        "PYTHONDONTWRITEBYTECODE=1",
        "PYTHONNOUSERSITE=1",
        "OMP_NUM_THREADS=1",
        python,
        "-B",
        "-c",
        source,
    ]


sender = endpoint(
    "evc33", "/tmp/rsna-environment-907632-7sr9a7i6/environment/bin/python", sources["send.py.txt"]
)
receiver = endpoint(
    "evc26",
    "/tmp/rsna-environment-917129-_r45a3dc/environment/bin/python",
    sources["receive.py.txt"],
)
cmd = shlex.join(
    ["bash", "-o", "pipefail", "-c", shlex.join(sender) + " | " + shlex.join(receiver)]
)
x = subprocess.run(
    ["ssh", "-S", "/tmp/rsna-research-20261001-stable.sock", "-o", "BatchMode=yes", "newton", cmd],
    capture_output=True,
    text=True,
    timeout=1800,
)
(L / "copy-transport.json").write_text(
    json.dumps({"returncode": x.returncode, "stdout": x.stdout, "stderr": x.stderr}, indent=2)
)
assert x.returncode == 0, (
    "Copy stopped/uncertain; reconcile original endpoints: " + x.stderr[-1200:]
)
(D / "actual-assets-accepted.json").write_text(json.dumps(json.loads(x.stdout), indent=2) + "\n")
print(x.stdout, flush=True)
