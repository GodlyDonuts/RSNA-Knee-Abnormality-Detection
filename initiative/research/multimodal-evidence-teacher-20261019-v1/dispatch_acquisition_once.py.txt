"""One CPU acquisition, credentials exclusively through the bounded SSH stdin."""

import hashlib
import importlib.util
import json
import shlex
import subprocess
from pathlib import Path

R = Path.cwd()
D = Path(__file__).resolve().parent
L = R / "artifacts/multimodal-evidence-teacher-20261019-v1"
L.mkdir(mode=0o700, exist_ok=True)
assert not (L / "acquisition-intent.json").exists()
git = subprocess.check_output(["git", "rev-parse", "HEAD"], text=True).strip()
p = D / "acquire_model_once.py.txt"
source = p.read_text()
assert source == subprocess.check_output(
    ["git", "show", git + ":" + p.relative_to(R).as_posix()], text=True
)
manifest = json.loads((D / "original-public-model-manifest.json").read_text())
env = R / ".env"
assert env.stat().st_mode & 0o077 == 0
values = {
    k.strip(): x.strip().strip("\"'")
    for line in env.read_text().splitlines()
    for k, sep, x in [line.partition("=")]
    if sep and not line.lstrip().startswith("#")
}
spec = importlib.util.spec_from_file_location(
    "stdin_transport",
    R / "initiative/research/paired-logit-head-gradient-20261018-v1/stdin_transport.py",
)
t = importlib.util.module_from_spec(spec)
spec.loader.exec_module(t)
a = [
    "srun",
    "--account=skattel",
    "--partition=preemptable",
    "--qos=preemptable",
    "--nodelist=evc33",
    "--ntasks=1",
    "--cpus-per-task=1",
    "--gres=none",
    "--mem=4G",
    "--time=00:30:00",
    "--job-name=rsna-medgemma15-original-acquisition",
    "env",
    "CUDA_VISIBLE_DEVICES=",
    "PYTHONNOUSERSITE=1",
    "PYTHONDONTWRITEBYTECODE=1",
    "OMP_NUM_THREADS=1",
    "/tmp/rsna-environment-907632-7sr9a7i6/environment/bin/python",
    "-B",
    "-c",
    t.COMMAND,
]
with (L / "acquisition-intent.json").open("x") as f:
    json.dump(
        {
            "source_git": git,
            "source_sha256": hashlib.sha256(source.encode()).hexdigest(),
            "model": manifest["model"],
            "revision": manifest["revision"],
            "new_GPU": 0,
            "automatic_retry": False,
        },
        f,
        indent=2,
    )
x = subprocess.run(
    [
        "ssh",
        "-S",
        "/tmp/rsna-research-20261001-stable.sock",
        "-o",
        "BatchMode=yes",
        "newton",
        shlex.join(a),
    ],
    input=t.envelope(source, {"token": values["HF_TOKEN"], "manifest": manifest}),
    capture_output=True,
    text=True,
    timeout=2400,
)
(L / "acquisition-transport.json").write_text(
    json.dumps({"returncode": x.returncode, "stdout": x.stdout, "stderr": x.stderr}, indent=2)
)
assert x.returncode == 0, (
    "Original acquisition stopped/uncertain; reconcile before any continuation: " + x.stderr[-1200:]
)
out = json.loads(x.stdout.splitlines()[-1])
assert out["status"] == "actual_original_medgemma15_model_downloaded_and_verified_private_node33"
(D / "actual-original-model-verified.json").write_text(json.dumps(out, indent=2) + "\n")
print(json.dumps(out), flush=True)
