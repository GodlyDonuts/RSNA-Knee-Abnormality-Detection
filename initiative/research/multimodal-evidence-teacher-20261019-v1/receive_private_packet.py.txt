"""One complete native-display packet, bounded and private; no report/label access."""

import base64
import hashlib
import io
import json
import os
import shutil
import sys
from pathlib import Path

from PIL import Image

assert os.uname().nodename == "evc33" and not os.environ.get("SLURM_JOB_GPUS")
os.umask(0o077)
R = Path("/tmp/rsna-medgemma15-first-packet-20261019-v1")
assert not R.exists()
assert shutil.disk_usage("/tmp").free > (20 + 9) * 2**30 + (32 + 128 + 256) * 2**20
v = json.load(sys.stdin)
raw = base64.b64decode(v["manifest_base64"], validate=True)
m = json.loads(raw)
assert m["source_complete"] and m["training_only"] and m["source_slices"] == 195
assert m["reports_labels_predictions_included"] is False
assert (
    m["fitting_roster_sha256"] == "e71eb9ba952b9e4bc68e32b548b131c457b5c86c9381eb089012404923dc68cf"
)
assert len(m["images"]) == len(v["files"]) == 195 and m["input_bytes"] < 64 * 2**20
seen, decoded, total = set(), {}, 0
for r in m["images"]:
    name = r["file"]
    assert Path(name).name == name and name.endswith(".png") and name not in seen
    seen.add(name)
    data = base64.b64decode(v["files"][name], validate=True)
    assert len(data) == r["bytes"] and hashlib.sha256(data).hexdigest() == r["sha256"]
    im = Image.open(io.BytesIO(data))
    assert im.size == (r["width"], r["height"]) and im.mode == "L"
    im.verify()
    total += len(data)
    assert total < 64 * 2**20
    decoded[name] = data
assert total == m["input_bytes"]
for si in range(1, 6):
    rows = [r for r in m["images"] if r["series_index"] == si]
    assert sorted(r["slice_index"] for r in rows) == list(range(1, rows[0]["series_slices"] + 1))
R.mkdir(mode=0o700)
assert hashlib.sha256(v["pilot_source"].encode()).hexdigest() == v["pilot_sha256"]
compile(v["pilot_source"], "frozen_multimodal_pilot", "exec")
with (R / "pilot.py").open("x") as f:
    f.write(v["pilot_source"])
for name, data in decoded.items():
    with (R / name).open("xb") as f:
        f.write(data)
with (R / "manifest.private.json").open("xb") as f:
    f.write(raw)
out = {
    "status": "actual_complete_native_MRI_packet_private_node33",
    "CPU_job": os.environ["SLURM_JOB_ID"],
    "slices": 195,
    "series": 5,
    "bytes": total,
    "manifest_sha256": hashlib.sha256(raw).hexdigest(),
    "pilot_sha256": v["pilot_sha256"],
    "originals_retained": True,
    "native_pixel_crops_no_spatial_resizing": True,
    "report_or_labels_or_model_predictions": False,
    "GPU_or_accuracy_claim": False,
}
(R / "verified.json").write_text(json.dumps(out, indent=2))
print(json.dumps(out))
