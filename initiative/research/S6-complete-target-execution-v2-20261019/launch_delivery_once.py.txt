"""Launch one corrected producer and its followers only after actual recovery dispatch."""

import hashlib
import json
import subprocess
import time
from pathlib import Path

R = Path.cwd()
D = Path(__file__).resolve().parent
L = R / "artifacts/S6-complete-target-execution-v2-20261019"
L.mkdir(exist_ok=True)
assert not (L / "delivery-owners.json").exists()
actual = json.loads((D / "actual-GPU-submitted.json").read_text())
assert actual["status"] == "actual_complete_teacher_oneH100_recovery_submitted_once"
git = subprocess.check_output(["git", "rev-parse", "HEAD"], text=True).strip()
for name in ("follow_production_once", "follow_packaging_once", "follow_scoring_once"):
    p = D / (name + ".py.txt")
    assert p.read_bytes() == subprocess.check_output(
        ["git", "show", git + ":" + str(p.relative_to(R))]
    )
started = {}


def spawn(role, name, args=()):
    p = D / (name + ".py.txt")
    with (
        (L / (role + ".stdout.log")).open("x") as out,
        (L / (role + ".stderr.log")).open("x") as err,
    ):
        child = subprocess.Popen(
            ["python3", str(p), *map(str, args)],
            cwd=R,
            stdin=subprocess.DEVNULL,
            stdout=out,
            stderr=err,
            start_new_session=True,
        )
    started[role] = {"PID": child.pid, "source_sha256": hashlib.sha256(p.read_bytes()).hexdigest()}
    (L / "delivery-startup.json").write_text(
        json.dumps({"source_git": git, "owners": started}, indent=2) + "\n"
    )
    return child


producer = spawn("production", "follow_production_once")
intent = L / "production-owner-v2/intent.json"
for _ in range(120):
    if intent.exists():
        break
    assert producer.poll() is None, "Original new producer stopped; reconcile, no restart"
    time.sleep(0.5)
assert intent.exists()
assert json.loads(intent.read_text())["PID"] == producer.pid
packager = spawn("packaging", "follow_packaging_once", (producer.pid,))
pack_intent = L / "packaging-owner-v2/owner-intent.json"
for _ in range(120):
    if pack_intent.exists():
        break
    assert packager.poll() is None, "Original new packager stopped; reconcile, no restart"
    time.sleep(0.5)
assert pack_intent.exists()
(D / "actual-packaging-owner.json").write_text(
    json.dumps({"PID": packager.pid, "source_git": git}, indent=2) + "\n"
)
scorer = spawn("scoring", "follow_scoring_once", (packager.pid,))
v = {
    "status": "actual_corrected_original_production_packaging_and_canonical_scoring_followers_started",
    "source_git": git,
    "actual_recovery_GPU": actual["GPU_job"],
    "owners": started,
    "numerical_recipes_unchanged": True,
    "original_failed_owners_and_intents_preserved": True,
    "automatic_retry": False,
}
(L / "delivery-owners.json").write_text(json.dumps(v, indent=2) + "\n")
(D / "actual-delivery-owners.json").write_text(json.dumps(v, indent=2) + "\n")
print(json.dumps(v))
