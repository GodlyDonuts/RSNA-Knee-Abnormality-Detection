"""One original32 pixel-only CPU measurement; never change a live loader."""

import hashlib
import importlib.util
import json
import shlex
import subprocess
from pathlib import Path

R = Path.cwd()
D = Path(__file__).resolve().parent
L = R / "artifacts/selected-window-rendering-20261020-v1"
L.mkdir(mode=0o700, exist_ok=True)
assert not (L / "intent.json").exists()
pin = subprocess.check_output(["git", "rev-parse", "HEAD"], text=True).strip()
for n in ("renderer.py.txt", "probe.py.txt", "probe_once.py.txt"):
    p = D / n
    assert p.read_bytes() == subprocess.check_output(
        ["git", "show", pin + ":" + p.relative_to(R).as_posix()]
    )
with (L / "intent.json").open("x") as f:
    json.dump(
        {
            "source_git": pin,
            "CPU_only": True,
            "GPU_models_or_labels": False,
            "maximum_private_32_study_probes": 1,
            "new_node_local_bytes": 2**20,
            "new_shared_bytes": 0,
            "automatic_retry": False,
        },
        f,
    )
s = importlib.util.spec_from_file_location(
    "transport", R / "initiative/research/paired-logit-head-gradient-20261018-v1/stdin_transport.py"
)
t = importlib.util.module_from_spec(s)
s.loader.exec_module(t)
a = [
    "srun",
    "--account=skattel",
    "--partition=normal",
    "--qos=normal",
    "--nodelist=evc45",
    "--ntasks=1",
    "--cpus-per-task=1",
    "--gres=none",
    "--mem=4G",
    "--time=00:10:00",
    "--job-name=rsna-original-selected-window-CPU",
    "env",
    "CUDA_VISIBLE_DEVICES=",
    "PYTHONNOUSERSITE=1",
    "PYTHONDONTWRITEBYTECODE=1",
    "OMP_NUM_THREADS=1",
    "/tmp/rsna-environment-971427-pyk8rld8/environment/bin/python",
    "-B",
    "-c",
    t.COMMAND,
]
x = subprocess.run(
    ["ssh", "-S", "/tmp/rsna-research-20261001-stable.sock", "newton", shlex.join(a)],
    input=t.envelope(
        (D / "probe.py.txt").read_text(),
        {
            "source_git": pin,
            "helper": (D / "renderer.py.txt").read_text(),
            "helper_sha256": hashlib.sha256((D / "renderer.py.txt").read_bytes()).hexdigest(),
        },
    ),
    capture_output=True,
    text=True,
    timeout=1200,
)
(L / "transport.json").write_text(
    json.dumps({"returncode": x.returncode, "stdout": x.stdout, "stderr": x.stderr}, indent=2)
    + "\n"
)
assert x.returncode == 0, "Inspect actual original CPU probe; no automatic repeat"
z = json.loads(x.stdout)
(D / "actual-CPU-measured.json").write_text(json.dumps(z, indent=2) + "\n")
print(json.dumps(z), flush=True)
