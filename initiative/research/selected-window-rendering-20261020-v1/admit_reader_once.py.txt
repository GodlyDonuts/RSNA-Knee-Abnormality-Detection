import hashlib
import importlib.machinery
import importlib.util
import json
import shlex
import subprocess
from pathlib import Path

R = Path.cwd()
D = Path(__file__).resolve().parent
L = R / "artifacts/selected-window-full-cohort-20261020-v1"
assert not (L / "reader-admission-intent.json").exists()
pin = subprocess.check_output(["git", "rev-parse", "HEAD"], text=True).strip()
for name in ("admit_reader.py.txt", "admit_reader_once.py.txt", "manifest_stage.py.txt"):
    p = D / name
    assert p.read_bytes() == subprocess.check_output(
        ["git", "show", pin + ":" + p.relative_to(R).as_posix()]
    )
loader = importlib.machinery.SourceFileLoader("admission", str(D / "admit_reader.py.txt"))
m = importlib.util.module_from_spec(importlib.util.spec_from_loader(loader.name, loader))
loader.exec_module(m)
program = m.build((D / "manifest_stage.py.txt").read_text())
files = json.loads(
    (
        R / "initiative/research/teacher-pair-earlier45-20261020-v1/actual-full-input-copy.json"
    ).read_text()
)["receiver"]["files"]
modules = {
    n: hashlib.sha256((R / n).read_bytes()).hexdigest()
    for n in (
        "rsna/competitive/licensed_parent_population.py",
        "rsna/competitive/licensed_pixel_bags.py",
        "rsna/competitive/licensed_pixel_cache.py",
    )
}
s = importlib.util.spec_from_file_location(
    "transport", R / "initiative/research/paired-logit-head-gradient-20261018-v1/stdin_transport.py"
)
t = importlib.util.module_from_spec(s)
s.loader.exec_module(t)
sources = {
    "reader": (R / "rsna/competitive/licensed_selected_pixel_bags.py").read_text(),
    "renderer": (D / "renderer.py.txt").read_text(),
}
with (L / "reader-admission-intent.json").open("x") as f:
    json.dump(
        {
            "source_git": pin,
            "original_CPU": "975656",
            "original_actual_terminal": "COMPLETE_full_manifest",
            "maximum_new_CPU_jobs": 1,
            "new_GPU_or_pixel_copies": 0,
            "same_original32MiB_draft_scope": True,
            "automatic_retry": False,
        },
        f,
    )
a = [
    "srun",
    "--account=skattel",
    "--partition=normal",
    "--qos=normal",
    "--nodelist=evc45",
    "--ntasks=1",
    "--cpus-per-task=1",
    "--gres=none",
    "--mem=4G",
    "--time=00:05:00",
    "--job-name=rsna-window-integrity-selected-reader-admission",
    "env",
    "CUDA_VISIBLE_DEVICES=",
    "PYTHONNOUSERSITE=1",
    "PYTHONDONTWRITEBYTECODE=1",
    "OMP_NUM_THREADS=1",
    "/tmp/rsna-environment-971427-pyk8rld8/environment/bin/python",
    "-B",
    "-c",
    t.COMMAND,
]
x = subprocess.run(
    ["ssh", "-S", "/tmp/rsna-research-20261001-stable.sock", "newton", shlex.join(a)],
    input=t.envelope(
        program,
        {
            "source_git": pin,
            "modules": modules,
            "original_files": files,
            "actual_integrity": json.loads((D / "actual-full-cohort-integrity.json").read_text()),
            "new_sources": sources,
            "new_source_SHA": {
                n: hashlib.sha256(z.encode()).hexdigest() for n, z in sources.items()
            },
        },
    ),
    capture_output=True,
    text=True,
    timeout=3600,
)
(L / "reader-admission-transport.json").write_text(
    json.dumps({"returncode": x.returncode, "stdout": x.stdout, "stderr": x.stderr}, indent=2)
    + "\n"
)
assert x.returncode == 0, "Inspect original continuation state; never repeat automatically"
z = json.loads(x.stdout)
assert (
    z["status"]
    == "actual_full_manifest_selected_reader_original_images_targets_views_and_RNG_equal"
)
(D / "actual-reader-admission.json").write_text(json.dumps(z, indent=2) + "\n")
