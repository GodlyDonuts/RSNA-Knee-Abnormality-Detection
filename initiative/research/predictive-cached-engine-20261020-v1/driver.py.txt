"""Single admitted H100 job: unchanged fresh worker phases; exact private equality."""

import hashlib
import json
import os
import subprocess
import sys
from pathlib import Path

import numpy as np
import torch

D = Path("/tmp/rsna-predictive-cached-engine-20261020-v1")
os.umask(0o077)
assert os.uname().nodename == "evc27" and os.environ.get("SLURM_JOB_GPUS")
assert torch.cuda.device_count() == 1 and "H100" in torch.cuda.get_device_name(0)
assert not (D / "GPU-run-start.json").exists()
binding = json.loads((D / "binding.private.json").read_text())
for n, h in binding["code_sha256"].items():
    assert hashlib.sha256((D / "code" / n).read_bytes()).hexdigest() == h
(D / "GPU-run-start.json").write_text(
    json.dumps(
        {
            "GPU_job": os.environ["SLURM_JOB_ID"],
            "source_git": binding["source_git"],
            "clinical_accuracy_or_submission_candidate": False,
        }
    )
    + "\n"
)


def equal(a, b):
    if isinstance(a, torch.Tensor):
        assert a.dtype == b.dtype and a.shape == b.shape and torch.equal(a, b)
    elif isinstance(a, np.ndarray):
        assert a.dtype == b.dtype and np.array_equal(a, b)
    elif isinstance(a, dict):
        assert a.keys() == b.keys()
        for k in a:
            equal(a[k], b[k])
    elif isinstance(a, (tuple, list)):
        assert type(a) is type(b) and len(a) == len(b)
        for x, y in zip(a, b):
            equal(x, y)
    else:
        assert a == b


results = []
for arm in ("B", "C", "C0"):
    for phase in ("full", "stop1", "resume"):
        subprocess.run(
            [sys.executable, "-B", str(D / "code/worker.py"), arm, phase],
            check=True,
            env={
                **os.environ,
                "PYTHONNOUSERSITE": "1",
                "PYTHONDONTWRITEBYTECODE": "1",
                "CUBLAS_WORKSPACE_CONFIG": ":4096:8",
            },
        )
    root = D / "engineering" / arm
    complete = torch.load(root / "full/final.pt", map_location="cpu", weights_only=False)
    resumed = torch.load(root / "resume/final.pt", map_location="cpu", weights_only=False)
    equal(complete, resumed)
    assert complete["step"] == 4 and len(complete["history"]) == 4
    assert all(
        r["encoder_gradient_norm"] > 0 and r["head_gradient_norm"] > 0 for r in complete["history"]
    )
    assert all(not x["encoder_updated"] for x in complete["history"][:1])
    assert all(r["encoder_updated"] for r in complete["history"][1:])
    assert [r["studies"] for r in complete["history"]] == [8, 4, 8, 4]
    assert all(max(r["LR"]) <= 3e-5 for r in complete["history"])
    assert (root / "full/prediction.private.csv").read_bytes() == (
        root / "resume/prediction.private.csv"
    ).read_bytes()
    results.append(
        {
            "arm": arm,
            "full_stop_resume_complete_state_and_prediction_bytes_equal": True,
            "terminal": json.loads((root / "full/terminal.json").read_text()),
            "real_optimizer_history": complete["history"],
        }
    )
    del complete, resumed
out = {
    "status": "actual_cached_B_C_C0_real_MRI_complete_recovery_and_throughput_accepted",
    "GPU_job": os.environ["SLURM_JOB_ID"],
    "source_git": binding["source_git"],
    "arms": results,
    "twelve_fitting_train_four_disjoint_engineering_only": True,
    "clinical_accuracy_or_full_fit_or_T4_acceptance": False,
    "no_old_teacher_or_inherited_classifier_logits": True,
    "all_original_private_assets_preserved": True,
}
(D / "GPU-accepted.json").write_text(json.dumps(out) + "\n")
print(json.dumps(out), flush=True)
