class FixedEncoder(torch.nn.Module):
    def __init__(self, backbone):
        super().__init__()
        self.backbone = backbone
        self.register_buffer("mean", torch.tensor([0.485, 0.456, 0.406])[None, :, None, None])
        self.register_buffer("std", torch.tensor([0.229, 0.224, 0.225])[None, :, None, None])

    def forward(self, images):
        return self.backbone((images - self.mean) / self.std)

class Case:
    load_seconds = 0.0
    loads = 0

    def __len__(self):
        return 12

    def case(self, index_number, epoch):
        assert 0 <= index_number < 16 and 0 <= epoch < 2
        row = index[index_number]
        began = time.monotonic()
        contexts = load(row["path"], row["proof"], provenance=row["provenance"])
        self.load_seconds += time.monotonic() - began
        self.loads += 1
        return {
            "contexts": contexts,
            "labels": labels[index_number : index_number + 1],
            "mask": masks[index_number : index_number + 1],
        }
