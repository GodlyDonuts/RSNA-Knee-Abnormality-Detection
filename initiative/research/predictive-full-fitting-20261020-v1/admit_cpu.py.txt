def require(ok, reason):
    if not ok:
        raise ValueError(reason)


def canonical(value):
    return json.dumps(value, sort_keys=True, separators=(",", ":")).encode()


def sha(path):
    with Path(path).open("rb") as handle:
        return hashlib.file_digest(handle, "sha256").hexdigest()


def stat(path):
    path = Path(path)
    s = path.stat()
    require(
        path.is_absolute() and path.is_file() and not path.is_symlink() and s.st_uid == os.getuid(),
        "Original owned regular file required",
    )
    return [s.st_ino, s.st_size, s.st_mtime_ns, s.st_ctime_ns]


def ref(path):
    path = Path(path)
    before = stat(path)
    digest = sha(path)
    require(stat(path) == before, "Original reference changed during read")
    return {"path": str(path), "sha256": digest}


def read_ref(value):
    actual = ref(value["path"])
    require(actual["sha256"] == value["sha256"], "Original small reference changed")
    return json.loads(Path(value["path"]).read_text())


def write_once(path, value):
    # Publish a complete small JSON atomically without overwriting any original.
    path = Path(path)
    tmp = path.with_name(path.name + ".writing")
    with tmp.open("xb") as handle:
        handle.write(canonical(value) + b"\n")
        handle.flush()
        os.fsync(handle.fileno())
    os.link(tmp, path)
    tmp.unlink()

def initialize(root, refs):

    import timm
    import torch

    from rsna.competitive.predictive.model import Predictive, ReconstructionControl

    require(torch.cuda.device_count() == 0, "Preparation must have zero allocated GPUs")
    torch.set_num_threads(1)
    torch.set_num_interop_threads(1)
    selected = [
        n
        for n in ast.parse(Path(refs["parent_source"]["path"]).read_text()).body
        if isinstance(n, (ast.FunctionDef, ast.ClassDef))
        and n.name in ("build_backbone", "RaptorClassifier", "load_model")
    ]
    require(len(selected) == 3, "Exact original parent constructor required")
    ns = dict(torch=torch, nn=torch.nn, timm=timm, gc=gc, hashlib=hashlib, INPUT_RES=320)
    exec(compile(ast.Module(body=selected, type_ignores=[]), "original_MRI_encoder", "exec"), ns)
    with contextlib.redirect_stdout(io.StringIO()):
        parent, resolution = ns["load_model"](
            refs["parent_weight"]["path"],
            "coatnet_rmlp_2_rw_384.sw_in12k_ft_in1k",
            384,
            torch.device("cpu"),
        )
    require(
        resolution == 320 and sum(p.numel() for p in parent.parameters()) == 73136096,
        "Exact original MRI backbone failed CPU loading",
    )

    class Encoder(torch.nn.Module):
        def __init__(self, backbone):
            super().__init__()
            self.backbone = backbone
            self.register_buffer("mean", torch.tensor([0.485, 0.456, 0.406])[None, :, None, None])
            self.register_buffer("std", torch.tensor([0.229, 0.224, 0.225])[None, :, None, None])

        def forward(self, images):
            return self.backbone((images - self.mean) / self.std)

    models = {}
    for arm in ("B", "C", "C0"):
        random.seed(20261020)
        np.random.seed(20261020)
        torch.manual_seed(20261020)
        cls = ReconstructionControl if arm == "B" else Predictive
        model = cls(Encoder(parent.backbone), parent.backbone.num_features, hidden=128)
        shared = hashlib.sha256()
        for name, tensor in sorted(model.state_dict().items()):
            if name.split(".")[0] in {
                "encoder",
                "project",
                "position",
                "attention",
                "norm",
                "prior",
            }:
                shared.update(canonical([name, str(tensor.dtype), list(tensor.shape)]))
                shared.update(tensor.contiguous().numpy().tobytes())
        schema = {
            name: {"shape": list(t.shape), "dtype": str(t.dtype)}
            for name, t in model.state_dict().items()
        }
        count = sum(p.numel() for p in model.parameters())
        state_bytes = sum(t.numel() * t.element_size() for t in model.state_dict().values())
        require(
            count == PARAMETERS[arm] and all(p.requires_grad for p in model.encoder.parameters()),
            "Complete new predictive arm must retain full trainable encoder",
        )
        mean = model.reconstruct if arm == "B" else model.negative
        require(
            not torch.count_nonzero(mean.weight) and not torch.count_nonzero(mean.bias),
            "Accepted zero density-mean initialization required",
        )
        if arm != "B":
            require(
                not torch.count_nonzero(model.shift.weight)
                and not torch.count_nonzero(model.shift.bias),
                "Accepted zero conditional shift required",
            )
        require(
            7 * (4 * state_bytes + 64 * MIB) + 7 * (state_bytes + 16 * MIB) + 64 * MIB <= 12 * GIB,
            "Seven full checkpoints/six exports/one download exceed 12GiB arm cap",
        )
        models[arm] = dict(
            shared_tensor_sha256=shared.hexdigest(),
            parameter_count=count,
            model_schema_sha256=hashlib.sha256(canonical(schema)).hexdigest(),
            model_state_bytes=state_bytes,
        )
        del model
    require(
        len({x["shared_tensor_sha256"] for x in models.values()}) == 1,
        "B/C/C0 initial encoder/context states differ",
    )
    require(
        stat(refs["parent_weight"]["path"]) == refs["parent_weight"]["stat"],
        "Parent changed while read",
    )
    return models, dict(torch=torch.__version__, timm=timm.__version__, cuda=torch.version.cuda)

def terminal_job(job, allowed=("COMPLETED",)):
    require(str(job).isdecimal(), "Exact original job ID required")
    accounting = subprocess.check_output(
        ["sacct", "-X", "-n", "-P", "-j", str(job), "--format=JobIDRaw,State,ExitCode"], text=True
    )
    live = subprocess.check_output(["squeue", "-h", "-j", str(job), "-o", "%i|%T"], text=True)
    exact = [row.split("|") for row in accounting.splitlines() if row.split("|")[0] == str(job)]
    require(
        not live.strip()
        and len(exact) == 1
        and exact[0][1] in allowed
        and (exact[0][1] != "COMPLETED" or exact[0][2] == "0:0"),
        "Original CPU producer must actually be complete before full fitting admission",
    )
    return {
        "job": str(job),
        "state": exact[0][1],
        "exit_code": exact[0][2],
        "accounting_sha256": hashlib.sha256(accounting.encode()).hexdigest(),
    }

def main(v):
    os.umask(0o077)
    sys.dont_write_bytecode = True
    require(
        os.uname().nodename == "evc33"
        and not os.environ.get("SLURM_JOB_GPUS")
        and os.environ.get("SLURM_JOB_ID")
        and Path(sys.executable).resolve() == PYTHON.resolve(),
        "Exact original evc33 runtime in a zero-GPU CPU allocation required",
    )
    os.environ.update(HF_HUB_OFFLINE="1", TRANSFORMERS_OFFLINE="1")
    root = Path(v["root"])
    require(
        root.is_absolute()
        and root.resolve().is_relative_to(Path("/tmp"))
        and not root.is_symlink(),
        "New owned CPU-control scope under /tmp required",
    )
    require(v["mode"] in ("prepare", "admit"), "Prepare or full admission only")
    return prepare(v, root) if v["mode"] == "prepare" else admit(v, root)


if __name__ == "__main__":
    try:

    except BaseException as error:

            tb = tb.tb_next
        print(
            json.dumps({"error_type": type(error).__name__, "site_line": tb.tb_lineno}),
            file=sys.stderr,
        )
        raise SystemExit(1)
