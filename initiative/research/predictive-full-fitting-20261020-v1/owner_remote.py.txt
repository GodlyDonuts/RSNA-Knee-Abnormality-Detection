def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def once(path, value):
    path = Path(path)
    tmp = path.with_name(path.name + ".pending-" + str(os.getpid()))
    with tmp.open("x") as stream:
        json.dump(value, stream, allow_nan=False, sort_keys=True)
        stream.write("\n")
        stream.flush()
        os.fsync(stream.fileno())
    try:
        os.link(tmp, path)
    finally:
        tmp.unlink()


def reference(path):
    return {"path": str(path), "sha256": sha(path)}

def accounting(jobs):
    assert jobs and all(str(j).isdecimal() for j in jobs)
    text = subprocess.check_output(
        [
            "sacct",
            "-X",
            "-n",
            "-P",
            "-j",
            ",".join(jobs),
            "--format=JobIDRaw,State,ExitCode,JobName%80,NodeList%80",
        ],
        text=True,
    )
    queue = subprocess.check_output(
        ["squeue", "-h", "-j", ",".join(jobs), "-o", "%i|%T"], text=True
    )
    queued = {line.split("|")[0] for line in queue.splitlines()}
    result = {}
    for job in jobs:
        rows = [line.split("|") for line in text.splitlines() if line.split("|")[0] == job]
        assert len(rows) <= 1
        result[job] = {"state": None, "queue_empty": job not in queued}
        if rows:
            row = rows[0]
            result[job].update(state=row[1], exit_code=row[2], name=row[3], node=row[4])
    return result

def remaining_cache(v):
    # Keep unused original caps reserved conservatively. Existing bytes are paid
    # once, including reusable members from the original sixteen-case cache.
    raw, context = v["raw_collected"], v["context_collected"]
    for role, receipt in (("raw", raw), ("context", context)):
        assert receipt["status"] == "actual_original_full_cache_terminal_collected"
        assert receipt["role"] == role and receipt["target_terminal"] == "COMPLETED"
        assert receipt["exit_code"] == "0:0" and receipt["studies"] == 3887
        assert receipt["fitting_roster_sha256"] == ROSTER
        assert receipt["existing_cache_stat_bindings_checked"] == 3887
    raw_roots = {
        "actual_complete3887_original_raw_fitting_cache_and_exact_context_target_support_parity_accepted": Path(
            "/tmp/rsna-predictive-full-raw-cache-20261020-v1"
        ),
        "actual_missing_only3887_original_raw_cache_continuation_accepted": Path(
            "/tmp/rsna-predictive-full-cache-continuation-20261020-v1"
        ),
    }
    raw_root = raw_roots[raw["original_actual"]["status"]]
    for receipt, root in ((raw, raw_root), (context, cache_roots(context))):
        assert sha(root / "actual.json") == receipt["original_actual_sha256"]
        assert receipt["original_actual"]["CPU_job"] == receipt["target_job"]
        assert receipt["original_actual"]["source_git"] == receipt["source_git"]
    assert 0 < raw["existing_cache_bytes"] <= 320 * GIB
    assert 0 < context["existing_cache_bytes"] <= 96 * GIB
    return max(0, 320 * GIB - raw["existing_cache_bytes"]) + max(
        0, 96 * GIB - context["existing_cache_bytes"]
    )


def base_other(v):

    # Remove selected36GiB only where explicit per-arm scopes are added below.
    return 24 * GIB + 574 * MIB + remaining_cache(v)

def check_node():
    text = subprocess.check_output(["scontrol", "show", "node", "evc33", "-o"], text=True)
    tokens = dict(token.split("=", 1) for token in text.split() if "=" in token)
    assert tokens["NodeName"] == "evc33"
    assert re.search(r"(?:^|,)gres/gpu=2(?:,|$)", tokens["CfgTRES"])
    assert "gpu:nvidia_h100_pcie:2" in tokens["Gres"]
    return hashlib.sha256(text.encode()).hexdigest()

    node_sha = check_node()

def log_progress(v):
    result = {"status": "actual_predictive_full_job_progress_observed", "arms": {}}
    jobs = v["dispatched"]["jobs"]
    states = accounting([j["GPU_job"] for j in jobs.values()])
    for arm, job in jobs.items():
        assert arm in ARMS and job["source_git"] == SCIENCE
        name, identity = "rsna-predictive-full-" + arm, job["GPU_job"]
        assert job["job_name"] == name and job["node"] == "evc33"
        out = LOG_ROOT / (name + "-" + identity + ".out")
        err = LOG_ROOT / (name + "-" + identity + ".err")
        assert str(out) == job["stdout"] and str(err) == job["stderr"]
        state = states[identity]
        if state.get("name"):
            assert state["name"] == name
            assert state.get("node") in ("evc33", "None assigned", "(null)", "", "None")
        log_bytes = sum(p.stat().st_size for p in (out, err) if p.exists())
        assert log_bytes <= 8 * MIB, "Original combined GPU log allowance exceeded; preserve job"
        records, terminal = [], None
        if out.exists():
            assert out.stat().st_size < 32 * MIB and not out.is_symlink()
            for line in out.read_text().splitlines():
                try:
                    value = json.loads(line)
                except (ValueError, TypeError):
                    continue
                if not isinstance(value, dict) or value.get("arm") != arm:
                    continue
                if type(value.get("step")) is int:
                    records.append(value)
                if value.get("status") == FINAL:
                    assert terminal is None
                    terminal = value
        assert [r["step"] for r in records] == list(range(1, len(records) + 1))
        assert len(records) <= 2916
        for r in records:
            assert all(
                math.isfinite(r[k])
                for k in (
                    "loss",
                    "exam_loss",
                    "density_loss",
                    "gradient_norm",
                    "head_gradient_norm",
                    "encoder_gradient_norm",
                )
            )
        first = records[:200]
        early = None
        if len(first) == 200:
            early = {
                "actual_updates": 200,
                "history_sha256": hashlib.sha256(
                    json.dumps(
                        first, sort_keys=True, separators=(",", ":"), allow_nan=False
                    ).encode()
                ).hexdigest(),
                "maxima": {
                    key: max(r[key] for r in first)
                    for key in (
                        "loss",
                        "exam_loss",
                        "head_gradient_norm",
                        "encoder_gradient_norm",
                        "gradient_norm",
                        "peak_GPU_bytes",
                        "peak_host_RSS_bytes",
                    )
                },
                "last": first[-1],
                "clinical_accuracy_claim": False,
            }
        if terminal:
            assert terminal["GPU_job"] == identity and terminal["source_git"] == SCIENCE
            assert terminal["binding_sha256"] == job["binding_sha256"]
            assert terminal["updates"] == 2916 and terminal["study_exposures"] == 23322
            assert terminal["immutable_epoch_sidecars"] == 6 and len(records) == 2916
        result["arms"][arm] = {
            "GPU_job": identity,
            "slurm": state,
            "observed_updates": len(records),
            "latest": records[-1] if records else None,
            "first200": early,
            "terminal": terminal,
            "stderr_bytes": err.stat().st_size if err.exists() else 0,
            "sealed_epochs": 6
            if terminal
            else (records[-1]["sealed_epoch_sidecars"] if records else 0),
        }
    return result


def used_bytes(root):
    total = 0
    if root.exists():
        for path in root.rglob("*"):
            try:
                if path.is_file():
                    total += path.stat().st_size
            except FileNotFoundError:
                assert "independent-download-" in str(path)
    return total


def publish(v):
    preparation(v)
    arm, epoch = v["arm"], v["epoch"]
    assert arm in ARMS and type(epoch) is int and 1 <= epoch <= 6
    binding = ROOT / f"full-admission/binding-{arm}.private.json"
    assert sha(binding) == v["dispatched"]["jobs"][arm]["binding_sha256"]
    side = json.loads((run_root(arm) / f"epoch-{epoch:03d}.json").read_text())
    assert side["GPU_job"] == v["dispatched"]["jobs"][arm]["GPU_job"]
    namespace = runpy.run_path(str(ROOT / "code/publish_epoch.py"), run_name="publisher_library")
    payload = {
        "mode": "inspect",
        "binding": reference(binding),
        "epoch": epoch,
        "source_git": SCIENCE,
    }
    check = namespace["main"](payload)
    if check.get("status") == VERIFIED:
        return check  # Read-only reconciliation of an already verified original.
    assert check["status"] == "sealed_epoch_ready_no_publication_intent"
    funding = {
        "status": "actual_predictive_epoch_publication_storage_funded",
        "CPU_job": os.environ["SLURM_JOB_ID"],
        "node": "evc33",
        "arm": arm,
        "run_root": str(run_root(arm)),
        "epoch": epoch,
        "arm_cap_bytes": 12 * GIB,
        "standing_free_reserve_bytes": 20 * GIB,
        "other_unpaid_node_liabilities_bytes": base_other(v)
        + 16 * MIB
        + sum(max(0, 12 * GIB - used_bytes(run_root(other))) for other in ARMS if other != arm),
        "shared_unpaid_bytes": SHARED_FUTURE,
        "shared_unpaid_files": 368,
        "all_other_obligations_retained": True,
        "no_deletion_credit": True,
    }
    # The actual publisher checks fresh df/lfs against this explicit complete scope.
    dest = ROOT / f"publication-funding-{arm}-{epoch:03d}.json"
    once(dest, funding)
    payload.update(mode="publish", publication_funding=reference(dest), token=v["token"])
    return namespace["main"](payload)


def main(v):
    os.umask(0o077)
    if v["mode"] == "progress":
        return log_progress(v)
    return {"admit": admission, "dispatch": dispatch, "publish": publish}[v["mode"]](v)


if __name__ == "__main__":
    try:
        print(json.dumps(main(json.load(sys.stdin)), allow_nan=False), flush=True)
    except BaseException as error:
        tb = error.__traceback__
        while tb.tb_next:
            tb = tb.tb_next
        print(
            json.dumps({"error_type": type(error).__name__, "site_line": tb.tb_lineno}),
            file=sys.stderr,
        )
        raise SystemExit(1)
