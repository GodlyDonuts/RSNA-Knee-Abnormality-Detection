"""Single CPU source smoke test, never a deployment or actual model admission."""

import hashlib
import importlib.util
import json
import runpy
import shlex
import subprocess
from pathlib import Path

R = Path.cwd()
D = Path(__file__).resolve().parent
L = R / "artifacts/S6-complete-target-serving-20261019-v1"
L.mkdir(mode=0o700, exist_ok=True)
assert not (L / "CPU-source-intent.json").exists()
pin = subprocess.check_output(["git", "rev-parse", "HEAD"], text=True).strip()
paths = {
    "compiler.py": D / "compiler.py.txt",
    "validate.py": D / "validate.py.txt",
    "legacy.py": R / "scripts/kaggle/licensed_importance_serving.py",
    "bounded.py": R / "rsna/competitive/bounded_preparation.py",
    "codec.py": R / "rsna/competitive/licensed_pixel_cache.py",
    "mixin.py": R / "rsna/competitive/licensed_full_bag.py",
    "importance.py": R / "rsna/competitive/licensed_importance.py",
}
sources = {n: p.read_text() for n, p in paths.items()}
for n, p in paths.items():
    assert (
        subprocess.check_output(["git", "show", pin + ":" + p.relative_to(R).as_posix()], text=True)
        == sources[n]
    )
fixture = runpy.run_path(str(D / "check.py.txt"))
program = (D / "actual_parent_CPU.py.txt").read_text()
assert (
    subprocess.check_output(
        ["git", "show", pin + ":" + (D / "actual_parent_CPU.py.txt").relative_to(R).as_posix()],
        text=True,
    )
    == program
)
spec = importlib.util.spec_from_file_location(
    "transport", R / "initiative/research/paired-logit-head-gradient-20261018-v1/stdin_transport.py"
)
t = importlib.util.module_from_spec(spec)
spec.loader.exec_module(t)
(L / "CPU-source-intent.json").write_text(
    json.dumps(
        {
            "source_git": pin,
            "source_SHA": {n: hashlib.sha256(s.encode()).hexdigest() for n, s in sources.items()},
            "synthetic_receipts_only": True,
            "new_GPU_Dataset_or_competition_mutations": 0,
            "automatic_retry": False,
        },
        indent=2,
    )
    + "\n"
)
a = [
    "srun",
    "--account=skattel",
    "--partition=preemptable",
    "--qos=preemptable",
    "--nodelist=evc33",
    "--ntasks=1",
    "--cpus-per-task=1",
    "--gres=none",
    "--mem=2G",
    "--time=00:05:00",
    "--job-name=rsna-H100-serving-source-CPU",
    "env",
    "CUDA_VISIBLE_DEVICES=",
    "PYTHONDONTWRITEBYTECODE=1",
    "PYTHONNOUSERSITE=1",
    "OMP_NUM_THREADS=1",
    "/tmp/rsna-environment-907632-7sr9a7i6/environment/bin/python",
    "-B",
    "-c",
    t.COMMAND,
]
x = subprocess.run(
    [
        "ssh",
        "-S",
        "/tmp/rsna-research-20261001-stable.sock",
        "-o",
        "BatchMode=yes",
        "newton",
        shlex.join(a),
    ],
    input=t.envelope(
        program,
        {
            "source_git": pin,
            "source": sources,
            "source_SHA": {n: hashlib.sha256(s.encode()).hexdigest() for n, s in sources.items()},
            "training": fixture["training"],
            "publication": fixture["publication"],
        },
    ),
    capture_output=True,
    text=True,
    timeout=1800,
)
(L / "CPU-source-transport.json").write_text(
    json.dumps({"returncode": x.returncode, "stdout": x.stdout, "stderr": x.stderr}, indent=2)
    + "\n"
)
assert x.returncode == 0, (
    "Original source compilation stopped; inspect intent, never automatically retry"
)
actual = json.loads(x.stdout)
(D / "actual-real-parent-source-CPU.json").write_text(json.dumps(actual, indent=2) + "\n")
print(x.stdout, flush=True)
