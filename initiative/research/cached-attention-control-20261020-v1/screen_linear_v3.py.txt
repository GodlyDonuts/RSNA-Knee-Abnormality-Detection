stats = {
    eps: {
        "sni_square": np.zeros(12),
        "cv_square": np.zeros(12),
        "valid": np.zeros(12, np.int64),

started = time.monotonic()
for ordinal, index in enumerate(selected):
    weights, votes, bias, _ = readout_moments(head, data[int(index)][0])
    q = proposal(weights, votes, bias)
    u = np.random.default_rng(20261020 + int(index)).random((draws, 16))
    indices = np.searchsorted(q.cumsum(), u, side="right").clip(max=93).astype(np.int64)
    wi = np.arange(94)[None, :, None]
    ti = np.arange(12)[None, None, :]
    vi = np.arange(2)[:, None, None]
    for eps in levels:
        ds = eps * np.sin((wi + 1) * 0.37 + (ti + 1) * 0.61 + vi * 0.4)
        dv = (
            eps * votes.std(1, keepdims=True) * np.cos((wi + 1) * 0.43 - (ti + 1) * 0.29 + vi * 0.7)
        )
        current_weights = weights * np.exp(ds)
        current_weights /= current_weights.sum(1, keepdims=True)
        current_votes = votes + dv
        exact = probability((current_weights * current_votes).sum(1) + bias).mean(0)
        sni = sampled_probabilities(current_weights, current_votes, bias, indices, q)
        # Batch the same draw across both views; no per-case arrays are written.
        w = torch.from_numpy(
            np.broadcast_to(weights, (draws, *weights.shape)).reshape(draws * 2, 94, 12).copy()
        )
        vv = torch.from_numpy(
            np.broadcast_to(votes, (draws, *votes.shape)).reshape(draws * 2, 94, 12).copy()
        )
        ii = torch.from_numpy(
            np.repeat(indices[:, None, :], 2, axis=1).reshape(draws * 2, 16).copy()
        )
        lq = torch.from_numpy(np.log(q[ii.numpy()]))
        ds_selected = np.stack([ds[:, row, :] for row in indices]).reshape(draws * 2, 16, 12)
        dv_selected = np.stack([dv[:, row, :] for row in indices]).reshape(draws * 2, 16, 12)
        logits = m.linear_readout(
            w,
            vv,
            ii,
            lq,
            torch.from_numpy(ds_selected.copy()),
            torch.from_numpy(dv_selected.copy()),
            torch.from_numpy(bias.copy()),
        )
        assert torch.isfinite(logits).all()
        cv = logits.sigmoid().numpy().reshape(draws, 2, 12).mean(1)
        valid = np.ones((draws, 12), dtype=bool)
        st = stats[eps]
        st["valid"] += draws
        st["sni_square"] += np.where(valid, (sni - exact) ** 2, 0).sum(0)
        st["cv_square"] += np.where(valid, (cv - exact) ** 2, 0).sum(0)
        st["max_absolute_error"] = max(
            st["max_absolute_error"], float(np.max(np.where(valid, np.abs(cv - exact), 0)))
        )
    (D / "progress.json").write_text(

    ratio = float(b.mean() / a.mean())
    invalid = int(st["invalid"].sum())
