"""Finite source-pinned follower; one recovery only after actual positive scout."""

import hashlib
import importlib.util
import json
import shlex
import subprocess
import time
from pathlib import Path

R = Path.cwd()
D = Path(__file__).resolve().parent
L = R / "artifacts/teacher-pair-real-recovery48-20261019-v2"
L.mkdir(mode=0o700, exist_ok=True)
assert not (L / "owner-intent.json").exists()
git = subprocess.check_output(["git", "rev-parse", "HEAD"], text=True).strip()
proof = json.loads(
    (
        R
        / "initiative/audits/hf-teacher-pair-recovery48-source-20261019-v1/actual-private-verified.json"
    ).read_text()
)
assert proof["source_git"] == git
programs = {
    name: (D / (name + ".py.txt")).read_text()
    for name in ("worker", "driver", "stage_once", "follow_once")
}
for name, text in programs.items():
    assert text.encode() == subprocess.check_output(
        ["git", "show", git + ":" + str((D / (name + ".py.txt")).relative_to(R))]
    )
(L / "owner-intent.json").write_text(
    json.dumps(
        {
            "source_git": git,
            "sources_sha256": {
                n: hashlib.sha256(t.encode()).hexdigest() for n, t in programs.items()
            },
            "maximum_new_GPU_jobs": 1,
            "wait_for": "actual original gradient GPU973601 COMPLETED0:0 and accepted real receipt",
            "automatic_retry": False,
        },
        indent=2,
    )
    + "\n"
)
s = importlib.util.spec_from_file_location(
    "transport", R / "initiative/research/paired-logit-head-gradient-20261018-v1/stdin_transport.py"
)
t = importlib.util.module_from_spec(s)
s.loader.exec_module(t)
original = (
    R
    / "initiative/research/teacher-pair-admitted48-fallback-20261019-v1/actual-gradient-terminal.json"
)
start = time.monotonic()
while not original.exists():
    assert time.monotonic() - start < 172800, (
        "Original scout observation deadline; not terminal/restart authority"
    )
    time.sleep(30)
observed = json.loads(original.read_text())
assert observed["GPU_job"] == "973601"
if "receipt" not in observed or observed["actual_Slurm_terminal"][1:3] != ["COMPLETED", "0:0"]:
    result = {
        "status": "original_gradient_probe_not_accepted_no_recovery_GPU_dispatched",
        "actual_original_scout": observed,
        "source_git": git,
        "automatic_retry": False,
    }
    (L / "result.json").write_text(json.dumps(result, indent=2) + "\n")
    print(json.dumps(result))
    raise SystemExit(0)
payload = {
    "source_git": git,
    "private_source_proof": proof,
    "gradient_receipt_sha256": observed["receipt_sha256"],
    "QA": {
        "full_suite_passed": 4942,
        "skipped": 8,
        "new_wrapper_cases_passed": 7,
        "recovery_cases_passed": 5,
    },
    "files": {"worker.py": programs["worker"], "driver.py": programs["driver"]},
}
args = [
    "srun",
    "--account=skattel",
    "--partition=normal",
    "--qos=normal",
    "--nodelist=evc48",
    "--ntasks=1",
    "--cpus-per-task=1",
    "--gres=none",
    "--mem=2G",
    "--time=00:05:00",
    "--job-name=rsna-teacher-pair-recovery-admission",
    "env",
    "CUDA_VISIBLE_DEVICES=",
    "PYTHONDONTWRITEBYTECODE=1",
    "PYTHONNOUSERSITE=1",
    "/tmp/rsna-environment-971427-pyk8rld8/environment/bin/python",
    "-B",
    "-c",
    t.COMMAND,
]
(L / "stage-intent.json").write_text(
    json.dumps(
        {
            "source_git": git,
            "original_gradient_GPU": "973601",
            "actual_gradient_receipt_sha256": observed["receipt_sha256"],
            "automatic_retry": False,
        },
        indent=2,
    )
    + "\n"
)
x = subprocess.run(
    [
        "ssh",
        "-S",
        "/tmp/rsna-research-20261001-stable.sock",
        "-o",
        "BatchMode=yes",
        "newton",
        shlex.join(args),
    ],
    input=t.envelope(programs["stage_once"], payload),
    capture_output=True,
    text=True,
    timeout=600,
)
(L / "stage-transport.json").write_text(
    json.dumps({"returncode": x.returncode, "stdout": x.stdout, "stderr": x.stderr}, indent=2)
    + "\n"
)
assert x.returncode == 0, "Reconcile original stage/intents read-only; do not retry"
submitted = json.loads(x.stdout)
assert (
    submitted["status"]
    == "actual_positive_gradient_source_one_H100_optimizer_recovery_submitted_once"
)
(D / "actual-submitted.json").write_text(json.dumps(submitted, indent=2) + "\n")
print(json.dumps(submitted), flush=True)
# This follower submits no full training and no competition candidate. Original
# recovery result remains private until a distinct CPU terminal observer reviews it.
(L / "result.json").write_text(json.dumps(submitted, indent=2) + "\n")
