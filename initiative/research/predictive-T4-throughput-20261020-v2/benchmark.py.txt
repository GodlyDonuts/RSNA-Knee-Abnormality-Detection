def atomic_json(path, value):
    """Persist complete finite JSON before publishing its new filename."""
    path = Path(path)
    temporary = path.with_name(path.name + ".tmp")
    encoded = json.dumps(value, sort_keys=True, indent=2, allow_nan=False) + "\n"
    with temporary.open("x") as output:
        output.write(encoded)
        output.flush()
        os.fsync(output.fileno())
    os.replace(temporary, path)


def stage(name, **metrics):
    """Only caller-supplied aggregate metrics; never IDs, source bodies or arrays."""
    with progress_lock:
        event = {
            "stage": name,
            "elapsed_seconds": time.monotonic() - start,
            "metrics": metrics,
        }
        json.dumps(event, allow_nan=False)  # Reject before contaminating retained progress.
        progress["events"].append(event)
        atomic_json(root / "progress.json", progress)
        atomic_json(Path("/kaggle/working/predictive-T4-progress-v2.json"), progress)
        print(
            json.dumps({"status": "predictive_T4_v2_progress", **event}, allow_nan=False),
            flush=True,
        )

class FixedEncoder(torch.nn.Module):
    def __init__(self, backbone):
        super().__init__()
        self.backbone = backbone
        self.register_buffer("mean", torch.tensor([0.485, 0.456, 0.406])[None, :, None, None])
        self.register_buffer("std", torch.tensor([0.229, 0.224, 0.225])[None, :, None, None])

    def forward(self, images):
        return self.backbone((images - self.mean) / self.std)

def nonzero_shift_precision_check(model, contexts):
    """Exercise evidence arithmetic separately from the zero-shift fresh head.

    This fixed synthetic perturbation is neither learned nor clinically useful.
    It is restored before returning. The unchanged fresh model is used for all
    timed measurements and prediction hashes.
    """
    assert not torch.count_nonzero(model.shift.weight) and not torch.count_nonzero(model.shift.bias)
    device = next(model.parameters()).device
    captures = []

    def capture(module, arguments, result):
        captures.append(result["evidence"].log_bayes_factor.detach().float().cpu())

    hook = model.register_forward_hook(capture)
    try:
        with torch.no_grad():
            model.shift.weight.copy_(
                (
                    (torch.arange(model.shift.weight.numel(), device=device) % 17 - 8).float()
                    * 1e-4
                ).reshape_as(model.shift.weight)
            )
            model.shift.bias.copy_(
                (
                    (torch.arange(model.shift.bias.numel(), device=device) % 13 - 6).float() * 1e-4
                ).reshape_as(model.shift.bias)
            )
        low = predict(model, contexts, precision="fp16")
        assert len(captures) == 2
        low_evidence = torch.cat(captures)
        captures.clear()
        reference = predict(model, contexts, precision="fp32")
        assert len(captures) == 2
        reference_evidence = torch.cat(captures)
        probability_gap = float((reference - low).abs().max())
        evidence_gap = float((reference_evidence - low_evidence).abs().max())
        maximum_evidence = float(reference_evidence.abs().max())
        assert torch.isfinite(low_evidence).all() and torch.isfinite(reference_evidence).all()
        assert maximum_evidence > 1e-6, "Precision stress must actually exercise nonzero evidence"
        assert probability_gap < 0.02
        return {
            "fixed_synthetic_shift_scale": 1e-4,
            "maximum_absolute_FP32_log_bayes_factor": maximum_evidence,
            "maximum_FP16_FP32_log_bayes_factor_gap": evidence_gap,
            "maximum_FP16_FP32_probability_gap": probability_gap,
            "trained_head_precision_acceptance": False,
        }
    finally:
        hook.remove()
        with torch.no_grad():
            model.shift.weight.zero_()
            model.shift.bias.zero_()


def run(device):
    torch.cuda.set_device(device)
    model = models[device]
    with torch.inference_mode():
        stage("device_warmup_started", device=device)
        warm_start = time.monotonic()
        warm = predict(model, cases[0], precision="fp16")
        torch.cuda.synchronize(device)
        stage("device_warmup_complete", device=device, seconds=time.monotonic() - warm_start)
        torch.cuda.reset_peak_memory_stats(device)
        began = time.monotonic()
        rows = []
        for case in cases:
            rows.append(predict(model, case, precision="fp16").cpu().numpy())
        torch.cuda.synchronize(device)
        elapsed = time.monotonic() - began
        throughput_peak_allocated = torch.cuda.max_memory_allocated(device)
        throughput_peak_reserved = torch.cuda.max_memory_reserved(device)
        measured = {
            "device": device,
            "GPU": torch.cuda.get_device_name(device),
            "studies": len(rows),
            "encoder_windows": sum(len(context["images"]) for case in cases for context in case),
            "seconds": elapsed,
            "throughput_FP16_peak_allocated_bytes": throughput_peak_allocated,
            "throughput_FP16_peak_reserved_bytes": throughput_peak_reserved,
            "finite_predictions": bool(np.isfinite(rows).all()),
            "prediction_sha256": hashlib.sha256(
                np.asarray(rows, dtype="<f4").tobytes()
            ).hexdigest(),
        }
        stage("device_FP16_timed_measurement_complete", **measured)
        precision_start = time.monotonic()
        reference = predict(model, cases[0], precision="fp32")
        max_gap = float((reference - warm).abs().max())
        stage(
            "device_zero_shift_precision_complete",
            device=device,
            seconds=time.monotonic() - precision_start,
            maximum_probability_gap=max_gap,
            context_prior_only=True,
        )
        assert max_gap < 0.02 and np.isfinite(rows).all()
        precision_start = time.monotonic()
        precision_stress = nonzero_shift_precision_check(model, cases[0])
        stage(
            "device_nonzero_shift_precision_complete",
            device=device,
            seconds=time.monotonic() - precision_start,
            **precision_stress,
        )
    return {
        **measured,
        "peak_allocated_bytes": torch.cuda.max_memory_allocated(device),
        "peak_reserved_bytes": torch.cuda.max_memory_reserved(device),
        "maximum_first_public_FP16_FP32_probability_gap": max_gap,
        "fresh_zero_shift_probability_comparison_checks_context_prior_only": True,
        "separate_nonzero_shift_precision_stress": precision_stress,
    }

def classify(receipt):
    complete = (
        receipt["completed_device_count"] == 2
        and not receipt["device_errors"]
        and receipt["host22GiB_device13GiB_memory_checks"]
        and receipt["submission_csv_absent"]
    )
    elapsed = receipt["elapsed_seconds"]
    return {
        "status": (
            "measured_incomplete_or_failed_checks"
            if not complete
            else "measured_slow"
            if elapsed >= 550
            else "measured_within_original_runtime_target"
        ),
        "original_runtime_target_seconds": 550,
        "session_cap_seconds": 1200,
        "measured_elapsed_seconds": elapsed,
        "session_headroom_seconds_at_measurement": 1200 - elapsed,
        "trained_model_acceptance_or_accuracy": False,
        "hidden_size_runtime_acceptance": False,
        "competition_entry": False,
    }
