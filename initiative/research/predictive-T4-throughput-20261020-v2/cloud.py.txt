def seconds(value):
    return float(value.removesuffix("s"))


def source_zero_observation(remote):
    """Fresh exact original v1 state; a dated prompt never retires its liability."""
    slug = "godlydonuts/rsna-zero-full-e3-a5c5b1d7f43c"
    kernel = remote.kernel("GetKernel", slug, 1)
    assert kernel["metadata"]["id"] == 139786921
    state = remote.kernel("GetKernelSessionStatus", slug, 1).get("status", "UNREPORTED")
    return {
        "observed_kernel_id": 139786921,
        "slug": slug,
        "version": 1,
        "state": state,
        "unfinished_source_zero_budget_seconds": 0 if state == "COMPLETE" else 600,
        "remaining_training_retired_only_on_actual_COMPLETE": state == "COMPLETE",
    }


def required_free_seconds(source_zero):
    return 1200 + 3000 + source_zero["unfinished_source_zero_budget_seconds"]


def capacity_observation(value):
    """Keep observed jobs from being recursively registered as this probe's jobs."""
    if isinstance(value, dict):
        return {
            ("observed_kernel_id" if key == "kernel_id" else key): capacity_observation(child)
            for key, child in value.items()
        }
    if isinstance(value, list):
        return [capacity_observation(child) for child in value]
    return value
