"""One normal allocation, two matched fresh fixed-final fits; no automatic retry."""

import hashlib
import json
import os
import shutil
import subprocess
import sys
import time
from pathlib import Path

R = Path("/tmp/rsna-teacher-pair-full-fitting45-20261020-v1")
meta = json.loads((R / "staged.json").read_text())
assert os.uname().nodename == "evc45" and os.environ.get("SLURM_JOB_GPUS")
assert hashlib.sha256(Path(__file__).read_bytes()).hexdigest() == meta["driver_sha256"]
assert hashlib.sha256((R / "worker.py").read_bytes()).hexdigest() == meta["worker_sha256"]
for name, h in meta["accepted_module_sha256"].items():
    assert (
        hashlib.sha256(
            (Path("/tmp/rsna-teacher-pair-gradient-evc45-20261019-v1/code") / name).read_bytes()
        ).hexdigest()
        == h
    )
assert not (R / "GPU-intent.json").exists()
with (R / "GPU-intent.json").open("x") as f:
    json.dump(
        {
            "GPU_job": os.environ["SLURM_JOB_ID"],
            "source_git": meta["source_git"],
            "objectives": ["pointwise", "teacher_rank"],
            "maximum_epochs_each": 3,
            "maximum_updates_each": 1458,
            "actual_GPU_recovery": "974940",
            "automatic_retry": False,
        },
        f,
    )
start = time.monotonic()
for objective in ("pointwise", "teacher_rank"):
    assert not (R / objective).exists()
    with (
        (R / (objective + ".stdout.private.log")).open("x") as out,
        (R / (objective + ".stderr.private.log")).open("x") as err,
    ):
        child = subprocess.Popen(
            [sys.executable, "-B", str(R / "worker.py"), objective], stdout=out, stderr=err
        )
        while child.poll() is None:
            paid = sum(
                p.stat().st_size
                for p in R.rglob("*")
                if p.is_file() and not p.is_symlink() and "inputs" not in p.relative_to(R).parts
            )
            # Both objectives' models, optimizer states and one write/verification
            # temporary share the already reserved24GiB future checkpoint cap.
            assert paid < 24 * 2**30
            assert shutil.disk_usage("/tmp").free > 20 * 2**30 + max(0, 24 * 2**30 - paid)
            assert time.monotonic() - start < 12 * 3600 - 120
            time.sleep(15)
        assert child.returncode == 0, "Preserve original failure; never retry or start next arm"
    row = json.loads((R / objective / "terminal.json").read_text())
    assert row["completed_updates"] == 1458 and row["objective"] == objective
(R / "terminal.json").write_text(
    json.dumps(
        {
            "status": "actual_both_matched_full_fits_complete",
            "GPU_job": os.environ["SLURM_JOB_ID"],
            "source_git": meta["source_git"],
            "objectives": ["pointwise", "teacher_rank"],
            "new_clinical_or_Kaggle_result": False,
            "automatic_retry": False,
        },
        indent=2,
    )
    + "\n"
)
