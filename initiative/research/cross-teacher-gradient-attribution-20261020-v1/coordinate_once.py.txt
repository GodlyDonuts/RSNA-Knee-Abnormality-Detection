"""One immutable fitting-only CPU diagnostic, no credential or remote mutation API."""

import hashlib
import importlib.util
import json
import shlex
import subprocess
from pathlib import Path

R = Path.cwd()
D = Path(__file__).resolve().parent
L = R / "artifacts/cross-teacher-gradient-attribution-20261020-v1"
L.mkdir(mode=0o700, exist_ok=True)
assert not (L / "intent.json").exists()
pin = subprocess.check_output(["git", "rev-parse", "HEAD"], text=True).strip()
module = (R / "rsna/competitive/supervision_gradient.py").read_text()
program = (D / "worker.py.txt").read_text()
for path in (
    D / "worker.py.txt",
    D / "coordinate_once.py.txt",
    R / "rsna/competitive/supervision_gradient.py",
):
    assert path.read_bytes() == subprocess.check_output(
        ["git", "show", pin + ":" + path.relative_to(R).as_posix()]
    )
with (L / "intent.json").open("x") as f:
    json.dump(
        {
            "source_git": pin,
            "maximum_CPU_jobs": 1,
            "GPU_model_optimizer_pixel_or_external_API_operations": 0,
            "private_output_cap": 2**20,
            "inside_existing24GiB_control_scope": True,
            "automatic_retry": False,
        },
        f,
        indent=2,
    )
s = importlib.util.spec_from_file_location(
    "transport", R / "initiative/research/paired-logit-head-gradient-20261018-v1/stdin_transport.py"
)
t = importlib.util.module_from_spec(s)
s.loader.exec_module(t)
args = [
    "srun",
    "--account=skattel",
    "--partition=normal",
    "--qos=normal",
    "--nodelist=evc48",
    "--ntasks=1",
    "--cpus-per-task=1",
    "--gres=none",
    "--mem=512M",
    "--time=00:03:00",
    "--job-name=rsna-cross-teacher-gradient-audit",
    "env",
    "CUDA_VISIBLE_DEVICES=",
    "PYTHONNOUSERSITE=1",
    "PYTHONDONTWRITEBYTECODE=1",
    "OMP_NUM_THREADS=1",
    "/tmp/rsna-environment-971427-pyk8rld8/environment/bin/python",
    "-B",
    "-c",
    t.COMMAND,
]
x = subprocess.run(
    [
        "ssh",
        "-S",
        "/tmp/rsna-research-20261001-stable.sock",
        "-o",
        "BatchMode=yes",
        "newton",
        shlex.join(args),
    ],
    input=t.envelope(
        program,
        {
            "source_git": pin,
            "module": module,
            "module_SHA": hashlib.sha256(module.encode()).hexdigest(),
        },
    ),
    capture_output=True,
    text=True,
    timeout=600,
)
(L / "transport.json").write_text(
    json.dumps({"returncode": x.returncode, "stdout": x.stdout, "stderr": x.stderr}, indent=2)
    + "\n"
)
assert x.returncode == 0, (
    "Inspect the original read-only source/data path; do not repeat automatically"
)
z = json.loads(x.stdout)
(D / "actual.json").write_text(json.dumps(z, indent=2) + "\n")
print(json.dumps(z), flush=True)
