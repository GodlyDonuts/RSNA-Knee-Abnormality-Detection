"""Physical relations for unchanged N3 windows, not clinical extent measurements.

Every token retains the three ORIGINAL source slices that generated it. Neighbor
windows overlap, so they are not independent corroborating evidence. Repeat
edges below require disjoint source support. Cross-series correspondence and
lesion area/volume cannot be inferred from this cache and are not invented.
"""

from __future__ import annotations

from dataclasses import dataclass

import numpy as np


@dataclass(frozen=True)
class WindowRelations:
    supports: np.ndarray  # N,3 original indices, private owning-series namespace
    centers_mm: np.ndarray  # N source-center projections in this series ONLY
    continuity_edges: np.ndarray  # E,2 selected-order edges, may share source slices
    continuity_gaps_mm: np.ndarray
    repeat_edges: np.ndarray  # E,2 nearest forward disjoint-support windows
    repeat_gaps_mm: np.ndarray
    geometry_known: bool


def within_series_relations(metadata: dict, selected: np.ndarray) -> WindowRelations:
    """Bind fixed selected-source triples without changing any image sampling.

    Physical edges cannot cross a native gap greater than 1.5 times its series
    median. This is an engineering rule, not a universal MRI/diagnostic cutoff.
    Selected-order neighbors need not be adjacent original acquired slices.
    No spacing, registration, slab thickness or disease label is inferred.
    """
    count = metadata.get("n_slices")
    sops = metadata.get("sop_uids")
    if (
        type(count) is not int
        or count < 1
        or not isinstance(sops, list)
        or len(sops) != count
        or any(not isinstance(x, str) or not x for x in sops)
        or len(set(sops)) != count
        or not isinstance(selected, np.ndarray)
        or selected.ndim != 1
        or not np.issubdtype(selected.dtype, np.integer)
        or not len(selected)
        or (selected < 0).any()
        or (selected >= count).any()
        or (np.diff(selected) <= 0).any()
    ):
        raise ValueError(
            "Unique original SOP membership and ordered distinct source indices required"
        )
    supports = np.array(
        [selected[j : j + 3] for j in range(max(0, len(selected) - 2))], dtype=np.int64
    ).reshape(-1, 3)
    empty = np.empty((0, 2), dtype=np.int64)
    gaps = np.empty(0, dtype=np.float64)
    if metadata.get("ordering") != "patient_geometry":
        return WindowRelations(supports, np.zeros(len(supports)), empty, gaps, empty, gaps, False)
    orientation = np.asarray(metadata.get("orientation", []), dtype=np.float64)
    positions = np.asarray(metadata.get("positions", []), dtype=np.float64)
    if (
        orientation.shape != (6,)
        or positions.shape != (count, 3)
        or not np.isfinite(orientation).all()
        or not np.isfinite(positions).all()
    ):
        raise ValueError(
            "Claimed patient geometry requires exact finite source orientation/positions"
        )
    row, col = orientation[:3], orientation[3:]
    if not np.allclose(
        [np.linalg.norm(row), np.linalg.norm(col), row @ col], [1, 1, 0], atol=1e-3, rtol=0
    ):
        raise ValueError("Orthonormal patient coordinate basis required")
    normal = np.cross(row, col)
    normal /= np.linalg.norm(normal)
    projected = positions @ normal
    native_gaps = np.diff(projected)
    if (native_gaps <= 0).any():
        raise ValueError("Claimed physical source order must be strictly increasing")
    centers = projected[supports[:, 1]]
    # A large acquisition gap cannot become continuous by sampling across it.
    native_ok = native_gaps <= 1.5 * np.median(native_gaps) if count > 1 else np.empty(0, bool)

    def edges(offset: int) -> tuple[np.ndarray, np.ndarray]:
        pairs = []
        for i in range(max(0, len(supports) - offset)):
            j = i + offset
            lo, hi = int(supports[i, 1]), int(supports[j, 1])
            if native_ok[lo:hi].all():
                pairs.append((i, j))
        pairs_array = np.asarray(pairs, dtype=np.int64).reshape(-1, 2)
        distances = centers[pairs_array[:, 1]] - centers[pairs_array[:, 0]]
        return pairs_array, distances

    continuity, continuity_gaps = edges(1)
    repeat, repeat_gaps = edges(3)
    if any(set(supports[i]) & set(supports[j]) for i, j in repeat):
        raise ValueError("Corroborating windows must not share an original source slice")
    return WindowRelations(
        supports, centers, continuity, continuity_gaps, repeat, repeat_gaps, True
    )
