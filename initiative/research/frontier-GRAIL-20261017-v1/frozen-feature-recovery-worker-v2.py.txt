    assert hashlib.sha256((Path(admitted['original_source']) / name).read_bytes()).hexdigest() == digest
torch.ones(1, device='cuda')

torch.set_num_threads(2)

assert hashlib.sha256((R / 'geometry.py').read_bytes()).hexdigest() == admitted['geometry_sha256']

assert set(membership) == {'fitting', 'outer'}
assert len(membership['fitting']) == 2247 and len(membership['outer']) == 1156

assert shutil.disk_usage('/tmp').free > 612022943803 + max(0, 3 * 2**30 - paid())
with (R / (tag + '-intent.json')).open('x') as f:
    json.dump({'GPU_job': os.environ['SLURM_JOB_ID'], 'source_git': admitted['source_git'], 'phase': phase, 'automatic_retry': False, 'optimizer_or_clinical_target_execution': False}, f)
features_root = R / 'features'
features_root.mkdir(mode=0o700, exist_ok=True)
captured = []
spatial = []

def capture_spatial(module, args):
    assert len(args) == 1
    value = args[0]
    assert value.ndim == 4 and value.shape[1] == 1024 and value.dtype == torch.float32
    assert torch.isfinite(value).all()
    spatial.append(value.detach().cpu())
    return None

spatial_hook = model.encoder.head.global_pool.register_forward_pre_hook(capture_spatial)
hook = model.encoder.register_forward_hook(lambda module, args, output: captured.append(output.detach().float().cpu()))
def graphs_for(uid, dataset, windows):
    graph_rows = []
    for slot, entry in enumerate(select_series(dataset.series[uid], 6, False)):
        sid = entry['SeriesInstanceUID']
        meta = dataset._metadata[(uid, sid)][1]
        selected = np.linspace(0, meta['n_slices'] - 1, min(16, meta['n_slices'])).round().astype(np.int64)
        graph = geometry.within_series_relations(meta, selected)
        assert [j for s, j in windows if s == slot] == list(range(len(graph.supports)))
        graph_rows.append({'slot': slot, 'series_uid': sid, 'source_metadata_sha256': dataset._metadata[(uid, sid)][0], 'selected_source_indices': selected.tolist(), 'supports': graph.supports.tolist(), 'centers_mm': graph.centers_mm.tolist(), 'continuity_edges': graph.continuity_edges.tolist(), 'continuity_gaps_mm': graph.continuity_gaps_mm.tolist(), 'repeat_edges': graph.repeat_edges.tolist(), 'repeat_gaps_mm': graph.repeat_gaps_mm.tolist(), 'geometry_known': graph.geometry_known, 'native_original_shape': meta['original_shape'], 'native_resized_shape': meta['resized_shape'], 'native_cache_side': meta['side'], 'pixel_spacing': meta['pixel_spacing'], 'image_orientation_patient': meta['orientation'], 'selected_image_positions_patient': [meta['positions'][int(j)] for j in selected], 'full_frame_letterbox_align_corners_false': True, 'input_augmentation_or_mirror': False})
    return graph_rows

try:
    with torch.inference_mode():
        for role, dataset in roles.items():
            count = 16
            for index in range(count):
                uid = membership[role][index]
                filename = hashlib.sha256(uid.encode()).hexdigest() + '.npz'
                destination = features_root / filename
                if destination.exists():
                    proof=original_files[filename]
                    stat=destination.stat()
                    assert stat.st_size==proof['bytes'] and stat.st_mtime_ns==proof['mtime_ns'] and stat.st_ino==proof['inode']
                    with np.load(destination,allow_pickle=False) as arrays:
                        features=torch.from_numpy(arrays['features'].copy())
                        maps=torch.from_numpy(arrays['spatial_maps'].copy())
                        logits=torch.from_numpy(arrays['base_logits'].copy())
                        windows=[tuple(map(int,w)) for w in arrays['windows']]
                    assert len(windows)==len(features)==len(maps) and features.shape[1]==1024
                    assert features.dtype==maps.dtype==logits.dtype==torch.float32
                    assert torch.isfinite(features).all() and torch.isfinite(maps).all() and torch.isfinite(logits).all()
                    repooled=model.pool(features.cuda()).float().cpu()
                    error=float(torch.max(torch.abs(logits-repooled)));assert error<=1e-6
                    maximum=max(maximum,error);map_shapes.add(tuple(maps.shape[1:]));spatial_uncompressed_bytes+=maps.numel()*maps.element_size()
                    row={'study_uid':uid,'role':role,'file':filename,'bytes':proof['bytes'],'sha256':proof['sha256'],'windows':len(windows),'actual_spatial_map_shape':list(maps.shape),'spatial_map_dtype':str(maps.dtype),'source_graphs':graphs_for(uid,dataset,windows),'actual_repool_logit_error':error,'original_949626_completed_forward_reused':True}
                    records.append(row)
                    (R/'records-private').mkdir(mode=0o700,exist_ok=True)
                    record=R/'records-private'/(filename+'.json');assert not record.exists();record.write_text(json.dumps(row)+'\n')
                    (R/(tag+'-progress.json')).write_text(json.dumps({'GPU_job':os.environ['SLURM_JOB_ID'],'completed_studies':len(records),'total_studies':32,'new_encoder_forwards':sum(not x.get('original_949626_completed_forward_reused',False) for x in records),'elapsed_seconds':time.monotonic()-start})+'\n')
                    continue
                assert not destination.exists()
                sample = dataset[index]
                assert not sample['label_mask'].any()
                windows = candidate_windows(sample['series_mask'], sample['slice_mask'])
                images = window_tensor(sample['images'], windows)
                captured.clear()
                spatial.clear()
                logits = model(images.cuda(), chunk_size=12).float().cpu()
                features = torch.cat(captured)
                maps = torch.cat(spatial)
                assert maps.shape[:2] == (len(windows), 1024)
                map_shapes.add(tuple(maps.shape[1:]))
                spatial_uncompressed_bytes += maps.numel() * maps.element_size()
                assert features.shape == (len(windows), 1024) and torch.isfinite(features).all()
                repooled = model.pool(features.cuda()).float().cpu()
                error = float(torch.max(torch.abs(logits - repooled)))
                maximum = max(maximum, error)
                assert error <= 1e-6
                graph_rows = graphs_for(uid, dataset, windows)
                buffer = io.BytesIO()
                np.savez_compressed(buffer, features=features.numpy(), base_logits=logits.numpy(), windows=np.asarray(windows, dtype=np.int64), spatial_maps=maps.numpy())
                raw = buffer.getvalue()
                assert len(raw) < 64 * 2**20
                assert sum(p.stat().st_size for p in features_root.iterdir()) + len(raw) < 2 * 2**30
                assert paid() + len(raw) + 512 * 2**20 < 3 * 2**30
                assert shutil.disk_usage('/tmp').free > 612022943803 + max(0, 3 * 2**30 - paid())
                with destination.open('xb') as f:
                    f.write(raw)
                row = {'study_uid': uid, 'role': role, 'file': filename, 'bytes': len(raw), 'sha256': hashlib.sha256(raw).hexdigest(), 'windows': len(windows), 'actual_spatial_map_shape': list(maps.shape), 'spatial_map_dtype': str(maps.dtype), 'source_graphs': graph_rows, 'actual_repool_logit_error': error}
                records.append(row)
