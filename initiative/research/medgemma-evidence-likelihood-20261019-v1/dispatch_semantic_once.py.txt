"""One actual strict model/full MRI CPU admission; original owners untouched."""

import hashlib
import importlib.util
import json
import shlex
import subprocess
from importlib.machinery import SourceFileLoader
from pathlib import Path

R = Path.cwd()
D = Path(__file__).resolve().parent
L = R / "artifacts/medgemma-evidence-likelihood-20261019-v1"
L.mkdir(mode=0o700, exist_ok=True)
assert not (L / "GPU-stage-intent-semantic-v1.json").exists()
pin = subprocess.check_output(["git", "rev-parse", "HEAD"], text=True).strip()


def module(name):
    path = D / (name + ".py.txt")
    assert (
        subprocess.check_output(["git", "show", pin + ":" + path.relative_to(R).as_posix()])
        == path.read_bytes()
    )
    spec = importlib.util.spec_from_loader(name, SourceFileLoader(name, str(path)))
    m = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(m)
    return m


stage = module("semantic_stage")
protocol = module("semantic_protocol")
original_stage = (D / "stage_pilot.py.txt").read_text()
program, program_SHA = stage.build(original_stage)
word_gate = json.loads((D / "actual-word-token-gate.json").read_text())
original_result = json.loads((D / "actual-terminal-verified.json").read_text())
assert (
    original_result["GPU_job"] == "972070"
    and original_result["predeclared_engineering_disposition"]
    == "do_not_use_as_teacher_without_protocol_rework"
)
actual = json.loads((D / "actual-input-gate.json").read_text())
assert (
    actual["status"]
    == "actual_node48_original_native_packet_processor_and_fixed_answer_CPU_admitted"
)
controls = {
    "pilot-semantic.py": protocol.build((D / "pilot.py.txt").read_text(), word_gate),
    "likelihood.py": (D / "likelihood.py.txt").read_text(),
}
for name in (
    "pilot.py.txt",
    "likelihood.py.txt",
    "stage_pilot.py.txt",
    "actual-word-token-gate.json",
):
    assert (
        subprocess.check_output(["git", "show", pin + ":" + (D / name).relative_to(R).as_posix()])
        == (D / name).read_bytes()
    )
hashes = {name: hashlib.sha256(body.encode()).hexdigest() for name, body in controls.items()}
with (L / "GPU-stage-intent-semantic-v1.json").open("x") as f:
    json.dump(
        {
            "source_git": pin,
            "program_SHA": hashlib.sha256(program.encode()).hexdigest(),
            "actual_auxiliary_CPU": actual["CPU_job"],
            "maximum_new_GPU_jobs": 1,
            "original_owners_unchanged": True,
            "automatic_retry": False,
        },
        f,
        indent=2,
    )
spec = importlib.util.spec_from_file_location(
    "transport", R / "initiative/research/paired-logit-head-gradient-20261018-v1/stdin_transport.py"
)
t = importlib.util.module_from_spec(spec)
spec.loader.exec_module(t)
a = [
    "srun",
    "--account=skattel",
    "--partition=preemptable",
    "--qos=preemptable",
    "--nodelist=evc48",
    "--ntasks=1",
    "--cpus-per-task=1",
    "--gres=none",
    "--mem=4G",
    "--time=00:05:00",
    "--job-name=rsna-medgemma-semantic-GPU-admission",
    "env",
    "CUDA_VISIBLE_DEVICES=",
    "PYTHONNOUSERSITE=1",
    "PYTHONDONTWRITEBYTECODE=1",
    "OMP_NUM_THREADS=1",
    "/tmp/rsna-environment-971427-pyk8rld8/environment/bin/python",
    "-B",
    "-c",
    t.COMMAND,
]
x = subprocess.run(
    [
        "ssh",
        "-S",
        "/tmp/rsna-research-20261001-stable.sock",
        "-o",
        "BatchMode=yes",
        "newton",
        shlex.join(a),
    ],
    input=t.envelope(
        program,
        {
            "source_git": pin,
            "inputs": actual,
            "controls": controls,
            "hashes": hashes,
            "word_token_gate": word_gate,
        },
    ),
    capture_output=True,
    text=True,
    timeout=1800,
)
(L / "GPU-stage-transport-semantic-v1.json").write_text(
    json.dumps({"returncode": x.returncode, "stdout": x.stdout, "stderr": x.stderr}, indent=2)
    + "\n"
)
assert x.returncode == 0, (
    "Original CPU model/input mutation requires read-only reconciliation, never retry"
)
actual = json.loads(x.stdout)
(D / "actual-semantic-GPU-submitted.json").write_text(json.dumps(actual, indent=2) + "\n")
print(x.stdout, flush=True)
