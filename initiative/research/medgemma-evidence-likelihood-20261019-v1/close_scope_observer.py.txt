"""Read actual terminal scopes/retained bytes; no deletion or clinical-score export."""

import json
import os
import shutil
import subprocess
from pathlib import Path

assert os.uname().nodename == "evc48" and not os.environ.get("SLURM_JOB_GPUS")
for job in ("972070", "972145"):
    t = subprocess.check_output(
        ["sacct", "-X", "-n", "-P", "-j", job, "--format=JobIDRaw,State,ExitCode"], text=True
    )
    assert job + "|COMPLETED|0:0" in t
E = Path("/tmp/rsna-medgemma-evidence-likelihood-20261019-v1")
seen = set()
paid = 0
files = 0
for p in E.rglob("*"):
    assert not p.is_symlink()
    if not p.is_file():
        continue
    st = p.stat()
    key = (st.st_dev, st.st_ino)
    if key not in seen:
        seen.add(key)
        paid += st.st_size
        files += 1
assert paid < 24 * 2**30
H = Path("/home/sa305415/rsna-knee/kaggle")
shared = []
for stem in ("medgemma-likelihood-972070", "medgemma-semantic-972145"):
    for suffix in ("stdout", "stderr"):
        p = H / (stem + "." + suffix)
        assert p.is_file()
        shared.append(p)
shared_bytes = sum(p.stat().st_size for p in shared)
assert len(shared) == 4 and shared_bytes < 4 * 2**20
print(
    json.dumps(
        {
            "status": "actual_two_rejected_likelihood_protocols_terminal_retained_storage_scope_verified",
            "CPU_job": os.environ["SLURM_JOB_ID"],
            "GPU_jobs": ["972070", "972145"],
            "original_assets_and_private_outputs_retained": True,
            "retained_node_unique_bytes": paid,
            "retained_node_unique_files": files,
            "original_node_scope_total_cap_bytes": 24 * 2**30,
            "unused_node_future_cap_after_terminal": 24 * 2**30 - paid,
            "actual_known_shared_log_files": 4,
            "actual_known_shared_log_bytes": shared_bytes,
            "original_shared_total_file_cap": 32,
            "unused_shared_future_files_after_terminal": 28,
            "remaining_likelihood_GPU_or_annotation_or_training_obligations": 0,
            "new_copy_or_download_or_delete_or_GPU_mutations": 0,
            "node_free_bytes_with_all_originals_retained": shutil.disk_usage("/tmp").free,
        }
    ),
    flush=True,
)
