import shutil

    inputs == v["inputs"]
    and inputs["status"]
    == "actual_node48_original_native_packet_processor_and_fixed_answer_CPU_admitted"
)
for job in (inputs["CPU_job"], inputs["asset_CPU"]):
    t = subprocess.check_output(
        ["sacct", "-X", "-n", "-P", "-j", job, "--format=JobIDRaw,State,ExitCode"], text=True
    )
    assert job + "|COMPLETED|0:0" in t
node = subprocess.check_output(["scontrol", "show", "node", "evc48", "-o"], text=True)

teacher = Path("/home/sa305415/rsna-knee/kaggle/S6-resident-owner-20261019-v1")
retained = [p for p in teacher.rglob("*") if p.is_file()]
paid_files = len(retained)
paid_bytes = sum(p.stat().st_size for p in retained)
assert paid_files < 256 and paid_bytes < 4 * 2**20
q = subprocess.check_output(["lfs", "quota", "-u", "sa305415", "/home/sa305415"], text=True)
a = next(x.split() for x in q.splitlines() if x.strip().startswith("/home/sa305415"))
assert int(a[6]) - int(a[5]) > 5000 + 128 + max(0, 256 - paid_files) + 128 + 32
assert (int(a[2]) - int(a[1])) * 1024 > 20 * 2**30 + 32 * 2**20 + (16 + 12 + 4) * 2**20 + max(
    0, 4 * 2**20 - paid_bytes
)
assert shutil.disk_usage("/tmp").free > (20 + 87 + 24) * 2**30 + 32 * 2**20
assert set(v["controls"]) == {"pilot.py", "likelihood.py"}
assert sum(len(s.encode()) for s in v["controls"].values()) < 256 * 2**10
assert not (E / "pilot-v1").exists()

    compile(source, name, "exec")
    with (E / name).open("x") as f:
        f.write(source)

    "entry_sha256": v["hashes"]["pilot.py"],
    "likelihood_sha256": v["hashes"]["likelihood.py"],
    "input_receipt_sha256": hashlib.sha256(
        (E / "inputs-verified-v1.json").read_bytes()
    ).hexdigest(),

    "native_slices": 195,
