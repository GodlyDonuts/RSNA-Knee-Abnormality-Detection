"""One canonical metadata phase for the existing R6 public run and eventual entry."""

import hashlib
import importlib.util
import json
import shlex
import subprocess
import sys
import time
from pathlib import Path

R = Path.cwd()
D = Path(__file__).resolve().parent
L = R / "artifacts/R6-postscore-member-execution-20261019-v1"
phase = sys.argv[1]
assert phase in ("status", "submit", "outcome")
git = subprocess.check_output(["git", "rev-parse", "HEAD"], text=True).strip()
p = D / "advance.py.txt"
program = p.read_text()
assert program == subprocess.check_output(
    ["git", "show", git + ":" + p.relative_to(R).as_posix()], text=True
)
if phase == "submit":
    with (L / "submit.intent.json").open("x") as f:
        json.dump(
            {
                "phase": phase,
                "source_git": git,
                "program_sha256": hashlib.sha256(program.encode()).hexdigest(),
                "maximum_code_submissions": 1,
                "notebook_version": 2,
                "public_CSV_upload": False,
                "automatic_retry": False,
            },
            f,
            indent=2,
        )
env = R / ".env"
assert env.stat().st_mode & 0o077 == 0
values = {
    k.strip(): x.strip().strip("\"'")
    for line in env.read_text().splitlines()
    for k, sep, x in [line.partition("=")]
    if sep
}
spec = importlib.util.spec_from_file_location(
    "bounded", R / "initiative/research/paired-logit-head-gradient-20261018-v1/stdin_transport.py"
)
t = importlib.util.module_from_spec(spec)
spec.loader.exec_module(t)
a = [
    "srun",
    "--account=skattel",
    "--partition=preemptable",
    "--qos=preemptable",
    "--nodelist=evc33",
    "--ntasks=1",
    "--cpus-per-task=1",
    "--gres=none",
    "--mem=2G",
    "--time=00:06:00",
    "--job-name=rsna-R6-paired-" + phase,
    "env",
    "CUDA_VISIBLE_DEVICES=",
    "PYTHONDONTWRITEBYTECODE=1",
    "PYTHONNOUSERSITE=1",
    "/tmp/rsna-environment-907632-7sr9a7i6/environment/bin/python",
    "-B",
    "-c",
    t.COMMAND,
]
columns = [
    "StudyInstanceUID",
    "ACL",
    "MCL",
    "Medial Meniscus",
    "Lateral Meniscus",
    "Medial OA",
    "Lateral OA",
    "PF OA",
    "Effusion",
    "Synovitis",
    "Baker's",
    "Contusion",
    "Fracture",
]
x = subprocess.run(
    [
        "ssh",
        "-S",
        "/tmp/rsna-research-20261001-stable.sock",
        "-o",
        "BatchMode=yes",
        "newton",
        shlex.join(a),
    ],
    input=t.envelope(
        program,
        {
            "phase": phase,
            "token": values["KAGGLE_API_TOKEN"],
            "columns": columns,
            "csv_source": (R / "scripts/kaggle/output_csv.py").read_text(),
        },
    ),
    capture_output=True,
    text=True,
    timeout=480,
)
(L / (phase + "-" + str(time.time_ns()) + ".json")).write_text(
    json.dumps({"returncode": x.returncode, "stdout": x.stdout, "stderr": x.stderr}, indent=2)
)
assert x.returncode == 0, (
    "Observe the same original handle/ledger; never repeat an uncertain mutation: "
    + x.stderr[-1200:]
)
actual = json.loads(x.stdout)
name = "actual-live.json"
if actual["status"] == "actual_R6_paired_public_MRI_and_exact_member_CSV_accepted":
    name = "actual-public-accepted.json"
elif actual["status"] == "one_exact_R6_paired_code_submission_accepted":
    name = "actual-submitted.json"
elif actual["status"] == "actual_R6_paired_hidden_outcome" and actual["outcome"] in (
    "scored",
    "failed",
):
    name = "actual-hidden-terminal.json"
(D / name).write_text(json.dumps(actual, indent=2) + "\n")
print(json.dumps(actual), flush=True)
