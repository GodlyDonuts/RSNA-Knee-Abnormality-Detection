"""Compile the completed qualified paired member into R6; no external mutation."""

import hashlib
import importlib.util
import json
import shlex
import subprocess
from pathlib import Path

R = Path.cwd()
D = Path(__file__).resolve().parent
L = R / "artifacts/R6-postscore-member-execution-20261019-v1"
L.mkdir(mode=0o700, parents=True, exist_ok=True)
assert not (L / "intent.json").exists()
original = (
    R
    / "initiative/experiments/competitive-v1/R6-paired-final-preparation-20261018-v1/program.py.txt"
)
text = original.read_text()
policy = (
    R / "initiative/research/reconnected-teacher-scoring-20261019-v1/resource_policy.py.txt"
).read_text()
git = subprocess.check_output(["git", "rev-parse", "HEAD"], text=True).strip()
program = """import hashlib,json,os,sys
from pathlib import Path
request=json.load(sys.stdin)
assert os.uname().nodename=='evc33' and not os.environ.get('SLURM_JOB_GPUS')
p=Path('/tmp/rsna-paired-full-fitting-20261018-v1/R6-final-preparation-v1/program.py.txt')
s=p.read_text();assert hashlib.sha256(s.encode()).hexdigest()==request['original_program_sha256']
assert not (p.parent/'prepare-intent.json').exists() and not (p.parent/'prepared.json').exists()
r={};exec(compile(request['policy'],'existing_current_CPU_RAM_policy','exec'),r);r['current_plan']()
old="assert shutil.disk_usage('/tmp').free>floor and paid(P)<32*2**20"
assert s.count(old)==1
s=s.replace(old,"assert current_plan()['new_bulk_disk_model_or_pixel_writes_allowed'] is False and paid(P)<32*2**20")
assert s.count('v=json.load(sys.stdin)')==1;s=s.replace('v=json.load(sys.stdin)','v=control_request')
control={'phase':'prepare','source_git':request['source_git'],'program_sha256':request['original_program_sha256']}
n={'current_plan':r['current_plan'],'control_request':control,'__name__':'__main__'}
exec(compile(s,'original_R6_candidate_current_resource_scope_only','exec'),n)
"""
(L / "intent.json").write_text(
    json.dumps(
        {
            "source_git": git,
            "original_program_sha256": hashlib.sha256(text.encode()).hexdigest(),
            "policy_sha256": hashlib.sha256(policy.encode()).hexdigest(),
            "CPU_only": True,
            "GPU_or_entry_or_Dataset_mutations": 0,
            "automatic_retry": False,
        },
        indent=2,
    )
)
spec = importlib.util.spec_from_file_location(
    "bounded", R / "initiative/research/paired-logit-head-gradient-20261018-v1/stdin_transport.py"
)
t = importlib.util.module_from_spec(spec)
spec.loader.exec_module(t)
a = [
    "srun",
    "--account=skattel",
    "--partition=preemptable",
    "--qos=preemptable",
    "--nodelist=evc33",
    "--ntasks=1",
    "--cpus-per-task=1",
    "--gres=none",
    "--mem=2G",
    "--time=00:05:00",
    "env",
    "CUDA_VISIBLE_DEVICES=",
    "PYTHONDONTWRITEBYTECODE=1",
    "PYTHONNOUSERSITE=1",
    "OMP_NUM_THREADS=1",
    "OPENBLAS_NUM_THREADS=1",
    "MKL_NUM_THREADS=1",
    "/tmp/rsna-environment-907632-7sr9a7i6/environment/bin/python",
    "-B",
    "-c",
    t.COMMAND,
]
x = subprocess.run(
    [
        "ssh",
        "-S",
        "/tmp/rsna-research-20261001-stable.sock",
        "-o",
        "BatchMode=yes",
        "newton",
        shlex.join(a),
    ],
    input=t.envelope(
        program,
        {
            "source_git": git,
            "original_program_sha256": hashlib.sha256(text.encode()).hexdigest(),
            "policy": policy,
        },
    ),
    capture_output=True,
    text=True,
    timeout=450,
)
(L / "transport.json").write_text(
    json.dumps({"returncode": x.returncode, "stdout": x.stdout, "stderr": x.stderr}, indent=2)
)
assert x.returncode == 0, (
    "Exact source preparation stopped; reconcile original without retry: " + x.stderr[-1000:]
)
actual = json.loads(x.stdout)
assert actual["status"] == "actual_final_qualified_paired_R6_member_candidate_CPU_compiled_only"
(D / "actual-prepared.json").write_text(json.dumps(actual, indent=2) + "\n")
print(json.dumps(actual), flush=True)
