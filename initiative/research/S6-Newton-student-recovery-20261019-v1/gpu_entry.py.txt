assert (
    hashlib.sha256((G / "recovery.H100.private.py").read_bytes()).hexdigest()

    == meta["runtime_input_receipt_sha256"]

env.update(
    HF_HUB_OFFLINE="1",

    PYTHONDONTWRITEBYTECODE="1",

child = None
started = time.monotonic()


def paid(root):
    size = 0
    for p in root.rglob("*"):
        try:
            if p.is_file() and not p.is_symlink():
                size += p.stat().st_size
        except FileNotFoundError:
            continue
    return size


try:
    with (G / "stdout.private.log").open("x") as out, (G / "stderr.private.log").open("x") as err:
        child = subprocess.Popen(
            [sys.executable, "-B", str(G / "recovery.H100.private.py")],
            env=env,
            stdout=out,
            stderr=err,
            start_new_session=True,
        )
        while child.poll() is None:
            assert paid(D / "work") < 12 * 2**30 and paid(D) < 12 * 2**30 + 512 * 2**20 + 16 * 2**20
            assert time.monotonic() - started < 3500
            time.sleep(2)
        assert child.returncode == 0, (
            "Original recovery failed; inspect its private logs, never repeat"
        )
    result = json.loads(
        (D / "work/licensed-importance-real-MRI-recovery-accepted.json").read_text()

    assert (
        result["status"]
        == "actual_complete_S6_target_change_full_encoder_oneH100_MRI_recovery_accepted"

    actual = {
        "status": "actual_complete_teacher_oneH100_recovery_accepted",

        "production_dispatched": False,

except BaseException as e:
    if child is not None and child.poll() is None:
        os.killpg(child.pid, signal.SIGTERM)
        try:
            child.wait(timeout=30)
        except subprocess.TimeoutExpired:
            os.killpg(child.pid, signal.SIGKILL)
            child.wait(timeout=10)
    (G / "terminal-failed.json").write_text(
        json.dumps(
            {
                "status": "original_oneH100_recovery_stopped_no_retry",
                "GPU_job": os.environ["SLURM_JOB_ID"],
                "type": type(e).__name__,
                "reason": str(e),
                "private_logs_and_checkpoints_preserved": True,
                "automatic_retry": False,
            },
            indent=2,
        )
        + "\n"
    )
    raise
