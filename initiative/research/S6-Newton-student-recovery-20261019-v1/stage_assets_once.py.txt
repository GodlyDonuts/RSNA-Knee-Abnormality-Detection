"""One CPU admission of exact retained inputs and the public parent weight."""

import hashlib
import importlib.util
import json
import shlex
import subprocess
from pathlib import Path

R = Path.cwd()
D = Path(__file__).resolve().parent
L = R / "artifacts/S6-Newton-student-recovery-20261019-v1"
L.mkdir(mode=0o700, exist_ok=True)
assert not (L / "asset-stage-intent.json").exists()
pin = subprocess.check_output(["git", "rev-parse", "HEAD"], text=True).strip()
for name in ("stage_assets.py.txt", "stage_assets_once.py.txt", "compiler.py.txt"):
    assert (
        subprocess.check_output(
            ["git", "show", pin + ":" + (D / name).relative_to(R).as_posix()], text=True
        )
        == (D / name).read_text()
    )
program = (D / "stage_assets.py.txt").read_text()
values = {
    key.strip(): value.strip().strip("\"'")
    for line in (R / ".env").read_text().splitlines()
    for key, sep, value in [line.partition("=")]
    if sep and not line.lstrip().startswith("#")
}
assert (R / ".env").stat().st_mode & 0o077 == 0
spec = importlib.util.spec_from_file_location(
    "transport", R / "initiative/research/paired-logit-head-gradient-20261018-v1/stdin_transport.py"
)
t = importlib.util.module_from_spec(spec)
spec.loader.exec_module(t)
args = [
    "srun",
    "--account=skattel",
    "--partition=preemptable",
    "--qos=preemptable",
    "--nodelist=evc33",
    "--ntasks=1",
    "--cpus-per-task=1",
    "--gres=none",
    "--mem=2G",
    "--time=00:12:00",
    "--job-name=rsna-S6-Newton-recovery-assets",
    "env",
    "CUDA_VISIBLE_DEVICES=",
    "PYTHONDONTWRITEBYTECODE=1",
    "PYTHONNOUSERSITE=1",
    "OMP_NUM_THREADS=1",
    "/tmp/rsna-environment-907632-7sr9a7i6/environment/bin/python",
    "-B",
    "-c",
    t.COMMAND,
]
with (L / "asset-stage-intent.json").open("x") as f:
    json.dump(
        {
            "source_git": pin,
            "program_sha256": hashlib.sha256(program.encode()).hexdigest(),
            "private_future_recovery_cap_bytes": 12 * 2**30 + 512 * 2**20 + 16 * 2**20,
            "new_CPU_assets_only": True,
            "GPU_jobs_or_competition_entries": 0,
            "automatic_retry": False,
        },
        f,
        indent=2,
    )
x = subprocess.run(
    [
        "ssh",
        "-S",
        "/tmp/rsna-research-20261001-stable.sock",
        "-o",
        "BatchMode=yes",
        "newton",
        shlex.join(args),
    ],
    input=t.envelope(
        program,
        {
            "source_git": pin,
            "token": values["KAGGLE_API_TOKEN"],
            "own_sources": {"compiler.py": (D / "compiler.py.txt").read_text()},
        },
    ),
    capture_output=True,
    text=True,
    timeout=7200,
)
(L / "asset-stage-transport.json").write_text(
    json.dumps({"returncode": x.returncode, "stdout": x.stdout, "stderr": x.stderr}, indent=2)
    + "\n"
)
assert x.returncode == 0, (
    "Original asset intent/CPU handle must be reconciled read-only; never retry"
)
actual = json.loads(x.stdout)
assert actual["status"] == "actual_Newton_oneH100_recovery_assets_and_private_funding_CPU_admitted"
(D / "actual-assets-admitted.json").write_text(json.dumps(actual, indent=2) + "\n")
print(json.dumps(actual), flush=True)
