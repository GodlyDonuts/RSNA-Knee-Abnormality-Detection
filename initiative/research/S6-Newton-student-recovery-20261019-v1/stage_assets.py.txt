def safe_request(function, *args, **kwargs):
    try:
        return function(*args, **kwargs)
    except w.requests.RequestException:
        raise RuntimeError("Original public parent request failed; no automatic retry") from None


def safe_blocks(response):
    try:
        yield from response.iter_content(1024 * 1024)
    except w.requests.RequestException:
        raise RuntimeError(
            "Original public parent stream interrupted; preserve original partial bytes"
        ) from None

            "ownerSlug": "nartaa",

    for _ in range(6):
        if response.status_code == 200:
            break
        assert response.status_code in (301, 302, 303, 307, 308), (
            "Pinned public parent download refused"
        )
        url = urljoin(response.url, response.headers.get("Location", ""))
        response.close()
        assert (
            urlparse(url).scheme == "https"
            and not urlparse(url).username
            and (urlparse(url).hostname or "").endswith(
                (".googleapis.com", ".googleusercontent.com", ".kaggle.com")
            )
        )
        # Storage redirects use a fresh session: never forward Kaggle credentials.
        response = safe_request(
            w.requests.get, url, stream=True, allow_redirects=False, timeout=(15, 90)
        )
    assert response.status_code == 200
    h = hashlib.sha256()
    size = 0
    with response, weight.open("xb") as f:
        for block in safe_blocks(response):
            size += len(block)
            assert (
                size <= 512 * 2**20
                and shutil.disk_usage("/tmp").free > 20 * 2**30 + 32 * 2**20 + 12 * 2**30
            )
            h.update(block)
            f.write(block)
        f.flush()
        os.fsync(f.fileno())
    assert h.hexdigest() == parent_sha
    downloaded = True
