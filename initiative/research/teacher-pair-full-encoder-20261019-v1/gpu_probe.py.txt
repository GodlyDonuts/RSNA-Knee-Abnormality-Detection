torch.backends.cuda.matmul.allow_tf32 = False

)
constants = {
    n.targets[0].id: ast.literal_eval(n.value)
    for n in ast.parse(old).body
    if isinstance(n, ast.Assign)
    and len(n.targets) == 1
    and isinstance(n.targets[0], ast.Name)
    and n.targets[0].id in ("WORKER", "TARGETS")
}
worker = ast.parse(base64.b64decode(constants["WORKER"], validate=True).decode())
boundary = next(
    i
    for i, n in enumerate(worker.body)
    if isinstance(n, ast.Import) and any(a.name == "timm" for a in n.names)
)
renderer = {
    "__name__": "original_renderer_for_new_teacher_pair_probe",
    "__file__": str(F / "original-renderer.private.py"),

exec(
    compile(
        ast.Module(body=worker.body[:boundary], type_ignores=[]), "original_exact_renderer", "exec"
    ),
    renderer,

    lambda volume, mask, mirror: renderer["eval_windows"](

assert (

    and hashlib.sha256(json.dumps(original.uids, separators=(",", ":")).encode()).hexdigest()

targets = json.loads(base64.b64decode(constants["TARGETS"], validate=True))
assert targets["eligible_fitting_only"] is True and len(targets["uids"]) == 32
mapping = {uid: i for i, uid in enumerate(original.uids)}

    and indices == json.loads((F / "private-input-binding.json").read_text())["indices"]

    base, res = ns["load_model"](
        str(A / "parent/raptor_ft_alldata_t16_blendjev_oai_d96_r384_swa.pt"),
        "coatnet_rmlp_2_rw_384.sw_in12k_ft_in1k",

    pass

    n: v.detach().clone()

    if any(k in n for k in ("running_mean", "running_var", "num_batches_tracked"))

initial = capture_rng(device)

for phase in ("sequential_pointwise", "joint_pointwise", "joint_teacher_rank"):
    assert time.monotonic() - start_all < 1000
    assert sum(p.stat().st_size for p in F.rglob("*") if p.is_file()) < 256 * 2**20
    restore_rng(initial, device)
    model.zero_grad(set_to_none=True)
    torch.cuda.empty_cache()
    torch.cuda.reset_peak_memory_stats(device)
    torch.cuda.synchronize()
    start = time.monotonic()
    recipe = Recipe(
        epochs=2, objective="teacher_rank" if phase == "joint_teacher_rank" else "pointwise"
    )
    if phase == "sequential_pointwise":
        value = 0.0
        for index in indices:
            positions, mirror, logq = sample(data.uids[index], recipe, 0, data.proposal(index))
            for view in (mirror, not mirror):
                case = data.case(index, positions, view)
                with torch.autocast("cuda", dtype=torch.float16):
                    z = model(case["images"].to(device)[None], logq.to(device)[None]).float()
                loss = anchored_loss(
                    z,
                    case["parent_logits"].to(device)[None].float(),
                    case["labels"].to(device)[None].float(),
                    case["mask"].to(device)[None],
                ) / (2 * len(indices))
                value += float(loss.detach())
                (loss * 1024.0).backward()
                del loss, z, case
        batch = {"studies": 8, "anatomical_views": 16}
    else:
        loss, batch = coupled_batch(model, data, indices, recipe, 0, device)
        value = float(loss.detach())
        (loss * 1024.0).backward()
        del loss
    for p in model.parameters():
        assert p.grad is not None and torch.isfinite(p.grad).all()
        p.grad.div_(1024.0)
    torch.cuda.synchronize()
    elapsed = time.monotonic() - start
    peak = torch.cuda.max_memory_allocated(device)
    assert peak < 64 * 2**30 and all(p._version == versions[n] for n, p in model.named_parameters())
    assert all(torch.equal(t, model.state_dict()[n]) for n, t in bn.items())
    end = capture_rng(device)
    if phase == "sequential_pointwise":
        reference = {n: p.grad.detach().cpu().clone() for n, p in model.named_parameters()}
        end_rng = end
    else:
        assert (
            torch.equal(end["torch"], end_rng["torch"])
            and torch.equal(end["cuda"], end_rng["cuda"])
            and end["python"] == end_rng["python"]
            and end["numpy"] == end_rng["numpy"]
        )
    sums = {group: [0.0, 0.0, 0.0, 0.0] for group in ("all", "encoder", "head")}
    for name, p in model.named_parameters():
        ref = reference[name].double()
        current = p.grad.detach().cpu().double()
        values = [
            float(ref.square().sum()),
            float(current.square().sum()),
            float((current - ref).square().sum()),
            float((current * ref).sum()),
        ]
        for group in ("all", "encoder" if name.startswith("backbone.") else "head"):
            sums[group] = [a + b for a, b in zip(sums[group], values, strict=True)]
    assert all(a > 0 and b > 0 for a, b, _, _ in sums.values())
    metrics = {
        g: {
            "reference_squared_norm": a,
            "current_squared_norm": b,
            "squared_error": c,
            "relative_L2_error": math.sqrt(c / a),
            "cosine": d / math.sqrt(a * b),
        }
        for g, (a, b, c, d) in sums.items()
    }
    assert all(
        v["reference_squared_norm"] > 0 and v["current_squared_norm"] > 0 for v in metrics.values()
    )
    if phase == "joint_pointwise":
        assert all(
            v["relative_L2_error"] <= 0.005 and v["cosine"] >= 0.99999 for v in metrics.values()
        )
    if phase == "joint_teacher_rank":
        assert metrics["all"]["squared_error"] > 0
    results[phase] = {
        "loss": value,
        "forward_backward_and_input_seconds": elapsed,
        "peak_allocated_bytes": peak,
        "gradient_comparison": metrics,
        **batch,
    }
    (F / (phase + "-aggregate.json")).write_text(json.dumps(results[phase], indent=2) + "\n")
