"""One small read-only owning-node collection after exact successful terminal."""

import hashlib
import json
import os
import stat
import subprocess
import sys
from pathlib import Path

v = json.load(sys.stdin)
assert os.uname().nodename == "evc33" and not os.environ.get("SLURM_JOB_GPUS")
job = v["target_job"]
assert job.isdecimal()
accounting = subprocess.check_output(
    ["sacct", "-X", "-n", "-P", "-j", job, "--format=JobIDRaw,State,ExitCode"], text=True
)
queue = subprocess.check_output(["squeue", "-h", "-u", "sa305415", "-o", "%i|%T"], text=True)
rows = [line.split("|") for line in accounting.splitlines() if line.split("|")[0] == job]
queued = [line for line in queue.splitlines() if line.strip() and line.split("|")[0] == job]
assert not queued and len(rows) == 1 and rows[0][1:3] == ["COMPLETED", "0:0"]
root = Path(v["root"])
assert root.is_dir() and root.stat().st_uid == os.getuid() and not root.stat().st_mode & 0o077
intent = json.loads((root / "intent.json").read_text())
actual_path = root / "actual.json"
actual = json.loads(actual_path.read_text())
assert intent["CPU_job"] == actual["CPU_job"] == job
assert intent["source_git"] == actual["source_git"] == v["expected_source_git"]
assert actual["studies"] == 3887
if v["role"] == "raw":
    name, key = "full-cache-index.private.json", "private_registry_sha256"
    cap = 320 * 2**30
    assert actual.get("both_masked_contexts_targets_and_full_support_priors_exact") or actual.get(
        "exact_original_decoding_masked_context_target_support_parity"
    )
else:
    assert v["role"] == "context"
    name, key = "context-index.private.json", "private_index_sha256"
    cap = 96 * 2**30
    assert actual["exact_both_view_images_targets_metadata_priors"] and actual["global_RNG_unchanged"]
index = root / name
body = index.read_bytes()
assert len(body) < 32 * 2**20 and hashlib.sha256(body).hexdigest() == actual[key]
records = json.loads(body)
uids = [row["uid"] for row in records]
assert len(records) == 3887 and len(set(uids)) == 3887
roster = hashlib.sha256(json.dumps(uids, separators=(",", ":")).encode()).hexdigest()
assert roster == v["fitting_roster_sha256"]
total = 0
for row in records:
    p = Path(row["path"])
    s = p.lstat()
    assert stat.S_ISREG(s.st_mode) and s.st_uid == os.getuid()
    assert [s.st_ino, s.st_size, s.st_mtime_ns, s.st_ctime_ns] == row["proof"]["stat"]
    assert row["proof"]["bytes"] == s.st_size
    total += s.st_size
assert total < cap
out = {
    "status": "actual_original_full_cache_terminal_collected",
    "role": v["role"], "target_job": job, "collection_CPU_job": os.environ["SLURM_JOB_ID"],
    "source_git": v["expected_source_git"], "target_terminal": "COMPLETED", "exit_code": "0:0",
    "original_actual_sha256": hashlib.sha256(actual_path.read_bytes()).hexdigest(),
    "private_index_sha256": actual[key], "fitting_roster_sha256": roster,
    "studies": len(records), "existing_cache_stat_bindings_checked": len(records),
    "existing_cache_bytes": total, "original_actual": actual,
    "pixel_reads_or_rehashes_or_writes": 0, "new_GPU_models_or_entries": 0,
}
print(json.dumps(out), flush=True)
