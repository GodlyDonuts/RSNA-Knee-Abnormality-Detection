def context_provenance(row, *, original=False):
    return {
        "original_raw_cache_sha256": row["proof"]["sha256"],
        "original_source_revision": "52f995f4568e36bfa5fbbaf69a89af41226b4f5d",
        "side": 320,
        "context_cap": 24,
        "halo": 8,
        "classes": 12,
        "module_sha256": original_renderer if original else renderer,
    }


def require_source_record(i, row):
    assert row["uid"] == records[i]["uid"]
    assert recon.signature(row["path"]) == row["proof"]["stat"]
    allowed = {
        new_raw / "cache" / f"case-{i:04d}.private.npz",
        old_raw / "cache" / f"case-{i:04d}.private.npz",
    }
    assert Path(row["path"]) in allowed
    proof_path = new_raw / "proofs" / f"case-{i:04d}.source-proof.private.json.gz"
    recon.regular(proof_path)
    body = proof_path.read_bytes()
    assert hashlib.sha256(body).hexdigest() == row["source_proof_sha256"]
    recon.source_proof_matches(json.loads(gzip.decompress(body)), records[i])
    return row


def producer_state():
    state = subprocess.check_output(
        ["sacct", "-X", "-n", "-P", "-j", producer, "--format=JobIDRaw,State,ExitCode"],
        text=True,
    )
    rows = [line.split("|") for line in state.splitlines() if line.split("|")[0] == producer]
    assert len(rows) == 1
    return rows[0][1:3]


def raw_record(i):
    if i in original["complete"]:
        return original["complete"][i]
    path = new_raw / "proofs" / f"case-{i:04d}.verified.private.json"
    while True:
        if path.exists():
            try:
                row = recon.bound_json(path)
            except json.JSONDecodeError:
                # The original numerical body writes this small JSON in place;
                # only a confirmed live producer permits waiting on that write.
                state, _ = producer_state()
                assert state in ("RUNNING", "COMPLETING")
                time.sleep(1)
                continue
            return require_source_record(i, row)
        state, code = producer_state()
        if state == "COMPLETED":
            assert code == "0:0" and path.exists(), "Completed producer lacks exact case proof"
            continue  # Proof may have appeared between the existence check and accounting.
        if state not in ("PENDING", "RUNNING", "COMPLETING"):
            raise RuntimeError("Successor source terminated; preserve all context prefixes")
        time.sleep(10)

def funding_gate():
    raw_paths = [Path(r["path"]) for r in raw16] + list((old_raw / "cache").glob("*.npz"))
    raw_paths += list((new_raw / "cache").glob("*.npz"))
    context_paths = [Path(r["path"]) for r in initial] + list(old_context.glob("context-*.npz"))
    if D.exists():
        context_paths += list(D.glob("context-*.npz"))
    raw_paid, context_paid = (recon.physical_bytes(raw_paths), recon.physical_bytes(context_paths))
    assert raw_paid < 320 * 2**30 and context_paid < 96 * 2**30
    other = v["funding"]["other_node_future_bytes"]
    assert type(other) is int and other >= (8 + 12 + 24) * 2**30 + 550 * 2**20
    assert shutil.disk_usage("/tmp").free > (
        20 * 2**30
        + other
        + 4 * 2**30
        + max(0, 320 * 2**30 - raw_paid)
        + max(0, 96 * 2**30 - context_paid)
    )
    return context_paid
