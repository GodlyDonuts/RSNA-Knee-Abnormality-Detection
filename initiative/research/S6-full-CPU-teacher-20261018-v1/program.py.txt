def atomic(path, value):
    t = path.with_suffix(path.suffix + ".temporary")
    t.write_text(json.dumps(value, sort_keys=True, allow_nan=False))
    t.replace(path)


def safe_paid(root):
    total = 0
    for p in root.rglob("*"):
        try:
            if p.is_file():
                total += p.stat().st_size
        except FileNotFoundError:
            pass  # Only concurrent removal of newly owned transient scratch.
    return total


stopped = threading.Event()
broker = contract.HeapBroker(80 * 2**30)
source_limit = threading.Semaphore(4)
pending = {}
pending_lock = threading.RLock()
slots = []
pool = concurrent.futures.ThreadPoolExecutor(max_workers=12)
futures = {}
succeeded = False
start = time.perf_counter()
deadline = time.monotonic() + 70 * 60


class Slot:
    def __init__(self, n):
        self.n = n
        self.root = run / ("slot-%02d" % n)
        self.root.mkdir(mode=0o700)
        self.err = (run / ("worker-%02d.private.log" % n)).open("x")
        env = {
            k: x
            for k, x in os.environ.items()
            if not any(t in k.upper() for t in ("TOKEN", "SECRET", "PASSWORD", "AUTHORIZATION"))
        }
        env.update(
            CUDA_VISIBLE_DEVICES="",
            PYTHONNOUSERSITE="1",
            PYTHONDONTWRITEBYTECODE="1",
            OMP_NUM_THREADS="2",
            MKL_NUM_THREADS="2",
            OPENBLAS_NUM_THREADS="1",
        )
        self.child = subprocess.Popen(
            [sys.executable, "-B", str(C / "worker.py")],
            stdin=subprocess.PIPE,
            stdout=subprocess.PIPE,
            stderr=self.err,
            text=True,
            env=env,
            start_new_session=True,
        )
        self.child.stdin.write(
            json.dumps(
                {
                    "scratch_root": str(self.root),
                    "HF_token": v["HF_token"],
                    "slot_cap_bytes": 5 * 2**30,
                    "source_buffer_helper": str(E / "source_buffer.py"),
                    "source_buffer_helper_sha256": expected["files"]["source_buffer.py"],
                    "classifier_worker": str(C / "classifier.py"),
                    "classifier_worker_sha256": stage["files"]["classifier.py"],
                }
            )
            + "\n"
        )
        self.child.stdin.flush()

    def consume(self, uid):
        lease = 0
        with pending_lock:
            pending[uid] = {"slot": self.n, "state": "requested"}
            atomic(D / "pending.private.json", pending)
        source_limit.acquire()
        try:
            self.child.stdin.write(json.dumps(records[uid]) + "\n")
            self.child.stdin.flush()
            line = self.child.stdout.readline()
            assert line, "Source worker exited"
            ready = json.loads(line)
            assert ready.get("uid") == uid and ready.get("stage") == "source_ready", (
                "Original source/identity admission failed"
            )
        finally:
            source_limit.release()
        with pending_lock:
            pending[uid] = {
                "slot": self.n,
                "state": "source_verified",
                "owned_scratch": ready["owned_scratch"],
            }
            atomic(D / "pending.private.json", pending)
        try:
            broker.acquire(ready["decoder_heap_bytes"], stopped)
            lease = ready["decoder_heap_bytes"]
            self.child.stdin.write(json.dumps({"uid": uid, "render": True}) + "\n")
            self.child.stdin.flush()
            line = self.child.stdout.readline()
            assert line, "Renderer/classifier worker exited"
            result = json.loads(line)
            assert (
                result.get("uid") == uid
                and result.get("error") is None
                and len(result.get("logits", [])) == 12
            ), "Original MRI/CPU teacher failed"
            return result
        finally:
            if lease:
                broker.release(lease)

    def close(self):
        if self.child.poll() is None:
            os.killpg(self.child.pid, signal.SIGTERM)
        try:
            self.child.wait(timeout=10)
        except subprocess.TimeoutExpired:
            os.killpg(self.child.pid, signal.SIGKILL)
            self.child.wait()
        self.err.close()


def terminate(signum, frame):
    stopped.set()
    raise Stop("Signal/resource stop: keep journal and private partial source/teacher outputs")


signal.signal(signal.SIGTERM, terminate)
signal.signal(signal.SIGINT, terminate)


def watch():
    while not stopped.wait(2):
        if (
            mem()["MemAvailable"] < 32 * 2**30
            or shutil.disk_usage("/dev/shm").free < 28 * 2**30
            or safe_paid(ram) > 64 * 2**30
            or safe_paid(P) > 8 * 2**20
        ):
            os.kill(os.getpid(), signal.SIGTERM)
            return


try:
    slots = [Slot(i) for i in range(12)]
    threading.Thread(target=watch, daemon=True).start()
    next_index = 0

    def submit(slot):
        global next_index
        if next_index < len(todo) and time.monotonic() < deadline:
            uid = todo[next_index]
            next_index += 1
            futures[pool.submit(slot.consume, uid)] = (slot, uid)

    for slot in slots:
        submit(slot)
    while futures:
        done, _ = concurrent.futures.wait(futures, return_when=concurrent.futures.FIRST_COMPLETED)
        for future in done:
            slot, uid = futures.pop(future)
            value = future.result()
            owned = Path(value["owned_scratch"]).resolve()
            assert owned.is_relative_to(run) and owned.name.startswith("fitting-study-")
            assert value["all_original_DICOM_identity_checks_passed"] is True
            result = {
                k: value[k]
                for k in (
                    "uid",
                    "logits",
                    "windows",
                    "origin",
                    "phase_role",
                    "CPU_job",
                    "image_tensor_sha256",
                    "metadata_tensor_sha256",
                    "source_proof_sha256",
                    "source_files",
                    "source_bytes",
                    "original_device_only_CPU_AST_sha256",
                    "original_CNN_mathematics_unchanged",
                    "seconds",
                )
            }
            result["phase"] = phase
            journal.append(result)
            # Only already consumed new scratch is removed, after its durable prediction/proof.
            shutil.rmtree(owned)
            with pending_lock:
                pending.pop(uid)
                atomic(D / "pending.private.json", pending)
            if len(journal.rows) % 8 == 0 or len(journal.rows) - before <= 3:
                print(
                    json.dumps(
                        {
                            "status": "actual_CPU_teacher_rows_preserved",
                            "completed": len(journal.rows),
                            "of": 3887,
                            "new_this_phase": len(journal.rows) - before,
                            "elapsed_seconds": round(time.perf_counter() - start, 1),
                        }
                    ),
                    flush=True,
                )
            submit(slot)
    assert not pending
    complete = len(journal.rows) == 3887
    if complete:
        logits = np.asarray([journal.rows[u]["logits"] for u in uids], dtype=np.float64)
        assert logits.shape == (3887, 12) and np.isfinite(logits).all()
        probabilities = 1 / (1 + np.exp(-logits))
        assert np.isfinite(probabilities).all()
        path = P / "teacher-final.private.npz"
        assert not path.exists()
        np.savez_compressed(path, uids=np.asarray(uids), logits=logits, probabilities=probabilities)
    out = {
        "status": "actual_complete_S6_fitting_teacher_ready_for_separate_target_admission"
        if complete
        else "actual_CPU_full_teacher_production_checkpoint_preserved",
        "CPU_job": os.environ["SLURM_JOB_ID"],
        "source_git": stage["source_git"],
        "phase": phase,
        "new_studies": len(journal.rows) - before,
        "completed": len(journal.rows),
        "of": 3887,
        "CPU_workers": 12,
        "CPU_threads_per_model": 2,
        "GPU": 0,
        "original_model_CPU_mathematics_preserved": True,
        "partial_rank_targets_or_clinical_labels_used": False,
        "seconds": time.perf_counter() - start,
        "journal_bytes": journal.path.stat().st_size,
        "journal_sha256": hashlib.sha256(journal.path.read_bytes()).hexdigest(),
        "teacher_file_sha256": hashlib.sha256(
            (P / "teacher-final.private.npz").read_bytes()
        ).hexdigest()
        if complete
        else None,
        "clinical_accuracy_or_student_training_claim": False,
        "automatic_retry": False,
    }
    atomic(D / "terminal.json", out)
    if complete:
        atomic(P / "teacher-complete.json", out)
    print(json.dumps(out), flush=True)
    succeeded = True
except BaseException as error:
    stopped.set()
    atomic(
        D / "failed.json",
        {
            "CPU_job": os.environ["SLURM_JOB_ID"],
            "phase": phase,
            "completed": len(journal.rows),
            "error_type": type(error).__name__,
            "automatic_retry": False,
        },
    )
    (run / "failure-traceback.private.txt").write_text(traceback.format_exc())
    raise
finally:
    stopped.set()
    for slot in slots:
        slot.close()
    pool.shutdown(wait=False, cancel_futures=True)
    if succeeded:
        (lock / "owner.json").unlink()
        lock.rmdir()
