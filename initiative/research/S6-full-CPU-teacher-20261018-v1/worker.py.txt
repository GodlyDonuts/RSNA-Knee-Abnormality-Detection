"""Persistent CPU renderer with explicit source-ready/memory-admission protocol.

Private request/response JSON travels through pipes, not public logs. A future
parent owns source-index/provenance admission, GPU scoring and scratch retirement.
This file is prepared source; no worker or production job is dispatched by it.
"""

from __future__ import annotations

import contextlib
import hashlib
import importlib.util
import inspect
import json
import os
import sys
import time
import traceback
from importlib.machinery import SourceFileLoader
from pathlib import Path


def main():
    assert not os.environ.get("SLURM_JOB_GPUS") or os.environ.get("CUDA_VISIBLE_DEVICES") == ""
    assert os.environ.get("CUDA_VISIBLE_DEVICES") == ""
    os.umask(0o077)
    protocol = sys.stdout
    sys.stdout = sys.stderr
    admission = json.loads(sys.stdin.readline())
    root = Path(admission["scratch_root"]).resolve()
    assert root.is_relative_to("/dev/shm") and root.is_dir()
    token = admission.pop("HF_token")
    slot_cap = int(admission["slot_cap_bytes"])
    assert slot_cap == 5 * 2**30
    os.environ.update(
        HF_HUB_OFFLINE="1",
        TRANSFORMERS_OFFLINE="1",
        PIP_NO_INDEX="1",
        HF_HOME=str(root / "hf"),
        TORCH_HOME=str(root / "torch"),
        XDG_CACHE_HOME=str(root / "cache"),
    )
    base = Path("/tmp/rsna-S6-checkpoint-admission-20261017-v1")
    sys.path.insert(0, str(base / "runtime-opencv-v1/site-packages"))
    sys.path.insert(0, "/tmp/rsna-compact-native-n3-assets-20261016-v1/runtime-overlay")
    import numpy as np
    import requests
    import torch

    assert not torch.cuda.is_available()
    torch.set_num_threads(2)
    torch.set_num_interop_threads(1)
    helper_path = Path(admission["source_buffer_helper"])
    assert (
        hashlib.sha256(helper_path.read_bytes()).hexdigest()
        == admission["source_buffer_helper_sha256"]
    )
    spec = importlib.util.spec_from_loader(
        "bounded_teacher_source", SourceFileLoader("bounded_teacher_source", str(helper_path))
    )
    helper = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(helper)
    spec = importlib.util.spec_from_file_location(
        "original_S6_stream_renderer", base / "source-tree-v1/launcher.private.py"
    )
    launch = importlib.util.module_from_spec(spec)
    sys.modules[spec.name] = launch
    spec.loader.exec_module(launch)
    assert (
        hashlib.sha256(inspect.getsource(launch.CI.main).encode()).hexdigest()
        == "7dac0d1616c2b9fc6e1f32a1846f03c3b5a051fd41665c9f15f6b185131d8946"
    )
    source_revision = "52f995f4568e36bfa5fbbaf69a89af41226b4f5d"
    geometry = json.loads(
        (base / "CPU-fitting8-UID-view-v2/collected/index.private.json").read_text()
    )["geometry"]
    dataset = launch.CI.CStudies(
        root,
        "train",
        [],
        root,
        base / "source-tree-v1/code/cloud_code",
        geometry["size"],
        eval_stride=geometry["eval_stride"],
        views=geometry["views"],
        slices=geometry["slices"],
        anchor_weights=base / "source-tree-v1/anatomy.pt",
        phys_mm=geometry["phys_mm"],
    )
    with np.load(
        "/tmp/rsna-licensed-parent-production-inputs-20261016-v1/targets.private.npz",
        allow_pickle=False,
    ) as data:
        fitting = set(data["uids"].tolist())
    reused = set(
        json.loads(
            (base / "CPU32-recovered-numerics-20261018-v1/selection.private.json").read_text()
        )["selected_uids"]
    )
    assert len(fitting) == 3887 and len(reused) == 32 and reused <= fitting
    seen = set()

    @contextlib.contextmanager
    def opener(row):
        from urllib.parse import quote

        url = (
            "https://huggingface.co/datasets/GodlyDonuts/RSNA-Data/resolve/"
            + source_revision
            + "/"
            + quote(row["path"], safe="/")
        )
        with requests.get(
            url, headers={"Authorization": "Bearer " + token}, timeout=(10, 90), stream=True
        ) as response:
            if response.status_code != 200:
                raise RuntimeError("Source HTTP status " + str(response.status_code))
            response.raw.decode_content = True
            yield response.raw

    for line in sys.stdin:
        request = json.loads(line)
        uid = request["uid"]
        assert uid in fitting and uid not in reused and uid not in seen
        seen.add(uid)
        owned = None
        started = time.perf_counter()
        try:
            if request.get("resume_owned"):
                owned = Path(request["resume_owned"]).resolve()
                assert owned.is_relative_to(root.parent) and owned.name.startswith("fitting-study-")
                data = owned / "data"
                proof = json.loads((owned / "source-proof.private.json").read_text())
                assert proof["study_uid_sha256"] == hashlib.sha256(uid.encode()).hexdigest()
                expected = [(r["path"], r["bytes"], r["sha256"]) for r in request["source_files"]]
                assert [
                    (r["source_path"], r["bytes"], r["sha256"]) for r in proof["files"]
                ] == expected
                assert (data / "train_series.csv").read_bytes().decode() == request["series_csv"]
                for row in request["source_files"]:
                    f = data / "train_series" / uid / row["series_uid"] / (row["sop_uid"] + ".dcm")
                    assert f.stat().st_size == row["bytes"]
            else:
                data, proof = helper.acquire(
                    root,
                    uid=uid,
                    rows=request["source_files"],
                    series_csv=request["series_csv"],
                    cap=2**30,
                    workers=6,
                    opener=opener,
                )
                owned = data.parent
            acquisition = time.perf_counter() - started
            import pydicom

            native_bytes = 0
            for row in request["source_files"]:
                f = data / "train_series" / uid / row["series_uid"] / (row["sop_uid"] + ".dcm")
                header = pydicom.dcmread(f, stop_before_pixels=True)
                assert (
                    str(header.StudyInstanceUID) == uid
                    and str(header.SeriesInstanceUID) == row["series_uid"]
                    and str(header.SOPInstanceUID) == row["sop_uid"]
                )
                native_bytes += (
                    2
                    * int(header.Rows)
                    * int(header.Columns)
                    * max(1, int(getattr(header, "NumberOfFrames", 1)))
                )
            # Explicit engineering caps, checked before native image decoding.
            image_upper = len(request["source_files"]) * 12 * 3 * 320 * 320
            assert proof["source_bytes"] + native_bytes + 16 * 2**20 < slot_cap
            assert proof["source_bytes"] + image_upper + 16 * 2**20 < slot_cap
            heap = max(8 * native_bytes + 2 * image_upper + 2 * 2**30, 4 * 2**30 + 6 * image_upper)
            assert 0 < heap <= 80 * 2**30
            ready = {
                "stage": "source_ready",
                "uid": uid,
                "owned_scratch": str(owned),
                "decoder_heap_bytes": heap,
                "native_bytes_upper": native_bytes,
                "new_source_acquisition_seconds": acquisition,
            }
            print(json.dumps(ready), file=protocol, flush=True)
            permission = json.loads(sys.stdin.readline())
            assert permission == {"uid": uid, "render": True}
            dataset.data, dataset.tmp, dataset.uids = str(data), str(owned / "native"), [uid]
            with (
                (owned / "renderer.private.log").open("x") as log,
                contextlib.redirect_stdout(log),
                contextlib.redirect_stderr(log),
            ):
                item = dataset[0]
            assert item["error"] is None and len(item["images"]) > 0
            images, meta = item["images"].numpy(), item["meta"].numpy()
            assert images.dtype == np.uint8 and images.shape[1:] == (3, 320, 320)
            assert len(images) <= 12 * len(request["source_files"])
            assert (
                meta.dtype == np.float32
                and meta.shape == (len(images), 6)
                and np.isfinite(meta).all()
            )
            tensor_file = owned / "rendered.private.npz"
            current_bytes = sum(p.stat().st_size for p in owned.rglob("*") if p.is_file())
            assert current_bytes + images.nbytes + meta.nbytes + 32768 < slot_cap
            np.savez(tensor_file, images=images, meta=meta)
            assert sum(p.stat().st_size for p in owned.rglob("*") if p.is_file()) < slot_cap
            result = {
                "uid": uid,
                "tensor_path": str(tensor_file),
                "owned_scratch": str(owned),
                "windows": len(images),
                "image_tensor_sha256": hashlib.sha256(memoryview(images).cast("B")).hexdigest(),
                "metadata_tensor_sha256": hashlib.sha256(memoryview(meta).cast("B")).hexdigest(),
                "source_files": len(proof["files"]),
                "source_bytes": proof["source_bytes"],
                "source_proof_path": str(owned / "source-proof.private.json"),
                "anchor": item.get("anchor"),
                "seconds": item["seconds"],
                "acquisition_seconds": acquisition,
                "whole_seconds": time.perf_counter() - started,
                "error": None,
            }
            result["all_original_DICOM_identity_checks_passed"] = True
            result["source_proof_sha256"] = hashlib.sha256(
                (owned / "source-proof.private.json").read_bytes()
            ).hexdigest()
            with (owned / "render-result.private.json").open("x") as f:
                json.dump(result, f)
            # Consume these exact tensors with the already executed original CPU port.
            import subprocess

            classifier = Path(admission["classifier_worker"])
            assert (
                hashlib.sha256(classifier.read_bytes()).hexdigest()
                == admission["classifier_worker_sha256"]
            )
            output = owned / "CPU-teacher"
            env = {
                k: v
                for k, v in os.environ.items()
                if not any(w in k.upper() for w in ("TOKEN", "SECRET", "PASSWORD", "AUTHORIZATION"))
            }
            with (
                (owned / "classifier-stdout.private.log").open("x") as out,
                (owned / "classifier-stderr.private.log").open("x") as err,
            ):
                child = subprocess.run(
                    [sys.executable, "-B", str(classifier)],
                    input=json.dumps({"render": result, "output": str(output)}),
                    text=True,
                    stdout=out,
                    stderr=err,
                    env=env,
                    timeout=1200,
                )
            assert child.returncode == 0, "Original CPU classifier stopped; retain without retry"
            scored = json.loads((output / "verified.private.json").read_text())
            assert scored["uid"] == uid and scored["windows"] == result["windows"]
            result.update(scored)
            result["phase_role"] = "full_fitting_CPU_FP32_stream"
            with (owned / "teacher-result.private.json").open("x") as f:
                json.dump(result, f)
            del images, meta, item
            import ctypes
            import gc

            gc.collect()
            ctypes.CDLL(None).malloc_trim(0)
            print(json.dumps(result), file=protocol, flush=True)
        except BaseException as error:
            if owned is not None:
                (owned / "worker-traceback.private.txt").write_text(traceback.format_exc())
            print(
                json.dumps(
                    {
                        "uid_sha256": hashlib.sha256(uid.encode()).hexdigest(),
                        "error": type(error).__name__,
                        "partial_bytes_retained": True,
                        "automatic_retry": False,
                    }
                ),
                file=protocol,
                flush=True,
            )
            raise


if __name__ == "__main__":
    main()
