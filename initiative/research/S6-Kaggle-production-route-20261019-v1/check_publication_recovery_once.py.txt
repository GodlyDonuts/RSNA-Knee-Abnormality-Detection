"""Once-only read-only publication reconciliation, credentials RAM/stdin only."""

import importlib.util
import json
import shlex
import subprocess
from pathlib import Path

R = Path.cwd()
D = Path(__file__).resolve().parent
L = R / "artifacts/S6-Kaggle-production-route-20261019-v1"
assert not (L / "publication-recovery-preflight-intent.json").exists()
for pid in (1613, 2264, 2861):
    assert subprocess.run(["ps", "-p", str(pid)], capture_output=True).returncode != 0
values = {
    k.strip(): z.strip().strip("\"'")
    for line in (R / ".env").read_text().splitlines()
    for k, sep, z in [line.partition("=")]
    if sep and not line.lstrip().startswith("#")
}
assert (R / ".env").stat().st_mode & 0o077 == 0
s = importlib.util.spec_from_file_location(
    "transport", R / "initiative/research/paired-logit-head-gradient-20261018-v1/stdin_transport.py"
)
t = importlib.util.module_from_spec(s)
s.loader.exec_module(t)
(L / "publication-recovery-preflight-intent.json").write_text(
    json.dumps(
        {
            "failed_CPU": 974799,
            "read_only": True,
            "model_transfers_or_HF_mutations": 0,
            "automatic_retry": False,
        }
    )
    + "\n"
)
a = [
    "srun",
    "--account=skattel",
    "--partition=normal",
    "--qos=normal",
    "--nodelist=evc48",
    "--ntasks=1",
    "--cpus-per-task=1",
    "--gres=none",
    "--mem=2G",
    "--time=00:06:00",
    "--job-name=rsna-epoch-publication-reconciliation",
    "env",
    "CUDA_VISIBLE_DEVICES=",
    "PYTHONNOUSERSITE=1",
    "PYTHONDONTWRITEBYTECODE=1",
    "/home/sa305415/rsna-knee/kaggle/management-cp312-v1/bin/python",
    "-B",
    "-c",
    t.COMMAND,
]
x = subprocess.run(
    ["ssh", "-S", "/tmp/rsna-research-20261001-stable.sock", "newton", shlex.join(a)],
    input=t.envelope(
        (D / "publication_preflight_recovery.py.txt").read_text(),
        {
            "actual": json.loads((D / "actual-terminal-epoch-1.json").read_text()),
            "token": values["KAGGLE_API_TOKEN"],
        },
    ),
    capture_output=True,
    text=True,
    timeout=900,
)
(L / "publication-recovery-preflight-transport.json").write_text(
    json.dumps({"returncode": x.returncode, "stdout": x.stdout, "stderr": x.stderr}, indent=2)
    + "\n"
)
assert x.returncode == 0, "Reconcile original preflight; no model mutation was authorized"
z = json.loads(x.stdout)
(D / "actual-publication-recovery-preflight.json").write_text(json.dumps(z, indent=2) + "\n")
print(json.dumps(z), flush=True)
