ROLE = "cross_lineage_equal_rank"
OVERLAY = "ca09ef0f2eb9712ec231d151b57769b6138e0d1d516918c8fdc1890c0034f18b"
POPULATION = "0485d1b1df7af2b8ef10207bfa37545e2c65f7dc68f47a254f2af0c475de7f42"
TARGET = "3d7014e7758c25bd4c39aa3676954fa6cd378e0aaf7c46d1510af05ad3c077ba"
PARENT = "7e5315dad125b99fc65b340b3de41de628e9be51ff5835355dd61c86472244ef"

STATUS = "actual_complete_S6_target_full_encoder_epoch_session_complete"


def verify(state, snapshot, receipt, actual_engine, *, expected_parameters=73136096):
    engine = actual_engine.get("receipt", {})
    ready = engine.get("complete_teacher_admission", {})
    b = state["binding"]
    modules = engine.get("source_module_sha256", {})
    overlay_key = "rsna/competitive/complete_teacher_overlay.py"
    if (
        actual_engine.get("status")
        != "actual_existing_twoT4_outputs_independent_CPU_replay_accepted"
        or actual_engine.get("kernel_id") != 139695847
        or actual_engine.get("CPU_job") != "974246"
        or actual_engine.get("original_kernel_status") != "ERROR"
        or actual_engine.get("source_sha256")
        != "6747a9deaff18122565bd0a6b9b58382be781d42dd9ab8c4f5ca18d066dd1ff5"
        or engine.get("status")
        != "actual_complete_S6_target_change_full_encoder_twoT4_MRI_recovery_accepted"
        or engine.get("world_size") != 2
        or engine.get("same_initial_complete_model_and_allrank_RNG_between_roles") is not True
        or engine.get("distinct_final_learned_tensors", 0) <= 0
        or modules.get(overlay_key) != OVERLAY
        or len(modules) != 14
        or ready.get("status")
        != "actual_complete3887_two_role_teacher_targets_private_CPU_constructed"
        or ready.get("population") != 3887
        or ready.get("population_sha256") != POPULATION
        or ready.get("target_file") != {"bytes": 794124, "sha256": TARGET}
        or ready.get("labels_masks_and_original_targets_modified") is not False
        or b.get("complete_teacher_admission") != ready
        or b.get("teacher_target_role") != ROLE
        or receipt.get("teacher_target_role") != ROLE
        or receipt.get("role") != "influence"
        or receipt.get("total") != 1458
        or receipt.get("fixed_final_complete") is not (receipt.get("segment") == 3)
        or b.get("teacher_transport") != ready.get("roles", {}).get(ROLE)
        or b.get("source_module_sha256") != {n: h for n, h in modules.items() if n != overlay_key}
        or b.get("teacher_overlay_source_sha256") != OVERLAY
        or b.get("production_plan") != "complete_S6_equal_rank_matched_full3887_v1"
        or b.get("parent_checkpoint_sha256") != PARENT
        or b.get("parent_source_sha256") != SOURCE
        or b.get("version") != VERSION
        or b.get("recipe") != asdict(Recipe(epochs=3, role="influence"))
        or b.get("role") != "influence"
        or b.get("population_sha256") != POPULATION
        or b.get("proposal", {}).get("proposal_table") != {"bytes": 2923152, "sha256": PROPOSAL}
        or b.get("actual_recovery", {}).get("kernel_id") != 139130771
        or b.get("diagnostic_only") is not False
        or b.get("protected_gold_confirmation_used") is not False
        or b.get("independent_internal_validation") is not False
    ):
        raise ValueError("Actual complete twoT4 engine/target/source/sampler provenance required")
    result = complete_state(
        state,
        snapshot,
        receipt,
        expected_parameters=expected_parameters,
        expected_windows=16,
        expected_status=STATUS,
    )
    devices = receipt.get("devices", [])
    if len(devices) != 2 or any(
        d.get("GPU") != "Tesla T4"
        or d.get("completed") != result["completed"]
        or d.get("trainable_parameters") != 73136096
        or d.get("all_BN_buffers_equal_parent") is not True
        for d in devices
    ):
        raise ValueError("Both actual T4s and complete original learned encoder required")
    return {
        **result,
        "teacher_target_role": ROLE,
        "sampling_role": "influence",
        "training_windows": 16,
        "complete_target_sha256": TARGET,
        "actual_target_engine_CPU": 974246,
        "actual_target_engine_kernel": 139695847,
    }
