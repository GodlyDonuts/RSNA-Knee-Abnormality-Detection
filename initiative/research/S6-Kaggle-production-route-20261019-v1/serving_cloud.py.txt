"""Canonical exact twoT4 target checkpoint qualification and code entry; no model re-upload."""

import hashlib
import importlib.util
import json
import os
import shutil
import subprocess
import sys
from pathlib import Path

D = Path("/tmp/rsna-S6-Kaggle-production-route-20261019-v1")
F = D / "serving-account-control-v1"


def read(path):
    return json.loads(Path(path).read_text())


def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def bounds():
    assert (
        os.uname().nodename == "evc33"
        and not os.environ.get("SLURM_JOB_GPUS")
        and os.environ.get("CUDA_VISIBLE_DEVICES") == ""
    )
    assert (
        sum(p.stat().st_size for p in D.rglob("*") if p.is_file() and not p.is_symlink()) + 2**20
        < 8 * 2**20
    )
    assert shutil.disk_usage("/tmp").free > (20 + 8) * 2**30 + 32 * 2**20
    q = subprocess.check_output(["lfs", "quota", "-u", "sa305415", "/home/sa305415"], text=True)
    z = next(x.split() for x in q.splitlines() if x.strip().startswith("/home/sa305415"))
    assert (int(z[2]) - int(z[1])) * 1024 > 20 * 2**30 + 32 * 2**20 and int(z[6]) - int(z[5]) > 5128


def load(name):
    s = importlib.util.spec_from_file_location(
        "complete_target_" + name, F / "code" / (name + ".py")
    )
    m = importlib.util.module_from_spec(s)
    s.loader.exec_module(m)
    return m


def execute(v):
    phase, epoch = v["phase"], v.get("epoch", 1)
    assert (
        phase in ("stage", "prepare", "push", "status", "submit", "outcome")
        and type(epoch) is int
        and epoch in (1, 3)
    )
    bounds()
    if phase == "stage":
        assert not F.exists()
        assert v["source_sha256"] == {
            n: hashlib.sha256(s.encode()).hexdigest() for n, s in v["sources"].items()
        }
        assert sum(len(s.encode()) for s in v["sources"].values()) < 256 * 2**10
        F.mkdir(mode=0o700)
        (F / "code").mkdir(mode=0o700)
        for name, source in v["sources"].items():
            assert name.endswith(".py") and "/" not in name
            compile(source, name, "exec")
            (F / "code" / name).write_text(source)
        out = {
            "status": "actual_twoT4_epoch_serving_account_control_staged",
            "CPU_job": os.environ["SLURM_JOB_ID"],
            "source_git": v["source_git"],
            "source_sha256": v["source_sha256"],
            "new_GPU_or_entry_mutations": 0,
        }
        (F / "staged.json").write_text(json.dumps(out) + "\n")
        return out
    stage = read(F / "staged.json")
    assert stage["source_git"] == v["source_git"] and stage["source_sha256"] == v["source_sha256"]
    assert all(sha(F / "code" / n) == h for n, h in stage["source_sha256"].items())
    w, policy = load("workflow"), load("policy")
    import requests

    remote = w.Remote.__new__(w.Remote)
    remote.session = requests.Session()
    remote.session.headers["Authorization"] = "Bearer " + v["token"]
    remote.username = "godlydonuts"
    who = remote.call("security.OAuthService", "IntrospectToken", {"token": v["token"]})
    assert who.get("active") is True and who.get("username") == "godlydonuts"
    E = D / ("epoch-" + str(epoch))
    S = E / "serving-v1"
    training = read(E / "actual-accepted.json")
    pub = read(E / "publication-private-verified.json")
    compiled = read(S / "compiled.json")
    assert (
        compiled["status"]
        == "actual_privately_preserved_complete_target_twoT4_epoch_serving_sources_compiled"
        and compiled["epoch"] == epoch
    )
    assert (
        training["status"] == "COMPLETE" and training["epoch"] == epoch and training["version"] == 1
    )
    assert pub["model_sha256"] == compiled["model_sha256"] == training["receipt"]["epoch"]["sha256"]
    assert (
        pub["kernel_id"] == training["kernel_id"]
        and pub["original_source_sha256"] == training["source_sha256"]
    )
    candidate = {
        "epoch": epoch,
        "updates": 486 * epoch,
        "role": "cross_lineage_equal_rank",
        "model_sha256": compiled["model_sha256"],
        "training_source_sha256": training["source_sha256"],
        "original_compilation": compiled,
    }
    role = candidate["role"]
    model_sha = candidate["model_sha256"]
    key = "licensed-parent-full-epoch-serving:" + model_sha
    target = S / "account-phase-v1"
    claim = Path("/home/sa305415/rsna-knee/kaggle/coordinator-claim")
    try:
        claim.mkdir(mode=0o700)
    except FileExistsError:
        return {"status": "waiting_existing_canonical_claim_no_mutation"}
    owner = {
        "kind": "actual_S6_complete_target_twoT4_epoch_serving",
        "CPU_job": os.environ["SLURM_JOB_ID"],
        "phase": phase,
        "role": role,
    }
    w.new_json(claim / "owner.json", owner)
    retain = False
    try:
        with w.ledger(
            Path("/home/sa305415/rsna-knee/kaggle/account-ledger.sqlite3"), "godlydonuts"
        ) as db:
            w.verify_account(db, remote)
            if phase == "prepare":
                assert (
                    not target.exists()
                    and db.execute("SELECT value FROM meta WHERE key=?", (key,)).fetchone() is None
                )
                sources = {}
                texts = {}
                for mode in policy.CAPS:
                    path = S / (mode + ".private.py")
                    assert {"bytes": path.stat().st_size, "sha256": sha(path)} == compiled[
                        "outputs"
                    ][mode]
                    texts[mode] = policy.public_bounded_source(path.read_text(), mode)
                    sources[mode] = {
                        "bytes": len(texts[mode].encode()),
                        "sha256": hashlib.sha256(texts[mode].encode()).hexdigest(),
                    }
                target.mkdir(mode=0o700)
                for mode, text in texts.items():
                    (target / (mode + ".private.py")).write_text(text)
                row = {
                    "state": "full_parent_epoch_model_verified",
                    "kind": "complete_target_twoT4_epoch_serving",
                    "role": role,
                    "epoch": epoch,
                    "model_sha256": model_sha,
                    "source_git": v["source_git"],
                    "candidate": candidate,
                    "source": sources,
                    "training": training,
                    "private_publication": pub,
                }
                db.execute("INSERT INTO meta VALUES (?,?)", (key, json.dumps(row, sort_keys=True)))
                db.commit()
                w.new_json(target / "prepared.json", row)
                return {
                    "status": "actual_preserved_epoch_serving_prepared_without_model_reupload",
                    "epoch": epoch,
                    "model_sha256": model_sha,
                    "new_GPU_or_entry_mutations": 0,
                }
            found = db.execute("SELECT value FROM meta WHERE key=?", (key,)).fetchone()
            assert found is not None
            row = json.loads(found[0])
            assert row["candidate"] == candidate and row["role"] == role

            def save():
                db.execute(
                    "UPDATE meta SET value=? WHERE key=?", (json.dumps(row, sort_keys=True), key)
                )
                db.execute(
                    "INSERT INTO events(at,sha,event) VALUES(?,?,?)",
                    (w.utcnow(), model_sha, row["state"]),
                )
                db.commit()

            mode = v.get("mode", "qualification")
            assert mode in policy.CAPS
            source = (target / (mode + ".private.py")).read_text()
            source_sha = hashlib.sha256(source.encode()).hexdigest()
            assert source_sha == row["source"][mode]["sha256"]
            request = policy.request(epoch, mode, source, training)
            slug = request["slug"]

            def exact():
                kernel = remote.kernel("GetKernel", slug, 1)
                m = kernel["metadata"]
                assert kernel["blob"]["source"] == source
                assert m.get("id") == row[mode]["kernel_id"]
                for name in ("isPrivate", "enableGpu", "machineShape", "dockerImage"):
                    assert m.get(name) == request[name]
                assert not m.get("enableInternet", False)
                for name in ("datasetDataSources", "kernelDataSources", "competitionDataSources"):
                    assert sorted(m.get(name, [])) == sorted(request[name])
                return remote.kernel("GetKernelSessionStatus", slug, 1).get("status")

            if phase == "push":
                assert mode not in row and not (target / (mode + "-push-intent.json")).exists()
                assert row["state"] == (
                    "full_parent_epoch_model_verified"
                    if mode == "qualification"
                    else "full_parent_epoch_qualification_accepted"
                )
                cap = load("gpu_capacity").admission(db, remote, caller="serving")
                quota = remote.call(
                    "kernels.KernelsApiService", "GetAcceleratorQuotaStatistics", {}
                )["gpuQuota"]
                complete = [
                    json.loads(raw).get("epoch", 0)
                    for k, raw in db.execute("SELECT key,value FROM meta")
                    if k.startswith("licensed-importance-full-fit:complete-S6-cross:")
                    and json.loads(raw).get("state")
                    == "complete_teacher_cross_full_fit_actual_complete"
                ]
                assert epoch in complete
                budget = policy.budget(quota, epoch, mode, latest_training_epoch=max(complete))
                if not cap["available"] or not budget["available"]:
                    return {
                        "status": "waiting_actual_GPU_capacity_or_budget_no_mutation",
                        "capacity": cap,
                        "budget": budget,
                    }
                for ref in request["kernelDataSources"]:
                    assert (
                        remote.kernel("GetKernelSessionStatus", ref, 1).get("status") == "COMPLETE"
                    )
                absence = remote.require_absent_kernel(slug)
                intent = {
                    "role": role,
                    "slug": slug,
                    "version": 1,
                    "source_sha256": source_sha,
                    "model_sha256": model_sha,
                    "budget": budget,
                    "capacity": cap,
                    "absence": absence,
                    "request_sha256": w.object_hash(request),
                    "automatic_retry": False,
                }
                w.new_json(target / (mode + "-push-intent.json"), intent)
                row.update(state="full_parent_epoch_" + mode + "_push_intent", **{mode: intent})
                save()
                retain = True
                response = remote.call("kernels.KernelsApiService", "SaveKernel", request)
                w.new_json(target / (mode + "-push-response.json"), response)
                assert (
                    response.get("ref") == "/code/" + slug
                    and response.get("versionNumber") == 1
                    and type(response.get("kernelId")) is int
                    and not response.get("error")
                )
                assert not any(response.get(name) for name in w.SAVE_KERNEL_INVALID)
                row[mode]["kernel_id"] = response["kernelId"]
                row["state"] = "full_parent_epoch_" + mode + "_pushed"
                save()
                retain = False
                return {
                    "status": "one_exact_target_epoch_" + mode + "_pushed",
                    "role": role,
                    "slug": slug,
                    "version": 1,
                    "kernel_id": response["kernelId"],
                    "model_sha256": model_sha,
                    "source_sha256": source_sha,
                    "competition_entry": False,
                }
            assert mode in row and row[mode]["source_sha256"] == source_sha
            actual_status = exact()
            if phase == "status":
                result = {
                    "status": actual_status,
                    "role": role,
                    "mode": mode,
                    "slug": slug,
                    "version": 1,
                    "kernel_id": row[mode]["kernel_id"],
                    "source_sha256": source_sha,
                    "model_sha256": model_sha,
                }
                if actual_status == "COMPLETE":
                    receipt = remote.output_json(
                        slug,
                        1,
                        "licensed-IID16-epoch-"
                        + ("qualification" if mode == "qualification" else "public")
                        + "-accepted.json",
                    )
                    policy.receipt(receipt, candidate, mode)
                    result["receipt"] = receipt
                    if mode == "competition":
                        csv = load("output_csv")
                        checked = csv.verify_csv(
                            csv.download_csv(remote, slug, 1), policy.COLUMNS, policy.ROSTER
                        )
                        assert (
                            checked["CSV_sha256"] == receipt["CSV_sha256"]
                            and checked["bytes"] == receipt["CSV_bytes"]
                        )
                        result["CSV_verified_in_RAM"] = checked
                    if not (target / (mode + "-actual-accepted.json")).exists():
                        w.new_json(target / (mode + "-actual-accepted.json"), result)
                        row[mode]["receipt"] = receipt
                        row["state"] = "full_parent_epoch_" + mode + "_accepted"
                        save()
                elif actual_status in ("ERROR", "CANCELLED", "CANCEL_ACKNOWLEDGED"):
                    if not (target / (mode + "-actual-failed.json")).exists():
                        w.new_json(target / (mode + "-actual-failed.json"), result)
                        row["state"] = "full_parent_epoch_" + mode + "_failed"
                        save()
                return result
            assert mode == "competition" and actual_status == "COMPLETE"
            if phase == "submit":
                assert (
                    row["state"] == "full_parent_epoch_competition_accepted"
                    and not row.get("submission_id")
                    and not (target / "entry-intent.json").exists()
                )
                public = read(target / "competition-actual-accepted.json")
                assert (
                    public["receipt"] == row["competition"]["receipt"]
                    and public["CSV_verified_in_RAM"]["test_roster_sha256"] == policy.ROSTER
                )
                account = remote.account()
                slots = w.allowance(db, account)
                if slots < 1:
                    return {
                        "status": "waiting_authoritative_live_account_allowance_no_mutation",
                        "account": account,
                    }
                subs = remote.submissions()
                description = "rsna-complete-twoT4-S6-e" + str(epoch) + ":" + model_sha
                assert not any(s.get("description") == description for s in subs)
                assert not any(
                    json.loads(raw)
                    .get("state", "")
                    .endswith(("_submit_intent", "_submit_uncertain"))
                    for (raw,) in db.execute("SELECT value FROM meta")
                    if raw.lstrip().startswith("{")
                )
                intent = {
                    "role": role,
                    "description": description,
                    "model_sha256": model_sha,
                    "source_sha256": source_sha,
                    "slug": slug,
                    "version": 1,
                    "kernel_id": row[mode]["kernel_id"],
                    "fileName": "submission.csv",
                    "fresh_account": account,
                    "effective_allowance": slots,
                    "automatic_retry": False,
                }
                w.new_json(target / "entry-intent.json", intent)
                row.update(
                    state="full_parent_epoch_submit_intent",
                    description=description,
                    attempted_at=w.utcnow(),
                )
                save()
                retain = True
                user, name = slug.split("/")
                response = remote.competition(
                    "CreateCodeSubmission",
                    {
                        "fileName": "submission.csv",
                        "kernelOwner": user,
                        "kernelSlug": name,
                        "kernelVersion": 1,
                        "submissionDescription": description,
                    },
                )
                w.new_json(target / "entry-response.json", response)
                assert type(response.get("ref")) is int and response["ref"] > 0
                row.update(state="full_parent_epoch_submitted", submission_id=response["ref"])
                save()
                retain = False
                result = {
                    "status": "one_exact_target_epoch_code_submission_accepted",
                    "role": role,
                    "submission_id": response["ref"],
                    "slug": slug,
                    "version": 1,
                    "public_example_CSV_uploaded": False,
                    "actual_hidden_score_known": False,
                }
                w.new_json(target / "actual-submitted.json", result)
                return result
            assert phase == "outcome" and row["state"] in (
                "full_parent_epoch_submitted",
                "full_parent_epoch_scored",
                "full_parent_epoch_hidden_failed",
            )
            found = [s for s in remote.submissions() if s.get("ref") == row["submission_id"]]
            assert len(found) == 1
            s = found[0]
            outcome = w.submission_outcome(s)
            result = {
                "status": "actual_target_epoch_hidden_outcome",
                "role": role,
                "submission_id": s["ref"],
                "actual_status": s.get("status"),
                "public_score": s.get("publicScore"),
                "outcome": outcome,
                "error_present": bool(s.get("errorDescription")),
                "new_mutations": 0,
            }
            if outcome in ("scored", "failed"):
                row.update(
                    state="full_parent_epoch_scored"
                    if outcome == "scored"
                    else "full_parent_epoch_hidden_failed",
                    actual_hidden_outcome=result,
                )
                save()
                if not (target / "actual-hidden-terminal.json").exists():
                    w.new_json(target / "actual-hidden-terminal.json", result)
            return result
    finally:
        if not retain:
            assert read(claim / "owner.json") == owner
            (claim / "owner.json").unlink()
            claim.rmdir()


if __name__ == "__main__":
    os.umask(0o077)
    value = json.load(sys.stdin)
    try:
        print(json.dumps(execute(value)), flush=True)
    except BaseException as error:
        import traceback

        frames = traceback.extract_tb(error.__traceback__)
        print(
            json.dumps(
                {
                    "status": "original_final_serving_phase_stopped_no_retry",
                    "phase": value.get("phase"),
                    "exception_type": type(error).__name__,
                    "site_line": frames[-1].lineno if frames else None,
                    "URLs_credentials_external_source_or_case_records_logged": False,
                }
            ),
            file=sys.stderr,
            flush=True,
        )
        raise SystemExit(1) from None
