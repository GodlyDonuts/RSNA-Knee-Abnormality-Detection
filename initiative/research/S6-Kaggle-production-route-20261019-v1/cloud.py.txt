def module(name, h):
    p = C / (name + ".py")
    assert hashlib.sha256(p.read_bytes()).hexdigest() == h
    s = importlib.util.spec_from_file_location("canonical_" + name, p)
    m = importlib.util.module_from_spec(s)
    s.loader.exec_module(m)
    return m
