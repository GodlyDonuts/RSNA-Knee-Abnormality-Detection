def provenance(training, publication, model_sha):
    r = training.get("receipt", {})
    epoch = training.get("epoch")
    b = r.get("binding", {})
    p = publication.get("independent_CPU_state_verification", {})
    if (
        training.get("status") != "COMPLETE"
        or type(epoch) is not int
        or not 1 <= epoch <= 3
        or training.get("role") != ROLE
        or r.get("teacher_target_role") != ROLE
        or r.get("role") != "influence"
        or r.get("status") != "actual_complete_S6_target_full_encoder_epoch_session_complete"
        or r.get("segment") != epoch
        or r.get("completed") != 486 * epoch
        or r.get("total") != 1458
        or r.get("fixed_final_complete") is not (epoch == 3)
        or publication.get("status")
        != "actual_complete_target_twoT4_epoch_private_all_objects_download_size_SHA_verified"
        or publication.get("epoch") != epoch
        or publication.get("completed") != 486 * epoch
        or publication.get("kernel_id") != training.get("kernel_id")
        or publication.get("version") != training.get("version")
        or publication.get("original_source_sha256") != training.get("source_sha256")
        or publication.get("source_git") != b.get("source_git")
        or len(b.get("source_git", "")) != 40
        or publication.get("prefix")
        != "checkpoints/complete-S6-target-twoT4-"
        + b.get("source_git", "")[:12]
        + "/"
        + ROLE
        + "/epoch-"
        + str(epoch).zfill(3)
        or publication.get("teacher_target_role") != ROLE
        or publication.get("objects") != 6
        or publication.get("canonical_shared_publication_lock_used") is not True
        or publication.get("model_sha256") != model_sha
        or publication.get("files", {}).get("model.pt") != r.get("epoch")
        or r.get("epoch", {}).get("sha256") != model_sha
        or not isinstance(model_sha, str)
        or len(model_sha) != 64
        or any(c not in "0123456789abcdef" for c in model_sha)
        or p.get("full_learned_graph_exact") is not True
        or p.get("finite_contiguous_updates") != 486 * epoch
        or p.get("actual_AdamW_parameters") != 73136096
        or p.get("all_two_rank_RNG_and_scaler_records_present") is not True
        or p.get("teacher_target_role") != ROLE
        or p.get("sampling_role") != "influence"
        or p.get("training_windows") != 16
        or p.get("complete_target_sha256") != TARGET
        or p.get("actual_target_engine_CPU") != 974246
        or p.get("actual_target_engine_kernel") != 139695847
    ):
        raise ValueError(
            "Actual exact changed-target epoch and six-object private byte verification required"
        )
    return epoch

        != "actual_complete3887_two_role_teacher_targets_private_CPU_constructed"
