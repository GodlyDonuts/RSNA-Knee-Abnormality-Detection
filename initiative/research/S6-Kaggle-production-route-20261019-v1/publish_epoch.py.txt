def shared_metadata_budget(row):
    """Retain original funded128 metadata inodes and four bounded transients.

    Models/journals/HF caches reside exclusively on the admitted node-local root.
    Four shared transients cover the directory lock/marker plus SQLite WAL/SHM.
    No original deletion, hard-quota borrowing or retirement of old liabilities.
    """
    free_files = int(row[6]) - int(row[5])
    free_bytes = (int(row[2]) - int(row[1])) * 1024
    if free_files <= 5000 + 128 + 4 or free_bytes <= 20 * 2**30 + 80 * 2**20:
        raise ValueError("Original shared reserve after funded metadata and transients required")
    return {
        "shared_free_files": free_files,
        "funded_metadata_liability": 128,
        "transient_inode_bound": 4,
        "shared_reserve_after_liabilities": free_files - 128 - 4,
    }

def download(name, dest, item):
    owner, slug = a["slug"].split("/")
    r.session.cookies.clear()
    response = r.session.post(
        "https://api.kaggle.com/v1/kernels.KernelsApiService/DownloadKernelOutput",
        json={"ownerSlug": owner, "kernelSlug": slug, "versionNumber": 1, "filePath": name},
        timeout=(10, 90),
        allow_redirects=False,
        stream=True,
    )
    for _ in range(6):
        if response.status_code == 200:
            break
        assert response.status_code in (301, 302, 303, 307, 308)
        url = urljoin(response.url, response.headers["Location"])
        response.close()
        u = urlparse(url)
        assert (
            u.scheme == "https"
            and u.hostname
            and (
                u.hostname
                in (
                    "storage.googleapis.com",
                    "api.kaggle.com",
                    "www.kaggle.com",
                    "www.kaggleusercontent.com",
                )
                or u.hostname.endswith(".googleusercontent.com")
            )
        )
        response = (r.session if u.hostname == "api.kaggle.com" else requests).get(
            url, timeout=(10, 90), allow_redirects=False, stream=True
        )
    assert response.status_code == 200
    h = hashlib.sha256()
    count = 0
    try:
        with dest.open("xb") as f:
            for block in response.iter_content(2**20):
                count += len(block)
                assert count <= item["bytes"]
                f.write(block)
                h.update(block)
    finally:
        response.close()
    assert count == item["bytes"] and h.hexdigest() == item["sha256"]
