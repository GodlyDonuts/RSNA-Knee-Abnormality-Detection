import importlib.util
import json
import os
import sqlite3
import subprocess
import sys

import requests

phase = v["phase"]
assert phase in ("observe", "accept", "absence", "draft", "draft_launch_intent", "observe_version1")
assert os.uname().nodename == "evc33" and not os.environ.get("SLURM_JOB_GPUS")
D = Path("/tmp/rsna-S6-Kaggle-production-route-20261019-v1/epoch-3")
source = (D / "source.private.py").read_text()
sha = hashlib.sha256(source.encode()).hexdigest()
assert sha == "c870ded987b8e006abc36f65ed9bfddd64cf9ceb6ed71148429aa3995e3f2410"
assert json.loads((D / "compiled.json").read_text())["source_sha256"] == sha
claim = Path("/home/sa305415/rsna-knee/kaggle/coordinator-claim")
owner = json.loads((claim / "owner.json").read_text())
assert owner == {
    "CPU_job": "975430",
    "epoch": 3,
    "kind": "complete_S6_changed_role_full_fit",
    "phase": "push",
}
assert "975430|FAILED|" in subprocess.check_output(
    ["sacct", "-X", "-n", "-P", "-j", "975430", "--format=JobIDRaw,State,ExitCode"], text=True
)
f = Path("/tmp/rsna-compact-T4-probe-20261016-v1/cloud-control/workflow.py")
assert (
    hashlib.sha256(f.read_bytes()).hexdigest()
    == "2d3074fbf677285687e80c0e6b58721407bc61ddea39a990797da0e1d4e3b900"
)
s = importlib.util.spec_from_file_location("canonical_saved_epoch_workflow", f)
w = importlib.util.module_from_spec(s)
s.loader.exec_module(w)
r = w.Remote.__new__(w.Remote)
r.username = "godlydonuts"
r.session = requests.Session()
r.session.headers["Authorization"] = "Bearer " + v["token"]
who = r.call("security.OAuthService", "IntrospectToken", {"token": v["token"]})
assert who.get("active") is True and who.get("username") == "godlydonuts"
key = "licensed-importance-full-fit:complete-S6-cross:" + sha
path = Path("/home/sa305415/rsna-knee/kaggle/account-ledger.sqlite3")
db = sqlite3.connect("file:" + str(path) + "?mode=ro", uri=True)
row = json.loads(db.execute("SELECT value FROM meta WHERE key=?", (key,)).fetchone()[0])
db.close()
assert row["state"] == "complete_teacher_cross_full_fit_push_intent" and "kernel_id" not in row
assert not (D / "push-response.json").exists() and not (D / "pushed.json").exists()
slug = row["slug"]
assert slug == "godlydonuts/rsna-s6-cross-full-e3-" + sha[:12]
if phase == "draft_metadata":
    k = r.call(
        "kernels.KernelsApiService",
        "GetKernel",
        {"userName": "godlydonuts", "kernelSlug": slug.split("/")[1]},
    )
    m = k["metadata"]
    b = k["blob"]["source"]
    print(
        json.dumps(
            {
                "status": "actual_E3_draft_metadata_read_only",
                "CPU_job": os.environ["SLURM_JOB_ID"],
                "metadata": m,
                "source_sha256": hashlib.sha256(b.encode()).hexdigest(),
                "source_equal_original": b == source,
                "source_equal_original_with_cell_header": b == "# %% [code]\n" + source,
                "source_bytes": len(b.encode()),
                "new_GPU_or_submission_mutations": 0,
            }
        ),
        flush=True,
    )
    raise SystemExit(0)
if phase in ("draft_launch_intent", "draft_launch_intent_v2"):
    assert v["UI_draft_version1_offline_twoT4_verified"] is True
    quota = r.call("kernels.KernelsApiService", "GetAcceleratorQuotaStatistics", {})["gpuQuota"]

    def seconds(z):
        return float(z.removesuffix("s"))

    assert seconds(quota.get("timeReserved", "0s")) == 0
    assert seconds(quota["totalTimeAllowed"]) - seconds(quota["timeUsed"]) >= 12000
    draft = r.call(
        "kernels.KernelsApiService",
        "GetKernel",
        {"userName": "godlydonuts", "kernelSlug": slug.split("/")[1]},
    )
    assert draft["metadata"].get("currentVersionNumber", 0) == 0, (
        "Existing executable version prohibits UI save"
    )
    assert draft["blob"]["source"] == source, "Exact original draft source required"
    for n in ("isPrivate", "enableGpu", "enableInternet", "machineShape", "dockerImage"):
        assert (
            draft["metadata"].get(n, False if n == "enableInternet" else None) == row["request"][n]
        )
    out = {
        "status": "actual_partial_draft_original_version1_UI_commit_intent",
        "CPU_job": os.environ["SLURM_JOB_ID"],
        "source_git": v["source_git"],
        "slug": slug,
        "source_sha256": sha,
        "fresh_GPU_quota": quota,
        "prior_SaveKernel_repeated": False,
        "UI_commit_maximum": 1,
        "expected_first_version": 1,
        "competition_submission": False,
    }
    w.new_json(D / "draft-version1-launch-intent.json", out)
    with w.ledger(path, "godlydonuts") as db:
        current = json.loads(db.execute("SELECT value FROM meta WHERE key=?", (key,)).fetchone()[0])
        assert current == row
        row["partial_draft_version1_UI_commit_intent"] = out
        db.execute("UPDATE meta SET value=? WHERE key=?", (json.dumps(row, sort_keys=True), key))
        db.commit()
    print(json.dumps(out), flush=True)
    raise SystemExit(0)
