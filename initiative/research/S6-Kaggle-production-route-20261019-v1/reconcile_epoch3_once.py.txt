"""Two explicit CPU phases, observe then reconcile existing version; zero new pushes."""

import importlib.util
import json
import shlex
import subprocess
import sys
from pathlib import Path

R = Path.cwd()
D = Path(__file__).resolve().parent
L = R / "artifacts/S6-epoch3-SaveKernel-reconciliation-20261020-v1"
L.mkdir(mode=0o700, exist_ok=True)
phase = sys.argv[1]
assert phase in ("observe", "accept", "absence")
assert not (L / (phase + "-intent.json")).exists()
pin = subprocess.check_output(["git", "rev-parse", "HEAD"], text=True).strip()
for name in ("reconcile_epoch3.py.txt", "reconcile_epoch3_once.py.txt"):
    p = D / name
    assert p.read_bytes() == subprocess.check_output(
        ["git", "show", pin + ":" + p.relative_to(R).as_posix()]
    )
env = R / ".env"
assert env.stat().st_mode & 0o077 == 0
values = {
    k.strip(): z.strip().strip("\"'")
    for line in env.read_text().splitlines()
    for k, sep, z in [line.partition("=")]
    if sep and not line.lstrip().startswith("#")
}
v = {"source_git": pin, "phase": phase, "token": values["KAGGLE_API_TOKEN"]}
if phase == "accept":
    v["prior_observation"] = json.loads((D / "actual-epoch3-remote-observed.json").read_text())
s = importlib.util.spec_from_file_location(
    "transport", R / "initiative/research/paired-logit-head-gradient-20261018-v1/stdin_transport.py"
)
t = importlib.util.module_from_spec(s)
s.loader.exec_module(t)
with (L / (phase + "-intent.json")).open("x") as f:
    json.dump(
        {
            "phase": phase,
            "source_git": pin,
            "new_SaveKernel_or_submission_requests": 0,
            "canonical_ledger_write": phase == "accept",
            "automatic_retry": False,
        },
        f,
    )
a = [
    "srun",
    "--account=skattel",
    "--partition=normal",
    "--qos=normal",
    "--nodelist=evc33",
    "--ntasks=1",
    "--cpus-per-task=1",
    "--gres=none",
    "--mem=2G",
    "--time=00:05:00",
    "--time-min=00:02:00",
    "--job-name=rsna-E3-reconcile-" + phase,
    "env",
    "CUDA_VISIBLE_DEVICES=",
    "PYTHONDONTWRITEBYTECODE=1",
    "PYTHONNOUSERSITE=1",
    "OMP_NUM_THREADS=1",
    "/home/sa305415/rsna-knee/kaggle/management-cp312-v1/bin/python",
    "-c",
    t.COMMAND,
]
x = subprocess.run(
    ["ssh", "-S", "/tmp/rsna-research-20261001-stable.sock", "newton", shlex.join(a)],
    input=t.envelope((D / "reconcile_epoch3.py.txt").read_text(), v),
    capture_output=True,
    text=True,
    timeout=900,
)
(L / (phase + "-transport.json")).write_text(
    json.dumps({"returncode": x.returncode, "stdout": x.stdout, "stderr": x.stderr}, indent=2)
    + "\n"
)
assert x.returncode == 0, "Reconcile only the original intent/version; never repeat SaveKernel"
z = json.loads(x.stdout)
(
    D
    / {
        "observe": "actual-epoch3-remote-observed.json",
        "accept": "actual-push-epoch-3.json",
        "absence": "actual-epoch3-absence-observed.json",
    }[phase]
).write_text(json.dumps(z, indent=2) + "\n")
print(json.dumps(z), flush=True)
