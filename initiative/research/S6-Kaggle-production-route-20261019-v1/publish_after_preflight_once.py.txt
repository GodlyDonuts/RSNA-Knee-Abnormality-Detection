def cpu(node, program, payload, name, minutes, memory, runtime):
    args = [
        "srun",
        "--account=skattel",
        "--partition=normal",
        "--qos=normal",
        "--nodelist=evc" + str(node),
        "--ntasks=1",
        "--cpus-per-task=1",
        "--gres=none",
        "--mem=" + memory,
        "--time=00:" + str(minutes).zfill(2) + ":00",
        "--job-name=" + name,
        "env",
        "CUDA_VISIBLE_DEVICES=",
        "PYTHONDONTWRITEBYTECODE=1",
        "PYTHONNOUSERSITE=1",
        "OMP_NUM_THREADS=1",
        runtime,
        "-c",
        t.COMMAND,
    ]
    return subprocess.run(
        ["ssh", "-S", "/tmp/rsna-research-20261001-stable.sock", "newton", shlex.join(args)],
        input=t.envelope(program, payload),
        capture_output=True,
        text=True,
        timeout=(minutes + 5) * 60,
    )
