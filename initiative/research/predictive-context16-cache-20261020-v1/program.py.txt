# ruff: noqa: E402
"""Exact retained FP32 masked contexts for16 admitted raw studies, CPU only."""

import hashlib
import json
import os
import shutil
import sys
import time
from pathlib import Path

import torch

v = json.load(sys.stdin)
assert os.uname().nodename == "evc33" and not os.environ.get("SLURM_JOB_GPUS")
os.umask(0o077)
torch.set_num_threads(1)
torch.set_num_interop_threads(1)
D = Path("/tmp/rsna-predictive-context16-cache-20261020-v1")
assert not D.exists()
raw = Path("/tmp/rsna-predictive-raw16-cache-20261020-v1")
actual = json.loads((raw / "actual.json").read_text())
assert (
    actual["fitting_studies"] == 16
    and actual["both_masked_contexts_targets_supports_priors_bit_equal"]
)
full = Path("/tmp/rsna-predictive-full-raw-cache-20261020-v1")
fund = json.loads((full / "intent.json").read_text())
assert (
    fund["raw_cache_total_cap_bytes"] == 320 * 2**30
    and fund["future_model_and_publication_scope_bytes"] == 24 * 2**30
)
paid_raw = actual["cache_bytes"] + sum(p.stat().st_size for p in (full / "cache").glob("*.npz"))
assert (
    shutil.disk_usage("/tmp").free
    > (20 + 8 + 12 + 4 + 24 + 2) * 2**30
    + max(0, 320 * 2**30 - paid_raw)
    + (166 + 64 + 4 + 16 + 256 + 8) * 2**20
)
D.mkdir(mode=0o700)
(D / "intent.json").write_text(
    json.dumps(
        {
            "CPU_job": os.environ["SLURM_JOB_ID"],
            "source_git": v["source_git"],
            "exact_FP32_context_cap_bytes": 2 * 2**30,
            "controls_cap_bytes": 8 * 2**20,
            "all_old_funding_retained": True,
            "no_new_source_downloads": True,
            "zeroGPU": True,
            "automatic_retry": False,
        }
    )
    + "\n"
)
for name, s in v["files"].items():
    p = D / "code" / name
    assert p.resolve().is_relative_to((D / "code").resolve())
    assert hashlib.sha256(s.encode()).hexdigest() == v["sha256"][name]
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text(s)
sys.path.insert(0, str(D / "code"))
from rsna.competitive.predictive import cache, context_cache
from rsna.competitive.predictive.cases import assemble

rows = json.loads((raw / "cache-index.private.json").read_text())
assert len(rows) == 16
index = []
paid = 0
render_seconds = []
write_seconds = []
read_seconds = []
rng = torch.get_rng_state().clone()
start = time.monotonic()
for i, row in enumerate(rows):
    t = time.monotonic()
    groups, _ = cache.load(Path(row["path"]), row["proof"])
    contexts = assemble(groups, side=320, context_cap=24, classes=12)
    render_seconds.append(time.monotonic() - t)
    provenance = {
        "original_raw_cache_sha256": row["proof"]["sha256"],
        "original_source_revision": "52f995f4568e36bfa5fbbaf69a89af41226b4f5d",
        "side": 320,
        "context_cap": 24,
        "halo": 8,
        "classes": 12,
        "module_sha256": {
            k: h
            for k, h in v["sha256"].items()
            if k
            in (
                "rsna/competitive/predictive/inputs.py",
                "rsna/competitive/predictive/cases.py",
                "rsna/competitive/predictive/supports.py",
            )
        },
    }
    p = D / ("context-" + str(i).zfill(3) + ".private.npz")
    t = time.monotonic()
    proof = context_cache.write(
        p, contexts, provenance, cap_bytes=min(128 * 2**20, 2 * 2**30 - paid)
    )
    write_seconds.append(time.monotonic() - t)
    t = time.monotonic()
    restored = context_cache.load(p, proof, provenance=provenance)
    read_seconds.append(time.monotonic() - t)
    for left, right in zip(contexts, restored):
        for key in context_cache.KEYS:
            assert torch.equal(left[key], right[key])
        for key in ("indptr", "regions", "log_prior"):
            assert torch.equal(getattr(left["supports"], key), getattr(right["supports"], key))
    assert torch.equal(rng, torch.get_rng_state())
    paid += proof["bytes"]
    index.append({"uid": row["uid"], "path": str(p), "proof": proof, "provenance": provenance})
    del groups, contexts, restored
(D / "context-index.private.json").write_text(json.dumps(index) + "\n")
out = {
    "status": "actual16_exact_FP32_context_target_prior_cache_and_RNG_parity_accepted",
    "CPU_job": os.environ["SLURM_JOB_ID"],
    "source_git": v["source_git"],
    "studies": 16,
    "context_cap": 24,
    "source_descriptor_regions": actual["descriptor_regions"],
    "cache_bytes": paid,
    "uncached_load_and_render_seconds": sum(render_seconds),
    "independent_cache_write_SHA_seconds": sum(write_seconds),
    "cached_load_seconds": sum(read_seconds),
    "measured_CPU_load_render_speed_ratio": sum(render_seconds) / sum(read_seconds),
    "exact_both_view_images_targets_metadata_priors": True,
    "global_RNG_unchanged": True,
    "new_raw_source_downloads": 0,
    "model_encoder_or_optimizer_or_GPU": False,
    "clinical_accuracy": False,
    "elapsed_seconds": time.monotonic() - start,
    "private_index_sha256": hashlib.sha256(
        (D / "context-index.private.json").read_bytes()
    ).hexdigest(),
}
(D / "actual.json").write_text(json.dumps(out) + "\n")
print(json.dumps(out), flush=True)
