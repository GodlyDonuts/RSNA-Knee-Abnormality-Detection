"""Reuse the exact accepted direct TLS endpoints for a fresh32-case packet on26."""

import ast
import hashlib
import importlib.util
import json
import shlex
import subprocess
from pathlib import Path

R = Path.cwd()
D = Path(__file__).resolve().parent
OLD = R / "initiative/research/S6-earlier-H100-route-20261019-v1"
L = R / "artifacts/teacher-pair-earlier26-20261019-v1"
L.mkdir(mode=0o700, exist_ok=True)
assert not (L / "auxiliary-intent.json").exists()
git = subprocess.check_output(["git", "rev-parse", "HEAD"], text=True).strip()
original = {
    n: (OLD / n).read_text()
    for n in (
        "auxiliary_common.py.txt",
        "auxiliary_sender_v2.py.txt",
        "auxiliary_receiver_v2.py.txt",
        "auxiliary_v2_once.py.txt",
    )
}
for name, body in original.items():
    assert body.encode() == subprocess.check_output(
        ["git", "show", git + ":" + str((OLD / name).relative_to(R))]
    )
TLS = (
    R / "initiative/research/S6-Newton-student-production-20261019-v1/tls_resume.py.txt"
).read_text()
assert TLS.encode() == subprocess.check_output(
    [
        "git",
        "show",
        "21e0598b:initiative/research/S6-Newton-student-production-20261019-v1/tls_resume.py.txt",
    ]
)
source = ast.parse(original["auxiliary_v2_once.py.txt"])
program = next(
    ast.literal_eval(n.value)
    for n in source.body
    if isinstance(n, ast.Assign)
    and any(isinstance(t, ast.Name) and t.id == "program" for t in n.targets)
)
a = program.index("for job,state in ")
b = program.index("def command(", a)
program = program[:a] + program[b:]
program = (
    program.replace("evc48", "evc26")
    .replace("--partition=preemptable", "--partition=normal")
    .replace("--qos=preemptable", "--qos=normal")
)
receiver = (
    original["auxiliary_receiver_v2.py.txt"].replace("evc48", "evc26").replace("ivc48", "ivc26")
)
anchor = 'E = Path("/tmp/rsna-S6-earlier-H100-route-20261019-v1")'
assert receiver.count(anchor) == 1
receiver = receiver.replace(anchor, anchor + "\nE.mkdir(mode=0o700,exist_ok=True)")
sender = original["auxiliary_sender_v2.py.txt"].replace("ivc48", "ivc26")
for name, body in {"head": program, "receiver": receiver, "sender": sender}.items():
    compile(body, name, "exec")
payload = {
    "common": original["auxiliary_common.py.txt"],
    "TLS": TLS,
    "receiver": receiver,
    "sender": sender,
    "pin": git,
}
(L / "auxiliary-intent.json").write_text(
    json.dumps(
        {
            "source_git": git,
            "original_sources_sha256": {
                n: hashlib.sha256(s.encode()).hexdigest() for n, s in original.items()
            },
            "derived_head_sha256": hashlib.sha256(program.encode()).hexdigest(),
            "source": "evc33",
            "receiver": "evc26",
            "maximum_bytes": 2 * 2**30,
            "old_copies_or_owners_changed": False,
            "automatic_retry": False,
        },
        indent=2,
    )
    + "\n"
)
s = importlib.util.spec_from_file_location(
    "transport", R / "initiative/research/paired-logit-head-gradient-20261018-v1/stdin_transport.py"
)
t = importlib.util.module_from_spec(s)
s.loader.exec_module(t)
x = subprocess.run(
    [
        "ssh",
        "-S",
        "/tmp/rsna-research-20261001-stable.sock",
        "newton",
        "/home/sa305415/rsna-knee/kaggle/management-cp312-v1/bin/python -B -c "
        + shlex.quote(t.COMMAND),
    ],
    input=t.envelope(program, payload),
    capture_output=True,
    text=True,
    timeout=2400,
)
(L / "auxiliary-transport.json").write_text(
    json.dumps({"returncode": x.returncode, "stdout": x.stdout, "stderr": x.stderr}, indent=2)
    + "\n"
)
assert x.returncode == 0, (
    "Existing endpoint intent/terminal must be reconciled read-only, never retry"
)
out = json.loads(x.stdout)
(D / "actual-auxiliary.json").write_text(json.dumps(out, indent=2) + "\n")
print(json.dumps(out), flush=True)
