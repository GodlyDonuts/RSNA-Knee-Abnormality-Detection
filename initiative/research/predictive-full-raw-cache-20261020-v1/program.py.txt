@contextlib.contextmanager
def opener(row):
    url = (
        "https://huggingface.co/datasets/GodlyDonuts/RSNA-Data/resolve/"
        + revision
        + "/"
        + quote(row["path"], safe="/")
    )
    with requests.get(
        url, headers={"Authorization": "Bearer " + token}, timeout=(10, 90), stream=True
    ) as response:
        if response.status_code != 200:
            raise RuntimeError("Source HTTP status " + str(response.status_code))
        response.raw.decode_content = True
        yield response.raw

def one(item):
    assert not stop.is_set()
    i, c = item
    uid, rows = c["uid"], c["source_files"]
    charged = paid_raw + sum(p.stat().st_size for p in (D / "cache").glob("*.npz"))
    assert charged + 2 * 2**30 < 320 * 2**30
    assert (
        shutil.disk_usage("/tmp").free
        > (20 + 8 + 12 + 4 + 24) * 2**30
        + max(0, 320 * 2**30 - charged)
        + (166 + 64 + 4 + 16 + 256) * 2**20
    )
    with (D / "proofs" / ("case-" + str(i).zfill(4) + ".intent.json")).open("x") as f:
        json.dump(
            {
                "source_ordinal": i,
                "source_bytes": sum(r["bytes"] for r in rows),
                "original_source_table_sha256": hashlib.sha256(
                    json.dumps(rows, sort_keys=True, separators=(",", ":")).encode()
                ).hexdigest(),
                "automatic_retry": False,
            },
            f,
        )
    data, source_proof = helper["acquire"](
        D / "scratch",
        uid=uid,
        rows=rows,
        series_csv=c["series_csv"],
        cap=1024 * 2**20,
        workers=8,
        opener=opener,
    )
    ram = data.parent
    namespace = {
        **globals(),
        "uid": uid,
        "rows": rows,
        "ram": ram,
        "np": np,
        "pydicom": pydicom,
        "m": m,
    }
    exec(compile(decoder, "same_original_integer_decoder", "exec"), namespace)
    exec(compile(v["conversion"], "same_original_group_conversion", "exec"), namespace)
    sources = namespace["sources"]
    original = [RawGroup(normalized(g), g.aliases, g.acquisition, g.ordinal) for g in sources]
    provenance = {
        "original_source_revision": revision,
        "original_fitting_case_sha256": hashlib.sha256(uid.encode()).hexdigest(),
        "source_file_table_sha256": hashlib.sha256(
            json.dumps(rows, sort_keys=True, separators=(",", ":")).encode()
        ).hexdigest(),
        "source_DICOMs": len(rows),
        "source_DICOM_bytes": sum(r["bytes"] for r in rows),
    }
    p = D / "cache" / ("case-" + str(i).zfill(4) + ".private.npz")
    proof = write(p, sources, provenance, cap_bytes=1024 * 2**20)
    restored, check = load(p, proof)
    assert check == provenance and all(
        torch.equal(a.frames, b.frames) for a, b in zip(original, restored)
    )
    left = assemble(original, side=320, context_cap=8, classes=12)
    right = assemble(restored, side=320, context_cap=8, classes=12)
    for x, y in zip(left, right):
        for key in ("images", "targets", "observed", "token_metadata", "region_metadata"):
            assert torch.equal(x[key], y[key])
        for key in ("indptr", "regions", "log_prior"):
            assert torch.equal(getattr(x["supports"], key), getattr(y["supports"], key))
    proof_path = D / "proofs" / ("case-" + str(i).zfill(4) + ".source-proof.private.json.gz")
    compressed = gzip.compress(json.dumps(source_proof).encode())
    record = {
        "uid": uid,
        "path": str(p),
        "proof": proof,
        "source_proof_sha256": hashlib.sha256(compressed).hexdigest(),
        "original16_reused": False,
    }
    encoded = json.dumps(record) + "\n"
    with proof_lock:
        charge = len(compressed) + len(encoded.encode()) + 4096
        assert control_committed[0] + charge + 8 * 2**20 < 256 * 2**20
        control_committed[0] += charge
        proof_path.write_bytes(compressed)
        (D / "proofs" / ("case-" + str(i).zfill(4) + ".verified.private.json")).write_text(encoded)
    assert ram.parent == D / "scratch"
    shutil.rmtree(ram)
    return i, record


iterator = iter(pending)
with ThreadPoolExecutor(max_workers=2) as pool:
    active = {pool.submit(one, next(iterator)) for _ in range(2)}
    try:
        while active:
            done, active = wait(active, return_when=FIRST_COMPLETED)
            for future in done:
                i, row = future.result()
                results[i] = row
                with (D / "progress.private.json").open("w") as f:
                    json.dump(
                        {
                            "completed": len(results),
                            "total": 3887,
                            "cache_bytes": sum(r["proof"]["bytes"] for r in results.values()),
                            "descriptor_regions": sum(
                                r["proof"]["regions"] for r in results.values()
                            ),
                            "elapsed_seconds": time.monotonic() - start,
                        },
                        f,
                    )
                if len(results) % 100 == 0:
                    print(
                        json.dumps(
                            {
                                "status": "raw_fitting_cache_progress",
                                "completed": len(results),
                                "total": 3887,
                                "elapsed_seconds": time.monotonic() - start,
                            }
                        ),
                        flush=True,
                    )
                item = next(iterator, None)
                if item is not None:
                    active.add(pool.submit(one, item))
    except BaseException:
        stop.set()
        for future in active:
            future.cancel()
        raise
