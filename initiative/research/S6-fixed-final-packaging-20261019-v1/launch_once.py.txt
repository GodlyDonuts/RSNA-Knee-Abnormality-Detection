"""Start one finite packaging follower; no cluster mutation at launch."""

import hashlib
import json
import os
import subprocess
import sys
import time
from pathlib import Path

R = Path.cwd()
D = Path(__file__).resolve().parent
L = R / "artifacts/S6-fixed-final-packaging-20261019-v1"
L.mkdir(mode=0o700, parents=True, exist_ok=True)
os.umask(0o077)
assert not (L / "launch-intent.json").exists() and not (L / "owner-intent.json").exists()
pin = subprocess.check_output(["git", "rev-parse", "HEAD"], text=True).strip()
source = D / "follow_once.py.txt"
assert (
    subprocess.check_output(["git", "show", pin + ":" + source.relative_to(R).as_posix()])
    == source.read_bytes()
)
assert not (
    R / "artifacts/S6-earlier-H100-execution-20261019-v1/production-owner-v1/result.json"
).exists()
x = subprocess.run(["ps", "-p", "67269", "-o", "command="], capture_output=True, text=True)
assert (
    x.returncode == 0
    and "S6-earlier-H100-execution-20261019-v1/follow_production_once.py.txt" in x.stdout
)
with (L / "launch-intent.json").open("x") as f:
    json.dump(
        {
            "source_git": pin,
            "follower_sha256": hashlib.sha256(source.read_bytes()).hexdigest(),
            "maximum_new_followers": 1,
            "maximum_GPU_Dataset_or_competition_mutations": 0,
            "automatic_retry": False,
        },
        f,
        indent=2,
    )
with (L / "stdout.log").open("x") as out, (L / "stderr.log").open("x") as err:
    child = subprocess.Popen(
        [sys.executable, str(source)], cwd=R, stdout=out, stderr=err, start_new_session=True
    )
    for _ in range(50):
        if (L / "owner-intent.json").exists() or child.poll() is not None:
            break
        time.sleep(0.1)
assert (
    child.poll() is None and json.loads((L / "owner-intent.json").read_text())["PID"] == child.pid
), "Inspect original launch; never retry"
x = subprocess.run(["ps", "-p", str(child.pid), "-o", "command="], capture_output=True, text=True)
assert x.returncode == 0 and str(source) in x.stdout
actual = {
    "status": "actual_sole_CPU_packaging_follower_live_waiting_original_production",
    "PID": child.pid,
    "source_git": pin,
    "original_producer_PID": 67269,
    "maximum_new_CPU_jobs": 1,
    "new_GPU_Dataset_or_competition_mutations": 0,
    "automatic_retry": False,
}
(D / "actual-owner-started.json").write_text(json.dumps(actual, indent=2) + "\n")
print(json.dumps(actual), flush=True)
