def load(name):
    spec = importlib.util.spec_from_file_location(
        "actual_fixed_final_" + name, D / "source" / (name + ".py")
    )
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module


package, validator, compiler = load("package"), load("validate"), load("compiler")

assert torch.cuda.device_count() == 0 and str(torch.__version__) == "2.14.0+cu130"
outputs = {}


def read(path):
    return json.loads(path.read_text())


for role in package.ROLES:
