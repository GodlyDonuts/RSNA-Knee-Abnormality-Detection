import hashlib
import importlib.util
import json
import shlex
import subprocess
from pathlib import Path

R = Path.cwd()
D = Path(__file__).resolve().parent
L = R / "artifacts/S6-prefilled4-CPU-teacher-20261018-v1"
L.mkdir(mode=0o700, parents=True, exist_ok=True)
assert not (L / "intent.json").exists()
source = subprocess.check_output(["git", "rev-parse", "HEAD"], text=True).strip()
files = {}
for name in ("program.py.txt", "worker.py.txt"):
    text = (D / name).read_text()
    assert (
        subprocess.check_output(
            ["git", "show", source + ":" + D.relative_to(R).as_posix() + "/" + name], text=True
        )
        == text
    )
    files[name] = text
with (L / "intent.json").open("x") as f:
    json.dump(
        {
            "source_git": source,
            "studies": 4,
            "existing_prefill": "961960",
            "GPU": 0,
            "CPU": 9,
            "memory_GiB": 64,
            "wall_minutes": 30,
            "automatic_retry": False,
            "new_raw_source_download_or_render": False,
            "files": {n: hashlib.sha256(t.encode()).hexdigest() for n, t in files.items()},
        },
        f,
        indent=2,
    )
p = R / "initiative/research/paired-logit-head-gradient-20261018-v1/stdin_transport.py"
s = importlib.util.spec_from_file_location("t", p)
t = importlib.util.module_from_spec(s)
s.loader.exec_module(t)
a = [
    "srun",
    "--account=skattel",
    "--partition=preemptable",
    "--qos=preemptable",
    "--nodelist=evc33",
    "--ntasks=1",
    "--cpus-per-task=9",
    "--gres=none",
    "--mem=64G",
    "--time=00:30:00",
    "--job-name=rsna-S6-CPU-teacher4",
    "env",
    "CUDA_VISIBLE_DEVICES=",
    "PYTHONDONTWRITEBYTECODE=1",
    "PYTHONNOUSERSITE=1",
    "OMP_NUM_THREADS=2",
    "OPENBLAS_NUM_THREADS=1",
    "/tmp/rsna-environment-907632-7sr9a7i6/environment/bin/python",
    "-B",
    "-c",
    t.COMMAND,
]
x = subprocess.run(
    [
        "ssh",
        "-S",
        "/tmp/rsna-research-20261001-stable.sock",
        "-o",
        "BatchMode=yes",
        "newton",
        shlex.join(a),
    ],
    input=t.envelope(
        files["program.py.txt"],
        {
            "source_git": source,
            "worker": files["worker.py.txt"],
            "worker_sha256": hashlib.sha256(files["worker.py.txt"].encode()).hexdigest(),
        },
    ),
    capture_output=True,
    text=True,
)
with (L / "response.json").open("x") as f:
    json.dump({"returncode": x.returncode, "stdout": x.stdout, "stderr": x.stderr}, f, indent=2)
assert x.returncode == 0, x.stderr[-1000:]
out = json.loads(x.stdout)
with (D / "actual-complete.json").open("x") as f:
    json.dump(out, f, indent=2)
    f.write("\n")
print(json.dumps(out), flush=True)
