"""Execute one frozen phase on an owning zero-GPU CPU allocation; token via stdin."""

import hashlib
import importlib.util
import json
import shlex
import subprocess
import sys
from pathlib import Path

R = Path.cwd()
D = Path(__file__).resolve().parent
L = R / "artifacts/S6-Kaggle-target-recovery-20261019-v1"
L.mkdir(exist_ok=True)


def run(phase):
    assert phase in ("prepare", "create", "dataset_status", "push", "status")
    git = subprocess.check_output(["git", "rev-parse", "HEAD"], text=True).strip()
    for name in ("client.py.txt", "program.py.txt"):
        p = D / name
        assert p.read_bytes() == subprocess.check_output(
            ["git", "show", git + ":" + str(p.relative_to(R))]
        )
    if phase in ("prepare", "create", "push"):
        intent = L / (phase + "-intent.json")
        assert not intent.exists(), (
            "Existing mutation-phase intent: reconcile original result, never retry"
        )
        intent.write_text(
            json.dumps(
                {
                    "source_git": git,
                    "program_sha256": hashlib.sha256(
                        (D / "program.py.txt").read_bytes()
                    ).hexdigest(),
                    "phase": phase,
                    "automatic_retry": False,
                },
                indent=2,
            )
            + "\n"
        )
    env = R / ".env"
    assert env.stat().st_mode & 0o077 == 0
    values = {
        k.strip(): z.strip().strip("\"'")
        for line in env.read_text().splitlines()
        for k, sep, z in [line.partition("=")]
        if sep and not line.lstrip().startswith("#")
    }
    s = importlib.util.spec_from_file_location(
        "canonical_stdin_transport",
        R / "initiative/research/paired-logit-head-gradient-20261018-v1/stdin_transport.py",
    )
    t = importlib.util.module_from_spec(s)
    s.loader.exec_module(t)
    args = [
        "srun",
        "--account=skattel",
        "--partition=normal",
        "--qos=normal",
        "--nodelist=evc48",
        "--ntasks=1",
        "--cpus-per-task=1",
        "--gres=none",
        "--mem=2G",
        "--time=00:12:00",
        "--job-name=rsna-S6-Kaggle-target-" + phase,
        "env",
        "CUDA_VISIBLE_DEVICES=",
        "PYTHONDONTWRITEBYTECODE=1",
        "PYTHONNOUSERSITE=1",
        "/home/sa305415/rsna-knee/kaggle/management-cp312-v1/bin/python",
        "-B",
        "-c",
        t.COMMAND,
    ]
    x = subprocess.run(
        ["ssh", "-S", "/tmp/rsna-research-20261001-stable.sock", "newton", shlex.join(args)],
        input=t.envelope(
            (D / "program.py.txt").read_text(),
            {"phase": phase, "source_git": git, "token": values["KAGGLE_API_TOKEN"]},
        ),
        capture_output=True,
        text=True,
        timeout=900,
    )
    (L / (phase + "-transport.json")).write_text(
        json.dumps({"returncode": x.returncode, "stdout": x.stdout, "stderr": x.stderr}, indent=2)
        + "\n"
    )
    assert x.returncode == 0, (
        "Original phase stopped or observation uncertain: inspect exact ledger/journal, do not repeat mutations"
    )
    out = json.loads(x.stdout)
    if out["status"].startswith("waiting_"):
        if phase in ("prepare", "create", "push"):
            raise RuntimeError(
                "No mutation after capacity/claim wait; inspect original before explicit new admission"
            )
        return out
    (D / ("actual-" + phase + ".json")).write_text(json.dumps(out, indent=2) + "\n")
    return out


if __name__ == "__main__":
    print(json.dumps(run(sys.argv[1])), flush=True)
