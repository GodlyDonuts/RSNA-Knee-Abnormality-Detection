"""One CPU-only prepare/push or exact-version status, credential through stdin."""

import importlib.util
import json
import shlex
import subprocess
import sys
from pathlib import Path

R = Path.cwd()
D = Path(__file__).resolve().parent
L = R / "artifacts/selected-window-Kaggle-reader-20261020-v1"
L.mkdir(mode=0o700, exist_ok=True)
phase = sys.argv[1]
assert phase in ("prepare", "push", "status")
pin = subprocess.check_output(["git", "rev-parse", "HEAD"], text=True).strip()
reader = R / "rsna/competitive/licensed_selected_pixel_bags.py"
for p in (
    D / "run_once.py.txt",
    D / "cloud.py.txt",
    D / "compile.py.txt",
    D / "worker.py.txt",
    reader,
):
    assert p.read_bytes() == subprocess.check_output(
        ["git", "show", pin + ":" + p.relative_to(R).as_posix()]
    )
v = {"phase": phase, "source_git": pin}
if phase == "prepare":
    v.update(
        compiler=(D / "compile.py.txt").read_text(),
        worker=(D / "worker.py.txt").read_text(),
        reader=reader.read_text(),
    )
else:
    env = R / ".env"
    assert env.stat().st_mode & 0o077 == 0
    values = {
        k.strip(): z.strip().strip("\"'")
        for line in env.read_text().splitlines()
        for k, sep, z in [line.partition("=")]
        if sep and not line.lstrip().startswith("#")
    }
    v["token"] = values["KAGGLE_API_TOKEN"]
if phase == "push":
    v["verified_input"] = json.loads(
        (
            R
            / "initiative/research/selected-window-Kaggle-input-20261020-v1/actual-private-verified.json"
        ).read_text()
    )
    assert (
        v["verified_input"]["status"]
        == "actual_original_private_window_integrity_dataset_reconciled_all_objects_download_size_SHA_verified"
    )
    assert (D / "actual-prepare.json").is_file()
if phase in ("prepare", "push"):
    with (L / (phase + "-intent.json")).open("x") as f:
        json.dump(
            {
                "source_git": pin,
                "maximum_SaveKernel_calls": int(phase == "push"),
                "maximum_new_GPU_jobs": 0,
                "competition_entries": 0,
                "automatic_retry": False,
            },
            f,
        )
spec = importlib.util.spec_from_file_location(
    "transport", R / "initiative/research/paired-logit-head-gradient-20261018-v1/stdin_transport.py"
)
t = importlib.util.module_from_spec(spec)
spec.loader.exec_module(t)
args = [
    "srun",
    "--account=skattel",
    "--partition=normal",
    "--qos=normal",
    "--nodelist=evc33",
    "--ntasks=1",
    "--cpus-per-task=1",
    "--gres=none",
    "--mem=1G",
    "--time=00:05:00",
    "--job-name=rsna-selected-CPU-" + phase,
    "env",
    "CUDA_VISIBLE_DEVICES=",
    "PYTHONNOUSERSITE=1",
    "PYTHONDONTWRITEBYTECODE=1",
    "/home/sa305415/rsna-knee/kaggle/management-cp312-v1/bin/python",
    "-c",
    t.COMMAND,
]
x = subprocess.run(
    ["ssh", "-S", "/tmp/rsna-research-20261001-stable.sock", "newton", shlex.join(args)],
    input=t.envelope((D / "cloud.py.txt").read_text(), v),
    capture_output=True,
    text=True,
    timeout=900,
)
(L / (phase + "-transport.json")).write_text(
    json.dumps({"returncode": x.returncode, "stdout": x.stdout, "stderr": x.stderr}, indent=2)
    + "\n"
)
assert x.returncode == 0, (
    "Observe exact original intent/version; never repeat an uncertain mutation"
)
out = json.loads(x.stdout)
(D / ("actual-" + phase + ".json")).write_text(json.dumps(out, indent=2) + "\n")
print(json.dumps(out), flush=True)
