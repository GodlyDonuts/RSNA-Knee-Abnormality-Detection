def rng():
    return hashlib.sha256(
        torch.get_rng_state().numpy().tobytes()
        + pickle.dumps(np.random.get_state())
        + pickle.dumps(random.getstate())
    ).hexdigest()


before = rng()
old_time = fast_time = 0.0
start = time.monotonic()
for order, index in enumerate(indices):
    for mirror in (False, True):
        values = {}
        for name, reader in (
            (("original", original), ("selected", selected))
            if order % 2 == 0
            else (("selected", selected), ("original", original))
        ):
            t = time.monotonic()
            values[name] = reader.case(index, positions, mirror)
            elapsed = time.monotonic() - t
            if name == "original":
                old_time += elapsed
            else:
                fast_time += elapsed
        assert torch.equal(values["original"]["images"], values["selected"]["images"])
        for key in ("labels", "mask", "parent_logits"):
            torch.testing.assert_close(
                values["original"][key], values["selected"][key], atol=0, rtol=0, equal_nan=True
            )
assert rng() == before
