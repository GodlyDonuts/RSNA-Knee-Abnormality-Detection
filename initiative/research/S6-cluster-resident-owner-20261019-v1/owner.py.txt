def fresh_bounds():
    files = [p for p in D.rglob("*") if p.is_file()]
    paid = sum(p.stat().st_size for p in files)
    assert paid < 4 * 2**20 and len(files) < 256
    q = subprocess.check_output(["lfs", "quota", "-u", "sa305415", "/home/sa305415"], text=True)
    a = next(x.split() for x in q.splitlines() if x.strip().startswith("/home/sa305415"))
    assert (int(a[2]) - int(a[1])) * 1024 > 20 * 2**30 + max(0, 4 * 2**20 - paid) + 32 * 2**20
    assert int(a[6]) - int(a[5]) > 5000 + max(0, 256 - len(files))


def write(name, value, exclusive=True):
    raw = (json.dumps(value, sort_keys=True, indent=2) + "\n").encode()
    fresh_bounds()
    assert len(raw) < 64 * 1024
    if exclusive:
        with (D / name).open("xb") as f:
            f.write(raw)
            f.flush()
            os.fsync(f.fileno())
    else:
        p = D / (name + ".next")
        p.write_bytes(raw)
        p.replace(D / name)
