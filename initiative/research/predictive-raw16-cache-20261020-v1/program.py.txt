@contextlib.contextmanager
def opener(row):
    url = (
        "https://huggingface.co/datasets/GodlyDonuts/RSNA-Data/resolve/"
        + revision
        + "/"
        + quote(row["path"], safe="/")
    )
    with requests.get(
        url, headers={"Authorization": "Bearer " + token}, timeout=(10, 90), stream=True
    ) as response:
        if response.status_code != 200:
            raise RuntimeError("Source HTTP status " + str(response.status_code))
        response.raw.decode_content = True
        yield response.raw

        torch.equal(normalized(g), r.frames) for g, r in zip(sources, restored)
    )
    left = assemble(

        side=320,

    right = assemble(restored, side=320, context_cap=8, classes=12)
    for x, y in zip(left, right):
        for key in ("images", "targets", "observed", "token_metadata", "region_metadata"):
            assert torch.equal(x[key], y[key])
        for key in ("indptr", "regions", "log_prior"):
            assert torch.equal(getattr(x["supports"], key), getattr(y["supports"], key))
    proofs.append(
