# ruff: noqa: E402
"""Fresh real-MRI process for the isolated source-weight-zero objective."""

import ast
import functools
import hashlib
import importlib.util
import json
import os
import sys
from pathlib import Path

R = Path("/tmp/rsna-teacher-only-recovery45-20261020-v1")
F = Path("/tmp/rsna-teacher-pair-gradient-evc45-20261019-v1")
meta = json.loads((R / "staged.json").read_text())
assert os.uname().nodename == "evc45" and os.environ.get("SLURM_JOB_GPUS")
assert hashlib.sha256(Path(__file__).read_bytes()).hexdigest() == meta["worker_sha256"]
probe = (F / "gpu_probe.py").read_bytes()
assert hashlib.sha256(probe).hexdigest() == meta["gradient_worker_sha256"]
tree = ast.parse(probe)
boundary = next(
    i
    for i, n in enumerate(tree.body)
    if isinstance(n, ast.Assign)
    and any(isinstance(t, ast.Name) and t.id == "versions" for t in n.targets)
)
context = {
    "__name__": "original_accepted_zero_source_bootstrap",
    "__file__": str(F / "gpu_probe.py"),
}
exec(
    compile(
        ast.Module(body=tree.body[:boundary], type_ignores=[]),
        "original_unchanged_GPU_bootstrap",
        "exec",
    ),
    context,
)
sys.path.insert(0, str(F / "code"))
import torch

torch.set_num_threads(1)
from rsna.competitive.teacher_pair_train import fit as reference_fit

from rsna.schema import TARGETS

path = R / "teacher_only.py"
assert hashlib.sha256(path.read_bytes()).hexdigest() == meta["teacher_only_sha256"]
s = importlib.util.spec_from_file_location("rsna.competitive.teacher_only", path)
module = importlib.util.module_from_spec(s)
sys.modules[s.name] = module
s.loader.exec_module(module)
protocol = (R / "original_worker.py").read_text()
assert hashlib.sha256(protocol.encode()).hexdigest() == meta["original_worker_sha256"]
before = "from rsna.competitive.teacher_pair import Recipe"
assert protocol.count(before) == 1
protocol = protocol.replace(before, "from rsna.competitive.teacher_only import Recipe")
parts = ast.parse(protocol).body
parts = [
    n
    for n in parts
    if isinstance(n, (ast.Import, ast.ImportFrom))
    or isinstance(n, ast.FunctionDef)
    and n.name in ("run", "digest_state")
]
ns = {"__name__": "unchanged_original_recovery_with_explicit_ablation_recipe"}
exec(
    compile(
        ast.Module(body=parts, type_ignores=[]), "original_protocol_ablation_recipe_import", "exec"
    ),
    ns,
)
mapping = {u: i for i, u in enumerate(context["original"].uids)}
context["engineering_indices"] = [mapping[u] for u in context["targets"]["uids"]]
context.update(
    fit=functools.partial(module.fit, engine=reference_fit),
    targets=TARGETS,
    source_binding=meta["source_binding"],
)
work = Path(os.environ["TEACHER_ONLY_RECOVERY_WORK"])
assert (
    work.parent == Path("/dev/shm")
    and work.name.startswith("rsna-teacher-only-recovery-")
    and work.is_dir()
)
result = ns["run"](context, sys.argv[1], "pointwise", work)
assert result["trainable_parameters"] == 73136096 and result["device_type"] == "cuda"
print(json.dumps(result), flush=True)
