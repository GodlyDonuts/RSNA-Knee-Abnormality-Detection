"""Bounded raw-MRI -> unchanged S6 FP32 teacher, one resumable phase.

Only original fitting identities are accepted. Completed raw logits are fsynced
per study; no disease labels, reports or optimizer are read. CPU source/render
processes use stdin-only credentials. Public stdout contains aggregates only.
"""

GIB = 2**30


def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def atomic(path, value):
    path = Path(path)
    temp = path.with_suffix(path.suffix + ".temporary")
    with temp.open("w") as f:
        json.dump(value, f, sort_keys=True, allow_nan=False)
        f.flush()
        os.fsync(f.fileno())
    temp.replace(path)

    broker = contract.HeapBroker(80 * GIB)
    lock = threading.RLock()

    active = None
    watchdog_error = []

    def stop(reason):
        stopped.set()
        if threading.current_thread() is not threading.main_thread():
            os.kill(os.getpid(), signal.SIGTERM)
        raise Stop(reason)

    def write_pending():
        atomic(pending_path, pending)

    def check_resources(initial=False):
        mem = {
            line.split(":")[0]: int(line.split()[1]) * 1024
            for line in Path("/proc/meminfo").read_text().splitlines()
            if ":" in line
        }
        memory_floor = (188 if mode == "prefill" else 252) * GIB if initial else 32 * GIB
        tmpfs_floor = 92 * GIB if initial else 28 * GIB
        if mem["MemAvailable"] < memory_floor or shutil.disk_usage("/dev/shm").free < tmpfs_floor:
            raise RuntimeError("Host or tmpfs reserve breached")
        if sum(p.stat().st_size for p in ram.rglob("*") if p.is_file()) > 64 * GIB:
            raise RuntimeError("Bounded pipeline RAM scratch exceeded")
        if sum(p.stat().st_size for p in P.rglob("*") if p.is_file()) > 8 * 2**20:
            raise RuntimeError("Original stream control/output envelope exceeded")

    def watchdog():
        while not stopped.wait(2):
            try:
                check_resources()
            except BaseException as error:
                watchdog_error.append(type(error).__name__)
                os.kill(os.getpid(), signal.SIGTERM)
                return

    def terminate(signum, frame):
        stop("Signal or resource-watchdog stop; retain completed journal and buffers")

    signal.signal(signal.SIGTERM, terminate)
    signal.signal(signal.SIGINT, terminate)

    class Slot:
        def __init__(self, n):
            self.n = n
            self.root = ram / ("slot-%02d" % n)
            self.root.mkdir(mode=0o700, exist_ok=True)
            self.err = (run / ("worker-%02d.private.log" % n)).open("x")
            env = {
                k: v
                for k, v in os.environ.items()
                if not any(s in k.upper() for s in ("TOKEN", "SECRET", "PASSWORD", "AUTHORIZATION"))
            }
            env.update(
                CUDA_VISIBLE_DEVICES="",
                OMP_NUM_THREADS="2",
                OPENBLAS_NUM_THREADS="1",
                MKL_NUM_THREADS="2",
            )
            self.child = subprocess.Popen(
                [sys.executable, "-B", str(E / "render_worker.py")],
                stdin=subprocess.PIPE,
                stdout=subprocess.PIPE,
                stderr=self.err,
                text=True,
                env=env,
            )
            self.child.stdin.write(
                json.dumps(
                    {
                        "scratch_root": str(self.root),
                        "HF_token": secret["HF_token"],
                        "slot_cap_bytes": 16 * GIB,
                        "source_buffer_helper": str(E / "source_buffer.py"),
                        "source_buffer_helper_sha256": staged["files"]["source_buffer.py"],
                    }
                )
                + "\n"
            )
            self.child.stdin.flush()

        def render(self, uid):
            request = dict(records[uid])
            lease = 0
            with lock:
                previous = pending.get(uid, {})
                owned = previous.get("owned_scratch")
                if owned:
                    q = Path(owned).resolve()
                    assert q.is_relative_to(ram) and q.name.startswith("fitting-study-")
                    ready_file = q / "render-result.private.json"
                    if ready_file.exists():
                        value = json.loads(ready_file.read_text())
                        assert value["uid"] == uid and Path(value["tensor_path"]).is_file()
                        return value
                    assert (q / "source-proof.private.json").exists(), (
                        "Partial source transport needs explicit reconciliation, never automatic replay"
                    )
                    # Reuse verified original bytes, irrespective of previous worker slot.
                    request["resume_owned"] = str(q)
                pending[uid] = {**previous, "slot": self.n, "state": "source_requested"}
                write_pending()
            try:
                self.child.stdin.write(json.dumps(request) + "\n")
                self.child.stdin.flush()
                line = self.child.stdout.readline()
                if not line:
                    stop("CPU source worker exited; preserve private logs")
                ready = json.loads(line)
                if ready.get("stage") != "source_ready" or ready.get("uid") != uid:
                    stop("CPU source acquisition/identity admission failed")
                with lock:
                    pending[uid] = {
                        "slot": self.n,
                        "state": "source_verified",
                        "owned_scratch": ready["owned_scratch"],
                    }
                    write_pending()
                broker.acquire(ready["decoder_heap_bytes"], stopped)
                lease = ready["decoder_heap_bytes"]
                self.child.stdin.write(json.dumps({"uid": uid, "render": True}) + "\n")
                self.child.stdin.flush()
                line = self.child.stdout.readline()
                if not line:
                    stop("CPU renderer exited; preserve private logs")
                value = json.loads(line)
                if value.get("error") is not None or value.get("uid") != uid:
                    stop("Original CPU MRI renderer stopped")
                with lock:
                    pending[uid] = {
                        "slot": self.n,
                        "state": "rendered",
                        "owned_scratch": value["owned_scratch"],
                    }
                    write_pending()
                return value
            finally:
                if lease:
                    broker.release(lease)

        def close(self):
            if self.child.poll() is None:
                self.child.terminate()
            try:
                self.child.wait(timeout=10)
            except subprocess.TimeoutExpired:
                self.child.kill()
                self.child.wait()
            self.err.close()

    def retire_active():
        nonlocal active
        if active is None:
            return
        uid = active["uid"]
        if uid not in journal.rows:
            stop("Previous MRI forward did not produce a durable finite logit record")
        owned = Path(active["owned_scratch"]).resolve()
        assert owned.is_relative_to(ram) and owned.name.startswith("fitting-study-")
        # Only new owned transient source/render buffers; original assets/index remain untouched.
        shutil.rmtree(owned)
        with lock:
            pending.pop(uid, None)
            write_pending()
        active = None
        gc.collect()

    def pump():
        nonlocal next_submit
        while len(futures) < 4 and next_submit < len(todo):
            i = next_submit
            uid = todo[i]
            futures[i] = pool.submit(slots[i % 4].render, uid)
            next_submit += 1

    CT = None
    original_net = None
    forward_observer = None
    try:
        check_resources(initial=True)
        slots.extend(Slot(i) for i in range(4))
        thread = threading.Thread(target=watchdog, daemon=True)
        thread.start()
        if mode == "prefill":
            pump()
            prepared = [future.result(timeout=1200) for future in futures.values()]
            assert len(prepared) == 4 and len({x["uid"] for x in prepared}) == 4
            pool.shutdown(wait=True)
            result = {
                "status": "actual_first4_new_fitting_MRI_studies_rendered_and_retained_once",
                "Slurm_job": os.environ["SLURM_JOB_ID"],
                "source_git": staged["source_git"],
                "mode": mode,
                "new_studies": 4,
                "new_DICOM_files": sum(x["source_files"] for x in prepared),
                "new_windows": sum(x["windows"] for x in prepared),
                "new_source_bytes": sum(x["source_bytes"] for x in prepared),
                "completed_teacher_rows": len(journal.rows),
                "new_disease_inference_or_GPU": False,
                "source_indices_or_original_32_inference_repeated": False,
                "seconds": time.perf_counter() - start,
            }
            atomic(phase_dir / "terminal.json", result)
            print(json.dumps(result), file=console, flush=True)
            succeeded = True
            return
        spec = importlib.util.spec_from_file_location(
            "original_S6_stream_classifier", B / "source-tree-v1/launcher.private.py"

            hashlib.sha256(inspect.getsource(launch.CI.main).encode()).hexdigest()
            == "7dac0d1616c2b9fc6e1f32a1846f03c3b5a051fd41665c9f15f6b185131d8946"
        )

        class StreamInputs(torch.utils.data.Dataset):
            def __init__(self, data, split, ids, *args, **kwargs):
                if split != "train" or list(ids) != todo:
                    stop("Unexpected source identity/recovery path in original classifier")
                self.uids = list(ids)

            def __len__(self):
                return len(self.uids)

            def __getitem__(self, i):
                nonlocal active
                retire_active()
                pump()
                value = futures.pop(i).result(timeout=2400)
                if value["uid"] != self.uids[i]:
                    stop("Prefetched study identity differs")
                with np.load(value["tensor_path"], allow_pickle=False) as z:
                    images = z["images"]
                    meta = z["meta"]
                if (
                    images.dtype != np.uint8
                    or images.shape != (value["windows"], 3, 320, 320)
                    or meta.shape != (len(images), 6)
                    or meta.dtype != np.float32
                ):
                    stop("Rendered MRI tensor contract differs")
                if (
                    hashlib.sha256(memoryview(images).cast("B")).hexdigest()
                    != value["image_tensor_sha256"]
                    or hashlib.sha256(memoryview(meta).cast("B")).hexdigest()
                    != value["metadata_tensor_sha256"]
                ):
                    stop("Private rendered tensor digest differs before GPU consumption")
                active = value
                return {
                    "uid": value["uid"],
                    "error": None,
                    "images": torch.from_numpy(images),
                    "meta": torch.from_numpy(meta),
                    "anchor": value["anchor"],
                    "side": None,
                    "seconds": value["seconds"],
                }

        launch.CI.CStudies = StreamInputs
        launch.install_paths()
        import compact_train as CT

            net_calls.add(id(module))

                active is None

            str(B),

            geometry["backbone"],

            str(geometry["size"]),
            "--crop",
            str(geometry["crop"]),
            "--views",
            geometry["views"],
            "--slices",
            str(geometry["slices"]),
            "--phys-mm",
            str(geometry["phys_mm"]),
            "--eval-stride",
            str(geometry["eval_stride"]),

            str(B / "source-tree-v1/anatomy.pt"),

            future.result(timeout=1200)
        pool.shutdown(wait=True)

        old_receipt = json.loads((B / "H100-teacher8-20261018-v1/fp32/receipt.json").read_text())

            target = P / "teacher-final.private.npz"
            assert not target.exists()
            np.savez_compressed(
                target, uids=np.asarray(uids), logits=logits, probabilities=probabilities
            )
        result = {
            "status": "actual_complete_S6_fitting_teacher_ready_for_separate_target_admission"
            if complete
            else "actual_planned_partial_teacher_prefix_preserved",
            "Slurm_job": os.environ["SLURM_JOB_ID"],
            "source_git": staged["source_git"],
            "phase": phase,

            "original_main_source_unmodified": True,

            "GPU": torch.cuda.get_device_name(0),
            "GPU_count": 1,

            "seconds": time.perf_counter() - start,

        )
        (run / "failure-traceback.private.txt").write_text(traceback.format_exc())
        raise
    finally:
        stopped.set()
        for slot in slots:
            slot.close()
        pool.shutdown(wait=False, cancel_futures=True)
        if forward_observer is not None:
            forward_observer.remove()
        if succeeded:
            assert (
                json.loads((execution_lock / "owner.json").read_text())["Slurm_job"]
                == os.environ["SLURM_JOB_ID"]
            )
            (execution_lock / "owner.json").unlink()
            execution_lock.rmdir()


if __name__ == "__main__":
    main()
