"""Both assigned ordinary-node GPUs; isolate each visible UUID without resets."""

import json
import os
import subprocess
import sys

import torch

assert os.uname().nodename == "evc47" and os.environ.get("SLURM_JOB_GPUS")
assert os.environ.get("SLURM_GPUS_ON_NODE") == "2"
mode = sys.argv[1]
assert mode in ("batch", "step")
x = subprocess.run(
    ["nvidia-smi", "--query-gpu=index,uuid,name", "--format=csv,noheader"],
    text=True,
    capture_output=True,
    timeout=20,
)
rows = [r.split(",") for r in x.stdout.strip().splitlines() if "," in r]
assert len(rows) <= 2
results = []
code = """import json,torch
out={"count":torch.cuda.device_count()}
try:
 assert out["count"]==1
 out["name"]=torch.cuda.get_device_name(0)
 out["sum"]=float(torch.ones(4,device="cuda").sum())
 torch.cuda.synchronize()
 out["context_pass"]=True
except Exception as e:
 out["context_pass"]=False;out["error_type"]=type(e).__name__;out["error"]=str(e)[:1000]
print(json.dumps(out))
"""
for row in rows:
    index, uuid, name = [v.strip() for v in row]
    assert uuid.startswith("GPU-") and "H100" in name
    child = subprocess.run(
        [sys.executable, "-B", "-c", code],
        env={**os.environ, "CUDA_VISIBLE_DEVICES": uuid},
        text=True,
        capture_output=True,
        timeout=45,
    )
    results.append(
        {
            "allocated_index": index,
            "uuid": uuid,
            "returncode": child.returncode,
            "result": json.loads(child.stdout) if child.returncode == 0 else None,
            "error": child.stderr[-1000:],
        }
    )
print(
    json.dumps(
        {
            "mode": mode,
            "GPU_job": os.environ["SLURM_JOB_ID"],
            "environment": {
                n: os.environ.get(n)
                for n in (
                    "CUDA_VISIBLE_DEVICES",
                    "SLURM_JOB_GPUS",
                    "SLURM_STEP_GPUS",
                    "SLURM_GPUS_ON_NODE",
                )
            },
            "default_torch_count": torch.cuda.device_count(),
            "NVML_returncode": x.returncode,
            "NVML_inventory": x.stdout.strip().splitlines(),
            "authorized_per_GPU_contexts": results,
            "resets_driver_changes_MRI_models_or_optimizers": False,
        }
    ),
    flush=True,
)
