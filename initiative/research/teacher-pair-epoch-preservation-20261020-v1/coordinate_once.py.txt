"""Single CPU publisher coordinator; credential only RAM/SSH stdin, never argv."""

import hashlib
import importlib.util
import json
import shlex
import subprocess
import sys
from pathlib import Path

R = Path.cwd()
D = Path(__file__).resolve().parent
L = R / "artifacts/teacher-pair-epoch-preservation-20261020-v1"
L.mkdir(mode=0o700, exist_ok=True)
mode, role, epoch = sys.argv[1], sys.argv[2], int(sys.argv[3])
assert (
    mode in ("inspect", "publish") and role in ("pointwise", "teacher_rank") and epoch in (1, 2, 3)
)
pin = subprocess.check_output(["git", "rev-parse", "HEAD"], text=True).strip()
for name in ("coordinate_once.py.txt", "publish.py.txt"):
    p = D / name
    assert p.read_bytes() == subprocess.check_output(
        ["git", "show", pin + ":" + p.relative_to(R).as_posix()]
    )
program = (D / "publish.py.txt").read_text()
v = {
    "mode": mode,
    "role": role,
    "epoch": epoch,
    "source_git": pin,
    "publisher_SHA": hashlib.sha256(program.encode()).hexdigest(),
}
if mode == "publish":
    admission = json.loads((D / "actual-publisher-admission.json").read_text())
    assert admission["publisher_SHA"] == v["publisher_SHA"] and admission["GPU_job"] == "975297"
    intent = L / (role + "-" + str(epoch) + "-intent.json")
    assert not intent.exists(), "Existing original attempt requires read-only reconciliation"
    env = R / ".env"
    assert env.stat().st_mode & 0o077 == 0
    values = {
        k.strip(): z.strip().strip("\"'")
        for line in env.read_text().splitlines()
        for k, sep, z in [line.partition("=")]
        if sep and not line.lstrip().startswith("#")
    }
    v["token"] = values["HF_TOKEN"]
    with intent.open("x") as f:
        json.dump(
            {
                "role": role,
                "epoch": epoch,
                "source_git": pin,
                "publisher_SHA": hashlib.sha256(program.encode()).hexdigest(),
                "maximum_new_CPU_publications": 1,
                "GPU_or_competition_mutations": 0,
                "automatic_retry": False,
            },
            f,
            indent=2,
        )
s = importlib.util.spec_from_file_location(
    "transport", R / "initiative/research/paired-logit-head-gradient-20261018-v1/stdin_transport.py"
)
t = importlib.util.module_from_spec(s)
s.loader.exec_module(t)
args = [
    "srun",
    "--account=skattel",
    "--partition=normal",
    "--qos=normal",
    "--nodelist=evc48",
    "--ntasks=1",
    "--cpus-per-task=1",
    "--gres=none",
    "--mem=4G",
    "--time=01:45:00" if mode == "publish" else "--time=00:02:00",
    "--job-name=rsna-pair-" + mode + "-" + role + "-e" + str(epoch),
    "env",
    "CUDA_VISIBLE_DEVICES=",
    "PYTHONDONTWRITEBYTECODE=1",
    "PYTHONNOUSERSITE=1",
    "OMP_NUM_THREADS=1",
    "/tmp/rsna-environment-971427-pyk8rld8/environment/bin/python",
    "-B",
    "-c",
    t.COMMAND,
]
x = subprocess.run(
    [
        "ssh",
        "-S",
        "/tmp/rsna-research-20261001-stable.sock",
        "-o",
        "BatchMode=yes",
        "newton",
        shlex.join(args),
    ],
    input=t.envelope(program, v),
    capture_output=True,
    text=True,
    timeout=6600 if mode == "publish" else 600,
)
(L / (role + "-" + str(epoch) + "-" + mode + "-transport.json")).write_text(
    json.dumps({"returncode": x.returncode, "stdout": x.stdout, "stderr": x.stderr}, indent=2)
    + "\n"
)
assert x.returncode == 0, (
    "Inspect original journal and actual Slurm state; no automatic mutation retry"
)
z = json.loads(x.stdout)
if mode == "publish":
    assert (
        z["status"]
        == "actual_teacher_pair_immutable_epoch_private_all_objects_download_size_SHA_verified"
    )
    with (D / ("actual-" + role + "-epoch-" + str(epoch) + ".json")).open("x") as f:
        json.dump(z, f, indent=2)
if mode == "admit":
    assert (
        z["status"]
        == "actual_running_production_source_recipe_and_private_publisher_scope_admitted"
    )
    (D / "actual-publisher-admission.json").write_text(json.dumps(z, indent=2) + "\n")
print(json.dumps(z), flush=True)
