def sha(path):
    h = hashlib.sha256()
    with path.open("rb") as f:
        for chunk in iter(lambda: f.read(4 * 2**20), b""):
            h.update(chunk)
    return h.hexdigest()


def validate_binding(snapshot, run, meta, role, epoch, recipe):
    if (
        role not in ("pointwise", "teacher_rank")
        or type(epoch) is not int
        or epoch not in (1, 2, 3)
    ):
        raise ValueError("Exact production objective and epoch required")
    if set(snapshot) != {"binding", "epoch", "completed", "model"}:
        raise ValueError("Complete original immutable model snapshot required")
    b = snapshot["binding"]
    required = {
        "parent_checkpoint_sha256": PARENT,
        "source_binding": meta["source_binding"],
        "initial_state_sha256": INITIAL,
        "fitting_only": True,
        "protected_or_gold_used": False,
        "diagnostic_only": False,
        "matched_two_view_three_epoch_comparison": True,
        "version": "full_encoder_two_view_teacher_pair_risk_v1",
        "teacher_proposal_sha256": PROPOSAL,
        "recipe": recipe,
        "population": 3887,
        "population_sha256": COHORT,
        "world_size": 1,
        "anatomical_views_per_study": 2,
        "study_coupled_batch_loss": True,
        "device_type": "cuda",
        "precision": "FP16_autocast_FP32_loss_dynamic_scaler",
        "BN_running_statistics": "frozen_at_accepted_parent",
        "total_steps": 1458,
    }
    if b != run.get("binding") or any(b.get(k) != value for k, value in required.items()):
        raise ValueError(
            "Exact original production recipe/source/population/runtime binding required"
        )
    if (
        recipe.get("objective") != role
        or snapshot["epoch"] != epoch
        or snapshot["completed"] != 486 * epoch
    ):
        raise ValueError("Exact objective and completed epoch boundary required")
    if meta["source_git"] != SOURCE or b["source_binding"].get("actual_GPU_recovery") != "974940":
        raise ValueError("Original full production and actual GPU recovery required")
    if not isinstance(b.get("torch_version"), str) or "H100" not in b.get("device_name", ""):
        raise ValueError("Actual original H100 runtime required")
    return b


def signature(path):
    s = path.stat()
    if path.is_symlink() or not path.is_file() or s.st_uid != os.getuid() or s.st_mode & 0o077:
        raise ValueError("Original owner-only regular model snapshot required")
    return (s.st_dev, s.st_ino, s.st_size, s.st_mtime_ns, s.st_ctime_ns)


def budget():
    paid = sum(
        p.stat().st_size
        for p in ROOT.rglob("*")
        if p.is_file() and not p.is_symlink() and "inputs" not in p.relative_to(ROOT).parts
    )
    if paid + 512 * 2**20 >= 24 * 2**30:
        raise ValueError(
            "Original24GiB model/publication scope including temporary download required"
        )
    if (
        shutil.disk_usage("/tmp").free
        <= (20 + 87 + 12) * 2**30 + max(0, 24 * 2**30 - paid) + 64 * 2**20
    ):
        raise ValueError("Original retained node reserves after all liabilities required")
    q = subprocess.check_output(["lfs", "quota", "-u", "sa305415", "/home/sa305415"], text=True)
    row = next(x.split() for x in q.splitlines() if x.strip().startswith("/home/sa305415"))
    if (int(row[2]) - int(row[1])) * 1024 <= 20 * 2**30 + 80 * 2**20 or int(row[6]) - int(
        row[5]
    ) <= 5000 + 128 + 4:
        raise ValueError("Original shared reserve plus funded metadata required")
