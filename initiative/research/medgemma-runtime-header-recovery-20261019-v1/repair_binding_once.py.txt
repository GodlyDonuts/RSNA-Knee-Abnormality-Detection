"""One distinct GPU dispatch after a proved zero-runtime CPU-binding failure."""

import hashlib
import importlib.util
import json
import shlex
import subprocess
from pathlib import Path

R = Path.cwd()
D = Path(__file__).resolve().parent
L = R / "artifacts/medgemma-runtime-header-recovery-20261019-v1"
L.mkdir(mode=0o700, exist_ok=True)
assert not (L / "binding-pilot-dispatch-intent.json").exists()
pin = subprocess.check_output(["git", "rev-parse", "HEAD"], text=True).strip()
p = D / "repair_binding_stage.py.txt"
program = p.read_text()
assert program == subprocess.check_output(
    ["git", "show", pin + ":" + p.relative_to(R).as_posix()], text=True
)
with (L / "binding-pilot-dispatch-intent.json").open("x") as f:
    json.dump(
        {
            "source_git": pin,
            "program_sha256": hashlib.sha256(program.encode()).hexdigest(),
            "same_model_prompt_images_and_private_headers": True,
            "original_zero_runtime_failed_GPU_job": "970072",
            "existing_dependency_cap_bytes": 128 * 2**20,
            "maximum_new_GPU_jobs": 1,
            "new_system_package_installation": False,
            "automatic_retry": False,
        },
        f,
        indent=2,
    )
s = importlib.util.spec_from_file_location(
    "bounded", R / "initiative/research/paired-logit-head-gradient-20261018-v1/stdin_transport.py"
)
t = importlib.util.module_from_spec(s)
s.loader.exec_module(t)
a = [
    "srun",
    "--account=skattel",
    "--partition=preemptable",
    "--qos=preemptable",
    "--nodelist=evc33",
    "--ntasks=1",
    "--cpus-per-task=1",
    "--gres=none",
    "--mem=2G",
    "--time=00:10:00",
    "--job-name=rsna-MedGemma-CPU-binding-repair",
    "env",
    "CUDA_VISIBLE_DEVICES=",
    "PYTHONDONTWRITEBYTECODE=1",
    "PYTHONNOUSERSITE=1",
    "/tmp/rsna-environment-907632-7sr9a7i6/environment/bin/python",
    "-B",
    "-c",
    t.COMMAND,
]
x = subprocess.run(
    [
        "ssh",
        "-S",
        "/tmp/rsna-research-20261001-stable.sock",
        "-o",
        "BatchMode=yes",
        "newton",
        shlex.join(a),
    ],
    input=t.envelope(program, {"source_git": pin}),
    capture_output=True,
    text=True,
    timeout=730,
)
(L / "binding-pilot-dispatch-transport.json").write_text(
    json.dumps({"returncode": x.returncode, "stdout": x.stdout, "stderr": x.stderr}, indent=2)
    + "\n"
)
assert x.returncode == 0, (
    "Preserve original acquisition/compile evidence; no retry: " + x.stderr[-1800:]
)
v = json.loads(x.stdout)
(D / "actual-binding-corrected-GPU-submitted.json").write_text(json.dumps(v, indent=2) + "\n")
print(json.dumps(v), flush=True)
