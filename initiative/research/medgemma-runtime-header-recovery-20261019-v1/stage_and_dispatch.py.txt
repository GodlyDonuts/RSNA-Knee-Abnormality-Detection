        and hashlib.sha256(p.read_bytes()).hexdigest() == item["sha256"]

    a[5]

old = 'assert proof["pilot_sha256"] == hashlib.sha256(Path(__file__).read_bytes()).hexdigest()'
new = 'assert proof["pilot_sha256"] == hashlib.sha256(Path("/tmp/rsna-medgemma15-first-packet-20261019-v1/pilot.py").read_bytes()).hexdigest()'
assert source.count(old) == 1

source = source.replace(old, 'OUTPUT = R / "actual-pilot-headers-v1"')
derived_tree = ast.parse(source)
changed = [
    (a, b)
    for a, b in zip(original_tree.body, derived_tree.body, strict=True)
    if ast.dump(a) != ast.dump(b)
]
assert len(changed) == 2
assert sum(isinstance(a, ast.Assert) for a, _ in changed) == 1
assert (
    sum(
        isinstance(a, ast.Assign)
        and len(a.targets) == 1
        and isinstance(a.targets[0], ast.Name)
        and a.targets[0].id == "OUTPUT"
        for a, _ in changed
    )
    == 1
)
