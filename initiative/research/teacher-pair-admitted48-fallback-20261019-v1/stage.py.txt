    ["sacct", "-X", "-n", "-P", "-j", "972928", "--format=JobIDRaw,State,ExitCode"], text=True
)
assert v["failed26"]["accepted"] is False and not v["failed26"]["partial_phase_aggregates"]

assert shutil.disk_usage("/tmp").free > (20 + 87 + 12) * 2**30 + 32 * 2**20
q = subprocess.check_output(["lfs", "quota", "-u", "sa305415", "/home/sa305415"], text=True)
a = next(x.split() for x in q.splitlines() if x.strip().startswith("/home/sa305415"))

F = Path("/tmp/rsna-teacher-pair-gradient-admitted48-20261019-v3")
assert not F.exists()
for name, h in prior["original_source_sha256"].items():
    assert hashlib.sha256((old / "code" / name).read_bytes()).hexdigest() == h
for name, body in v["files"].items():
    assert hashlib.sha256(body.encode()).hexdigest() == v["hashes"][name]
F.mkdir(mode=0o700)
shutil.copytree(old / "code", F / "code", symlinks=True)
for name, body in v["files"].items():
    p = F / name
    if p.is_symlink():
        p.unlink()
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text(body)
sys.path[:0] = [str(F / "code")]
from rsna.competitive.teacher_pair_inputs32 import load32

A = Path("/tmp/rsna-S6-earlier-H100-route-20261019-v1/auxiliary-v2")
old_source = (A / "old_recovery/recovery.private.py").read_text()
assert (
    hashlib.sha256(old_source.encode()).hexdigest()
    == "a45d2a00322d84e412c4bd28235fc49d81a42f3ec3e733f9b7da76f68ec57b39"
)
constants = {
    n.targets[0].id: ast.literal_eval(n.value)
    for n in ast.parse(old_source).body
    if isinstance(n, ast.Assign)
    and len(n.targets) == 1
    and isinstance(n.targets[0], ast.Name)
    and n.targets[0].id == "TARGETS"
}
targets = json.loads(base64.b64decode(constants["TARGETS"], validate=True))
data, proposals, binding = load32(A, targets, None)

(F / "private-input-binding.json").write_text(json.dumps({"indices": list(range(8))}) + "\n")

    "new_source_sha256": v["hashes"],
    "original_source_sha256": prior["original_source_sha256"],
    "proposal_array_sha256": binding["selected_proposal_array_sha256"],
    "original_failed_job": "972928",
    "input_binding": binding,

    Path("/tmp/rsna-medgemma15-runtime-headers-20261019-v1/verified.json").read_text()

    env[name] = str(p)
