def load(path, name):
    s = importlib.util.spec_from_loader(name, SourceFileLoader(name, str(path)))
    m = importlib.util.module_from_spec(s)
    s.loader.exec_module(m)
    return m

C = R / "initiative/research/teacher-pair-full-encoder-20261019-v1"

    "/tmp/rsna-teacher-pair-gradient-admitted48-20261019-v3",
).replace('"evc26"', '"evc48"')
original = (C / "gpu_probe.py.txt").read_text()
