def validate_rejected_draft(raw, intent, metadata, source):
    assert hashlib.sha256(raw).hexdigest() == REJECT_SHA
    assert json.loads(raw) == {'error': 'Maximum batch GPU session count of 2 reached.'}
    assert intent['slug'] == DRAFT_SLUG
    assert metadata.get('id') == 139417644 and metadata.get('currentVersionNumber', 0) == 0
    assert hashlib.sha256(source.encode()).hexdigest() == DRAFT_SHA
    return True


def phase_requirement(mode, lr_fit_future, other_future, paired_future):
    assert mode in ('qualification', 'competition')
    assert all(type(n) is int and n >= 0 for n in (lr_fit_future, other_future, paired_future))
    assert paired_future == 0
    # Only the immediately admitted execution is funded now. Competition execution
    # remains forbidden until qualification passes AND its fresh 9h cap is funded.
    return lr_fit_future + other_future + 3600 + (1800 if mode == 'qualification' else 32400)


def other_future(root, rejected_live_verified):
    total = 0
    for mode, cap in (('qualification', 1800), ('competition', 32400)):
        intent, response = root / (mode + '-push-intent.json'), root / (mode + '-push-response.json')
        if intent.exists() and not response.exists():
            raise ValueError('Uncertain original execution requires reconciliation')
        if not response.exists():
            continue
        assert intent.exists()
        raw = response.read_bytes()
        if root == FINDING and mode == 'competition':
            assert rejected_live_verified and hashlib.sha256(raw).hexdigest() == REJECT_SHA
            assert json.loads(raw) == {'error': 'Maximum batch GPU session count of 2 reached.'}
            assert json.loads(intent.read_text())['slug'] == DRAFT_SLUG
            continue
        actual = json.loads(raw)
        assert type(actual.get('kernelId')) is int and actual['kernelId'] > 0
        assert actual.get('versionNumber') == 1 and not actual.get('error')
        terminals = []
        for suffix, statuses in (('-actual-accepted.json', ('COMPLETE',)), ('-actual-failed.json', ('ERROR', 'CANCELLED', 'CANCEL_ACKNOWLEDGED'))):
            p = root / (mode + suffix)
            if p.exists():
                receipt = json.loads(p.read_text())
                assert receipt['kernel_id'] == actual['kernelId'] and receipt['version'] == 1
                assert receipt['status'] in statuses
                terminals.append(receipt)
        assert len(terminals) <= 1
        if not terminals:
            total += cap
    return total


def qualification_training_provenance(receipt, base, model_sha, scope):
    assert receipt['fixed_final_only'] is True and receipt['epoch'] == 3
    assert scope.RECIPE['encoder_lr'] == 3e-6 and scope.RECIPE['head_lr'] == 3e-5
    assert scope.RECIPE['epochs'] == 3 and scope.RECIPE['role'] == 'influence'
    # Exact original source + checkpoint/publication proof, not fabricated defaults
    # for receipt fields the actual numerical notebook never emits.
    assert scope.provenance(base['training'], base['publication'], model_sha) == 3
    assert base['training']['receipt']['completed'] == 1458
    assert base['training']['receipt']['binding']['source_git'] == scope.SOURCE
    return True
