def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def load(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module

def existing_record(db, key):
    found = db.execute('SELECT value FROM meta WHERE key=?', (key,)).fetchone()

def entry_eligibility(w, db, account, submissions, description, candidate):
    assert candidate=='family'
    assert not any(x.get('description')==description for x in submissions),'Exact original description exists; reconcile without retry'
    records=[json.loads(x[0]) for x in db.execute('SELECT value FROM meta') if x[0].lstrip().startswith('{')]
    assert not any('submit' in str(x.get('state','')) and str(x.get('state','')).endswith(('_intent','_uncertain')) for x in records),'Existing uncertain code entry needs read-only reconciliation'
    model=description.split(':')[-1]
    assert not any(x.get('model_sha256')==model and x.get('submission_id') for x in records),'Exact model has an existing entry'
    reference=[x for x in submissions if x.get('ref')==57332456]
    assert len(reference)==1 and w.submission_outcome(reference[0])=='scored'
    slots=w.allowance(db,account)
    return {'eligible':slots>0,'effective_allowance':slots,'active_unresolved_submission_ids':[x['ref'] for x in submissions if w.submission_outcome(x) in ('pending_or_unknown','complete_unscored')],'reference_submission':57332456,'reference_actual_outcome':'scored','unrelated_reference_outcome_not_a_blocker':True,'account':account}
