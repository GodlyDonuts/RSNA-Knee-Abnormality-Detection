def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def load(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module

def entry_eligibility(w, db, account, submissions, description, candidate):
    assert candidate=='paired'
    assert not any(x.get('description')==description for x in submissions),'Exact original description exists; reconcile without retry'

    slots=w.allowance(db,account)

def vm_requirement(mode, lr_fit_future, other_started_scoring_future, paired_future):
    assert mode in ('qualification','competition')
    assert all(type(n) is int and n>=0 for n in (lr_fit_future,other_started_scoring_future,paired_future))
    assert paired_future==0, 'All three actual paired epochs must be complete'
    return lr_fit_future+other_started_scoring_future+3600+(1800 if mode=='qualification' else 0)+32400


def actual_started_other_scoring_future(root):
    # Receipt content must prove terminal; file existence alone never retires a cap.
    total=0
    for mode,cap in (('qualification',1800),('competition',32400)):
        intent=root/(mode+'-push-intent.json');response=root/(mode+'-push-response.json')
        if intent.exists() and not response.exists():
            raise ValueError('Original uncertain other scoring push needs read-only reconciliation')
        if not response.exists():
            continue
        assert intent.exists(), 'Original response requires its retained intent'
        result=json.loads(response.read_text())
        assert type(result.get('kernelId')) is int and result['kernelId']>0 and result.get('versionNumber')==1 and not result.get('error')
        terminal=[]
        for suffix,statuses in (('-actual-accepted.json',('COMPLETE',)),('-actual-failed.json',('ERROR','CANCELLED','CANCEL_ACKNOWLEDGED'))):
            path=root/(mode+suffix)
            if path.exists():
                receipt=json.loads(path.read_text())
                assert receipt['kernel_id']==result['kernelId'] and receipt['version']==1 and receipt['status'] in statuses
                terminal.append(receipt)
        assert len(terminal)<=1
        if not terminal:
            total+=cap
    return total

                        assert receipt['prediction_counts'] == {'started': 6, 'completed': 6, 'failed': 0, 'decode_errors': 0, 'empty_views': 0}
                        csv = load('matched_final_strict_csv', S / 'scripts/kaggle/output_csv.py')
                        checked = csv.verify_csv(csv.download_csv(r, slug, 1), meta['columns'], meta['expected_public_roster_sha256'])
                        assert checked['CSV_sha256'] == receipt['CSV_sha256'] and checked['bytes'] == receipt['CSV_bytes']
                        out['CSV_verified_in_RAM'] = checked
                    out['receipt'] = receipt
                    if not (D / (mode + '-actual-accepted.json')).exists():
                        w.new_json(D / (mode + '-actual-accepted.json'), out)
                        row[mode]['receipt'] = receipt; row['state'] = 'full_parent_epoch_' + mode + '_accepted'; save()
                elif actual_status in ('ERROR', 'CANCELLED', 'CANCEL_ACKNOWLEDGED'):
                    if not (D / (mode + '-actual-failed.json')).exists():
                        w.new_json(D / (mode + '-actual-failed.json'), out)
                        row['state'] = 'full_parent_epoch_' + mode + '_failed'; save()
                return out
            assert mode == 'competition' and actual_status == 'COMPLETE'
            description = 'rsna-final-' + candidate + '-e3:' + model_sha
            if phase in ('eligibility', 'submit'):
                assert row['state'] == 'full_parent_epoch_competition_accepted'
                assert not row.get('submission_id') and not (D / 'entry-submit-intent.json').exists()
                accepted_public = json.loads((D / 'competition-actual-accepted.json').read_text())
                assert accepted_public['receipt'] == row['competition']['receipt']
                assert r.output_json(slug, 1, 'licensed-IID16-epoch-public-accepted.json') == row['competition']['receipt']
                assert accepted_public['CSV_verified_in_RAM']['test_roster_sha256'] == meta['expected_public_roster_sha256']
                fresh = entry_eligibility(w, db, r.account(), r.submissions(), description, candidate)
                out = {'status': 'actual_paired_final_entry_eligibility', 'candidate': candidate, 'fresh': fresh, 'new_mutations': 0}
                if phase == 'submit':
                    assert fresh['eligible'], 'Fresh quota/outcome changed; no entry intent'
                    intent = {'candidate': candidate, 'model_sha256': model_sha, 'source_sha256': source_sha,
                              'kernel_id': row['competition']['kernel_id'], 'slug': slug, 'version': 1,
                              'fileName': 'submission.csv', 'description': description, 'fresh': fresh, 'automatic_retry': False}
                    w.new_json(D / 'entry-submit-intent.json', intent)
                    row.update(state='full_parent_epoch_submit_intent', description=description, attempted_at=w.utcnow()); save()
                    try:
                        owner_name, kernel_slug = slug.split('/')
                        response = r.competition('CreateCodeSubmission', {'fileName': 'submission.csv', 'kernelOwner': owner_name,
                            'kernelSlug': kernel_slug, 'kernelVersion': 1, 'submissionDescription': description})
                        w.new_json(D / 'entry-submit-response.json', response)
                        assert type(response.get('ref')) is int and response['ref'] > 0
                    except BaseException:
                        row['state'] = 'full_parent_epoch_submit_uncertain'; save(); retain = True; raise
                    row.update(state='full_parent_epoch_submitted', submission_id=response['ref']); save()
                    out = {'status': 'one_exact_paired_final_code_entry_accepted', 'candidate': candidate, 'submission_id': response['ref'],
                           'slug': slug, 'version': 1, 'public_CSV_uploaded': False, 'actual_score_known': False}
                    w.new_json(D / 'actual-submitted.json', out)
                return out
            assert phase == 'outcome' and row['state'] in ('full_parent_epoch_submitted', 'full_parent_epoch_scored', 'full_parent_epoch_hidden_failed')
            found = [x for x in r.submissions() if x['ref'] == row['submission_id']]
            assert len(found) == 1
            f = found[0]; outcome = w.submission_outcome(f)
            out = {'status': 'actual_paired_final_hidden_outcome', 'candidate': candidate,
                   'submission_id': f['ref'], 'actual_status': f.get('status'), 'public_score': f.get('publicScore'),
                   'outcome': outcome, 'error_present': bool(f.get('errorDescription')), 'new_mutations': 0}
            if outcome in ('scored', 'failed'):
                row.update(state='full_parent_epoch_scored' if outcome == 'scored' else 'full_parent_epoch_hidden_failed',
                           public_score=f.get('publicScore'), actual_hidden_error_present=bool(f.get('errorDescription'))); save()
                if not (D / 'actual-hidden-terminal.json').exists(): w.new_json(D / 'actual-hidden-terminal.json', out)
            return out
    finally:
        if not retain:
            assert json.loads((claim / 'owner.json').read_text()) == owner
            (claim / 'owner.json').unlink(); claim.rmdir()
