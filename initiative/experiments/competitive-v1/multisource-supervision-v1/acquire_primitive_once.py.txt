"""One private small-file acquisition; no images, fitting, gold targets or entry."""
import hashlib
import json
from pathlib import Path
import shlex
import subprocess

D=Path(__file__).resolve().parent
L=Path('artifacts/multisource-primitives-20261018-v1')
L.mkdir(mode=0o700,parents=True,exist_ok=True)
assert not (L/'intent.json').exists()
git=subprocess.check_output(['git','rev-parse','HEAD'],text=True).strip()
relative=D.relative_to(Path.cwd()).as_posix()
program=subprocess.check_output(['git','show',git+':'+relative+'/primitive_worker.py.txt'],text=True)
compile(program,'private_primitive_report_label_admission','exec')
env=Path('.env');assert env.stat().st_mode&0o077==0
values={k.strip():v.strip().strip('\"\'') for line in env.read_text().splitlines() for k,sep,v in [line.partition('=')] if sep}
with (L/'intent.json').open('x') as f:
    json.dump({'source_git':git,'program_sha256':hashlib.sha256(program.encode()).hexdigest(),'one_CPU_zero_GPU':True,'private_small_files_only':True,'no_automatic_retry':True},f,indent=2)
args=['srun','--account=skattel','--partition=preemptable','--qos=preemptable','--nodelist=evc33','--ntasks=1','--cpus-per-task=1','--gres=none','--mem=2G','--time=00:05:00','--job-name=rsna-primitive-report-source','env','PYTHONNOUSERSITE=1','PYTHONDONTWRITEBYTECODE=1','OMP_NUM_THREADS=1','OPENBLAS_NUM_THREADS=1','/tmp/rsna-environment-907632-7sr9a7i6/environment/bin/python','-B','-c',program]
x=subprocess.run(['ssh','-S','/tmp/rsna-research-20261001-stable.sock','-o','BatchMode=yes','newton',shlex.join(args)],input=json.dumps({'token':values['KAGGLE_API_TOKEN'],'source_git':git}),capture_output=True,text=True,timeout=430)
(L/'result.json').write_text(json.dumps({'returncode':x.returncode,'stdout':x.stdout,'stderr':x.stderr},indent=2)+'\n')
assert x.returncode==0,'Reconcile original acquisition; no duplicate transfer'
out=json.loads(x.stdout)
(D/'actual-primitive-source.json').write_text(json.dumps(out,indent=2)+'\n')
print(json.dumps(out),flush=True)
