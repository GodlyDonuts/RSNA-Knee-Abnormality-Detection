def budget(
    quota,
    *,
    original_main,
    original_sampling,
    reference_unfinished,
    blend_unfinished,
    scout_unfinished,
):
    def seconds(v):
        if not isinstance(v, str) or not v.endswith("s"):
            raise ValueError("Fresh authoritative seconds required")
        n = float(v[:-1])
        if not math.isfinite(n) or n < 0:
            raise ValueError("Finite nonnegative quota required")
        return n

    if any(type(n) is not int or n < 0 for n in (original_main, original_sampling)):
        raise ValueError("All unfinished original liabilities required")
    if any(type(b) is not bool for b in (reference_unfinished, blend_unfinished, scout_unfinished)):
        raise ValueError("Explicit actual sibling state required")
    total, minimum, used, reserved = [
        seconds(quota[k])
        for k in ("totalTimeAllowed", "minimumTimeAllowed", "timeUsed", "timeReserved")
    ]
    if not total >= minimum > 0:
        raise ValueError("Actual quota allowance required")
    allowed = total if quota.get("isPayToScaleEnabled") is True else minimum
    # No speculative reserved-time credit. Retain all actual unfinished caps.
    available = allowed - used - reserved
    required = original_main + original_sampling + 3600 + 24000
    required += 32400 * (reference_unfinished + blend_unfinished) + 1800 * scout_unfinished
    return {
        "allowed": available >= required,
        "available_VM_seconds": available,
        "required_all_future_VM_seconds": required,
        "new_label_campaign_seconds": 24000,
        "new_recovery_seconds": 1800,
        "future_fixed3_fit_seconds": 18000,
        "future_label_serving_seconds": 3600,
        "new_margin_seconds": 600,
        "old_serving_reserve_seconds": 3600,
        "reference_future_seconds": 32400 if reference_unfinished else 0,
        "original_main_seconds": original_main,
        "original_sampling_seconds": original_sampling,
        "old_blend_seconds": 32400 if blend_unfinished else 0,
        "old_scout_seconds": 1800 if scout_unfinished else 0,
        "fresh_quota": quota,
        "no_account_or_billing_change": True,
    }
