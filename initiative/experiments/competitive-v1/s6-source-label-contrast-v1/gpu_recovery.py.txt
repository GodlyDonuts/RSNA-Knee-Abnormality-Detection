def digest(text):
    return hashlib.sha256(text.encode()).hexdigest()


def change_once(text, old, new):
    if text.count(old) != 1:
        raise ValueError("Original accepted template changed; no ambiguous source substitution")
    return text.replace(old, new)

def rewrite_worker(worker):
    """Make only explicit data-role, reader-wrapper and output-root substitutions."""
    worker = change_once(
        worker,
        "parser.add_argument('role',choices=('uniform','influence'))",
        "parser.add_argument('role',choices=('original_source','released_S6_source'))",
    )
    worker = change_once(worker, "def main():\n", HELPER + "def main():\n")
    worker = change_once(
        worker,
        "original_data=PixelCases(targets,list(range(16)));data=ProposalCases(original_data,_teacher_proposals(model,original_data));initial_identity=_initial_identity(model,device);engineering=",
        "original_data=PixelCases(targets,list(range(16)));proposals=_teacher_proposals(model,original_data);data=ProposalCases(_source_labels(original_data,a.role,root),proposals);initial_identity=_initial_identity(model,device);engineering=",
    )
    worker = change_once(worker, "Recipe(epochs=2,role=a.role)", "Recipe(epochs=2,role='uniform')")
    # Original feature and target lookup remains exact. Only this owned output changes.
    worker = change_once(
        worker,
        "Path('/kaggle/working/importance-recovery')",
        "Path('/kaggle/working/s6-source-label-recovery')",
    )
    compile(worker, "actual_S6_label_only_GPU_worker", "exec")
    return worker


def build(original, overlay, sampled):
    if digest(original) != ORIGINAL_SHA:
        raise ValueError("Exact accepted original IID16 two-T4 diagnostic required")
    if digest(overlay) != OVERLAY_SHA or digest(sampled) != SAMPLED_SHA:
        raise ValueError("Exact actual-array/CPU-engine-admitted source label helpers required")
    values = {}
    for n in ast.parse(original).body:
        if isinstance(n, ast.Assign) and len(n.targets) == 1:
            target = n.targets[0]
            if isinstance(target, ast.Name) and target.id in (
                "FILES",
                "WORKER",
                "TARGETS",
                "PERMISSION",
            ):
                values[target.id] = ast.literal_eval(n.value)
    if set(values) != {"FILES", "WORKER", "TARGETS", "PERMISSION"}:
        raise ValueError("Complete original private closure/targets/permission literals required")
    worker = base64.b64decode(values["WORKER"], validate=True).decode()
    if digest(worker) != ORIGINAL_WORKER_SHA:
        raise ValueError("Exact actually accepted original numerical worker required")
    for name, expected in {
        "rsna/competitive/licensed_importance_train.py": "e024a8ee91d5d64967e6128bfeac1ab2774468613ccaa43d13fc8ea569a5de7d",
        "rsna/competitive/licensed_importance.py": "fc3fcedfc52fc922b82f7cb8e2034b6d7526d65c536c2ae8163aa8de1d3dafe4",
        "rsna/competitive/licensed_parent_train.py": "cf238d9154806a51cd92c80a61e39ca344ef865558305db51cbfdc86f89eeec0",
    }.items():
        if digest(values["FILES"].get(name, "")) != expected:
            raise ValueError("Unchanged accepted optimization/sampling engines required")
    worker = rewrite_worker(worker)
    files = {
        **values["FILES"],
        "s6_labels/source_overlay.py.txt": overlay,
        "s6_labels/sampled_fit.py.txt": sampled,
    }
    bootstrap = BOOTSTRAP
    for key, value in {

        "CONTROL_PREDICTION_SHA_VALUE": CONTROL_PREDICTION_SHA,
    }.items():

            raise ValueError("Unique complete bootstrap literal required")
        bootstrap = bootstrap.replace(key, repr(value))
    compile(bootstrap, "actual_private_S6_label_recovery", "exec")
    return bootstrap, {
        "source_sha256": digest(bootstrap),
        "worker_sha256": digest(worker),

        "original_module_sha256": {k: digest(v) for k, v in values["FILES"].items()},
        "overlay_sha256": OVERLAY_SHA,
        "sampled_adapter_sha256": SAMPLED_SHA,
        "released_S6_target_sha256": TARGET_SHA,
