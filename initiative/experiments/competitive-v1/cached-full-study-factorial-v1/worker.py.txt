def digest(path):return hashlib.sha256(Path(path).read_bytes()).hexdigest()
def write_new(path,value):

class Cases:
    def __init__(self,base,uid_list,label,mask,teacher,indices):
        self.base=base;self.indices=list(indices);self.uids=[uid_list[i] for i in self.indices]
        self.labels=torch.from_numpy(label.copy());self.masks=torch.from_numpy(mask.copy());self.teacher=torch.from_numpy(teacher.copy())
    def __len__(self):return len(self.indices)
    def case(self,index,positions,mirror):
        i=self.indices[index];raw=self.base[i]
        assert torch.equal(raw[1],self.teacher[i])
        return {'features':raw[0][int(mirror),positions].contiguous(),'parent_logits':self.teacher[i,int(mirror)],
                'labels':self.labels[i],'mask':self.masks[i]}

def equal(a,b):
    if isinstance(a,torch.Tensor):return isinstance(b,torch.Tensor) and a.dtype==b.dtype and a.shape==b.shape and torch.equal(a,b)
    if isinstance(a,dict):return isinstance(b,dict) and a.keys()==b.keys() and all(equal(a[k],b[k]) for k in a)
    if isinstance(a,(tuple,list)):return type(a) is type(b) and len(a)==len(b) and all(equal(x,y) for x,y in zip(a,b))
    return type(a) is type(b) and a==b


def main(phase,source_git):

    staged=json.loads((R/'staged.json').read_text());binding['source_files_sha256']=staged['source_files_sha256']

        for i in range(32):
            f=base[i][0];actual=model(f).sigmoid().mean(0).detach().numpy();reference=readout_moments(original,f)[3]
            assert np.array_equal(actual,reference);cached=1/(1+np.exp(-teacher[i].astype(np.float64)));maximum=max(maximum,float(np.max(np.abs(actual-cached.mean(0)))))
        assert maximum<=.003
        out={'status':'actual_full3887_cached_feature_and_old_new_label_binding_with32_FP32_initial_readout_accepted','CPU_job':os.environ['SLURM_JOB_ID'],

            'encoder_optimizer_or_fit_executed':False,'new_weights_pixels_features_or_dataset_transfers':0,'source_files_sha256':staged['source_files_sha256'],
            'new_shared_bytes_files':0,'clinical_or_Kaggle_accuracy_claim':False,'binding':binding}
        write_new(R/'input-admitted.json',out);return out
    assert (R/'input-admitted.json').exists()
    role,window=sys.argv[3],int(sys.argv[4]);assert role in labels and window in (16,94)
    mode=sys.argv[5]
    engineering=phase=='engineering'
    assert phase in ('engineering','full')
    recipe=engine.Recipe(role,window,epochs=2 if engineering else 3)
    D=R/('engineering' if engineering else 'full')/(role+'-'+str(window))/mode

    binding.update(diagnostic_only=engineering,source_label_role=role,source_label_array_sha256=hashlib.sha256(labels[role].tobytes()).hexdigest(),

        __import__('shutil').copy2(D/'run.json',resume/'run.json')
